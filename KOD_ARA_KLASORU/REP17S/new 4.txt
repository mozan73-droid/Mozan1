Function CreateReportRowSet()
	Dim RaporRowSet
	Dim Rep17SE
	Dim AlanBilgisi

	Set RaporRowSet=Doc.CreateRowSetObject("REPORT1=STOK10A","REPORT")

	Set Rep17SE=Doc.GetTableObject("REP17SE")

	If Rep17SE.BELLEKTEISLENECEKR>0 Then RaporRowSet.Set_Recs_on_Memory Rep17SE.BELLEKTEISLENECEKR

	Set AlanBilgisi=Doc.CreateSObject("FieldDefinition")

	AlanBilgisi.Clear()
	AlanBilgisi.Name="KOD->STOK00..."

	RaporRowSet.AddField(AlanBilgisi)

	AlanBilgisi.Clear()
	AlanBilgisi.Name="TL_BIRIMMALIYET"
	AlanBilgisi.DataType="N"
	AlanBilgisi.Length=15
	AlanBilgisi.SigDig=0
	RaporRowSet.AddField(AlanBilgisi)
	AlanBilgisi.Clear()
	AlanBilgisi.Name="MALIYET"
	AlanBilgisi.DataType="N"
	AlanBilgisi.Length=16
	AlanBilgisi.SigDig=0
	RaporRowSet.AddField(AlanBilgisi)

	Set CreateReportRowSet=RaporRowSet
End Function

Sub CreateKriterTable(RaporRowSet)

	If Not IsObject(TablodakiAlanBilgisi) Then Exit Sub

	Alanismi=TablodakiAlanBilgisi.Name
	TablodakiAlanBilgisi.Name="E_"&Alanismi
	TablodakiAlanBilgisi.FTFFnc=""

	Call Guess_If_Date_Field(TablodakiAlanBilgisi)

	KriterGirisTablosu.AddField(TablodakiAlanBilgisi)

	TablodakiAlanBilgisi.Name="EQ_"&Alanismi
	TablodakiAlanBilgisi.FTFFnc=""
	KriterGirisTablosu.AddField(TablodakiAlanBilgisi)

Sub PrintReport(RT, RO)
	Dim pe1
	Dim satirSayisi
	Dim I
	Dim Pdf_File_Name
	Dim Html_File_Name

	Doc.SetDocParameterValue "SONYARATILANEXCELCIKTISI",""
	Doc.SetDocParameterValue "SONYARATILANPDFCIKTISI",""
	Doc.SetDocParameterValue "SONYARATILANSMSCIKTISI",""
	Doc.SetDocParameterValue "SONYARATILANHTMLCIKTISI",""
	Doc.SetDocParameterValue "SONYARATILANODSCIKTISI",""	
	
	If Len(PV_OUTPUT)=1 Then
		If PV_OUTPUT="E" Then
			PrintReport_Excel RT, RO

			Exit Sub

		ElseIf PV_OUTPUT="O" Then
			PrintReport_Ods RT, RO
			Exit Sub	

		ElseIf PV_OUTPUT="S" Then
			PrintReport_SMSText RT, RO
			Exit Sub

		ElseIf PV_OUTPUT="X" Then
			Exit Sub
		End If
	End If

	Set pe1 = doc.CreateSObject("PrintEngineOutput")

	pe1.MsgOpt=0

	Set PV_RT=RT
	Set PV_RO=RO
	Set PV_PE1=pe1

	PV_SayfaNo=0
	PV_SayfaIciSatirNo=0
	PV_SayfaBasindanBasilanDataRowSayisi=0
	PV_SatirdaBasilacakAlanVarmi_Data=1

	ret = PV_PE1.Open(0,"",-1,0,"Portrait")

	Call SetFormatAttributes("V")
	Call SayfaBasi()

	satirSayisi = PV_RT.GetRecCount()

	For I=1 To satirSayisi
		If Int(I/20)*20=I Then
			If Doc.EscPressed() Then PV_PE1.Close

			Doc.StatusBarMessage("Çıktı Hazırlanıyor... (%%" & CStr(Int(I*100/satirSayisi))  & ")" )
		End If

		If PV_RT.REPORT___SUBTOTAL_B(I)>="1" And PV_RT.REPORT___SUBTOTAL_B(I)<="5" Then
			If PV_RT.REPORT___SUBTOTAL_N_1(I)>=0 Then Call HeaderRowYazdir("1",PV_RT.REPORT___SUBTOTAL_N_1(I))
			If PV_RT.REPORT___SUBTOTAL_N_2(I)>=0 Then Call HeaderRowYazdir("2",PV_RT.REPORT___SUBTOTAL_N_2(I))
			If PV_RT.REPORT___SUBTOTAL_N_3(I)>=0 Then Call HeaderRowYazdir("3",PV_RT.REPORT___SUBTOTAL_N_3(I))
			If PV_RT.REPORT___SUBTOTAL_N_4(I)>=0 Then Call HeaderRowYazdir("4",PV_RT.REPORT___SUBTOTAL_N_4(I))
			If PV_RT.REPORT___SUBTOTAL_N_5(I)>=0 Then Call HeaderRowYazdir("5",PV_RT.REPORT___SUBTOTAL_N_5(I))
		End If

		Call SatirYazdir(I)
	Next

	Call SayfaSonu()
	Call RaporSonu()

	If PV_OUTPUT="P" Then
		Pdf_File_Name=Doc.TempFileName_Client("pdf")

		If PV_PE1.WriteDoc(Pdf_File_Name)<>0 Then
			PV_PE1.Close
			Doc.SetDocParameterValue "SONYARATILANPDFCIKTISI",Pdf_File_Name

	ElseIf PV_OUTPUT="H" Then

	End If

Sub Excel_SatirYazdir(satirNo)
	Dim RowType
	Dim RowTypePrev
	Dim AlanSayisi
	Dim F
	Dim FieldDef
	Dim OutputStr
	Dim ncol

	RowType=PV_RT.REPORT___ROWTYPE(satirNo)
	RowTypePrev=" "
	If satirNo>1 Then RowTypePrev=PV_RT.REPORT___ROWTYPE(satirNo-1)

	If (RowType>="1" AND RowType<= "9") Then
		Select Case RowType
		Case "1" :
				If REP17SE.ST1_PRINTOPT="0" Then Exit Sub
		Case "2" :
				If REP17SE.ST2_PRINTOPT="0" Then Exit Sub
		Case "3" :
				If REP17SE.ST3_PRINTOPT="0" Then Exit Sub
		Case "4" :
				If REP17SE.ST4_PRINTOPT="0" Then Exit Sub
		Case "5" :
				If REP17SE.ST5_PRINTOPT="0" Then Exit Sub
		End Select
	End If

	If RowType="T" Then

		If RowTypePrev="T" Then Exit Sub

		Select Case RowTypePrev
		Case "T" : Exit Sub
		Case "A" :
				If REP17SE.OT1_GENELTOPLAM="0" Then Exit Sub
		Case "B" :
				If REP17SE.OT2_GENELTOPLAM="0" Then Exit Sub
		Case "C" :
				If REP17SE.OT3_GENELTOPLAM="0" Then Exit Sub
		Case "D" :
				If REP17SE.OT4_GENELTOPLAM="0" Then Exit Sub
		Case "E" :
				If REP17SE.OT5_GENELTOPLAM="0" Then Exit Sub
		Case Else :
				If REP17SE.R_OUTF_GENELTOPLAM="0" Then Exit Sub
		End Select

	End If

	AlanSayisi=PV_RO.GetNumberOfFields()

	If Not SatirdaBasilacakAlanVarmi(AlanSayisi,RowType) Then
		If RowType=" " Then PV_SatirdaBasilacakAlanVarmi_Data=0
		PV_EXCEL_PE2.InitRowValues
		If PV_SatirdaBasilacakAlanVarmi_Data>0 Then
			PV_EXCEL_PE2.SetRowValue_String -1,""
			Select Case RowType
			Case "1","2","3","4","5"		PV_EXCEL_PE2.SubTotalOut
			Case "A","B","C","D","E","T"	PV_EXCEL_PE2.GrandTotalOut
			Case Else						PV_EXCEL_PE2.LineOut
			End Select
		End If
		Exit Sub
	End If

	PV_EXCEL_PE2.InitRowValues
	PV_EXCEL_PE2.SetRowValue_String -1,""

	ncol=2

    For F=1 To AlanSayisi
		If PV_RO.BuAlanBasilacakmi(F,RowType)=1 Then
			Set FieldDef=PV_RO.GetFieldDefN(F)
			If FieldDef.PRINTOPT=1 Then
				Dim OutputStr2
				If FieldDef.IsNumeric() Then
					OutputStr=PV_RT.GetFieldValue(FieldDef.Name,satirNo)
					If FieldDef.TEKRARBASILMAMAOPT>0 Then
						If satirNo>1 And RowTypePrev=" " And RowType=" " Then
							OutputStr2=Doc.NumFormat_Gen(PV_RT.GetFieldValue(FieldDef.Name,satirNo-1),FieldDef.Length,FieldDef.SigDig,0)
							If OutputStr2=OutputStr Then OutputStr=0
						End If
					End If
					'PV_EXCEL_PE2.SetRowValue_Double -1,PV_RT.GetFieldValue(FieldDef.Name,satirNo)
					PV_EXCEL_PE2.SetRowValue_Double -1,OutputStr
				ElseIf FieldDef.IsDate() Then
					OutputStr=Doc.StrToNorDate(PV_RT.GetFieldValue(FieldDef.Name,satirNo))
					OutputStr=Left(Trim(OutputStr),FieldDef.Length)
					If FieldDef.TEKRARBASILMAMAOPT>0 Then
						If satirNo>1 And RowTypePrev=" " And RowType=" " Then
							OutputStr2=Doc.StrToNorDate(PV_RT.GetFieldValue(FieldDef.Name,satirNo-1))
							OutputStr2=Left(Trim(OutputStr2),FieldDef.Length)
							If OutputStr2=OutputStr Then OutputStr=""
						End If
					End If
				
					'OutputStr=Doc.StrToNorDate(PV_RT.GetFieldValue(FieldDef.Name,satirNo))
					'OutputStr=Left(Trim(OutputStr),FieldDef.Length)
					
					PV_EXCEL_PE2.SetRowValue_String -1,OutputStr
				Else
					
					OutputStr=Left(Trim(PV_RT.GetFieldValue(FieldDef.Name,satirNo)),FieldDef.Length)
					If FieldDef.TEKRARBASILMAMAOPT>0 Then
						If satirNo>1 And RowTypePrev=" " And RowType=" " Then
							OutputStr2=Left(Trim(PV_RT.GetFieldValue(FieldDef.Name,satirNo-1)),FieldDef.Length)
							If OutputStr2=OutputStr Then OutputStr=""
						End If
					End If
					
					PV_EXCEL_PE2.SetRowValue_String -1,OutputStr
				End If
				ncol=ncol+1
			End If
		ElseIf PV_RO.BuAlanBasilacakmi(F,"*")=1 Then
			PV_EXCEL_PE2.SetRowValue_String -1,""
		End If
	Next

	Select Case RowType
		Case "1","2","3","4","5"		PV_EXCEL_PE2.SubTotalOut
		Case "A","B","C","D","E","T"	PV_EXCEL_PE2.GrandTotalOut
		Case Else					PV_EXCEL_PE2.LineOut
	End Select

	PV_EXCEL_ROW=PV_EXCEL_ROW+1
	PV_EXCEL_ROW_SHEET=PV_EXCEL_ROW_SHEET+1

	If RowType=" " Or ( RowType>="A" AND RowType<="E" ) Then PV_SayfaBasindanBasilanDataRowSayisi=PV_SayfaBasindanBasilanDataRowSayisi+1
End Sub

Sub Excel_HeaderRowYazdir(rowtypest,satirNo)
	Dim SayfaBasiGereklimi
	Dim RowType
	Dim AlanSayisi
	Dim F
	Dim FieldDef
	Dim OutputStr
	Dim AraToplamaBasilacakAlanVarmi
	Dim ATHeaderString
	Dim FBaslik

	Select Case rowtypest
		Case "1" :
				If REP17SE.ST1H_PRINTOPT="0" Then Exit Sub
		Case "2" :
				If REP17SE.ST2H_PRINTOPT="0" Then Exit Sub
		Case "3" :
				If REP17SE.ST3H_PRINTOPT="0" Then Exit Sub
		Case "4" :
				If REP17SE.ST4H_PRINTOPT="0" Then Exit Sub
		Case "5" :
				If REP17SE.ST5H_PRINTOPT="0" Then Exit Sub
	End Select

	SayfaBasiGereklimi=0

	If SayfaBasiGereklimi=0 Then
		Select Case rowtypest
		Case "1" If REP17SE.ST1_YENISAYFA="1" Then SayfaBasiGereklimi=1
		Case "2" If REP17SE.ST2_YENISAYFA="1" Then SayfaBasiGereklimi=1
		Case "3" If REP17SE.ST3_YENISAYFA="1" Then SayfaBasiGereklimi=1
		Case "4" If REP17SE.ST4_YENISAYFA="1" Then SayfaBasiGereklimi=1
		Case "5" If REP17SE.ST5_YENISAYFA="1" Then SayfaBasiGereklimi=1
		End Select
	End If

	RowType=rowtypest
	RowType=Trim(RowType)&"H"

	AlanSayisi=PV_RO.GetNumberOfFields()

	AraToplamaBasilacakAlanVarmi=0

	ATHeaderString=" "

	For F=1 To AlanSayisi
		If PV_RO.BuAlanBasilacakmi(F,RowType)=1 Then
			AraToplamaBasilacakAlanVarmi=AraToplamaBasilacakAlanVarmi+1

			Set FieldDef=PV_RO.GetFieldDefN(F)
		
			FBaslik=Trim(FieldDef.ARaporAciklamasi)

			If FieldDef.IsNumeric() Then
				OutputStr=Doc.NumFormat_Gen(PV_RT.GetFieldValue(FieldDef.Name,satirNo),FieldDef.Length,FieldDef.SigDig,0)
				OutputStr=Trim(OutputStr)
				If AraToplamaBasilacakAlanVarmi>1 Then ATHeaderString=ATHeaderString+",  "
				ATHeaderString=ATHeaderString+"["
				ATHeaderString=ATHeaderString+FBaslik
				ATHeaderString=ATHeaderString+"="
				ATHeaderString=ATHeaderString+OutputStr
				ATHeaderString=ATHeaderString+"]"

			ElseIf FieldDef.IsDate() Then
				OutputStr=Doc.StrToNorDate(PV_RT.GetFieldValue(FieldDef.Name,satirNo))
				OutputStr=Trim(OutputStr)
				If AraToplamaBasilacakAlanVarmi>1 Then ATHeaderString=ATHeaderString+",  "
				ATHeaderString=ATHeaderString+"["
				ATHeaderString=ATHeaderString+FBaslik
				ATHeaderString=ATHeaderString+"="
				ATHeaderString=ATHeaderString+OutputStr
				ATHeaderString=ATHeaderString+"]"

			Else
				If AraToplamaBasilacakAlanVarmi>1 Then ATHeaderString=ATHeaderString+",  "
				ATHeaderString=ATHeaderString+"["
				ATHeaderString=ATHeaderString+FBaslik
				ATHeaderString=ATHeaderString+"="
				ATHeaderString=ATHeaderString+Trim(PV_RT.GetFieldValue(FieldDef.Name,satirNo))
				ATHeaderString=ATHeaderString+"]"
			End If
		End If
	Next

	If AraToplamaBasilacakAlanVarmi=0 Then Exit Sub

	PV_EXCEL_PE2.InitRowValues
	PV_EXCEL_PE2.SetRowValue_String -1,ATHeaderString
	PV_EXCEL_PE2.SubTotalHeaderOut
	PV_EXCEL_ROW=PV_EXCEL_ROW+1
	PV_EXCEL_ROW_SHEET=PV_EXCEL_ROW_SHEET+1
End Sub

'Kullanıcı Tanımlı Kriterler Rapor başlığında çıkmıyor idi
Sub AddReportHeader_CriQ(RRS,FN,CT,FV1,FV2,KRITERFIELDNAME,KRITERROWSET,DefAciklama)

	Dim str
	Dim AlanBilgisi
	Set AlanBilgisi=KRITERROWSET.GetFieldDef_E(KRITERFIELDNAME)
	If Not IsObject(AlanBilgisi) Then 
		Set AlanBilgisi=RRS.GetFieldDef_E(FN)
		If Not IsObject(AlanBilgisi) Then Exit Sub
	End If
	Dim AAciklama
	AAciklama=""
	If IsObject(AlanBilgisi) Then
		AAciklama=Trim(AlanBilgisi.ARaporAciklamasi)
	End If
	If Not Doc.Isblank(DefAciklama) Then AAciklama=DefAciklama

	If Len(AAciklama)>0 Then
		str=AAciklama&"("&KRITERFIELDNAME&")"
	Else
		str=KRITERFIELDNAME
	End If

	Select Case CT
	Case "><"
		If Doc.Isblank(FV1) And Doc.Isblank(FV2) Then Exit Sub
		If Doc.Isblank(FV1) Or Doc.Isblank(FV2) Then
			AddReportHeader_Cri2 RRS,FN,">=",FV1,""
			AddReportHeader_Cri2 RRS,FN,"<=",FV2,""
			Exit Sub
		End If
		If FV1=FV2 Then
			AddReportHeader_Cri2 RRS,FN,"=",FV1,""
			Exit Sub
		End If
		str=str&" "&"="&" "
		str=str&"["&Trim(FV1)&"]"
		str=str&" -> "
		str=str&"["&Trim(FV2)&"]"
	Case "BOS"
		str=str&" = Boş Olanlar"
	Case "!BOS"
		str=str&" = Dolu Olanlar"
	Case Else
		If Doc.Isblank(FV1) Then Exit Sub
		str=str&" "&CT&" "
		str=str&"["&Trim(FV1)&"]"
	End Select
	If Len(PV_KRITERACIKLAMASI)>0 Then
		PV_KRITERACIKLAMASI=PV_KRITERACIKLAMASI&CHR(13)&CHR(10)
	End If
	PV_KRITERACIKLAMASI=PV_KRITERACIKLAMASI&str
End Sub

Function SatirdaBasilacakAlanVarmi(AlanSayisi,RowType)
	SatirdaBasilacakAlanVarmi=False
	Dim F
    For F=1 To AlanSayisi
		If PV_RO.BuAlanBasilacakmi(F,RowType)=1 Then

Function PrintReport_SMSText_WriteToFile(ByRef FileName,ByRef ResultText)
	PrintReport_SMSText_WriteToFile=0
	On Error Resume Next
    Err.Clear
    Err.Number = 0
	Dim FSO1
	Dim TF1
	Set FSO1=CreateObject("Scripting.FileSystemObject")
	Set TF1=FSO1.CreateTextFile(FileName,True)
	TF1.Write(ResultText)
	TF1.Close
	Set TF1=Nothing
	Set FSO1=Nothing
    On Error GoTo 0
    If Err.Number <> 0 Then PrintReport_SMSText_WriteToFile=Err.Number
    Err.Clear
    Err.Number = 0
End Function

Sub SMSText_SatirYazdir(ByVal satirNo,ByRef ResultText)
	Dim RowType
	Dim RowTypePrev
	RowType=PV_RT.REPORT___ROWTYPE(satirNo)
	RowTypePrev=" "
	If satirNo>1 Then RowTypePrev=PV_RT.REPORT___ROWTYPE(satirNo-1)
	If RowType<>"T" And RowType<>" " Then Exit Sub

	If RowType="T" Then
		If RowTypePrev="T" Then Exit Sub
		Select Case RowTypePrev
		Case "T" : Exit Sub
		Case "A" : Exit Sub
		Case "B" : Exit Sub
		Case "C" : Exit Sub
		Case "D" : Exit Sub
		Case "E" : Exit Sub
		Case Else :
				If REP17SE.R_OUTF_GENELTOPLAM="0" Then Exit Sub
		End Select
	End If

	Dim AlanSayisi
	AlanSayisi=PV_RO.GetNumberOfFields()
	Dim F
	Dim FieldDef
	Dim OutputStr

	If Not SatirdaBasilacakAlanVarmi(AlanSayisi,RowType) Then
		If RowType=" " Then PV_SatirdaBasilacakAlanVarmi_Data=0
		Exit Sub
	End If
	Dim ResultText_Dolu
	ResultText_Dolu=False
	Dim ResultText_Satir
    For F=1 To AlanSayisi
		If PV_RO.BuAlanBasilacakmi(F,RowType)=1 Then
			Set FieldDef=PV_RO.GetFieldDefN(F)
			Dim FBaslik
			FBaslik=Trim(FieldDef.ARaporAciklamasi)
			FBaslik=Trim(FieldDef.OUTF_ACIKLAMA)
			FBaslik=Replace(FBaslik,"@@@LF@@@",vbLf)
			FBaslik=Replace(FBaslik,"@@@YENISATIR@@@",vbLf)
			If FieldDef.PRINTOPT=1 Then
				If FieldDef.IsNumeric() Then
					OutputStr=Doc.NumFormat_Gen(PV_RT.GetFieldValue(FieldDef.Name,satirNo),FieldDef.Length,FieldDef.SigDig,0)
					OutputStr=Trim(OutputStr)
				ElseIf FieldDef.IsDate() Then
					OutputStr=Doc.StrToNorDate(PV_RT.GetFieldValue(FieldDef.Name,satirNo))
					OutputStr=Left(Trim(OutputStr),FieldDef.Length)
				Else
					OutputStr=Left(Trim(PV_RT.GetFieldValue(FieldDef.Name,satirNo)),FieldDef.Length)
				End If
				If Not ResultText_Dolu Then
					ResultText_Satir=ResultText_Satir & vbLf
				Else
					ResultText_Satir=ResultText_Satir & " "
				End If
				If RowType="T" And Len(FBaslik)>0 Then
					If Left(FBaslik,1) = vbLf Then
						FBaslik=vbLf & "Toplam " & Trim(Mid(FBaslik,2))
					Else
						FBaslik="Toplam " & FBaslik
					End If
				End If
				ResultText_Satir=ResultText_Satir & FBaslik
				ResultText_Satir=ResultText_Satir & OutputStr
				ResultText_Dolu=True
			End If
		End If
	Next
	ResultText=ResultText & ResultText_Satir
'2025/07/08
'	If RowType=" " Then PV_SayfaBasindanBasilanDataRowSayisi=PV_SayfaBasindanBasilanDataRowSayisi+1
	If RowType=" " Or ( RowType>="A" AND RowType<="E" ) Then PV_SayfaBasindanBasilanDataRowSayisi=PV_SayfaBasindanBasilanDataRowSayisi+1
End Sub
Sub PrintReport_SMSText(RT,RO)
	Dim SMSText_File_Name
	SMSText_File_Name=Doc.TempFileName_Client("txt")
	Set PV_RT=RT
	Set PV_RO=RO
	Call SetFormatAttributes("S")
	Dim ResultText
	ResultText=PV_RAPORACIKLAMASI
	If Len(ResultText)>0 Then ResultText=ResultText & " "
	If REP17SE.HX_TARIHFORMAT="1" Then
		Dim ct
		Dim pTime
		ct=Now
		pTime=Right("0" & Day(ct),2) & "/" & Right("0" & Month(ct),2) & "/" & Year(ct)
		ResultText=ResultText & pTime & " " & FormatDateTime(ct,4)
	End If


	Dim satirSayisi
	satirSayisi = PV_RT.GetRecCount()
	Dim I
	For I=1 To satirSayisi
		If Int(I/20)*20=I Then
			If Doc.EscPressed() Then
				Exit Sub
			End If
			Doc.StatusBarMessage("Çıktı Hazırlanıyor... (%%" & CStr(Int(I*100/satirSayisi))  & ")" )
		End If
		Call SMSText_SatirYazdir(I,ResultText)
	Next
	ResultText=Replace(ResultText,vbLf,vbCrLf)
	If PrintReport_SMSText_WriteToFile(SMSText_File_Name,ResultText)=0 Then
		Doc.SetDocParameterValue "SONYARATILANSMSCIKTISI",SMSText_File_Name





	End If
End Sub

		CizgiGenisligi=REP17SE.H1_SONRACIZGI_CM*10
		CizgiRengi=RGB(192,192,192)
		If Trim(REP17SE.H1_SONRACIZGI_CLR)<>"" Then CizgiRengi=Doc.C9ToCRGB(REP17SE.H1_SONRACIZGI_CLR)

		PV_PE1.SetPenColor CizgiRengi
		PV_PE1.Line PV_H1L2VPOS,PV_mmleft_margin,0,PV_MMPRINTABLEPAGEWIDTHMM

		CizgiGenisligi=REP17SE.ALTUSTCIZGI_CM*10

		PV_PE1.SetPenSize CizgiGenisligi
		PV_PE1.SetPenColor CizgiRengi
		PV_PE1.Line PV_ULVPOS,PV_mmleft_margin,0,PV_MMPRINTABLEPAGEWIDTHMM

Sub AddStringParameterToString(ByRef paremeters,ByVal parametername,ByVal parametervalue)
	parametervalue=Replace(parametervalue,"'"," ")
	parametervalue=Trim(parametervalue)
'2009/12/21
	parametervalue=Doc.ReplaceDocValues(parametervalue)
	parametername=Trim(parametername)
	If parametervalue="" Then Exit Sub
	If Len(paremeters)>0 Then paremeters=paremeters & ", "
	paremeters=paremeters & parametername
	paremeters=paremeters & "="
	paremeters=paremeters & "'"
	paremeters=paremeters & parametervalue
	paremeters=paremeters & "'"
End Sub
Sub AddNumericParameterToString(ByRef paremeters,ByVal parametername,ByVal parametervalue)
	parametername=Trim(parametername)
	If parametervalue=0 Then Exit Sub
	If Len(paremeters)>0 Then paremeters=paremeters & ", "
	paremeters=paremeters & parametername
	paremeters=paremeters & "="
	paremeters=paremeters & "'"
	paremeters=paremeters & Doc.CStr(parametervalue)
	paremeters=paremeters & "'"
End Sub

Sub BirCizelge(Tbl_REP17CT,Tbl_REP17CT1)
		Dim Cizelge_Turu
		Cizelge_Turu=Trim(Tbl_REP17CT.CHART_TYPE)
		Dim Cizelge_Row_Type
		Cizelge_Row_Type=Trim(Tbl_REP17CT.REPORT_ROW_TYPE)
		Dim PAGE_AUTO_PORT_LANDS
		PAGE_AUTO_PORT_LANDS=Tbl_REP17CT.PAGE_AUTO_PORT_LANDS
		Dim Cizelge_X_Alan
		Cizelge_X_Alan=Trim(Tbl_REP17CT.FIELDNAME1)
		Dim Cizelge_Y_Alanlar
		Cizelge_Y_Alanlar=""
		Dim J
		For J=1 To Tbl_REP17CT1.GetRowCount
			If Tbl_REP17CT1.AP10="1" Then
				If Len(Cizelge_Y_Alanlar)>0 Then Cizelge_Y_Alanlar=Cizelge_Y_Alanlar&","
				Cizelge_Y_Alanlar=Cizelge_Y_Alanlar&Trim(Tbl_REP17CT1.FIELDNAME1(J))
			End If
		Next

		If Trim(Cizelge_X_Alan)="" Then Exit Sub
		If Trim(Cizelge_Y_Alanlar)="" Then Exit Sub
		If Tbl_REP17CT.PAGE_AUTO_BREAK="1" Then
			PV_PE1.PageBreak()
			If PAGE_AUTO_PORT_LANDS="L" Then PV_PE1.SetPageOrientation "LANDSCAPE"
			If PAGE_AUTO_PORT_LANDS="P" Then PV_PE1.SetPageOrientation "PORTRAIT"
			Call SayfaBasi0
		End If

		If Cizelge_Row_Type="R" Then Cizelge_Row_Type=" "
		Dim Tbl
		Dim KriterString
		KriterString="REPORT___ROWTYPE=='"&Cizelge_Row_Type&"'"

'2015/02/14
		If Not Doc.IsBlank(Tbl_REP17CT.DATA_FILTER_CRITERIA) Then 
			KriterString=KriterString & " & " & Trim(Tbl_REP17CT.DATA_FILTER_CRITERIA)
		End If

		If Cizelge_Row_Type="T" Then
			Set Tbl=Doc.Select2("__1","REPORT1",KriterString,"",1)
		Else
			Set Tbl=Doc.Select2("__1","REPORT1",KriterString,"",-1)
		End If
		Dim Alanlar
		Alanlar=Cizelge_X_Alan
		Alanlar=Alanlar&","
		Alanlar=Alanlar&Cizelge_Y_Alanlar
		Dim hData
		Dim PrepdataParameters
		If Tbl_REP17CT.DATAX_CR="C" Then
			PrepdataParameters="TRANSPOSE=1"
			If Tbl_REP17CT.DATAX_ROWNO>0 Then
				PrepdataParameters=PrepdataParameters & ",ROWNUMBER="&CStr(CInt(Tbl_REP17CT.DATAX_ROWNO))
			End If
			PrepdataParameters=PrepdataParameters & ", CREATELEGENDS=1"
		End If

		hData=PV_PE1.CreateChartData_RowSet(Tbl,Alanlar,PrepdataParameters)
		If hData<0 Then Exit Sub
		Dim Label1
		If Tbl_REP17CT.DATAX_CR="C" Then
'CreateChartData_RowSet legentları yaratıyor
			For J=1 To Tbl_REP17CT1.GetRowCount
				Label1=FormatAlanBasligi1(Trim(Tbl_REP17CT1.FIELDNAME1(J)))
'2009/12/21
				Label1=Doc.ReplaceDocValues(Label1)
				If Tbl_REP17CT1.AP10="1" Then
					PV_PE1.ChartDataAddXLabel hData,Label1
				End If
			Next
		Else
			For J=1 To Tbl_REP17CT1.GetRowCount
				Label1=Trim(Tbl_REP17CT1.ACIKLAMA(J))
				If Label1="" Then  Label1=FormatAlanBasligi1(Trim(Tbl_REP17CT1.FIELDNAME1(J)))
				If Tbl_REP17CT1.AP10="1" Then
					PV_PE1.ChartDataAddLegend hData,Label1
					Dim CProperties
					CProperties=""
					Call AddStringParameterToString(CProperties,"COLOR",Tbl_REP17CT1.COLOR1(J))
					Call AddStringParameterToString(CProperties,"LINESTYLE",Tbl_REP17CT1.LINESTYLE(J))
					Call AddStringParameterToString(CProperties,"HATCHSTYLE",Tbl_REP17CT1.HATCHSTYLE(J))
					PV_PE1.SetChartColumnProperties hData,J,CProperties
				End If
			Next

Function OzetTabloBasligi(satirNo,RowType,satirpozr)
	Call SetOTAAttributes
	Dim outmsg
	Select Case RowType
	Case "A" : outmsg=REP17SE.OT1_ACIKLAMA
	Case "B" : outmsg=REP17SE.OT2_ACIKLAMA
	Case "C" : outmsg=REP17SE.OT3_ACIKLAMA
	Case "D" : outmsg=REP17SE.OT4_ACIKLAMA
	Case "E" : outmsg=REP17SE.OT5_ACIKLAMA
	End Select
	outmsg=RTrim(outmsg)
	Dim OldTop
	OldTop=PV_PE1.GetTop()

	Dim OTAHEIGHTMM
	Dim OTAHEIGHTMMBKG
	OTAHEIGHTMM=REP17SE.OTAROWHEIGHTCM*10

	If OTAHEIGHTMM=0 Then
		ret = PV_PE1.RenderWrite(1,1,-2,PV_MMPRINTABLEPAGEWIDTHMM,outmsg)
		OTAHEIGHTMM=PV_PE1.GetRenderHeight()
	End If
	OTAHEIGHTMMBKG=(REP17SE.OTA_ONCECMBOSLUK*10)+OTAHEIGHTMM+(REP17SE.OTA_SONRACMBOSLUK*10)
	Dim CizgiGenisligi
	Dim CizgiRengi
	CizgiGenisligi=0.0
	CizgiRengi=RGB(192,192,192)
	If REP17SE.OTA_OUTF_CIZGI Then
		CizgiGenisligi=REP17SE.OTA_OUTF_CIZGI_CM*10
		CizgiRengi=RGB(192,192,192)
		If Trim(REP17SE.OTA_OUTF_CIZGI_CLR)<>"" Then CizgiRengi=Doc.C9ToCRGB(REP17SE.OTA_OUTF_CIZGI_CLR)
	End If
	If CizgiGenisligi>0 Then
		OTAHEIGHTMMBKG=OTAHEIGHTMMBKG+CizgiGenisligi+1
	End If

	If Trim(REP17SE.OTA_OUTF_BCKCLR)<>"" Then PV_PE1.SetBkgcolor Doc.C9ToCRGB(REP17SE.OTA_OUTF_BCKCLR)
	If PV_PE1.GetBkgcolor()<>RGB(255,255,255) Then
		PV_PE1.NoPen()
		PV_PE1.SetTransparentMode 0
		PV_PE1.SetBkgcolor PV_PE1.GetBkgcolor()

		PV_PE1.Box satirpozr,PV_mmleft_margin-(PV_BACKGROUNDSAGSOLFAZLALIKMM/2),OTAHEIGHTMMBKG, PV_MMPRINTABLEPAGEWIDTHMM+PV_BACKGROUNDSAGSOLFAZLALIKMM
		PV_PE1.SetTransparentMode 1
		PV_PE1.SetBkgcolor(RGB(255,255,255))
	End If
	SatirPozr=SatirPozr+(REP17SE.OTA_ONCECMBOSLUK*10)
	Dim BottomLine
	BottomLine = PV_PE1.Write(SatirPozr,PV_mmleft_margin,-2,PV_MMPRINTABLEPAGEWIDTHMM,outmsg)

	If CizgiGenisligi>0 Then
		PV_PE1.SetPenSize CizgiGenisligi
		PV_PE1.SetPenColor CizgiRengi
		PV_PE1.Line BottomLine+1,PV_mmleft_margin,0,PV_MMPRINTABLEPAGEWIDTHMM
	End If

	PV_PE1.SetTop(OldTop+OTAHEIGHTMMBKG)
	PV_PE1.SetBkgcolor(RGB(255,255,255))
	OzetTabloBasligi=OTAHEIGHTMMBKG
End Function

Sub SatirYazdir(satirNo)
	Dim pagewidth
	pagewidth=PV_PE1.GetPageWidth()
	Dim sonSatir
	Dim satirpozr
	Dim satirpozr_bck
'ğğğ sonSatir PV_PE1'nin SatirYazdir çağrıldındaki kaldığı yer(bottom)
	sonSatir = PV_PE1.GetBottom()
	Dim pageheight
	pageheight=PV_PE1.GetPageHeight()-PV_mmbottom_margin
	Dim BuRowmmsatirYuksekligi_YaziIcin
	BuRowmmsatirYuksekligi_YaziIcin=PV_mmsatirYuksekligi

	Dim RowType
	Dim RowTypePrev
	RowType=PV_RT.REPORT___ROWTYPE(satirNo)
	RowTypePrev=" "
	If satirNo>1 Then RowTypePrev=PV_RT.REPORT___ROWTYPE(satirNo-1)

'ğğğ Ara toplamlar Basılmayacak ise Dön
	If (RowType>="1" AND RowType<= "9") Then
		Select Case RowType
		Case "1" :
				If REP17SE.ST1_PRINTOPT="0" Then Exit Sub
		Case "2" :
				If REP17SE.ST2_PRINTOPT="0" Then Exit Sub
		Case "3" :
				If REP17SE.ST3_PRINTOPT="0" Then Exit Sub
		Case "4" :
				If REP17SE.ST4_PRINTOPT="0" Then Exit Sub
		Case "5" :
				If REP17SE.ST5_PRINTOPT="0" Then Exit Sub
		End Select
	End If
'ğğğ Özet Raporlar İçi Rapor Toplamı Basılmayacak ise Dön
	If RowType="T" Then
		If RowTypePrev="T" Then Exit Sub
		Select Case RowTypePrev
		Case "T" : Exit Sub
		Case "A" :
				If REP17SE.OT1_GENELTOPLAM="0" Then Exit Sub
		Case "B" :
				If REP17SE.OT2_GENELTOPLAM="0" Then Exit Sub
		Case "C" :
				If REP17SE.OT3_GENELTOPLAM="0" Then Exit Sub
		Case "D" :
				If REP17SE.OT4_GENELTOPLAM="0" Then Exit Sub
		Case "E" :
				If REP17SE.OT5_GENELTOPLAM="0" Then Exit Sub
		Case Else :
				If REP17SE.R_OUTF_GENELTOPLAM="0" Then Exit Sub
		End Select
	End If

	Dim AlanSayisi
	AlanSayisi=PV_RO.GetNumberOfFields()
	Dim F
	Dim FieldDef
	Dim OutputStr
	Dim OutputStr2
	Dim Align

	If Not SatirdaBasilacakAlanVarmi(AlanSayisi,RowType) Then
		If RowType=" " Then PV_SatirdaBasilacakAlanVarmi_Data=0
		Exit Sub
	End If
'ğğğ BuRowmmsatirYuksekligi_YaziIcin hesaplaması
	For F=1 To AlanSayisi
		If PV_RO.BuAlanBasilacakmi(F,RowType)=1 Or True Then
			Set FieldDef=PV_RO.GetFieldDefN(F)
			If FieldDef.PRINTOPT=1 Then
				PV_RO.SetFieldFontAttributes PV_PE1,F,RowType
				PV_PE1.SetBkgcolor(RGB(255,255,255))
				If FieldDef.IsNumeric() Then
					Align="RIGHT"
					PV_PE1.SetAlign "RIGHT"
				Else
					Align=FieldDef.DataAlignment
					Select Case Align
						Case "LEFT"			PV_PE1.SetAlign "LEFT"
						Case "RIGHT"		PV_PE1.SetAlign "RIGHT"
						Case "CENTER"		PV_PE1.SetAlign "CENTER"
						Case Else			PV_PE1.SetAlign "LEFT"
					End Select
				End If
				If FieldDef.OUTF_WRAPTEXT=1 Then
					If FieldDef.IsNumeric() Then
						OutputStr=Doc.NumFormat_Gen(PV_RT.GetFieldValue(FieldDef.Name,satirNo),FieldDef.Length,FieldDef.SigDig,0)
						OutputStr=Trim(OutputStr)
						ret = PV_PE1.RenderWrite(1,FieldDef.INTF_HESAPLANANYER_X_MM,-2,FieldDef.INTF_HESAPLANANGENISLIK_X_MM,OutputStr)
					ElseIf FieldDef.IsDate() Then
						OutputStr=Doc.StrToNorDate(PV_RT.GetFieldValue(FieldDef.Name,satirNo))
						OutputStr=Left(Trim(OutputStr),FieldDef.Length)
						ret = PV_PE1.RenderWrite(1,FieldDef.INTF_HESAPLANANYER_X_MM,-2,FieldDef.INTF_HESAPLANANGENISLIK_X_MM,OutputStr)
					Else
						OutputStr=Left(Trim(PV_RT.GetFieldValue(FieldDef.Name,satirNo)),FieldDef.Length)
						ret = PV_PE1.RenderWrite(1,FieldDef.INTF_HESAPLANANYER_X_MM,-2,FieldDef.INTF_HESAPLANANGENISLIK_X_MM,OutputStr)
					End If
				Else
					ret = PV_PE1.RenderWrite(1,FieldDef.INTF_HESAPLANANYER_X_MM,-2,FieldDef.INTF_HESAPLANANGENISLIK_X_MM,"g")
				End If
				If PV_PE1.GetRenderHeight()+1>BuRowmmsatirYuksekligi_YaziIcin Then
					BuRowmmsatirYuksekligi_YaziIcin=PV_PE1.GetRenderHeight()+1
				End If
			End If
		End If
	Next
	Dim SayfaBasindanBasilanDataRowSayisi
	SayfaBasindanBasilanDataRowSayisi=PV_SayfaBasindanBasilanDataRowSayisi
'2025/07/08
'	If RowType=" " Then PV_SayfaBasindanBasilanDataRowSayisi=PV_SayfaBasindanBasilanDataRowSayisi+1
	If RowType=" " Or ( RowType>="A" AND RowType<="E" ) Then PV_SayfaBasindanBasilanDataRowSayisi=PV_SayfaBasindanBasilanDataRowSayisi+1

'ğğğ SonSatirdanSonraBirakilacakExtraBosluk ve SayfaBasiGereklimi beraber hesaplanıyor çünkü sayfabaşı yapılır ise boşluğa gerek yok
	Dim SonSatirdanSonraBirakilacakExtraBosluk
	SonSatirdanSonraBirakilacakExtraBosluk=0
	Dim SayfaBasiGereklimi
	SayfaBasiGereklimi=0

'ğğğ özet raporlardan sonra boşluk ve sayfa sonu
	If RowType>="A" AND RowType<="E" Then
		If RowTypePrev="T" Then
			Select Case RowType
			Case "A" : SonSatirdanSonraBirakilacakExtraBosluk=REP17SE.CMOT1DENONCEBOSLUK*10
			Case "B" : SonSatirdanSonraBirakilacakExtraBosluk=REP17SE.CMOT2DENONCEBOSLUK*10
			Case "C" : SonSatirdanSonraBirakilacakExtraBosluk=REP17SE.CMOT3DENONCEBOSLUK*10
			Case "D" : SonSatirdanSonraBirakilacakExtraBosluk=REP17SE.CMOT4DENONCEBOSLUK*10
			Case "E" : SonSatirdanSonraBirakilacakExtraBosluk=REP17SE.CMOT5DENONCEBOSLUK*10
			End Select

'ğğğ önceki satır 'T' bu satır özet ise yeni sayfa gerekli mi?
			Select Case RowType
			Case "A" : If REP17SE.OT1_YENISAYFAYA="1" Then SayfaBasiGereklimi=1
			Case "B" : If REP17SE.OT2_YENISAYFAYA="1" Then SayfaBasiGereklimi=1
			Case "C" : If REP17SE.OT3_YENISAYFAYA="1" Then SayfaBasiGereklimi=1
			Case "D" : If REP17SE.OT4_YENISAYFAYA="1" Then SayfaBasiGereklimi=1
			Case "E" : If REP17SE.OT5_YENISAYFAYA="1" Then SayfaBasiGereklimi=1
			End Select
		End If
	End If
'ğğğ ara toplamlardan sonra sayfa basi gerekli mi?
	If SayfaBasiGereklimi=0 Then
		Select Case RowTypePrev
		Case "1" If REP17SE.ST1_YENISAYFA="1" Then SayfaBasiGereklimi=1
		Case "2" If REP17SE.ST2_YENISAYFA="1" Then SayfaBasiGereklimi=1
		Case "3" If REP17SE.ST3_YENISAYFA="1" Then SayfaBasiGereklimi=1
		Case "4" If REP17SE.ST4_YENISAYFA="1" Then SayfaBasiGereklimi=1
		Case "5" If REP17SE.ST5_YENISAYFA="1" Then SayfaBasiGereklimi=1
		End Select
	End If

'ğğğ her özet rapor ayrı sayfadan başlasın veya sayfa ortasına başlıklar yazılsın
	Dim OTSayfaBasi
	OTSayfaBasi=0

	sonSatir=sonSatir+SonSatirdanSonraBirakilacakExtraBosluk
	If sonSatir+BuRowmmsatirYuksekligi_YaziIcin >= pageheight Then SayfaBasiGereklimi=1

	If RowType>="A" AND RowType<="E" AND RowType<>RowTypePrev Then
		If PV_SatirdaBasilacakAlanVarmi_Data<>0 Then
			If SayfaBasindanBasilanDataRowSayisi=0 Then SayfaBasiGereklimi=0
		End If

		If SayfaBasiGereklimi=1 Then
			Call SayfaSonu()
			ret = PV_PE1.PageBreak()
			Call SayfaBasi()
			satirpozr = PV_vsatir1pozr
			SayfaBasiGereklimi=0
		Else
			If PV_SayfaIciSatirNo=1 Then
				satirpozr = PV_vsatir1pozr
			Else
				satirpozr = PV_PE1.GetTop() + PV_mmSonBasilanRowsatirYuksekligi
			End If
		End If
		satirpozr=satirpozr+SonSatirdanSonraBirakilacakExtraBosluk
		Dim OTAHEIGHTMMBKG
		OTAHEIGHTMMBKG=OzetTabloBasligi(satirNo,RowType,satirpozr)
		sonSatir=sonSatir+OTAHEIGHTMMBKG
		If sonSatir+BuRowmmsatirYuksekligi_YaziIcin >= pageheight Then SayfaBasiGereklimi=1
		OTSayfaBasi=1
	End If

	If PV_SatirdaBasilacakAlanVarmi_Data<>0 Then
		If SayfaBasindanBasilanDataRowSayisi=0 Then SayfaBasiGereklimi=0
	End If

	If SayfaBasiGereklimi=1 Then
		Call SayfaSonu()
		ret = PV_PE1.PageBreak()
		Call SayfaBasi()
		satirpozr = PV_vsatir1pozr
	Else
		If PV_SayfaIciSatirNo=1 AND OTSayfaBasi=0 Then
			satirpozr=PV_vsatir1pozr
		Else
			satirpozr=PV_PE1.GetTop() + PV_mmSonBasilanRowsatirYuksekligi
		End If
	End If
'ğğğ PV_mmSonBasilanRowsatirYuksekligi
	PV_mmSonBasilanRowsatirYuksekligi=BuRowmmsatirYuksekligi_YaziIcin


	Dim BckRenkYuksekligi
	BckRenkYuksekligi=BuRowmmsatirYuksekligi_YaziIcin
	Dim MMSTX_ONCEBOSLUK
	Dim MMSTX_SONRABOSLUK
	MMSTX_ONCEBOSLUK=0
	MMSTX_SONRABOSLUK=0
'ğğğ önce ve sonra boşluk
	Select Case RowType
		Case "1"		MMSTX_ONCEBOSLUK=REP17SE.ST1_ONCECMBOSLUK*10
		Case "2"		MMSTX_ONCEBOSLUK=REP17SE.ST2_ONCECMBOSLUK*10
		Case "3"		MMSTX_ONCEBOSLUK=REP17SE.ST3_ONCECMBOSLUK*10
		Case "4"		MMSTX_ONCEBOSLUK=REP17SE.ST4_ONCECMBOSLUK*10
		Case "5"		MMSTX_ONCEBOSLUK=REP17SE.ST5_ONCECMBOSLUK*10
		Case "T"		MMSTX_ONCEBOSLUK=REP17SE.GT_ONCECMBOSLUK*10
		Case Else		MMSTX_ONCEBOSLUK=REP17SE.DT_ONCECMBOSLUK*10
	End Select
	Select Case RowType
		Case "1"		MMSTX_SONRABOSLUK=REP17SE.ST1_SONRACMBOSLUK*10
		Case "2"		MMSTX_SONRABOSLUK=REP17SE.ST2_SONRACMBOSLUK*10
		Case "3"		MMSTX_SONRABOSLUK=REP17SE.ST3_SONRACMBOSLUK*10
		Case "4"		MMSTX_SONRABOSLUK=REP17SE.ST4_SONRACMBOSLUK*10
		Case "5"		MMSTX_SONRABOSLUK=REP17SE.ST5_SONRACMBOSLUK*10
		Case "T"		MMSTX_SONRABOSLUK=REP17SE.GT_SONRACMBOSLUK*10
		Case Else		MMSTX_SONRABOSLUK=REP17SE.DT_SONRACMBOSLUK*10
	End Select

	BckRenkYuksekligi=BckRenkYuksekligi+MMSTX_ONCEBOSLUK+MMSTX_SONRABOSLUK
	satirpozr_bck=satirpozr

	PV_PE1.SetBkgcolor(RGB(255,255,255))
	PV_RO.SetRowFontAttributes PV_PE1,RowType

'ğğğ zemin rengi varsa....
	If PV_PE1.GetBkgcolor()<>RGB(255,255,255) Then
		PV_PE1.NoPen()
		PV_PE1.SetTransparentMode 0
		PV_PE1.SetBkgcolor PV_PE1.GetBkgcolor()
		PV_PE1.Box satirpozr_bck,PV_mmleft_margin-(PV_BACKGROUNDSAGSOLFAZLALIKMM/2),BckRenkYuksekligi,PV_MMPRINTABLEPAGEWIDTHMM+PV_BACKGROUNDSAGSOLFAZLALIKMM
		PV_PE1.SetTransparentMode 1
		PV_PE1.SetBkgcolor(RGB(255,255,255))
	Else
		If PV_ZebraPrintOpt Then
			If Int(PV_SayfaIciSatirNo/2)*2<>PV_SayfaIciSatirNo Then
				If PV_Zebra_Satir_Rengi1<>RGB(255, 255, 255) Then
					PV_PE1.NoPen()
					PV_PE1.SetBkgcolor PV_Zebra_Satir_Rengi1
					PV_PE1.SetTransparentMode 0
					PV_PE1.Box satirpozr_bck,PV_mmleft_margin-(PV_BACKGROUNDSAGSOLFAZLALIKMM/2),BckRenkYuksekligi,PV_MMPRINTABLEPAGEWIDTHMM+PV_BACKGROUNDSAGSOLFAZLALIKMM
					PV_PE1.SetTransparentMode 1
					PV_PE1.SetBkgcolor(RGB(255,255,255))
				End If
			Else
				If PV_Zebra_Satir_Rengi2<>RGB(255, 255, 255) Then
					PV_PE1.NoPen()
					PV_PE1.SetBkgcolor PV_Zebra_Satir_Rengi2
					PV_PE1.SetTransparentMode 0
					PV_PE1.Box satirpozr_bck,PV_mmleft_margin-(PV_BACKGROUNDSAGSOLFAZLALIKMM/2),BckRenkYuksekligi,PV_MMPRINTABLEPAGEWIDTHMM+PV_BACKGROUNDSAGSOLFAZLALIKMM
					PV_PE1.SetTransparentMode 1
					PV_PE1.SetBkgcolor(RGB(255,255,255))
				End If
			End If
		End If
	End If

'ğğğ ara toplam ve toplamdan önce çizgi
	Dim CizgiGenisligi
	CizgiGenisligi=0.0
	Dim CizgiRengi
	CizgiRengi=RGB(192,192,192)
	Select Case RowType
		Case "1"		If REP17SE.ST1_OUTF_CIZGI="1" Then CizgiGenisligi=REP17SE.ST1_OUTF_CIZGI_CM*10
						If Trim(REP17SE.ST1_OUTF_CIZGI_CLR)<>"" Then CizgiRengi=Doc.C9ToCRGB(REP17SE.ST1_OUTF_CIZGI_CLR)
		Case "2"		If REP17SE.ST2_OUTF_CIZGI="1" Then CizgiGenisligi=REP17SE.ST2_OUTF_CIZGI_CM*10
						If Trim(REP17SE.ST2_OUTF_CIZGI_CLR)<>"" Then CizgiRengi=Doc.C9ToCRGB(REP17SE.ST2_OUTF_CIZGI_CLR)
		Case "3"		If REP17SE.ST3_OUTF_CIZGI="1" Then CizgiGenisligi=REP17SE.ST3_OUTF_CIZGI_CM*10
						If Trim(REP17SE.ST3_OUTF_CIZGI_CLR)<>"" Then CizgiRengi=Doc.C9ToCRGB(REP17SE.ST3_OUTF_CIZGI_CLR)
		Case "4"		If REP17SE.ST4_OUTF_CIZGI="1" Then CizgiGenisligi=REP17SE.ST4_OUTF_CIZGI_CM*10
						If Trim(REP17SE.ST4_OUTF_CIZGI_CLR)<>"" Then CizgiRengi=Doc.C9ToCRGB(REP17SE.ST4_OUTF_CIZGI_CLR)

						If Trim(REP17SE.ST5_OUTF_CIZGI_CLR)<>"" Then CizgiRengi=Doc.C9ToCRGB(REP17SE.ST5_OUTF_CIZGI_CLR)

						If Trim(REP17SE.DT_OUTF_CIZGI_CLR)<>"" Then CizgiRengi=Doc.C9ToCRGB(REP17SE.DT_OUTF_CIZGI_CLR)
	End Select
	If PV_SayfaIciSatirNo=1 Then CizgiGenisligi=0.0
	PV_SayfaIciSatirNo=PV_SayfaIciSatirNo+1

	satirpozr=satirpozr+MMSTX_ONCEBOSLUK+CizgiGenisligi
	satirpozr_bck=satirpozr_bck+CizgiGenisligi

    For F=1 To AlanSayisi
		If PV_RO.BuAlanBasilacakmi(F,RowType)=1 Then
			Set FieldDef=PV_RO.GetFieldDefN(F)
			If FieldDef.PRINTOPT=1 Then
				PV_RO.SetFieldFontAttributes PV_PE1,F,RowType
				If PV_PE1.GetBkgcolor()<>RGB(255,255,255) Then
					PV_PE1.NoPen()
					PV_PE1.SetTransparentMode 0
					PV_PE1.Box satirpozr_bck,FieldDef.INTF_HESAPLANANYER_X_MM,BckRenkYuksekligi,FieldDef.INTF_HESAPLANANGENISLIK_X_MM
					PV_PE1.SetTransparentMode 1
					PV_PE1.SetBkgcolor(RGB(255,255,255))
'					If PV_PE1.GetBkgcolor()<>RGB(255,255,255) Then ?
				End If
				If FieldDef.IsNumeric() Then
					Align="RIGHT"
					PV_PE1.SetAlign "RIGHT"
				Else
					Align=FieldDef.DataAlignment
					Select Case Align
						Case "LEFT"			PV_PE1.SetAlign "LEFT"
						Case "RIGHT"		PV_PE1.SetAlign "RIGHT"
						Case "CENTER"		PV_PE1.SetAlign "CENTER"
						Case Else			PV_PE1.SetAlign "LEFT"
					End Select
				End If
				If FieldDef.IsNumeric() Then
					OutputStr=Doc.NumFormat_Gen(PV_RT.GetFieldValue(FieldDef.Name,satirNo),FieldDef.Length,FieldDef.SigDig,0)
					OutputStr=Trim(OutputStr)
					If FieldDef.TEKRARBASILMAMAOPT>0 Then
						If satirNo>1 And RowTypePrev=" " And RowType=" " Then
							OutputStr2=Doc.NumFormat_Gen(PV_RT.GetFieldValue(FieldDef.Name,satirNo-1),FieldDef.Length,FieldDef.SigDig,0)
							OutputStr2=Trim(OutputStr2)
							If OutputStr2=OutputStr Then OutputStr=""
						End If
					End If

					ret = PV_PE1.Write(SatirPozr,FieldDef.INTF_HESAPLANANYER_X_MM,BuRowmmsatirYuksekligi_YaziIcin,FieldDef.INTF_HESAPLANANGENISLIK_X_MM,OutputStr)
				ElseIf FieldDef.IsDate() Then
					OutputStr=Doc.StrToNorDate(PV_RT.GetFieldValue(FieldDef.Name,satirNo))
					OutputStr=Left(Trim(OutputStr),FieldDef.Length)
					If FieldDef.TEKRARBASILMAMAOPT>0 Then
						If satirNo>1 And RowTypePrev=" " And RowType=" " Then
							OutputStr2=Doc.StrToNorDate(PV_RT.GetFieldValue(FieldDef.Name,satirNo-1))
							OutputStr2=Left(Trim(OutputStr2),FieldDef.Length)
							If OutputStr2=OutputStr Then OutputStr=""
						End If
					End If

Sub PrintReport_Ods(RT,RO)
	
	Dim fso
	Dim temp_folder
    Const TemporaryFolder = 2
	Set fso = CreateObject("Scripting.FileSystemObject")
    Set temp_folder = fso.GetSpecialFolder(TemporaryFolder)

	Dim UUID
	UUID = Doc.GenerateUUID()
	
	Dim FolderName
	FolderName =temp_folder &"\"& UUID
	If Not fso.FolderExists(FolderName) Then
		fso.CreateFolder(FolderName)
	End If

	Dim metainf_dir
	metainf_dir = FolderName &"\META-INF"
	
	If fso.FolderExists(FolderName) Then
		If Not fso.FolderExists(metainf_dir) Then
			fso.CreateFolder(metainf_dir)
		End If
		If Not fso.FolderExists(metainf_dir) Then
			Doc.d7MsgBox "ODS Dosyaları Oluşturulamıyor.","",-1
			Exit Sub 
		End If 
		
		FSO.CopyFile Doc.Getd7Directory_SwRoot() & "wx\ODS\settings.xml" , FolderName & "\settings.xml" ,true
		
		ods_content = FolderName &  "\content.xml" 
		ods_style = FolderName &  "\styles.xml" 
		manifest = metainf_dir &  "\manifest.xml" 
		
	Else
		Doc.d7MsgBox "ODS Dosyaları Oluşturulamıyor.","",-1
		Exit Sub 
	End If 
	
	GenerateStyleXML
	GenerateManifestXML
	GenerateContentXML RT , RO
	
	Styles_objXMLDoc.Save(Trim(ods_style))
	Manifest_objXMLDoc.Save(Trim(manifest))
	Content_objXMLDoc.Save(Trim(ods_content))
	
	
	Dim objTS
	Dim strods
	Dim fileSpec 
	'Dim objFSO
	fileSpec = ods_style
    'Set objFSO = CreateObject("Scripting.FileSystemObject")
    Set objTS = fso.OpenTextFile(fileSpec, 1, -2)
    strods = objTS.ReadAll
    strods = Replace(strods, "&amp;", "&")
    Set objTS = fso.OpenTextFile(fileSpec, 2, -2)
    objTS.Write strods
    objTS.Close
	
	Dim strContents
	Set objTS = fso.OpenTextFile(ods_content, 1, -2)
    strContents = objTS.ReadAll
	strContents = Replace(strContents, "dinamo-cell-type=""string""", "calcext:value-type=""string"" office:value-type=""string""")
	strContents = Replace(strContents, "dinamo-cell-type=""date""", "office:value-type=""date"" calcext:value-type=""date""")
	strContents = Replace(strContents, "dinamo-cell-type=""float""", "calcext:value-type=""float"" office:value-type=""float""")
    Set objTS = fso.OpenTextFile(ods_content, 2, -2)
    objTS.Write strContents
    objTS.Close
	
	
	Dim zip7za
	zip7za = Doc.Getd7Directory_SwRoot() & "\wx\eInvoice\7za.exe"
	
	dim runCmd
	runCmd =zip7za & " a -tzip " &  FolderName & ".ods " &  FolderName & "\*"

	Dim objShell
	Set objShell = CreateObject("Wscript.Shell")
	
	Dim exec : Set exec = objShell.Exec(runCmd)

	While exec.Status = 0
		Doc.Sleep(100)
	Wend
	
	If exec.ExitCode = 1 Then
		Dim mes1 
		mes1 = Doc.FormatMessage1(" [%1] Dosyası sıkıştırılırken hata ile karşılaşıldı.", FolderName   )
		Dim y 
		y = Doc.d7MsgBox(mes1,"Dinamo",MB_OK+MB_ICONSTOP)
	End If 
	
	Doc.SetDocParameterValue "SONYARATILANODSCIKTISI",FolderName & ".ods "
	fso.DeleteFolder FolderName, False
	
'2021/04/23 Metin
'	objShell.Run(FolderName & ".ods " )
	Doc.ShellOpen FolderName & ".ods "
	
End sub

Sub ODS_SatirYazdir(satirNo)

	Dim RowType
	Dim RowTypePrev
	RowType=PV_RT.REPORT___ROWTYPE(satirNo)
	RowTypePrev=" "
	If satirNo>1 Then RowTypePrev=PV_RT.REPORT___ROWTYPE(satirNo-1)

	If (RowType>="1" AND RowType<= "9") Then
		Select Case RowType
		Case "1" :
				If REP17SE.ST1_PRINTOPT="0" Then Exit Sub
		Case "2" :
				If REP17SE.ST2_PRINTOPT="0" Then Exit Sub
		Case "3" :
				If REP17SE.ST3_PRINTOPT="0" Then Exit Sub
		Case "4" :
				If REP17SE.ST4_PRINTOPT="0" Then Exit Sub
		Case "5" :
				If REP17SE.ST5_PRINTOPT="0" Then Exit Sub
		End Select
	End If
	If RowType="T" Then
		If RowTypePrev="T" Then Exit Sub
		Select Case RowTypePrev
		Case "T" : Exit Sub
		Case "A" :
				If REP17SE.OT1_GENELTOPLAM="0" Then Exit Sub
		Case "B" :
				If REP17SE.OT2_GENELTOPLAM="0" Then Exit Sub
		Case "C" :
				If REP17SE.OT3_GENELTOPLAM="0" Then Exit Sub
		Case "D" :
				If REP17SE.OT4_GENELTOPLAM="0" Then Exit Sub
		Case "E" :
				If REP17SE.OT5_GENELTOPLAM="0" Then Exit Sub
		Case Else :
				If REP17SE.R_OUTF_GENELTOPLAM="0" Then Exit Sub
		End Select
	End If

	Dim AlanSayisi
	AlanSayisi=PV_RO.GetNumberOfFields()
	Dim F
	Dim FieldDef
	Dim OutputStr
	
    For F=1 To AlanSayisi
		If PV_RO.BuAlanBasilacakmi(F,RowType)=1 Then
			Set FieldDef=PV_RO.GetFieldDefN(F)
			If FieldDef.PRINTOPT=1 Then
				Dim OutputStr2
				If FieldDef.IsNumeric() Then
					OutputStr=0
					OutputStr=FormatNumber(PV_RT.GetFieldValue(FieldDef.Name,satirNo) ,6,,,0)
					
					If FieldDef.TEKRARBASILMAMAOPT>0 Then
						If satirNo>1 And RowTypePrev=" " And RowType=" " Then
							OutputStr2=Doc.NumFormat_Gen(PV_RT.GetFieldValue(FieldDef.Name,satirNo-1),FieldDef.Length,FieldDef.SigDig,0)
							If OutputStr2=OutputStr Then OutputStr=0
						End If
					End If
					
					Call AddColumn("office:value=" & OutputStr & ",dinamo-cell-type=float",Doc.ChangeNumberStyleCD(OutputStr))
				ElseIf FieldDef.IsDate() Then
					OutputStr=""
					OutputStr=Doc.StrToNorDate(PV_RT.GetFieldValue(FieldDef.Name,satirNo))
					OutputStr=Left(Trim(OutputStr),FieldDef.Length)
					
					If FieldDef.TEKRARBASILMAMAOPT>0 Then
						If satirNo>1 And RowTypePrev=" " And RowType=" " Then
							OutputStr2=Doc.StrToNorDate(PV_RT.GetFieldValue(FieldDef.Name,satirNo-1))
							OutputStr2=Left(Trim(OutputStr2),FieldDef.Length)
							If OutputStr2=OutputStr Then OutputStr=""
						End If
					End If
					
					Call AddColumn("dinamo-cell-type=string",OutputStr)
				Else
					OutputStr=Left(Trim(PV_RT.GetFieldValue(FieldDef.Name,satirNo)),FieldDef.Length)
					If FieldDef.TEKRARBASILMAMAOPT>0 Then
						If satirNo>1 And RowTypePrev=" " And RowType=" " Then
							OutputStr2=Left(Trim(PV_RT.GetFieldValue(FieldDef.Name,satirNo-1)),FieldDef.Length)
							If OutputStr2=OutputStr Then OutputStr=""
						End If
					End If
				
					Call AddColumn("dinamo-cell-type=string",OutputStr)
				End If
			End If
			
			Select Case RowType
				Case "1","2","3","4","5"		Call AddAttributeArray(Content_objXMLDoc, o_table_cell, "table:style-name=subtotal_col"&F)  'PV_EXCEL_PE2.SubTotalOut
				Case "A","B","C","D","E","T"	Call AddAttributeArray(Content_objXMLDoc, o_table_cell, "table:style-name=GrandTotalOut_col"&F)'PV_EXCEL_PE2.GrandTotalOut
				Case Else						'PV_EXCEL_PE2.LineOut
			End Select
			
		ElseIf PV_RO.BuAlanBasilacakmi(F,"*")=1 Then
			'PV_EXCEL_PE2.SetRowValue_String -1,""
		End If
		
	Next
	
	
End Sub

Sub AddAttribute(objXMLDoc, o_Node, AttributeName, AttributeValue)
    Dim o_Att
    Set o_Att = objXMLDoc.createAttribute(AttributeName)
    o_Att.Value = Trim(AttributeValue)
    o_Node.setAttributeNode (o_Att)
End Sub

Sub AddAttributeArray(objXMLDoc, o_Node, Attribute)
    
	Attribute = Split(Attribute,",")
	
	Dim strAttributeName , strAttributeValue
	Dim i 	
	For i = LBound(Attribute) To UBound(Attribute)
		
		strAttributeName = ""
		strAttributeValue = ""
		
		strAttributeName = TRIM(Left(Attribute(i),instr(1,Attribute(i),"=")-1))
		strAttributeValue =  TRIM(Replace(Replace(Right(Attribute(i),Len(Attribute(i)) - instr(1,Attribute(i),"=")),"'",""),"""",""))
		
		Dim o_Att
		Set o_Att = objXMLDoc.createAttribute(strAttributeName)
		o_Att.Value = Trim(strAttributeValue)
		o_Node.setAttributeNode (o_Att)
		
	Next  
End Sub

Function RGBTOHEX(ByVal RGB)
	
	Dim s_rgb(3)
	s_rgb(0) = Left(RGB,3)
	s_rgb(1) = MID(RGB,4,3)
	s_rgb(2) = RIGHT(RGB,3)
	
	Dim hex1
	Dim hex2
	Dim hex3
	
	hex1=hex(s_rgb(0))
	hex2=hex(s_rgb(1))
	hex3=hex(s_rgb(2))
	
	If Len(hex1) = 1 Then hex1 = "0" & hex1
	If Len(hex2) = 1 Then hex2 = "0" & hex2
	If Len(hex3) = 1 Then hex3 = "0" & hex3
	
	RGBTOHEX = "#"  & hex1 & hex2 & hex3

End Function

'BYMOZAN
Sub SendTableToExcel(Tablo)

	dim pe2, Alanlar,ncol,i,AlanAdi,AlanTuru,Align,satirNo,ExcelSatir,OutputStr,Deger

	Set pe2 = Doc.CreateSObject("ExcelOutput")
	PE2.SetOrientation "P"
	PE2.PageB_ReportHead_std1 Tablo.GetAppName

	Alanlar = Tablo.GetFields()
	ncol = 0
	For i=0 To uBound(Alanlar,1)
		AlanAdi = Alanlar(i,0)
		if Cikacak(AlanAdi) Then
			ncol = ncol+1
			if ncol > 256 Then Exit For
			AlanTuru = Alanlar(i,2)
				If AlanTuru = "INT16" or AlanTuru = "INT32" or AlanTuru = "DOUBLE" Then
				Align = "RIGHT"
			Else
				Align = "LEFT"
			End if
			pe2.AddColumnDef Alanlar(i,0), Alanlar(i,3), Alanlar(i,4), Align
			pe2.SetRowValue_String -1, AlanAdi
		End if
	Next
	PE2.LineOut

	For satirNo = 1 to Tablo.GetRowCount()
		Doc.StatusBarMessage "Excel 'e gönderiliyor : " & Tablo.GetAppName & " - " & i & " / " & Tablo.GetRowCount
		PE2.InitRowValues

	ExcelSatir = -1
	ncol = 0

	For i=0 To uBound(Alanlar,1)
		AlanAdi = Alanlar(i,0)
		if Cikacak(AlanAdi) Then
			ncol = ncol+1
			if ncol > 256 Then Exit For
			AlanTuru = Alanlar(i,2)
			If AlanTuru = "INT16" or AlanTuru = "INT32" or AlanTuru = "DOUBLE" Then
				Deger = Tablo.GetFieldValue(AlanAdi,satirNo)
				pe2.SetRowValue_Double ExcelSatir, Deger
			ElseIf AlanTuru = "DATE" Then
				OutputStr=Doc.StrToNorDate(Tablo.GetFieldValue(AlanAdi,satirNo))
				pe2.SetRowValue_String ExcelSatir,OutputStr
			Else
				pe2.SetRowValue_String ExcelSatir,RTrim(Tablo.GetFieldValue(AlanAdi,satirNo))
			End if
		End if
	Next
	PE2.LineOut
	Next
	pe2.Show
End Sub
'---------------------------
Function Cikacak(AlanAdi)
	Cikacak = True
	if InStr(1, AlanAdi, "D7ROWVERSION") > 0 Then
		Cikacak = False
	End if
End Function
