Sub PrintForm()

	SayfaEni = 210
	SayfaBoyu = 297
	
	res=pe1.Open(0,"Mutabakat Mektubu",SayfaBoyu,SayfaEni,"")
	Call Print_Form_imza(pe1)
	Donem = Inputbox("Form almak istediðiniz dönemi yazýnýz","Dinamo","01")

End Sub

Function FontAyarla(BasilacakAlan,ToplamGen,Font)
	Oran = BasilacakAlan / ToplamGen
	If Oran >= 1 Then
		FontAyarla = Font
	Else
		If oran <=0.8 Then
			FontAyarla = Round(0.8 * Font,0)
		Else
			FontAyarla = Round(Oran * Font,0)
		End If
	End If
End Function
Function SayiBicimle(Sayi, OndalikSayisi, SondaSifir, Ayirac)
	If trim(sayi) = "" Then sayi = 0
	If trim(Ondaliksayisi) = "" Then Ondaliksayisi = 0

	If OndalikSayisi >= 0 Then
		Sayi = Round(Sayi, OndalikSayisi)
	End If
	Nokta = InStr(1, Sayi, ".")
	If Nokta = 0 Then
		ondalik = ""
	Else
		ondalik = Right(CStr(Sayi), Len(CStr(Sayi)) - Nokta)
	End If
	TamSayi = CStr(Fix(Sayi))
	If SondaSifir > 0 Then
		ondalik = SifirEkle(ondalik, OndalikSayisi)
	ElseIf SondaSifir = -1 Then
		ondalik = ondalik
	Else
		ondalik = SifirSil(ondalik)
	End If
	If Ayirac = 1 Then
		TamSayi = VirgulEkle(TamSayi)
	End If
	If ondalik = "" Then
		SayiBicimle = TamSayi
	Else
		SayiBicimle = TamSayi & "." & ondalik
	End If
End Function
Function SifirEkle(Sayi, OndalikHane)
	Sayi = CStr(Sayi)
	If OndalikHane - Len(Sayi) > 0 Then
		SifirEkle = Sayi & String(OndalikHane - Len(Sayi), "0")
	Else
		SifirEkle = Sayi
	End If
End Function
Function SifirSil(Sayi)
	Sayi = Trim(CStr(Sayi))
	While Right(Sayi, 1) = "0"
		Sayi = Left(Sayi, Len(Sayi) - 1)
	Wend
	SifirSil = Sayi
End Function
Function VirgulEkle(Sayi)
	If Sayi = 0 Then Exit Function
	Sayi = CStr(Sayi)
	k = Len(Sayi)
	For i = 0 To k - 1
		If i > 0 And i Mod 3 = 0 Then
			Sagdan = "," & Sagdan
		End If
		Sagdan = Mid(Sayi, k - i, 1) & Sagdan
	Next
	VirgulEkle = Sagdan
End Function
Function DecimalSayisi(Sayi)
	sayi = Cstr(Sayi)
	Nokta = InStr(1, Sayi, ".")
	If Nokta = 0 Then
		DecimalSayisi = 0
	Else
		ondalik = Right(CStr(Sayi), Len(CStr(Sayi)) - Nokta)
		DecimalSayisi = len(SifirSil(Ondalik))
	End If
End Function

	Call SayfaSayisiYazdir(pe1)

Function GetLokasyon(StokKodu)
	QStr = "SELECT Sum(SF_MIKTAR) SF_MIKTAR, (LOCATION1 || '-' || trim(LOCATION2) || '-' || trim(LOCATION3) || '-' || trim(LOCATION4))  LOKASYON"
	QStr = QStr & " FROM STOK01"
	QStr = QStr & " WHERE KOD='" & StokKodu & "' AND AMBCODE='01'"
	QStr = QStr & " GROUP BY LOCATION1, LOCATION2, LOCATION3, LOCATION4"
	Set S01 = Doc.RunSQLQuery("sdfgadfajh", QStr)
	Lokasyon = ""
	For i = 1 to S01.GetRowCount()
		if i > 1 Then Lokasyon = Lokasyon & ", "

	Next
	GetLokasyon = Lokasyon
End Function

Sub PrintForm()
	
	Dim objDoc, objWord, Selection
	Set objWord = CreateObject( "Word.Application" )
	objWord.visible = True
	Set objDoc = objWord.Documents.Add()
	Set Selection = objWord.Selection
	'objWord.Selection.EndKey 6,0
	FirmKodu = Trim(Doc.Getd7FirmCode())
	Uygulama = Trim(Doc.GetDocType)

	Set PFR7SE = Doc.Select1("dd333d", "PFRM7SE", "", "EVRAKNO", -1)
	'Set MKR7SE = Doc.Select1("dd333d", "MAKR7SE", "EVRAKNO = '_GNPRG1'", "", -1)
	
	KayitSayisi = PFR7SE.GetRecCount()
	'-------------------------------------------------
	'Satýr boþluk ayarýný yapar.
	Selection.ParagraphFormat.LineSpacing = 10
	objWord.ActiveDocument.ShowSpellingErrors = False
	'----------------------------------------------
	'Tab ayarýný yapar.
	objWord.ActiveDocument.DefaultTabStop = 15
	'---------------------------------------------
	For i = 1 To KayitSayisi
		EVRAKKODU_ORJ = Trim(PFR7SE.EVRAKNO(i))
		If Left(EVRAKKODU_ORJ, 2) <> "++" Then
			EVRAKKODU = Trim(PFR7SE.EVRAKNO(i))

			Kodlar = ""
			EKodlar = ""
			Selection.Paragraphs.Alignment = 1   '1:wdAlignParagraphCenter yani üst satýrdakini ortalar.
			Selection.TypeText "Form Kaynak Kodu"
			Selection.Typeparagraph
		
			Selection.Paragraphs.Alignment = 3
			Selection.TypeText "Firma Kodu : " & FirmKodu  & " / " & "Uygulama : " &  Uygulama
			Selection.Typeparagraph
			Selection.TypeText "Makro Kodu : " & EVRAKKODU 
			'------------------------
		
			'Selection.Paragraphs.Alignment = 1 wdAlignParagraphCenter yani üst satýrdakini ortalar.
			'2: wdAlignParagraphRight yani üst satýrdakini saga yaslar.
			'3: wdAlignParagraphRight yani üst satýrdakini sola yaslar.
		
			Selection.Paragraphs.Alignment = 3
			Selection.Typeparagraph
			Selection.TypeText "Hangi Tablo için : " & Trim(PFR7SE.DOCTYPE(i))
			Selection.Typeparagraph
			Selection.TypeText "Ýlave Hangi Uygulamalar için : " & Trim(PFR7SE.ADDDOCTYPES(i))
			Selection.Typeparagraph
			Selection.TypeText "Açýklama : " & Trim(PFR7SE.ACIKLAMA(i))
			Selection.Typeparagraph
			Selection.TypeText "Kod : "
			Selection.Typeparagraph

			Set PFR7ST = Doc.Select1("dd333d", "PFRM7ST", "EVRAKNO='"&EVRAKKODU&"'",">SRNUM", -1)

			For k = 1 To PFR7ST.GetRecCount()
				Selection.Font.Color = vbBlue
				Kodlar = Kodlar & PFR7ST.SCRIPTTEXT_RO(k)
			Next

			Selection.TypeText Kodlar
			Selection.Typeparagraph
			Selection.Typeparagraph
			'-----------------------------------
			Selection.Typeparagraph
			'-----------------------------------
			Selection.Font.Color = vbBlack
			Set PFR7SE= Doc.Select1("dd333d", "PFRM7SE", "", "", -1)
		
			Selection.TypeText "----------------------------------------------------------------"
			Selection.Typeparagraph
		End If
	Next
	'--------------------------------------------------------------------------
	'Font lara makyaj
	For j = 1  To 8
		strfind(1) = "Makro Kaynak Kodu"
		strfind(2) = "Makro Kodu : "
		strfind(3) = "Hangi Tablo için : "
		strfind(4) = "Ýlave Hangi Uygulamalar için : "
		strfind(5) = "Açýklama : "
		strfind(6) = "Kod : "
		strfind(7) = "Firma Kodu : "
		strfind(8) = "Uygulama : "

		Set objRange = objWord.ActiveDocument.Content
		
		With objRange.Find
		.Text = strfind(j)
		Do
			.Execute
			If .Found Then
				objRange.Font.Color = vbRed
				objRange.Font.Bold = True
				'If j = 1 Then objWord.ParagraphFormat.Alignment = wdAlignParagraphCenter
			End If
			Loop While .Found
			End With
		Next
		Selection.WholeStory
		'----------------------------------------------------
		'Satýr sonlarýndaki boþluðu alma
		Selection.ParagraphFormat.SpaceAfter = 0.5

		'Sayfanýn kenarlarýný daralt
		With objWord.ActiveDocument.PageSetup

			bosluk = 15
			.TopMargin = bosluk
			.BottomMargin = bosluk
			.LeftMargin = 43
			.RightMargin = bosluk
		End With
		'--------------------------------------------------------------------------
		'Satýr numaralarý ekler
    With Selection.PageSetup
        With .LineNumbering
            .Active = True
            .StartingNumber = 1
            .CountBy = 1
            .RestartMode = 1
            .DistanceFromText = 0
        End With
		End With
		'--------------------------------------------------------------------------
		objWord.visible = True
		desktoppath = CreateObject("WScript.Shell").SpecialFolders("Desktop")
		objDoc.SaveAs(desktoppath & "\" & FirmKodu & "_" & Uygulama)
		Msgbox "Masaüstüne kaydedildi..."
		Set objDoc = Nothing

		Set objWord = Nothing
