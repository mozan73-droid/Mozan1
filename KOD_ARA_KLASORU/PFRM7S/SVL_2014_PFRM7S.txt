Sub Print_form_sayfabasi(pe1)
	res = pe1.SetAlign("LEFT")
	res = pe1.Selectfont("Courier New", 7)
	CHKodu = STOK70E.CH_KODU
	CHAdi = Doc.GetDBFieldValue("CARI00", "AD", CHKodu, "")
	CHAdi2 = Doc.GetDBFieldValue("CARI00", "NAME2", CHKodu, "")
	CHAdres1 = Doc.GetDBFieldValue("CARI00", "ADRES_1", CHKodu, "")
	CHAdres2 = Doc.GetDBFieldValue("CARI00", "ADRES_2", CHKodu, "")
	CHAdres3 = Doc.GetDBFieldValue("CARI00", "ADRES_3", CHKodu, "")
	VergiD = Doc.GetDBFieldValue("CARI00", "VERGIDAIRESI", CHKodu, "")
	VergiNo = Doc.GetDBFieldValue("CARI00", "VERGINO", CHKodu, "")
		
	faturaTarihi = Doc.StrToNorDate(STOK70E.TARIH)

	
	
	IrsNolar = IrsStr(ST70)
	IrsTarihleri = IrsTarih(ST70)
	res = pe1.write(UstMargin + 10, solmargin+1, 5, 70, CHAdi)
	res = pe1.write(pe1.GetTop + satirYuksekligi, solmargin+1, 5, 80, CHAdi2)
	res = pe1.write(pe1.GetTop + satirYuksekligi, solmargin+1, 5, 80, CHAdres1)
	res = pe1.write(pe1.GetTop + satirYuksekligi, pe1.GetLeft, 5, 80, CHAdres2)
	res = pe1.write(pe1.GetTop + satirYuksekligi, pe1.GetLeft, 5, 80, CHAdres3)

	res = pe1.write(UstMargin + 60 , SolMargin + 50, 6, 30, trim(VergiNo))
	
	'res = pe1.write(pe1.GetTop , SolMargin + 65, 6, 30, trim(IrsTarihleri))
End Sub

Sub Print_form_satir(pe1, satirNo)
Dim satirpozr
		If trim(S70T.KOD(satirNo)) = "" Then Exit Sub
		Degisken = ""
		sonSatir = pe1.GetBottom()
		If sonSatir >=  satirSonu Then
			Call AraSayfaSonu(pe1)
			siraNo = 1
			res = pe1.PageBreak()
			Call Print_form_sayfabasi(pe1)
			satirpozr = satirBaslangic
		Else
			If satirNo = 1 Then
				satirpozr = satirBaslangic
			Else
				satirpozr = SatirBaslangic + siraNo * satiryuksekligi
			End If
			siraNo = siraNo + 1
		End If		
		res = pe1.setbold(0)
		if pe1.GetPageCount() > 1 and sirano=1 then 
			res= pe1.Setalign("LEFT")
			res = pe1.SetBold(1)
			res= pe1.write(satirpozr , solTutar - 30, satirYuksekligi, 30, "NAKLÝ YEKÜN") 
			res= pe1.Setalign("RIGHT")
			res= pe1.write(satirpozr, solTutar, satirYuksekligi, GenTutar, Doc.NumFormat(AraToplam,20,2)) 		
			res = pe1.SetBold(0)
			satirpozr = satirpozr + satirYuksekligi
			sirano = sirano + 1
		end if
		Sol = SolMargin + 1
		res = pe1.setalign("LEFT")
			For i = 0 To UBound(Grid, 1)
				SutunGenisligi = Grid(i, 1) 
				DegiskenAdi = Grid(i, 0)				

				if degiskenAdi = "TL_TUTAR" then
					AraToplam = AraToplam + S70T.GetFieldValue(DegiskenAdi, SatirNo)
					solTutar = sol
					GenTutar = SutunGenisligi
				End if
				if  degiskenAdi = "*" then
					iskontotutari=S70T.TL_XISKONTO_TUTARI(satirNo)
					istutar=S70T.TL_TUTAR(satirNo)
					sonuc=Doc.Div(iskontotutari,istutar) * 100
					Degisken=sonuc
				else
				Degisken = S70T.GetFieldValue(DegiskenAdi, SatirNo)
				end if
				if  degiskenAdi = "TL_FIYAT" then
					Miktar = S70T.SF_MIKTAR(satirNo)
					Tutar = S70T.TL_TUTAR(satirNo)
					BirimFiyat = doc.div(Tutar,Miktar)
					Degisken=BirimFiyat
				else
					Degisken = S70T.GetFieldValue(DegiskenAdi, SatirNo)
				end if

				If Grid(i, 4) = "N" Then
					res = pe1.SetAlign("RIGHT")
					Ondalik = Grid(i,6)			
					if degiskenadi="TL_FIYAT" then	
						Degisken = Doc.NumFormat(Degisken, 18, 4)
					else
						Degisken = Doc.NumFormat(Degisken, 18, Ondalik)
					end if
				End If

				res = pe1.write(satirpozr, Sol, satiryuksekligi, SutunGenisligi - 2, Degisken)			
				Sol = Sol + SutunGenisligi
			Next					 
		sonSatir = pe1.GetBottom()
End Sub

Sub AraSayfaSonu(pe1)
	sol1 = SolMargin + 1
	for i = 0 to uBound(Grid,1)
		if Grid(i,0) = "TL_TUTAR" then
			GenTutar = Grid(i,1) - 2
			solTutar = Sol1			
		elseif Grid(i,0) = "TL_FIYAT" then
			GenFiyat = Grid(i,1) + Grid(i+1,1) - 2
			solFiyat = Sol1
		end if
		sol1 = sol1 + Grid(i,1)
	Next
	res = pe1.SetAlign("LEFT")
	res = pe1.Write(pe1.GetTop + satirYuksekligi,solFiyat,5,genFiyat,"ARA TOPLAM")
	res = pe1.SetAlign("RIGHT")
	res = pe1.Write(pe1.GetTop,solTutar,5,genTutar,Doc.NumFormat(AraToplam,20,2))	
End Sub

Function IrsStr(ST70)
	For i = 1 to ST70.GetRecCount
		irsaliyeNo = ST70.IRSALIYENO(i)
		if trim(irsaliyeNo) <> "" then
			Istr = Istr & trim(irsaliyeNo) & ", "		
		end if
	Next
	if trim(Istr) <> "" then
		Istr = left(Istr, len(Istr)-2)
	end if
	IrsStr = Istr
End Function

Function IrsTarih(ST70)
	For i = 1 to ST70.GetRecCount
		irsaliyeNo = ST70.IRSALIYENO(i)
		if trim(irsaliyeNo) <> "" then
			Tarih = Doc.GetDBFieldValue("STOK60E", "TARIH", irsaliyeNo, "")
			Tarih = Doc.StrToNorDate(Tarih)
			Itarih = Itarih & trim(Tarih) & ", "		
		end if
	Next
	if trim(Itarih) <> "" then
		Itarih = left(Itarih, len(Itarih)-2)
	end if
	Irstarih = Itarih
End Function

Function DipToplamSatirSayisiHesapla()

	satirSayisi = 7

	Iskonto1 = STOK70E.TL_ISKONTO1TOPLAMI
	Iskonto2 = STOK70E.TL_ISKONTO2TOPLAMI
	Iskonto3 = STOK70E.TL_ISKONTO3TOPLAMI
	Iskonto4 = STOK70E.TL_ISKONTO4TOPLAMI

	IskontoTutari = STOK70E.TL_ISKONTOLARTOPLAMI
	OTVTutari = STOK70E.TL_VERART1TOPLAMI

'***********

	if trim(Iskonto1) > 0 then
		satirSayisi = satirSayisi + 1
	end if
	if trim(Iskonto2) > 0 then
		satirSayisi = satirSayisi + 1
	end if
	if trim(Iskonto3) > 0 then
		satirSayisi = satirSayisi + 1
	end if
	if trim(Iskonto4) > 0 then
		satirSayisi = satirSayisi + 1
	end if

	if trim(IskontoTutari) > 0 then
		satirSayisi = satirSayisi + 1
	end if

	if trim(OTVTutari) > 0 then
		satirSayisi = satirSayisi + 2
	end if
	DipToplamSatirSayisiHesapla = satirSayisi
End Function

Sub Alanlar()

	ReDim Grid(6, 7)


	Grid(0, 0) = "KOD"
	Grid(1, 0) = "KOD_STOK00_AD"
	Grid(2, 0) = "SF_SF_UNIT"
	Grid(3, 0) = "SF_MIKTAR"
	Grid(4, 0) = "TL_FIYAT"
	Grid(5, 0) = "TL_TUTAR"

	Grid(0, 1) = 18
	Grid(1, 1) = 73
	Grid(2, 1) = 10
	Grid(3, 1) = 15
	Grid(4, 1) = 25
	Grid(5, 1) = 30

	Grid(0, 4) = "C"
	Grid(1, 4) = "C"
	Grid(2, 4) = "C"
	Grid(3, 4) = "N"
	Grid(4, 4) = "N"
	Grid(5, 4) = "N"

	
	For i = 0 To UBound(Grid, 1) 
		Degisken = Grid(i, 0)
		If Grid(i, 4) = "N" Then
			ondalik = ST70.MaxSigDig(Degisken)	
			Grid(i, 6) = Ondalik		
		End If
	Next	
	Grid(2, 6) = 0
	Grid(4, 6) = 2
End Sub
Function FiyatBul(StokKodu)

Sub Print_form_satir(pe1, satirNo)

		If trim(S70T.KOD(satirNo)) = "" Then Exit Sub

Function IrsStr(ST70)
	For i = 1 to ST70.GetRecCount
		irsaliyeNo = ST70.IRSALIYENO(i)
		if trim(irsaliyeNo) <> "" then
			Istr = Istr & trim(irsaliyeNo) & ", "		
		end if
	Next
	if trim(Istr) <> "" then
		Istr = left(Istr, len(Istr)-2)
	end if
	IrsStr = Istr
End Function

Function IrsTarih(ST70)
	For i = 1 to ST70.GetRecCount
		irsaliyeNo = ST70.IRSALIYENO(i)
		if trim(irsaliyeNo) <> "" then
			Tarih = Doc.GetDBFieldValue("STOK60E", "TARIH", irsaliyeNo, "")
			Tarih = Doc.StrToNorDate(Tarih)
			Itarih = Itarih & trim(Tarih) & ", "		
		end if
	Next
	if trim(Itarih) <> "" then
		Itarih = left(Itarih, len(Itarih)-2)
	end if
	Irstarih = Itarih
End Function

Sub Print_form_satir(pe1, satirNo)
Dim satirpozr
		If trim(S70T.KOD(satirNo)) = "" Then Exit Sub
		Degisken = ""
		sonSatir = pe1.GetBottom()
		If sonSatir >=  satirSonu Then
			Call AraSayfaSonu(pe1)
			siraNo = 1

			Call Print_form_sayfabasi(pe1)
			satirpozr = satirBaslangic
		Else
			If satirNo = 1 Then
				satirpozr = satirBaslangic
			Else
				satirpozr = SatirBaslangic + siraNo * satiryuksekligi
			End If
			siraNo = siraNo + 1
		End If		
		res = pe1.setbold(0)
		if pe1.GetPageCount() > 1 and sirano=1 then 
			res= pe1.Setalign("LEFT")
			res = pe1.SetBold(1)
			res= pe1.write(satirpozr , solTutar - 30, satirYuksekligi, 30, "NAKLÝ YEKÜN") 
			res= pe1.Setalign("RIGHT")
			res= pe1.write(satirpozr, solTutar, satirYuksekligi, GenTutar, Doc.NumFormat(AraToplam,20,2)) 		
			res = pe1.SetBold(0)
			satirpozr = satirpozr + satirYuksekligi
			sirano = sirano + 1
		end if
		Sol = SolMargin + 1
		res = pe1.setalign("LEFT")
			For i = 0 To UBound(Grid, 1)
				SutunGenisligi = Grid(i, 1) 
				DegiskenAdi = Grid(i, 0)				

				if degiskenAdi = "TL_TUTAR" then
					AraToplam = AraToplam + S70T.GetFieldValue(DegiskenAdi, SatirNo)
					Degisken = Doc.NumFormat(Degisken, 18, 2)
					solTutar = sol
					GenTutar = SutunGenisligi
				End if
				if  degiskenAdi = "*" then
					iskontotutari=S70T.TL_XISKONTO_TUTARI(satirNo)
					istutar=S70T.TL_TUTAR(satirNo)
					sonuc=Doc.Div(iskontotutari,istutar) * 100
					Degisken=sonuc
				else
				Degisken = S70T.GetFieldValue(DegiskenAdi, SatirNo)
				end if
				if  degiskenAdi = "TL_FIYAT" then
					Miktar = S70T.SF_MIKTAR(satirNo)
					Tutar = S70T.TL_TUTAR(satirNo)
					BirimFiyat = doc.div(Tutar,Miktar)
					Degisken=BirimFiyat
				else
					Degisken = S70T.GetFieldValue(DegiskenAdi, SatirNo)
				end if

				If Grid(i, 4) = "N" Then
					res = pe1.SetAlign("RIGHT")
					Ondalik = Grid(i,6)			
					if degiskenadi="TL_FIYAT" then	
						Degisken = Doc.NumFormat(Degisken, 18, 4)
					else
						Degisken = Doc.NumFormat(Degisken, 18, Ondalik)
					end if
				End If

				res = pe1.write(satirpozr-8, Sol, satiryuksekligi, SutunGenisligi - 2, Degisken)			
				Sol = Sol + SutunGenisligi
			Next					 
		sonSatir = pe1.GetBottom()
End Sub

Function IrsStr(ST70)
	For i = 1 to ST70.GetRecCount
		irsaliyeNo = ST70.IRSALIYENO(i)
		if trim(irsaliyeNo) <> "" then
			Istr = Istr & trim(irsaliyeNo) & ", "		
		end if
	Next
	if trim(Istr) <> "" then
		Istr = left(Istr, len(Istr)-2)
	end if
	IrsStr = Istr
End Function

Function IrsTarih(ST70)
	For i = 1 to ST70.GetRecCount
		irsaliyeNo = ST70.IRSALIYENO(i)
		if trim(irsaliyeNo) <> "" then
			Tarih = Doc.GetDBFieldValue("STOK60E", "TARIH", irsaliyeNo, "")

			Itarih = Itarih & trim(Tarih) & ", "		
		end if
	Next
	if trim(Itarih) <> "" then
		Itarih = left(Itarih, len(Itarih)-2)
	end if
	Irstarih = Itarih
End Function

Sub Print_form_satir(pe1, satirNo, xFLAG)

			For i=0 To UBound(Grid,1)

					artno=STOK40T.ARTNO(satirNo)
					Set siparis = Doc.Select1("ABC","STOK40T1","ARTNO='"+artno+"'","",-1)

				res = pe1.SetAlign("LEFT")

Sub Print_form_satir(pe1, satirNo)

Function IrsStr(ST70)
	For i = 1 to ST70.GetRecCount
		irsaliyeNo = ST70.IRSALIYENO(i)
		if trim(irsaliyeNo) <> "" then
			Istr = Istr & trim(irsaliyeNo) & ", "		
		end if
	Next
	if trim(Istr) <> "" then
		Istr = left(Istr, len(Istr)-2)
	end if
	IrsStr = Istr
End Function

Function IrsTarih(ST70)
	For i = 1 to ST70.GetRecCount
		irsaliyeNo = ST70.IRSALIYENO(i)
		if trim(irsaliyeNo) <> "" then
			Tarih = Doc.GetDBFieldValue("STOK60E", "TARIH", irsaliyeNo, "")
			Tarih = Doc.StrToNorDate(Tarih)
			Itarih = Itarih & trim(Tarih) & ", "		
		end if
	Next
	if trim(Itarih) <> "" then
		Itarih = left(Itarih, len(Itarih)-2)
	end if
	Irstarih = Itarih
End Function

Sub Print_form_satir(pe1, satirNo)
Dim satirpozr
		If trim(S70T.KOD(satirNo)) = "" Then Exit Sub
		Degisken = ""
		sonSatir = pe1.GetBottom()
		If sonSatir >=  satirSonu Then
			Call AraSayfaSonu(pe1)
			siraNo = 1
			res = pe1.PageBreak()
			Call Print_form_sayfabasi(pe1)
			satirpozr = satirBaslangic
		Else
			If satirNo = 1 Then
				satirpozr = satirBaslangic
			Else
				satirpozr = SatirBaslangic + siraNo * satiryuksekligi
			End If
			siraNo = siraNo + 1
		End If		
		res = pe1.setbold(0)
		if pe1.GetPageCount() > 1 and sirano=1 then 
			res= pe1.Setalign("LEFT")
			res = pe1.SetBold(1)
			res= pe1.write(satirpozr , solTutar - 30, satirYuksekligi, 30, "NAKLÝ YEKÜN") 
			res= pe1.Setalign("RIGHT")
			res= pe1.write(satirpozr, solTutar, satirYuksekligi, GenTutar, Doc.NumFormat(AraToplam,20,2)) 		
			res = pe1.SetBold(0)
			satirpozr = satirpozr + satirYuksekligi
			sirano = sirano + 1
		end if
		Sol = SolMargin + 1
		res = pe1.setalign("LEFT")
			For i = 0 To UBound(Grid, 1)
				SutunGenisligi = Grid(i, 1) 
				DegiskenAdi = Grid(i, 0)				

				if degiskenAdi = "OR_TUTAR" then
					AraToplam = AraToplam + S70T.GetFieldValue(DegiskenAdi, SatirNo)
					solTutar = sol
					GenTutar = SutunGenisligi
				End if
				if  degiskenAdi = "*" then
					iskontotutari = S70T.OR_XISKONTO_TUTARI(satirNo)
					istutar = S70T.OR_TUTAR(satirNo)
					sonuc = Doc.Div(iskontotutari,istutar) * 100
					Degisken = sonuc
				else
				Degisken = S70T.GetFieldValue(DegiskenAdi, SatirNo)
				end if
				if  degiskenAdi = "OR_FIYAT" then
					Miktar = S70T.SF_MIKTAR(satirNo)
					Tutar = S70T.OR_TUTAR(satirNo)
					BirimFiyat = Doc.Div(Tutar,Miktar)
					Degisken = BirimFiyat
				else
					Degisken = S70T.GetFieldValue(DegiskenAdi, SatirNo)
				end if
				If Grid(i, 4) = "N" Then
					res = pe1.SetAlign("RIGHT")
					Ondalik = Grid(i,6)			
					if degiskenadi="OR_FIYAT" then	
						Degisken = Doc.NumFormat(Degisken, 18, 3)
					elseif degiskenadi="OR_TUTAR" then	
						Degisken = Doc.NumFormat(Degisken, 18, 2)

					else
						Degisken = Doc.NumFormat(Degisken, 18, Ondalik)
					end if
				End If

				res = pe1.write(satirpozr, Sol, satiryuksekligi, SutunGenisligi - 2, Degisken)			
				Sol = Sol + SutunGenisligi
			Next					 
		sonSatir = pe1.GetBottom()
End Sub

Sub AraSayfaSonu(pe1)

			GenTutar = Grid(i,1) - 2

		elseif Grid(i,0) = "OR_FIYAT" then

		sol1 = sol1 + Grid(i,1)
	Next
	res = pe1.SetAlign("LEFT")
	res = pe1.Write(pe1.GetTop + satirYuksekligi,solFiyat,5,genFiyat,"ARA TOPLAM")
	res = pe1.SetAlign("RIGHT")
	res = pe1.Write(pe1.GetTop,solTutar,5,genTutar,Doc.NumFormat(AraToplam,20,2))	
End Sub

Function IrsStr(ST70)
	For i = 1 to ST70.GetRecCount
		irsaliyeNo = ST70.IRSALIYENO(i)
		if trim(irsaliyeNo) <> "" then
			Istr = Istr & trim(irsaliyeNo) & ", "		
		end if
	Next
	if trim(Istr) <> "" then
		Istr = left(Istr, len(Istr)-2)
	end if
	IrsStr = Istr
End Function

Function IrsTarih(ST70)
	For i = 1 to ST70.GetRecCount
		irsaliyeNo = ST70.IRSALIYENO(i)
		if trim(irsaliyeNo) <> "" then
			Tarih = Doc.GetDBFieldValue("STOK60E", "TARIH", irsaliyeNo, "")
			Tarih = Doc.StrToNorDate(Tarih)
			Itarih = Itarih & trim(Tarih) & ", "		
		end if
	Next
	if trim(Itarih) <> "" then
		Itarih = left(Itarih, len(Itarih)-2)
	end if
	Irstarih = Itarih
End Function

Function IrsStr(ST70)
	For i = 1 to ST70.GetRecCount
		irsaliyeNo = ST70.IRSALIYENO(i)
		if trim(irsaliyeNo) <> "" then
			Istr = Istr & trim(irsaliyeNo) & ", "		
		end if
	Next
	if trim(Istr) <> "" then
		Istr = left(Istr, len(Istr)-2)
	end if
	IrsStr = Istr
End Function

Function IrsTarih(ST70)
	For i = 1 to ST70.GetRecCount
		irsaliyeNo = ST70.IRSALIYENO(i)
		if trim(irsaliyeNo) <> "" then
			Tarih = Doc.GetDBFieldValue("STOK60E", "TARIH", irsaliyeNo, "")
			Tarih = Doc.StrToNorDate(Tarih)
			Itarih = Itarih & trim(Tarih) & ", "		
		end if
	Next
	if trim(Itarih) <> "" then
		Itarih = left(Itarih, len(Itarih)-2)
	end if
	Irstarih = Itarih
End Function

Sub Print_form_satir(pe1, satirNo)
Dim satirpozr
		If trim(S70T.KOD(satirNo)) = "" Then Exit Sub

					Miktar = S70T.SF_MIKTAR(satirNo)

					else
						Degisken = Doc.NumFormat(Degisken, 18, Ondalik)
					end if

				Sol = Sol + SutunGenisligi
			Next					 
		sonSatir = pe1.GetBottom()
End Sub

Function IrsStr(ST70)
	For i = 1 to ST70.GetRecCount
		irsaliyeNo = ST70.IRSALIYENO(i)
		if trim(irsaliyeNo) <> "" then
			Istr = Istr & trim(irsaliyeNo) & ", "		
		end if
	Next
	if trim(Istr) <> "" then
		Istr = left(Istr, len(Istr)-2)
	end if
	IrsStr = Istr
End Function

Function IrsTarih(ST70)
	For i = 1 to ST70.GetRecCount
		irsaliyeNo = ST70.IRSALIYENO(i)
		if trim(irsaliyeNo) <> "" then
			Tarih = Doc.GetDBFieldValue("STOK60E", "TARIH", irsaliyeNo, "")
			Tarih = Doc.StrToNorDate(Tarih)
			Itarih = Itarih & trim(Tarih) & ", "		
		end if
	Next
	if trim(Itarih) <> "" then
		Itarih = left(Itarih, len(Itarih)-2)
	end if

Sub PrintForm()
	
	Dim objDoc, objWord, Selection
	Set objWord = CreateObject( "Word.Application" )
	objWord.visible = True
	Set objDoc = objWord.Documents.Add()
	Set Selection = objWord.Selection
	'objWord.Selection.EndKey 6,0
	FirmKodu = Trim(Doc.Getd7FirmCode())
	Uygulama = Trim(Doc.GetDocType)

	Set PFR7SE = Doc.Select1("dd333d", "PFRM7SE", "", "EVRAKNO", -1)
	'Set MKR7SE = Doc.Select1("dd333d", "MAKR7SE", "EVRAKNO = '_GNPRG1'", "", -1)
	
	KayitSayisi = PFR7SE.GetRecCount()
	'-------------------------------------------------
	'Satýr boþluk ayarýný yapar.
	Selection.ParagraphFormat.LineSpacing = 10
	objWord.ActiveDocument.ShowSpellingErrors = False
	'----------------------------------------------
	'Tab ayarýný yapar.
	objWord.ActiveDocument.DefaultTabStop = 15
	'---------------------------------------------
	For i = 1 To KayitSayisi
		EVRAKKODU_ORJ = Trim(PFR7SE.EVRAKNO(i))
		If Left(EVRAKKODU_ORJ, 2) <> "++" Then
			EVRAKKODU = Trim(PFR7SE.EVRAKNO(i))

			Kodlar = ""
			EKodlar = ""
			Selection.Paragraphs.Alignment = 1   '1:wdAlignParagraphCenter yani üst satýrdakini ortalar.
			Selection.TypeText "Form Kaynak Kodu"
			Selection.Typeparagraph
		
			Selection.Paragraphs.Alignment = 3
			Selection.TypeText "Firma Kodu : " & FirmKodu  & " / " & "Uygulama : " &  Uygulama
			Selection.Typeparagraph
			Selection.TypeText "Makro Kodu : " & EVRAKKODU 
			'------------------------
		
			'Selection.Paragraphs.Alignment = 1 wdAlignParagraphCenter yani üst satýrdakini ortalar.
			'2: wdAlignParagraphRight yani üst satýrdakini saga yaslar.
			'3: wdAlignParagraphRight yani üst satýrdakini sola yaslar.
		
			Selection.Paragraphs.Alignment = 3
			Selection.Typeparagraph
			Selection.TypeText "Hangi Tablo için : " & Trim(PFR7SE.DOCTYPE(i))
			Selection.Typeparagraph
			Selection.TypeText "Ýlave Hangi Uygulamalar için : " & Trim(PFR7SE.ADDDOCTYPES(i))
			Selection.Typeparagraph
			Selection.TypeText "Açýklama : " & Trim(PFR7SE.ACIKLAMA(i))
			Selection.Typeparagraph
			Selection.TypeText "Kod : "
			Selection.Typeparagraph

			Set PFR7ST = Doc.Select1("dd333d", "PFRM7ST", "EVRAKNO='"&EVRAKKODU&"'",">SRNUM", -1)

			For k = 1 To PFR7ST.GetRecCount()
				Selection.Font.Color = vbBlue
				Kodlar = Kodlar & PFR7ST.SCRIPTTEXT_RO(k)
			Next

			Selection.TypeText Kodlar
			Selection.Typeparagraph
			Selection.Typeparagraph
			'-----------------------------------
			Selection.Typeparagraph
			'-----------------------------------
			Selection.Font.Color = vbBlack
			Set PFR7SE= Doc.Select1("dd333d", "PFRM7SE", "", "", -1)
		
			Selection.TypeText "----------------------------------------------------------------"
			Selection.Typeparagraph
		End If
	Next
	'--------------------------------------------------------------------------
	'Font lara makyaj
	For j = 1  To 8
		strfind(1) = "Makro Kaynak Kodu"
		strfind(2) = "Makro Kodu : "
		strfind(3) = "Hangi Tablo için : "
		strfind(4) = "Ýlave Hangi Uygulamalar için : "
		strfind(5) = "Açýklama : "
		strfind(6) = "Kod : "
		strfind(7) = "Firma Kodu : "
		strfind(8) = "Uygulama : "

		Set objRange = objWord.ActiveDocument.Content
		
		With objRange.Find
		.Text = strfind(j)
		Do
			.Execute
			If .Found Then
				objRange.Font.Color = vbRed
				objRange.Font.Bold = True
				'If j = 1 Then objWord.ParagraphFormat.Alignment = wdAlignParagraphCenter
			End If
			Loop While .Found
			End With
		Next
		Selection.WholeStory
		'----------------------------------------------------
		'Satýr sonlarýndaki boþluðu alma
		Selection.ParagraphFormat.SpaceAfter = 0.5
		'---------------------------------------------------
		'Sayfanýn kenarlarýný daralt
		With objWord.ActiveDocument.PageSetup
			.Orientation = wdOrientLandscape
			bosluk = 15
			.TopMargin = bosluk
			.BottomMargin = bosluk
			.LeftMargin = 43
			.RightMargin = bosluk
		End With
		'--------------------------------------------------------------------------
		'Satýr numaralarý ekler
    With Selection.PageSetup
        With .LineNumbering
            .Active = True
            .StartingNumber = 1
            .CountBy = 1
            .RestartMode = 1
            .DistanceFromText = 0
        End With
		End With
		'--------------------------------------------------------------------------
		objWord.visible = True
		desktoppath = CreateObject("WScript.Shell").SpecialFolders("Desktop")
		objDoc.SaveAs(desktoppath & "\" & FirmKodu & "_" & Uygulama)
		Msgbox "Masaüstüne kaydedildi..."
		Set objDoc = Nothing
		Set oWords = Nothing
		Set objWord = Nothing
End Sub
