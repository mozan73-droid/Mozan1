Sub D7KIDSLB_Print()
	dim i,D7SPL,pe1
	sayfaBoyu = 75
	sayfaEni = 260
	satiryuksekligi = 3
	sayac = 1

	Call KTA_Yaz()
	
	Set pe1 = doc.CreatePrintEngineOutput()
	res = pe1.Open(0, "Paletleme", SayfaBoyu, SayfaEni, "")

	set D7SPL = Doc.GetTableObject("D7KIDSPL")
	BARKODNO = trim(D7SPL.KOD(1))

	D7KIDSPL.SafePack()
	D7KIDSPL.Sort("SKOD")
	PALET_KAYITSATIR_SAYISI=D7KIDSPL.GetRecCount()

	Call ALANLAR(D7SPL)

	If doc.ErrorNumber Then
		MsgBox ("Hata main : " & doc.ErrorNumber)
		Exit Sub
	End If

	Call Print_form_sayfabasi(pe1)

	For i = 1 To PALET_KAYITSATIR_SAYISI
		Call Print_form_satir(pe1, i, D7SPL)
	Next

	'res = pe1.Show()
	res	=	pe1.PrintDoc(0)
	res = Doc.ActivateAfterMacro(0)
End Sub

Sub Print_form_satir(pe1,satirNo, D7SPL)

	Dim satirpozr

	If trim(D7SPL.KOD(satirNo)) = "" Then Exit Sub
	ReDim xCizgi(6)
	Degisken = ""
	sonSatir = pe1.GetBottom()

	If satirNo = 1 Then
		satirpozr = satirBaslangic
	Else
		satirpozr = pe1.GetTop() + 1
	End If
	siraNo = siraNo + 1
	'**********************************************************************************
	res = pe1.selectfont("Times New Roman", 6)
	res = pe1.setbold(0)

	Sol = CERCEVE_SOL_KONUM + 1
	res = pe1.setalign("LEFT")

	'DEÐERLER
	For j=1 To 7
		SutunGenisligi = Grid(j,1) 
		DegiskenAdi = Grid(j,0)

		res = pe1.selectfont("Times New Roman", 6)

		If left(DegiskenAdi,1) <> "_" Then
			Degisken = D7SPL.GetFieldValue(DegiskenAdi,SatirNo)
		Else
			If DegiskenAdi = "_POS" Then
				Degisken = SatirNo
			Else
				Degisken = ""
			End If
		End If	

		res = pe1.SetAlign("LEFT")
		
		If Grid(j,2) = "MÝKTAR" Then
			res = pe1.SetAlign("RIGHT")
			'Ondalik = Grid(j,2)
			Degisken = Doc.NumFormat(Degisken, 18, Ondalik)
			Sol = Sol - 2

		ElseIf Grid(j,2)= "BÝRÝM" Then
			res = pe1.SetAlign("RIGHT")

		ElseIf Grid(j,2)= "PARTÝ NO" Then
			res = pe1.SetAlign("CENTER")

		ElseIf Grid(j,2)= "BARKOD NO" Then
			res = pe1.SetAlign("CENTER")
		
		ElseIf Grid(j,2)= "PARÇA KODU" Then
			res = pe1.SetAlign("LEFT")
			Degisken = Trim(Degisken)
		End If
		
		res = pe1.write(satirpozr, sol, satiryuksekligi, SutunGenisligi, Degisken)
		res = pe1.SetItalic(0)
		Sol = Sol + SutunGenisligi 
		res = pe1.Line(pe1.GetBottom, CERCEVE_SOL_KONUM, 0, sayfaeni-(solmargin+sagmargin))
	Next	

	If	sayac = PALET_KAYITSATIR_SAYISI Then

		DETAY_ALT_KONUM = pe1.GetBottom

		'ÖZET BAÞLIK
		res = pe1.SelectFont("Arial", 10)
		res = pe1.SetBold(1)
		res = pe1.SetAlign("Center")
		res = pe1.pensize(0.5)
		res = pe1.Line(pe1.GetBottom, CERCEVE_SOL_KONUM, 0, sayfaeni-(solmargin+sagmargin))
		res = pe1.write(DETAY_ALT_KONUM+0.5, CERCEVE_SOL_KONUM+1, 4, 96, "PARÇA BAZINDA TOPLAM ÖZETÝ")
		res = pe1.Line(pe1.GetBottom+0.5, CERCEVE_SOL_KONUM, 0, sayfaeni-(solmargin+sagmargin))
		TOPLAMOZETLERI_ALT_KONUM = pe1.GetBottom
		res = pe1.pensize(0.2)
		res = pe1.SelectFont("Arial", 9)
		res = pe1.SetBold(0)
		res = pe1.SetAlign("Left")

		'TOPLAMLARI ICIN
		Set D7KPALET= Doc.Select2("XXXSTOK63T", "D7KIDSPL", "", "", -1)
		D7KPALET.Sort("SKOD")
		D7KPALET.GroupBy2("SKOD")

		satirBaslangic = TOPLAMOZETLERI_ALT_KONUM
		For u = 1 To D7KPALET.GetRowCount()
			if u=1 then 
				satirpozr_OZET = satirBaslangic
			else
				satirpozr_OZET = pe1.GetTop() + 0.5
			end if
			siraNo = siraNo + 1

			OPSKODLARI(u) = D7KPALET.SKOD(u)
			MUSTERI_PARCA_KODLARI(u) = D7KPALET.SKOD_STOK00_AD(u)
			KODAGORE_MIKTAR(u) = D7KPALET.SF_MIKTAR(u)
			KODAGOREBIRIM(u) = D7KPALET.SF_SF_UNIT(u)

			res = pe1.write(satirpozr_OZET, CERCEVE_SOL_KONUM+1, 4, 24, OPSKODLARI(u))
			OPSKODLARI_SAG_KONUM = pe1.GetRight
			res = pe1.write(satirpozr_OZET, pe1.GetRight+1, 4, 53, MUSTERI_PARCA_KODLARI(u))
			MUSTERI_PARCA_KODLARI_SAG_KONUM = pe1.GetRight
			res = pe1.SetAlign("Right")
			KODAGORE_MIKTAR(u) = Doc.NumFormat(KODAGORE_MIKTAR(u), 16, Ondalik)
			res = pe1.write(satirpozr_OZET, pe1.GetRight+1, 4, 10, KODAGORE_MIKTAR(u))
			res = pe1.write(satirpozr_OZET, pe1.GetRight+1, 4, 5, KODAGOREBIRIM(u))
			res = pe1.Line(pe1.GetBottom, CERCEVE_SOL_KONUM, 0, sayfaeni-(solmargin+sagmargin))
		Next
		TOPLAM_KALEM_ALTI_KONUM = pe1.GetBottom
		res = pe1.Line(TOPLAMOZETLERI_ALT_KONUM, OPSKODLARI_SAG_KONUM, 4.4*D7KPALET.GetRowCount ,0)
		res = pe1.Line(TOPLAMOZETLERI_ALT_KONUM, MUSTERI_PARCA_KODLARI_SAG_KONUM, 4.4*D7KPALET.GetRowCount ,0)
	End if
	sayac = sayac+1
End Sub

Sub D7KIDSLB_Print1()
	dim i
	PALET_KAYITSATIR_SAYISI=D7KIDSPL.GetRecCount()
	
	Set Doc1 = Doc.CreateDocumentNoUI("STOK25", "")

	NRET = Doc1.New_Voucher()
	Set S25E = Doc1.GetTableObject("STOK25E")
	Set S25T = Doc1.GetTableObject("STOK25T")
	ETIKETTEKI_DEPO = GetS01Value(D7KIDSLB.SERINO(1), "AMBCODE")

	S25E.AMBCODE = ETIKETTEKI_DEPO
	S25E.TARGETAMBCODE = ETIKETTEKI_DEPO

	S25T.KOD(1) = D7KIDSPL.SKOD(1)
	S25T.LOTNUMBER(1) = D7KIDSPL.LOTNUMBER(1)
	S25T.SERINO(1) = D7KIDSPL.SERINO(1)
	S25T.SF_MIKTAR(1) = D7KIDSPL.SF_MIKTAR(1)
	BIRIM = Doc.GetKartValue("STOK00","IUNIT",S25T.KOD(1),-1)
	S25T.OSF_SF_UNIT(1) = BIRIM

	S25T.NEWCODE(1) = D7KIDSPL.SKOD(1)
	S25T.NEWLOTNUMBER(1) = S25T.LOTNUMBER(1) 

	'yeni barkod no alma
	Set Doc2 = Doc.CreateDocumentNoUI("D7KIDSLB", "")
	Set D7LB = Doc2.GetTableObject("D7KIDSLB")
	YeniSeriNo = Doc.Sayac0danAl("SERINO")

	NRET = Doc2.Load_Voucher(YeniSeriNo)

	'yeni barkod no yu kot lot seri no transferdeki yeniserino bölümüne atama
	S25T.NEWSERINO(1) = YENISERINO
	'yeni barkod etiketi sayfasýný kaydedip kapatma
	Doc2.Save_Voucher
	Doc2.LoadEmpty_Voucher()

	For i = 2 To PALET_KAYITSATIR_SAYISI
		'SATIR EKLEME
		recno=S25T.addrow
		'HÜCREYE TIKLAMA (setcurrentrow ile oluyor)
		S25T.setCURRENTROW(recno)
		S25T.KOD(recno) 				= D7KIDSPL.SKOD(i)
		S25T.LOTNUMBER(recno) 	= D7KIDSPL.LOTNUMBER(i)
		S25T.SERINO(recno) 			= D7KIDSPL.SERINO(i)

		S25T.NEWCODE(recno)			= D7KIDSPL.SKOD(i)
		S25T.NEWLOTNUMBER(recno)= S25T.LOTNUMBER(1) 
		S25T.NEWSERINO(recno) 	= YENISERINO

		S25T.SF_MIKTAR(recno) 	= D7KIDSPL.SF_MIKTAR(i)
		BIRIM = Doc.GetKartValue("STOK00","IUNIT",S25T.KOD(recno),-1)
		S25T.OSF_SF_UNIT(recno) = BIRIM
	Next

	Doc1.Save_Voucher
	Doc1.RunMacro0 "PFRM7S.ETIKET2", "PrintLabel"

	Doc1.LoadEmpty_Voucher
	Doc1.DeleteDocumentNoUI "D7KIDSLB", ""
End Sub

Sub Print_form_satir(pe1,satirNo, D7SPL)

	Dim satirpozr

	If trim(D7SPL.KOD(satirNo)) = "" Then Exit Sub
	ReDim xCizgi(6)
	Degisken = ""
	sonSatir = pe1.GetBottom()

	If satirNo = 1 Then
		satirpozr = satirBaslangic
	Else
		satirpozr = pe1.GetTop() + 1
	End If
	siraNo = siraNo + 1
	'**********************************************************************************
	res = pe1.selectfont("Times New Roman", 6)
	res = pe1.setbold(0)

	Sol = CERCEVE_SOL_KONUM + 1
	res = pe1.setalign("LEFT")

	'DEÐERLER
	For j=1 To 7
		SutunGenisligi = Grid(j,1) 
		DegiskenAdi = Grid(j,0)

		res = pe1.selectfont("Times New Roman", 6)

		If left(DegiskenAdi,1) <> "_" Then
			Degisken = D7SPL.GetFieldValue(DegiskenAdi,SatirNo)
		Else
			If DegiskenAdi = "_POS" Then
				Degisken = SatirNo
			Else
				Degisken = ""
			End If
		End If	

		res = pe1.SetAlign("LEFT")
		
		If Grid(j,2) = "MÝKTAR" Then
			res = pe1.SetAlign("RIGHT")
			'Ondalik = Grid(j,2)
			Degisken = Doc.NumFormat(Degisken, 18, Ondalik)
			Sol = Sol - 2

		ElseIf Grid(j,2)= "BÝRÝM" Then
			res = pe1.SetAlign("RIGHT")

		ElseIf Grid(j,2)= "PARTÝ NO" Then
			res = pe1.SetAlign("CENTER")

		ElseIf Grid(j,2)= "BARKOD NO" Then
			res = pe1.SetAlign("CENTER")
		
		ElseIf Grid(j,2)= "PARÇA KODU" Then
			res = pe1.SetAlign("LEFT")
			Degisken = Trim(Degisken)
		End If
		
		res = pe1.write(satirpozr, sol, satiryuksekligi, SutunGenisligi, Degisken)
		res = pe1.SetItalic(0)
		Sol = Sol + SutunGenisligi 
		res = pe1.Line(pe1.GetBottom, CERCEVE_SOL_KONUM, 0, sayfaeni-(solmargin+sagmargin))
	Next	

	If	sayac = PALET_KAYITSATIR_SAYISI Then

		DETAY_ALT_KONUM = pe1.GetBottom

		'ÖZET BAÞLIK
		res = pe1.SelectFont("Arial", 10)

		res = pe1.pensize(0.2)
		res = pe1.SelectFont("Arial", 9)
		res = pe1.SetBold(0)

		'TOPLAMLARI ICIN
		Set D7KPALET= Doc.Select2("XXXSTOK63T", "D7KIDSPL", "", "", -1)
		D7KPALET.Sort("SKOD")
		D7KPALET.GroupBy2("SKOD")

			if u=1 then

			else
				satirpozr_OZET = pe1.GetTop() + 0.5
			end if
			siraNo = siraNo + 1

			OPSKODLARI(u) = D7KPALET.SKOD(u)

			KODAGORE_MIKTAR(u) = D7KPALET.SF_MIKTAR(u)

			OPSKODLARI_SAG_KONUM = pe1.GetRight
			res = pe1.write(satirpozr_OZET, pe1.GetRight+1, 4, 53, MUSTERI_PARCA_KODLARI(u))

			res = pe1.Line(pe1.GetBottom, CERCEVE_SOL_KONUM, 0, sayfaeni-(solmargin+sagmargin))

		TOPLAM_KALEM_ALTI_KONUM = pe1.GetBottom

Sub Print_form_satir(pe1,satirNo, D7SPL)

	Dim satirpozr

	If trim(D7SPL.KOD(satirNo)) = "" Then Exit Sub
	ReDim xCizgi(6)
	Degisken = ""
	sonSatir = pe1.GetBottom()

	If satirNo = 1 Then
		satirpozr = satirBaslangic
	Else
		satirpozr = pe1.GetTop() + 1
	End If
	siraNo = siraNo + 1
	'**********************************************************************************
	res = pe1.selectfont("Times New Roman", 6)
	res = pe1.setbold(0)

	Sol = CERCEVE_SOL_KONUM + 1
	res = pe1.setalign("LEFT")

	'DEÐERLER
	For j=1 To 7
		SutunGenisligi = Grid(j,1) 
		DegiskenAdi = Grid(j,0)

		res = pe1.selectfont("Times New Roman", 6)

		If left(DegiskenAdi,1) <> "_" Then
			Degisken = D7SPL.GetFieldValue(DegiskenAdi,SatirNo)
		Else
			If DegiskenAdi = "_POS" Then
				Degisken = SatirNo
			Else
				Degisken = ""
			End If
		End If	

		res = pe1.SetAlign("LEFT")
		
		If Grid(j,2) = "MÝKTAR" Then
			res = pe1.SetAlign("RIGHT")
			'Ondalik = Grid(j,2)
			Degisken = Doc.NumFormat(Degisken, 18, Ondalik)

		ElseIf Grid(j,2)= "BÝRÝM" Then
			res = pe1.SetAlign("RIGHT")

		ElseIf Grid(j,2)= "PARTÝ NO" Then
			res = pe1.SetAlign("CENTER")

		ElseIf Grid(j,2)= "BARKOD NO" Then
			res = pe1.SetAlign("CENTER")
		
		ElseIf Grid(j,2)= "PARÇA KODU" Then
			res = pe1.SetAlign("LEFT")
			Degisken = Trim(Degisken)
		End If
		
		res = pe1.write(satirpozr, sol, satiryuksekligi, SutunGenisligi, Degisken)
		res = pe1.SetItalic(0)
		Sol = Sol + SutunGenisligi

	Next	

	If	sayac = PALET_KAYITSATIR_SAYISI Then

	For i = 1 To KayitSayisi

			Selection.Typeparagraph

			'Selection.Paragraphs.Alignment = 1 wdAlignParagraphCenter yani üst satýrdakini ortalar.

			'3: wdAlignParagraphRight yani üst satýrdakini sola yaslar.
