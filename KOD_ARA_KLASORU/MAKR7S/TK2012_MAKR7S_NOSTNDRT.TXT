Sub Makro1()
	Kod = PFRM7SE_B.SCRIPTTEXT
	KodArray = Split(Kod, vbcrlf)
	PFRM7SE_B.SCRIPTTEXT = ""
	Girinti = 0
	For i = 0 to uBound(KodArray)
		Doc.StatusBarMessage i & " / " & uBound(KodArray)
		satir = KodArray(i)
		satir = Replace(satir, Chr(9), " ")
		satir = trim(Satir)
				
		if uCase(Left(satir, 7)) = "ELSEIF " Then
			Girinti = Girinti - 1
		Elseif uCase(satir) = "ELSE" Then
			Girinti = Girinti - 1
		Elseif uCase(satir) = "END SUB" Then
			Girinti = 0
		Elseif uCase(satir) = "END FUNCTION" Then
			Girinti = 0
		Elseif uCase(satir) = "END IF" Then
			Girinti = Girinti - 1
		Elseif uCase(satir) = "NEXT" Then
			Girinti = Girinti - 1
		Elseif uCase(satir) = "WEND" Then
			Girinti = Girinti - 1
		Elseif uCase(satir) = "LOOP" Then
			Girinti = Girinti - 1
		Elseif uCase(Left(satir, 5)) = "CASE " Then
			Girinti = Girinti - 1
		Elseif uCase(satir) = "END SELECT" Then
			Girinti = Girinti - 2
		End if
		YeniSatir = Girintilisatir(satir, Girinti)
		
		PFRM7SE_B.SCRIPTTEXT = trim(PFRM7SE_B.SCRIPTTEXT) & YeniSatir & vbcrlf
		'  PFRM7SE_B.SCRIPTTEXT = YeniSatir
		
		if Ucase(Left(satir, 4)) = "SUB " Then
			Girinti = 1
		Elseif Ucase(Left(satir, 9)) = "FUNCTION " Then
			Girinti = 1
		Elseif uCase(Left(satir, 3)) = "IF " and Not TekSatirIf(satir) Then
			Girinti = Girinti  + 1
		Elseif uCase(Left(satir, 5)) = "CASE " Then
			Girinti = Girinti  + 1
		Elseif uCase(Left(satir, 12)) = "SELECT CASE " Then
			Girinti = Girinti  + 2
		Elseif uCase(Left(satir, 7)) = "ELSEIF " Then
			Girinti = Girinti  + 1
		Elseif uCase(satir) = "ELSE" Then
			Girinti = Girinti  + 1
		Elseif uCase(Left(satir, 4)) = "FOR " Then
			Girinti = Girinti  + 1
		Elseif uCase(Left(satir, 6)) = "WHILE " Then
			Girinti = Girinti  + 1
		Elseif uCase(Left(satir, 3)) = "DO " or trim(uCase(satir)) = "DO" Then
			Girinti = Girinti  + 1
		End if
	Next
End Sub
Function Girintilisatir(satir, Girinti)
	Girintilisatir = satir
	For i = 1 to Girinti
		Girintilisatir = Chr(9) & Girintilisatir
	Next
End Function
Function TekSatirIf(satir)
	TekSatirIf = False
	k = InStr(1, uCase(satir), "THEN ")
	if k = 0 Then Exit Function
	if trim(Mid(satir, k + 5)) <> "" Then
		TekSatirIf = True
	End if
End Function
Sub TabTrim(satir)
	While Left(satir, 1) = Chr(9)
		satir = Mid(satir, 2)
	Wend
	While Right(satir, 1) = Chr(9)
		satir = Left(satir, Len(satir) - 1)
	Wend
End Sub
