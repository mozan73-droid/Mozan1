Sub AlternatiflereDagit(List, TI, CRPTable1, satirNo)
	Set fso = CreateObject("Scripting.FileSystemObject")
	Set TFile = fso.OpenTextFile("c:\crplog.txt", 2, true)
	BaslangicTarihi = CRPTable1.REV_RTESTARIH(satirNo)
	BitisTarihi = CRPTable1.REV_RTESTARIH2(satirNo)
	List.Sort("KAYNAKMIKTART")
	BaslangicTarihi1 = Doc.StrtoNorDate(BaslangicTarihi)
	For i = 1 to List.GetRowCount()	
		Doc.StatusBarMessage "Alternatif Tezgahlara Daðýtýlýyor : " & trim(List.BOMREC_KAYNAKCODE(i)) & " - " & BaslangicTarihi1 & " - %%" & Round(i/List.GetRowCount* 100,0)
		if trim(List.BOMREC_ALTKODU(i)) <> "" Then
			AnaTezgah = List.BOMREC_KAYNAKCODE(i)
			AlternatifArr = GetAlternatifTezgahList(List.BOMREC_CODE(i), List.BOMREC_KAYNAKCODE(i), List.BOMREC_ALTKODU(i))
			AnaBos = GetBosKapasite(AnaTezgah, CRPTable1, BaslangicTarihi, BitisTarihi)
			EnBos = AnaBos + List.KAYNAKMIKTART(i)
			BosTezgah = AnaTezgah
			if AlternatifArr(0) <> "" Then
				For each AltTezgah in AlternatifArr
					AltBosKapasite = GetBosKapasite(AltTezgah, CRPTable1, BaslangicTarihi, BitisTarihi)
					logStr = "StokKodu : " & List.BOMREC_CODE(i) & VBCRLF
					logStr = logStr & ", Ana Tezgah : " & AnaTezgah & VBCRLF
					logStr = logStr & ", Yük : " & List.KAYNAKMIKTART(i) & VBCRLF
					logStr = logStr & ", Boþ Kap. : " & AnaBos & VBCRLF
					logStr = logStr & ", Alt. : " & AltTezgah & VBCRLF
					logStr = logStr & ", Alt Boþ. Kap. : " & AltBosKapasite & VBCRLF
'					MSGBOX logStr
					TFile.WriteLine LogStr
					if AltBosKapasite > EnBos Then
						EnBos = AltBosKapasite
						BosTezgah = AltTezgah
						TFile.WriteLine "Alternatif eklendi : " & BosTezgah
					End if
				Next
				if BosTezgah <> List.BOMREC_KAYNAKCODE(i) Then
					List.BOMREC_KAYNAKCODE(i) = BosTezgah
					Call KapasiteleriGuncelle(CRPTable1, AnaTezgah, BosTezgah, List.KAYNAKMIKTART(i), BaslangicTarihi)
				End if
			End if
		End if
	Next
	TI.AppendTable(List)
	TFile.Close
End Sub
Sub KapasiteleriGuncelle(CRPTable1, AnaTezgah, BosTezgah, Yuk, BaslangicTarihi)
	k = 0
	For i = 1 to CRPTable1.GetRowCount()
		if CRPTable1.BOMREC_KAYNAKCODE(i) = AnaTezgah and CRPTable1.REV_RTESTARIH(i) = BaslangicTarihi Then
			CRPTable1.TALEP(i) = CRPTable1.TALEP(i) - Yuk
			CRPTable1.FARK(i) = CRPTable1.FARK(i) + Yuk
			CRPTable1.YUZDE(i) = Doc.Div(CRPTable1.TALEP(i), CRPTable1.KAPASITE(i)) * 100
'			MSGBOX CRPTable1.BOMREC_KAYNAKCODE(i) & ", " & CRPTable1.FARK(i)
			k = k + 1
			if k = 2 Then Exit Sub
		End if
		if CRPTable1.BOMREC_KAYNAKCODE(i) = BosTezgah and CRPTable1.REV_RTESTARIH(i) = BaslangicTarihi Then
			CRPTable1.TALEP(i) = CRPTable1.TALEP(i) + Yuk
			CRPTable1.FARK(i) = CRPTable1.FARK(i) - Yuk
			CRPTable1.YUZDE(i) = Doc.Div(CRPTable1.TALEP(i), CRPTable1.KAPASITE(i)) * 100
'			MSGBOX CRPTable1.BOMREC_KAYNAKCODE(i) & ", " & CRPTable1.FARK(i)
			k = k + 1
			if k = 2 Then Exit Sub
		End if
	Next
End Sub
Function GetBosKapasite(AltTezgah, CRPTable1, BaslangicTarihi, BitisTarihi)
	QStr = "BOMREC_KAYNAKCODE='"+AltTezgah+"'&REV_RTESTARIH='"+BaslangicTarihi+"'"
	Set CRP1 = Doc.Select2("ddeerr33", "CRPTable", QStr, "", -1)
	if CRP1.GetRowCount() = 0 Then
		GetBosKapasite = AddAltTezgahtoCRP(AltTezgah, CRPTable1, BaslangicTarihi, BitisTarihi)
	Else
		GetBosKapasite = CRP1.FARK(1)
	End if
End Function
Function AddAltTezgahtoCRP(AltTezgah, CRPTable1, BaslangicTarihi, BitisTarihi)
	satirNo = CRPTable1.AddRow
	CRPTable1.BOMREC_KAYNAKCODE(satirNo) = AltTezgah
	CRPTable1.REV_RTESTARIH(satirNo) = BaslangicTarihi
	CRPTable1.REV_RTESTARIH2(satirNo) = BitisTarihi
	CRPTable1.TALEP(satirNo) = 0
	CRPTable1.YUZDE(satirNo) = 0
	DonemlikKapasite = GetDonemlikKapasite(AltTezgah, BaslangicTarihi, BitisTarihi)
	CRPTable1.KAPASITE(satirNo) = DonemlikKapasite
	CRPTable1.FARK(satirNo) = DonemlikKapasite

	AddAltTezgahtoCRP = DonemlikKapasite
End Function
Function GetDonemlikKapasite(AltTezgah, BaslangicTarihi, BitisTarihi)
	HaftaIci = Doc.GetDbFieldValue("IMLT00", "B_KAPASITE", AltTezgah, "")
	Cumartesi = Doc.GetDbFieldValue("IMLT00", "B_KAPASITE6", AltTezgah, "")
	Pazar = Doc.GetDbFieldValue("IMLT00", "B_KAPASITE7", AltTezgah, "")

	ToplamKapasite = 0
	Aralik = Doc.CDateToInt(BitisTarihi) - Doc.CDateToInt(BaslangicTarihi)
	Set d7Date = Doc.CreateSObject("d7Date")
	For i = 0 to Aralik
		Tarih1 = Doc.StrDate_Increment(BaslangicTarihi, i)
		d7Date.SetDate(Tarih1)	
		haftaGun = d7Date.GetDOW()
		if haftaGun >= 1 and haftaGun <= 5 Then
			ToplamKapasite = ToplamKapasite + HaftaIci
		Elseif haftaGun = 6 Then
			ToplamKapasite = ToplamKapasite + Cumartesi
		Elseif haftaGun = 7 Then
			ToplamKapasite = ToplamKapasite + Pazar
		End if
	Next
	GetDonemlikKapasite = ToplamKapasite
End Function
Function GetAlternatifTezgahList(MamulKodu, TezgahKodu, AlternatifKodu)
	ReDim AltArr(0)
	QStr = "AKTIF_PASIF='*'&BOMREC_CODE='"+MamulKodu+"'"
	QStr = QStr & "&BOMREC_ALTKODU='"+AlternatifKodu+"'"
	QStr = QStr & "&BOMREC_KAYNAKCODE<>'"+TezgahKodu+"'"
	Set B01T = Doc.Select1("dddeerr", "BOMU01T", QStr, "", -1)
	if B01T.GetRowCount > 0 Then ReDim AltArr(B01T.GetRowCount - 1)
	For i = 1 to B01T.GetRowCount
		AltArr(i-1) = B01T.BOMREC_KAYNAKCODE(i)
	Next
	GetAlternatifTezgahList = AltArr
End Function
Function GetBaslaBitis(Grid, AlanAdi)
	Dim TarihArr(1)
	GetBaslaBitis = TarihArr
	For i = 0 to uBound(Grid, 1)
		if Grid(i,0) = AlanAdi Then
			Baslik = Grid(i,2)
			BaslangicTarihi = Right(Baslik,8)
			BaslangicTarihi = Left(BaslangicTarihi, 6) & "20" & Right(BaslangicTarihi, 2)
			if InStr(1, Baslik, "Gün") > 0 Then
				BitisTarihi = BaslangicTarihi
			Elseif InStr(1, Baslik, "Hafta") > 0 Then
				Set d7Date = Doc.CreateSObject("d7Date")
				d7Date.SetDate(BaslangicTarihi)
				d7Date.HaftaninSonGununeGit()
				BitisTarihi = d7Date.GetNorDate()
			Elseif InStr(1, Baslik, "Ay") > 0 Then
				Set d7Date = Doc.CreateSObject("d7Date")
				d7Date.SetDate(BaslangicTarihi)
				d7Date.AyinSonGununeGit()
				BitisTarihi = d7Date.GetNorDate()
			End if
			TarihArr(0) = BaslangicTarihi
			TarihArr(1) = BitisTarihi
			GetBaslaBitis = TarihArr
			Exit Function
		End if
	Next
End Function
Function GetCRPTable(TableStr)
	Dim MTable
	Set MTable = Doc.CreateTransactionalTableObject(TableStr & "=STACK","IO")

	MTable.Addfield2 "REV_RTESTARIH","DATE",10,0,"L","",""
	MTable.Addfield2 "REV_RTESTARIH2","DATE",10,0,"L","",""
	MTable.Addfield2 "BOMREC_KAYNAKCODE","CHAR",24,0,"L","",""
	MTable.Addfield2 "TALEP","DOUBLE",10,2,"R","",""
	MTable.Addfield2 "KAPASITE","DOUBLE",10,2,"R","",""
	MTable.Addfield2 "FARK","DOUBLE",10,2,"R","",""
	MTable.Addfield2 "YUZDE","DOUBLE",10,2,"R","",""

	For i = 1 to TABLO6.GetRowCount()
		TezgahKodu = TABLO6.KOD(i)
		if trim(TezgahKodu) <> "" and InStr(1, TezgahKodu, "======") = 0 Then
			For j = 2 to 40
				BaslaBitis = GetBaslaBitis(Grid, "REPCOL" & j)
				satirNo = MTable.AddRow			
				KapasiteTalebi = TABLO6.GetFieldValue("REPCOL" & j, i+3)
				Fark = TABLO6.GetFieldValue("REPCOL" & j, i+5)
				DolulukYuzdesi = TABLO6.GetFieldValue("REPCOL" & j, i+7)

				MTable.REV_RTESTARIH(satirNo) = BaslaBitis(0)
				MTable.REV_RTESTARIH2(satirNo) = BaslaBitis(1)
				MTable.BOMREC_KAYNAKCODE(satirNo) = TezgahKodu
				MTable.TALEP(satirNo) = KapasiteTalebi
				MTable.FARK(satirNo) = Fark
				MTable.YUZDE(satirNo) = DolulukYuzdesi
			Next
		End if
	Next
	Set GetCRPTable = MTable
End Function

Function FOSNConnect(KullaniciAdi, KullaniciSifre)
	Dim urlx, strHeaders, sReturn, nlenHead, iStart, j, nFoundAt, nEndOfcookie, postvars
	urlx = "https://login.ford.com.tr/authenticate4OutPrg.aspx?userid=" & KullaniciAdi & "&password=" & KullaniciSifre
'	postvars = "id=" & uid & "&password=" & pwd
	XMLHTTP.Open "POST", urlx, False
	XMLHTTP.setRequestHeader "Content-Type", "application/x-www-form-urlencoded"
	XMLHTTP.setRequestHeader "User-Agent", "Mozilla/4.0 (compatible; MSIE 6.0; Windows NT 5.1)"
	XMLHTTP.setRequestHeader "Referer", "http://targetsite.com/"
	XMLHTTP.setRequestHeader "ACCEPT-LANGUAGE", "tr"
	XMLHTTP.setRequestHeader "Accept", "image/gif, image/x-xbitmap, image/jpeg, image/pjpeg, application/vnd.ms-powerpoint, application/vnd.ms-excel, application/msword, */*"
	XMLHTTP.send postvars
	While XMLHTTP.readyState <> 4
		XMLHTTP.waitForResponse 1000
	Wend
	strHeaders = XMLHTTP.getAllResponseHeaders()
	sReturn = XMLHTTP.responseText
'	Set XMLHTTP = Nothing
	nlenHead = Len(strHeaders)
	iStart = 1

	j = 0
	ReDim arrcookie(0)
	Do While iStart < nlenHead
		nFoundAt = InStr(iStart, strHeaders, "Set-Cookie:")
		If nFoundAt = 0 Then
			Exit Do
		End If
		ReDim Preserve arrcookie(j)
		nEndOfcookie = InStr(nFoundAt, strHeaders, ";")
		If (nEndOfcookie > 0) Then
			arrcookie(j) = Mid(strHeaders, nFoundAt + 12, nEndOfcookie - nFoundAt - 12)
		End If
		iStart = nEndOfcookie + 1
		j = j + 1
	Loop

	FOSNConnect = arrcookie
End Function

Function sendData(xmlDoc, CookieArray)
	sRef = "https://fosn.ford.com.tr/"
'	Set XMLHTTP = CreateObject("Microsoft.XMLHTTP")

	XMLHTTP.Open "POST", postURL, false

	'we need to setrequestheaders twice due to KB article Q234486.
	XMLHTTP.setRequestHeader "cookie", "x=y"
	XMLHTTP.setRequestHeader "User-Agent", "Mozilla/4.0 (compatible; MSIE 6.0; Windows NT 5.1)"
	XMLHTTP.setRequestHeader "Referer", sRef
	XMLHTTP.setRequestHeader "accept-language", "tr"
	XMLHTTP.setRequestHeader "content-type", "text/html"
	XMLHTTP.setRequestHeader "content-length", ""

	For j = 0 To UBound(CookieArray)
	'set the outgoing request header with the cookies received above
		XMLHTTP.setRequestHeader "cookie", CookieArray(j)
	Next
	XMLHTTP.send xmlDoc

	sendData = xmlhttp.responseText

Function FOSNConnect(KullaniciAdi, KullaniciSifre)
	Dim urlx, strHeaders, sReturn, nlenHead, iStart, j, nFoundAt, nEndOfcookie, postvars
	urlx = "https://login.ford.com.tr/authenticate4OutPrg.aspx?userid=" & KullaniciAdi & "&password=" & KullaniciSifre
'	postvars = "id=" & uid & "&password=" & pwd
	XMLHTTP.Open "POST", urlx, False
	XMLHTTP.setRequestHeader "Content-Type", "application/x-www-form-urlencoded"
	XMLHTTP.setRequestHeader "User-Agent", "Mozilla/4.0 (compatible; MSIE 6.0; Windows NT 5.1)"
	XMLHTTP.setRequestHeader "Referer", "http://targetsite.com/"
	XMLHTTP.setRequestHeader "ACCEPT-LANGUAGE", "tr"
	XMLHTTP.setRequestHeader "Accept", "image/gif, image/x-xbitmap, image/jpeg, image/pjpeg, application/vnd.ms-powerpoint, application/vnd.ms-excel, application/msword, */*"
	XMLHTTP.send postvars
	While XMLHTTP.readyState <> 4
		XMLHTTP.waitForResponse 1000
	Wend
	strHeaders = XMLHTTP.getAllResponseHeaders()
	sReturn = XMLHTTP.responseText
'	Set XMLHTTP = Nothing
	nlenHead = Len(strHeaders)
	iStart = 1

	j = 0
	ReDim arrcookie(0)
	Do While iStart < nlenHead
		nFoundAt = InStr(iStart, strHeaders, "Set-Cookie:")
		If nFoundAt = 0 Then
			Exit Do
		End If
		ReDim Preserve arrcookie(j)
		nEndOfcookie = InStr(nFoundAt, strHeaders, ";")
		If (nEndOfcookie > 0) Then
			arrcookie(j) = Mid(strHeaders, nFoundAt + 12, nEndOfcookie - nFoundAt - 12)
		End If
		iStart = nEndOfcookie + 1
		j = j + 1
	Loop

	FOSNConnect = arrcookie
End Function

Sub SatirYaz(ASN)

			Ambalaj = S60.FATURASIZMI(i)

			PART_PRIOR_CUM_QTY = 0

			MusteriSipNo = Doc.GetDBFieldValue("STOK40E","CHSIPNO",SiparisNo,"")
			PART_PO_NO = trim(MusteriSipNo)

				KontSayisi = S60.BRUTAGIRLIK(i)

					ParcaKontSayisi = Doc.Ceil(ParcaKontSayisi)
				else
					ParcaKontSayisi = Doc.CDbl(KontSayisi)
					kontParcaSayisi = Doc.Div(Miktar,ParcaKontSayisi)
					kontParcaSayisi = Doc.Ceil(kontParcaSayisi)
				end if
				KonteynerTipi = "BOX90"
			else

				ParcaKodu = " " & trim(StokKodu)
				kontParcaSayisi = "0000"
				ParcaKontSayisi = "0000"
				KonteynerTipi = ""

Function GetMiktarTarih(StokKodu, Miktar, Tarih, Attrib)
	Dim MiktarArray()
	Redim MiktarArray(1,0)
	if Attrib = "G" then
		MiktarArray(0,0) = Miktar
		MiktarArray(1,0) = Tarih
		GetMiktarTarih = MiktarArray
		exit Function
	end if

	Gun = GetHaftaninGunu(Tarih)
'	PartiBuyuklugu = Doc.GetDBFieldValue("STOK00","B_MINPARTIBUYUKLUGU", StokKodu, "")
'	if PartiBuyuklugu = 0 then PartiBuyuklugu = Miktar
'	if PartiBuyuklugu >= Miktar then
'		MiktarArray(0,0) = Miktar
'		MiktarArray(1,0) = Tarih
'		GetMiktarTarih = MiktarArray
'		exit Function
'	end if

	if Gun > 5 then Gun = 5
	KacaBolunecek = 5 - Gun + 1

	GunlukMiktar = Doc.Ceil(Doc.Div(Miktar, KacaBolunecek))

	KalanMiktar = Miktar
	While KalanMiktar > 0
		sirano = siraNo + 1
		if KalanMiktar > GunlukMiktar then
			SipMiktari = GunlukMiktar
		Else
			SipMiktari = KalanMiktar
		end if
		KalanMiktar = KalanMiktar - SipMiktari

		MiktarArray(0, uBound(MiktarArray,2)) = SipMiktari
		TeslimTarihi = GetTarihGun(Tarih, Gun + SiraNo - 1)
		MiktarArray(1, uBound(MiktarArray,2)) = TeslimTarihi
		if KalanMiktar > 0 then
			Redim Preserve MiktarArray(1, uBound(MiktarArray,2) + 1)
		end if
		if sirano > 100 then
			msgbox "Döngü 100 'den fazla devam etti. Hatayý kontrol ediniz." & vbcrlf & "Ýþlem Kesilmiþtir" 
			Exit Function
		end if
	Wend
	GetMiktarTarih = MiktarArray
End Function
Function GetHaftaninGunu(Tarih)

	Set d7Date = Doc.CreateSObject("d7Date")

	GetHaftaninGunu = d7Date.GetDOW

Function GetTarihGun(Tarih, KacinciGun)

	GetTarihGun = Doc.StrDate_Increment(Tarih, Fark)

Sub MGKaliteOnayiVer()

		if KalanTable.GetRowCount() > 0 Then
			if trim(KalanStatu) = "HURDA" Then

			Elseif trim(KalanStatu) = "RED" Then

			End if
		End if
	Next
	Call UretimKaliteGoster()
End Sub

Sub SIKaliteGoster()
	Set MTable = Doc.GettableObject("MAKROTABLE")
	Set MTableT = Doc.GettableObject("MAKROTABLESI")
	Set S01 = Doc.Select1("w3dede", "STOK01", "AMBCODE='" & MTable.AMBCODE_SI & "'", "KOD+LOTNUMBER", -1)
	if S01.GetRowCount() = 0 Then Exit Sub
	MTableT.Empty()

	Alanlar = S01.GetFieldNames("")

	MTableT.SetRowAID 1,1,1
	For i = 1 to S01.GetRowCount()

		IrsaliyeNo = ""
		TedarikciIrsNo = ""
		TedarikciCHKodu = ""

		Doc.StatusBarMessage "Onaylanmamýþlar getiriliyor : " & i & " / " & S01.GetRowCount()
		Set S29T = Doc.Select1("dde33", "STOK29T", QStr, "", -1)
		if S29T.GetRowCount() > 0 Then
			IrsaliyeNo = S29T.EVRAKNO
			TedarikciIrsNo = S29T.XIRSALIYENO
			TedarikciCHKodu = S29T.CARIHESAPCODE
		Else
			Set S68T = Doc.Select1("dde33", "STOK68T", QStr, "", -1)
			if S68T.GetRowCount() > 0 Then
				IrsaliyeNo = S68T.EVRAKNO
				TedarikciIrsNo = Doc.GetDBFieldValue("STOK68E", "XIRSALIYENO", S68T.EVRAKNO, "")
				TedarikciCHKodu = S68T.CHCODE
			End if
		End if
		if trim(TedarikciIrsNo) = trim(MTable.XIRSALIYENO) or trim(MTable.XIRSALIYENO) = "" Then
			satirNo = MTableT.AddRow
			MTableT.SetCurrentRow(satirNo)
			For each alan in Alanlar
				alanDegeri = S01.GetFieldValue(Alan, i)
				if Alan <> "STOK_MIKTAR" Then
					MTableT.SetFieldValue Alan, satirNo, alanDegeri
				End if
			Next
			MTableT.STOK_MIKTAR1 = S01.STOK_MIKTAR(i)
			MTableT.KOD_STOK00_AD = Doc.GetDBFieldValue("STOK00", "AD", MTableT.KOD, "")
			MTableT.ONAYLANAN_MIKTAR = S01.SF_MIKTAR(i)
			MTableT.IRSALIYENO = IrsaliyeNo
			MTableT.XIRSALIYENO = TedarikciIrsNo
			MTableT.CARIHESAPCODE = TedarikciCHKodu
			MTableT.CARIHESAPCODE_CARI00_AD = Doc.GetDBFieldValue("CARI00", "AD", TedarikciCHKodu, "")

Sub Fatura(TFile, i, FaturaNo)
	FaturaTarihi = STOK70E.TARIH(i)
	FaturaTarihi = TTarih(FaturaTarihi)
	FaturaNo = STOK70E.EVRAKNO(i)
	FaturaNo1 = Sag(FaturaNo, 6)
	FaturaTutari = Doc.NumFormat(Doc.GetDBFieldValue("STOK70E","TL_ISKONTOLUTOPLAM",FaturaNo,""),15,2)
	FaturaTutari = Sag(trim(FaturaTutari), 15)
	KDVTutari = Doc.NumFormat(Doc.GetDBFieldValue("STOK70E","TL_KDVLERTOPLAM",FaturaNo,""),13,2)
	KDVTutari = Sag(trim(KDVTutari), 13)

	IrsArray = Irsaliyeler(FaturaNo)
	For each Irsaliye in IrsArray
		IrsaliyeNo = IrsaliyeNo + Sag(Irsaliye,6)
	Next
	IrsaliyeNo = Sol(IrsaliyeNo, 120)

	rows = Doc.SelectEQ("STOK70T","EVRAKNO",FaturaNo)
	set STOK70 = Doc.GetTableObject("AU_STOK70T_SELECT")

	FaturaBaslik = "2FH" & FirmaKodu & FaturaTarihi & FaturaNo1 & FaturaTutari & KDVTutari & IrsaliyeNo & Space(11)
	TFile.WriteLine FaturaBaslik
	IrsaliyeNo = ""
	SiraNo = 0
	for j = 1 to STOK70.GetRecCount()
		gk10 = doc.getdbfieldvalue("STOK00", "GK_10", STOK70.KOD(j),"")
		if trim(gk10)<>"TX" then
			SiraNo = SiraNo + 1
			Call FaturaDetay(TFile,j, SiraNo, STOK70)
		end if
	Next
	FaturaSonKayit = "4FF" & FirmaKodu & FaturaTarihi & FaturaNo1 & Sag(STOK70.GetRecCount, 3) & Space(156)
	TFile.WriteLine FaturaSonKayit
End Sub
'-----------------------------------------------------------
Sub FaturaDetay(TFile,j, SiraNo, STOK70)
	FaturaTarihi = STOK70.TARIH(j)
	FaturaTarihi = TTarih(FaturaTarihi)
	FaturaNo = STOK70.EVRAKNO(j)
	FaturaNo = Sag(FaturaNo, 6)
	CHKodu = STOK70.CH_KODU(j)
	ResimNo = Doc.GetCHStokKodu(STOK70.KOD(j),CHKodu)
'	ResimNo = STOK70.KOD(j)
	ResimNo = Sag(ResimNo, 11)
	ResimMiktari = STOK70.SF_MIKTAR(j)
	ResimTutari = STOK70.TL_ISKONTOLU_TUTAR(j)
	ResimBirimFiyati = Doc.NumFormat(Doc.Div(ResimTutari,ResimMiktari),13,2)
	ResimKDVTutari = Doc.NumFormat(STOK70.TL_KDVTUT_EFEKTIF(j),13,2)
	ResimTutari = Doc.NumFormat(ResimTutari,15,2)
	ResimMiktari = Doc.NumFormat(ResimMiktari,11,2)

	ResimMiktari = Sag(ResimMiktari, 11)
	ResimTutari = Sag(ResimTutari, 15)
	ResimBirimFiyati = Sag(cstr(ResimBirimFiyati), 13)
	ResimKDVTutari = Sag(ResimKDVTutari, 13)
	SiraNo = Sag(SiraNo, 3)
	FaturaDetayi = "3DE" & FirmaKodu & FaturaTarihi & FaturaNo & ResimNo & ResimBirimFiyati & ResimMiktari & ResimTutari & _
                       ResimKDVTutari & SiraNo & String(93, " ")
                        
	TFile.WriteLine (FaturaDetayi)
End Sub
Function TTarih(Tarih)
	Tarih = doc.StrtoNorDate(Tarih)
	Yil = Doc.StrDate_Year(Tarih)
	Ay = Right("0" & Doc.StrDate_Month(Tarih), 2)
	Gun = Right("0" & Doc.StrDate_Day(Tarih), 2)
	TTarih = Yil & Ay & Gun
End Function
Function Sol(Yazi, KarakterSayisi)
	Sol = Left(Trim(Yazi) & String(KarakterSayisi, " "), KarakterSayisi)
End Function
'-------------------------------------------------------------------
Function Sag(Yazi, KarakterSayisi)
	Yazi = Replace(Yazi, ",", "")
	Sag = Right(String(KarakterSayisi, "0") & Trim(Yazi), KarakterSayisi)
End Function
'-------------------------------------------------------------------------------
Function Irsaliyeler(FaturaNo)
	dim IrsArray()
	Redim IrsArray(0)
	rows = Doc.SelectEQ("STOK70T","EVRAKNO",FaturaNo)
	Set STOK70 = Doc.GetTableObject("AU_STOK70T_SELECT")
	STOK70.GroupBy("IRSALIYENO")
	
	For i = 1 to STOK70.GetRecCount()
		gk10 = doc.getdbfieldvalue("STOK00", "GK_10", STOK70.KOD(i),"")
		if trim(gk10)<>"TX" then
			IrsArray(uBound(IrsArray)) = Sag(STOK70.IRSALIYENO(i),6)
			if i <> STOK70.GetRecCount then
				Redim Preserve IrsArray(uBound(IrsArray)+1)
			end if
		end if
	Next
	Irsaliyeler = IrsArray
End Function
