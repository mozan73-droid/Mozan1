Function CreateIfNotExistDir(vDr)
	Dim fs
	Dim newfolder
	CreateIfNotExistDir=False
	Set fs=CreateObject("Scripting.FileSystemObject")
	if Not fs.FolderExists(vDr) then
		Set newfolder=fs.CreateFolder(vDr)
		if Not IsObject(newfolder) then
			CreateIfNotExistDir=TRUE
			Doc.d7MsgBox vDr + " Dizin Oluþturulamiyor.","Freedom",MB_OK
			Exit Function
		end if
	end if
	set fs=nothing
End Function

Sub WriteCel(ObjSheet,RowNum, ColNum, pRowHeight, pColumnWidth, FontName, FontSize, FontBold, pWrapText,pHorizontalAlignment,pNumberFormat,pAciklama)
	With ObjSheet.Cells(RowNum, ColNum)
	if Trim(pNumberFormat)<>"" then
		.NumberFormat = pNumberFormat
	end if
	'        .FormulaR1C1 = pAciklama
	.Formula = pAciklama
	'        .RowHeight = pRowHeight '76.75
	.HorizontalAlignment = pHorizontalAlignment 'xlCenter
	.VerticalAlignment = xlCenter
	.WrapText = pWrapText 'True
	.Orientation = 0
	.AddIndent = False
	.IndentLevel = 0
	.ShrinkToFit = False
	.ReadingOrder = xlContext
	.MergeCells = False
	.Font.Name = FontName '"Arial"
	.Font.Size = FontSize '9
	.Font.Bold = FontBold 'True
	.ColumnWidth = pColumnWidth '3.71

Sub AlanlariDoldur(ByVal Doc1,ByVal ExcelSheet,ByVal Satir,ByVal KOD)
	Dim Tbl
	Set Tbl=Doc1.GetTableObject(TabloAdi)

	Dim AlanIsmi
	Dim AlanDegeri
	Dim Sutun
	For Sutun=2 To 100
		AlanIsmi=Trim(ExcelSheet.Cells(2, Sutun))
		AlanDegeri=ExcelSheet.Cells(Satir, Sutun)
		If AlanIsmi<>"" And AlanIsmi<>"KOD" Then
			Tbl.SetFieldValue AlanIsmi,0,AlanDegeri
		End If
	Next
End Sub

Sub ActivatedSub()
ACount=0
for i= 1 to LISTET.GetRecCount()
	RowNum=LISTET.KOD(i)
	if LISTET.AP10=0 then
   active=Doc.Load_Voucher(RowNum)
	STOK00.AP10=1
	active=Doc.Save_Voucher()
	aCount=aCount+1
end if
next
End Sub

   passive=Doc.Load_Voucher(RowNum)
	STOK00.AP10=0
	passive=Doc.Save_Voucher()
	pCount=pCount+1
end if
next
End Sub
