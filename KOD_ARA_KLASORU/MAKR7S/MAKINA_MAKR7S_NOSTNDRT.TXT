Sub SatirYaz(ASN)
	SevkAdresKodu = STOK60E.SHIPTOCODE
	if trim(SevkAdresKodu) = "FORD 1" then 'GÖLCÜK
		Birim = "EA"
	else
		Birim = "AD"
	end if
	CHKodu = STOK60E.CH_KODU
	for i = 1 to S60.GetRecCount()
		if trim(S60.KOD(i)) <> "" then
			Ambalaj = S60.FATURASIZMI(i)
			Kod = S60.KOD(i)
			if trim(Ambalaj) = "E" then
				StokKodu = Kod
			else
				StokKodu = trim(Doc.GetCHStokKodu(Kod,CHKodu))
			end if
			Miktar = S60.SF_MIKTAR(i)
'			Birim = S60.SF_SF_UNIT(i)
			PART_CUM_QTY = 0
			PART_PRIOR_CUM_QTY = 0 
			SiparisNo = S60.SIPNO(i)
			MusteriSipNo = Doc.GetDBFieldValue("STOK40E","CHSIPNO",SiparisNo,"")
			PART_PO_NO = trim(MusteriSipNo)
			if trim(Ambalaj) <> "E" then
				KontSayisi = S60.BRUTAGIRLIK(i)
				ParcaTuru = "PP"
				if trim(cstr(KontSayisi)) = "" or trim(cstr(KontSayisi)) = "0" then
'					kontParcaSayisi = Doc.GetDBFieldValue("STOK00","B_B_TECH_1",Kod,"")
					kontParcaSayisi = Doc.GetDBFieldValue("STOK00","B_MINPARTIBUYUKLUGU",Kod,"")
					kontParcaSayisi = Doc.Round(kontParcaSayisi,0)
					ParcaKontSayisi = Doc.Div(Miktar,kontParcaSayisi)
					ParcaKontSayisi = Doc.Ceil(ParcaKontSayisi)
				else
					ParcaKontSayisi = Doc.CDbl(KontSayisi)
					kontParcaSayisi = Doc.Div(Miktar,ParcaKontSayisi)
					kontParcaSayisi = Doc.Ceil(kontParcaSayisi)
				end if
				KonteynerTipi = "BOX90"
			else
				ParcaTuru = "CT"
				kontParcaSayisi = "0000"
				ParcaKontSayisi = "0000"
				KonteynerTipi = ""
			end if
			ASN.WriteLine(cTab(3) & "<PART>")        
			ASN.WriteLine(cTab(4) & "<NO_PART>" & trim(StokKodu) & "</NO_PART>")
			ASN.WriteLine(cTab(4) & "<PART_NO_TYPE>" & ParcaTuru & "</PART_NO_TYPE>")
			ASN.WriteLine(cTab(4) & "<PART_QUANTITY>" & Miktar & "</PART_QUANTITY>")
			ASN.WriteLine(cTab(4) & "<PART_UOM>" & Birim & "</PART_UOM>")
			ASN.WriteLine(cTab(4) & "<PART_CUM_QTY>" & PART_CUM_QTY & "</PART_CUM_QTY>")
			ASN.WriteLine(cTab(4) & "<PART_PRIOR_CUM_QTY>" & PART_PRIOR_CUM_QTY & "</PART_PRIOR_CUM_QTY>")
			ASN.WriteLine(cTab(4) & "<PART_PO_NO>" & PART_PO_NO & "</PART_PO_NO>")
			ASN.WriteLine(cTab(4) & "<CONTAINERS>")
			ASN.WriteLine(cTab(5) & "<CONTAINER>")
			ASN.WriteLine(cTab(6) & "<CONTAINER_TYPE_CODE>" & KonteynerTipi & "</CONTAINER_TYPE_CODE>")
			ASN.WriteLine(cTab(6) & "<PARTS_PER_CONT>" & KontParcaSayisi & "</PARTS_PER_CONT>")
			ASN.WriteLine(cTab(6) & "<CONTS_PER_PART>" & ParcaKontSayisi & "</CONTS_PER_PART>")
			ASN.WriteLine(cTab(5) & "</CONTAINER>")
			ASN.WriteLine(cTab(4) & "</CONTAINERS>")
			ASN.WriteLine(cTab(3) & "</PART>")        
		end if
	Next
End Sub

Function sendData(xmlDoc)
	Set xmlhttp = CreateObject("Microsoft.XMLHTTP")
	call xmlhttp.Open("POST", "https://fosn.ford.com.tr/asnposter/postxmlasn.asp", false)
	call xmlhttp.Send(xmlDoc)
	sendData = xmlhttp.responseText
End Function

Function AmbalajVarMi()
	AmbalajVarMi = False
	for i = 1 to STOK60T.GetRecCount()
		Ambalaj = STOK60T.FATURASIZMI(i)
		if trim(Ambalaj) = "E" then
			AmbalajVarMi = True
			Exit Function
		end if
	Next

Function NetAgirlikBul()
	ToplamNet = 0
	for i = 1 to S60.GetRecCount()
		StokKodu = S60.KOD(i)
		SevkMiktari = S60.SF_MIKTAR(i)

		KasaNetAgirligi = Doc.GetDBFieldValue("STOK00","B_B_TECH_3",StokKodu,"")
		KasaNetAgirligi = doc.Round(KasaNetAgirligi,0)
		KasaIciMiktar = Doc.GetDBFieldValue("STOK00","B_MINPARTIBUYUKLUGU",StokKodu,"")
		KasaAdedi = Doc.Ceil(Doc.Div(SevkMiktari,KasaIciMiktar))
		ToplamNet = ToplamNet + KasaAdedi * KasaNetAgirligi
	Next
	NetAgirlikBul = ToplamNet
End Function
Function BrutAgirlikBul()
	ToplamBrut = 0
	for i = 1 to S60.GetRecCount()
		StokKodu = S60.KOD(i)
		SevkMiktari = S60.SF_MIKTAR(i)

		KasaBrutAgirligi = Doc.GetDBFieldValue("STOK00","B_B_TECH_4",StokKodu,"")
		KasaBrutAgirligi = doc.Round(KasaBrutAgirligi,0)
		KasaIciMiktar = Doc.GetDBFieldValue("STOK00","B_MINPARTIBUYUKLUGU",StokKodu,"")
		KasaAdedi = Doc.Ceil(Doc.Div(SevkMiktari,KasaIciMiktar))
		ToplamBrut = ToplamBrut + KasaAdedi * KasaBrutAgirligi
	Next
	BrutAgirlikBul = ToplamBrut
End Function

Sub ToplamHesapla()
	For i = 1 to OTOSP1T.GetRecCount()
		GunlukToplam = 0
		HaftalikToplam = 0
		AylikToplam = 0
		For j = 1 to 40
			Ozellik = OTOSP1T.GetFieldValue("ATTRIB" & j,i)
			Deger = OTOSP1T.GetFieldValue("REPCOL" & j,i)
			if Ozellik = "G" then
				GunlukToplam = GunlukToplam + Deger
			Elseif Ozellik = "H" then
				HaftalikToplam = HaftalikToplam + Deger
			Elseif Ozellik = "A" then
				AylikToplam = AylikToplam + Deger			
			end if
		Next
		OTOSP1T.GUN_TOPLAMI(i) = GunlukToplam
		OTOSP1T.HAFTA_TOPLAMI(i) = HaftalikToplam
		OTOSP1T.AY_TOPLAMI(i) = AylikToplam
		OTOSP1T.GENEL_TOPLAM(i) = GunlukToplam + HaftalikToplam + AylikToplam 
	Next
End Sub

Function AlanBul(Tarih,Bastarih)
	If trim(BasTarih) = "" Then BasTarih = Doc.BuGun
	Baslangic = Doc.CDateToInt(BasTarih)
	TesTarih = Doc.CDateToInt(Tarih)
	Fark = TesTarih - Baslangic
	If Fark < 0 Then Fark = 0
	AlanBul = "REPCOL" & Fark + 1
End Function
Function AlanBulHafta(Tarih,Bastarih,BasCol)
	If trim(BasTarih) = "" Then BasTarih = Doc.BuGun
	Baslangic = Doc.CDateToInt(BasTarih)
	TesTarih = Doc.CDateToInt(Tarih)
	Fark = (TesTarih - Baslangic)/7
	If Fark < 0 or Fark > 9 Then
		AlanBulHafta = -1
		exit function
	end if
	AlanBulHafta = "REPCOL" & Fark + BasCol
End Function
Function AlanBulAy(Tarih,Bastarih,BasCol)
	If trim(BasTarih) = "" Then BasTarih = Doc.BuGun
	BaslangicAy = Cint(Mid(BasTarih,6,2))
	TesTarih = Cint(Mid(Tarih,6,2))
	Fark = TesTarih - BaslangicAy
	If Fark < 0 Then
		AlanBulAy = -1
		exit function
	end if
	AlanBulAy = "REPCOL" & Fark + BasCol
End Function
Function IlkTarih(Table)
	Table.Sort("RTESTARIH")
	For i = 1 To Table.GetRecCount()
		If Table.SF_MIKTAR(i) > 0 Then
			IlkTarih = Table.RTESTARIH(i)
			Exit Function
		End If
	Next
End Function

	Next
End Function

Sub ToplamHesapla()

		For j = 1 to 40

Sub ToplamHesapla()
	Dim i,GunlukToplam, HaftalikToplam, AylikToplam, j, Ozellik, Deger
	For i = 1 to OTOSP1T.GetRecCount()
		GunlukToplam = 0
		HaftalikToplam = 0
		AylikToplam = 0
		For j = 1 to 40
			Ozellik = OTOSP1T.GetFieldValue("ATTRIB" & j,i)
			Deger = OTOSP1T.GetFieldValue("REPCOL" & j,i)
			if Ozellik = "G" then
				GunlukToplam = GunlukToplam + Deger
			Elseif Ozellik = "H" then
				HaftalikToplam = HaftalikToplam + Deger
			Elseif Ozellik = "A" then
				AylikToplam = AylikToplam + Deger			
			end if
		Next
		OTOSP1T.GUN_TOPLAMI(i) = GunlukToplam
		OTOSP1T.HAFTA_TOPLAMI(i) = HaftalikToplam
		OTOSP1T.AY_TOPLAMI(i) = AylikToplam
		OTOSP1T.GENEL_TOPLAM(i) = GunlukToplam + HaftalikToplam + AylikToplam 
	Next
End Sub

Function FmtTarih(Tarih)
	dim Yil, Ay, Gun

	Gun = Right(Tarih,2)

		For Each Sc In Schedule
