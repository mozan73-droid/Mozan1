Sub Makro1()
	REP17RT.SafePack
	REP17JT.SafePack
	REP17KT.SafePack
	SQLStr = ""

	SelectStr = GetSelectStr()
	SQLStr = SQLStr & SelectStr

	JoinStr = GetJoinStr()
	if trim(JoinStr) <> "" Then
		SQLStr = SQLStr & vbcrlf & JoinStr
	End if

	WhereStr = GetWhereStr()
	if trim(WhereStr) <> "" Then
		SQLStr = SQLStr & vbcrlf & WhereStr
	End if

	REP17SE_B.SQLTEXT = SQLStr
End Sub
Function GetSelectStr()
	SQLStr = "SELECT "
	k = 1
	For i = 1 to REP17RT.GetRowCount()
		if IcmaldenOnceBaglantimi(REP17RT.FIELDNAME1(i)) and not EklenenAlanmi(REP17RT.FIELDNAME1(i)) Then
			if BaglantidanGelenAlanmi(REP17RT.FIELDNAME1(i)) Then
				Alan = GetOrijinalAlanAdi(REP17RT.FIELDNAME1(i)) & " " & trim(REP17RT.FIELDNAME1(i))
			Else
				Alan = trim(REP17SE.AU_TABLE) & "." & trim(REP17RT.FIELDNAME1(i))
			End if
			if Len(SQLStr) > k * 90 Then
				SQLStr = SQLStr & vbcrlf & Chr(9)
				k = k + 1
			End if
			SQLStr = SQLStr & trim(Alan) & ", "
		End if
	Next
	For i = 1 to REP17KT.GetRowCount()
		if not RaporIcerigindeVarmi(REP17KT.FIELDNAME1(i)) Then
			if IcmaldenOnceBaglantimi(REP17KT.FIELDNAME1(i)) and not EklenenAlanmi(REP17KT.FIELDNAME1(i)) Then
				if BaglantidanGelenAlanmi(REP17KT.FIELDNAME1(i)) Then
					Alan = GetOrijinalAlanAdi(REP17KT.FIELDNAME1(i)) & " " & trim(REP17KT.FIELDNAME1(i))
				Else
					Alan = trim(REP17SE.AU_TABLE) & "." & trim(REP17KT.FIELDNAME1(i))
				End if
				if Len(SQLStr) > k * 90 Then
					SQLStr = SQLStr & vbcrlf & Chr(9)
					k = k + 1
				End if
				SQLStr = SQLStr & trim(Alan) & ", "
			End if
		End if
	Next

	if Right(trim(SQLStr), 1) = "," Then
		SQLStr = Left(SQLStr, Len(SQLStr) - 2)
	End if
	
	GetSelectStr = SQLStr & vbcrlf & "FROM " & trim(REP17SE.AU_TABLE)
End Function
Function GetWhereStr()
	if REP17KT.GetRowCount = 0 Then
		GetWhereStr = ""
		Exit Function
	End if
	WhereStr = "WHERE 1=1 "
	For i = 1 to REP17KT.GetRowCount()
		if IcmaldenOnceBaglantimi(REP17KT.FIELDNAME1(i)) and not EklenenAlanmi(REP17KT.FIELDNAME1(i)) _
				and Doc.Is_EY1(REP17KT.BEFOREGROUPBY(i)) Then
			WStr = CreateWhereStr(REP17KT.FIELDNAME1(i), REP17KT.KRITERTIPI(i))
			if WStr <> "" Then
				WhereStr = WhereStr & vbcrlf & WStr
			End if
		End if
	Next
	GetWhereStr = WhereStr
End Function
Function CreateWhereStr(AlanAdi, KriterTipi)
	if not BaglantidanGelenAlanmi(AlanAdi) Then
		AlanAdi1 = trim(REP17SE.AU_TABLE) & "." & trim(AlanAdi)
	Else
'		AlanAdi1 = trim(AlanAdi)
		AlanAdi1 = GetOrijinalAlanAdi(AlanAdi)
	End if
	if trim(KriterTipi) = "><" Then
		CreateWhereStr = " AND " & AlanAdi1 & ">=###KRITERGIRIS.B_" & trim(Alanadi) & "###" & vbcrlf & _
					"###IF NOTEMPTY KRITERGIRIS.E_" & trim(Alanadi) & "###" & vbcrlf & _
					"AND " & AlanAdi1 & "<=###KRITERGIRIS.E_" & trim(AlanAdi) & "###" & vbcrlf & _
					"###ENDIF###"
	Elseif trim(KriterTipi) = ">=" Then
		CreateWhereStr = " AND " & AlanAdi1 & ">=###KRITERGIRIS.B_" & trim(Alanadi) & "###"
	Elseif trim(KriterTipi) = ">" Then
		CreateWhereStr = " AND " & AlanAdi1 & ">###KRITERGIRIS.B_" & trim(Alanadi) & "###"
	Elseif trim(KriterTipi) = "<=" Then
		CreateWhereStr = "###IF NOTEMPTY KRITERGIRIS.E_" & trim(Alanadi) & "###" & vbcrlf & _
					"AND " & AlanAdi1 & "<=###KRITERGIRIS.E_" & trim(AlanAdi) & "###" & vbcrlf & _
					"###ENDIF###"
	Elseif trim(KriterTipi) = "<" Then
		CreateWhereStr = "###IF NOTEMPTY KRITERGIRIS.E_" & trim(Alanadi) & "###" & vbcrlf & _
					"AND " & AlanAdi1 & "<###KRITERGIRIS.E_" & trim(AlanAdi) & "###" & vbcrlf & _
					"###ENDIF###"
	Elseif trim(KriterTipi) = "=" Then
		CreateWhereStr = "###IF NOTEMPTY KRITERGIRIS.EQ_" & trim(Alanadi) & "###" & vbcrlf & _
					"AND " & AlanAdi1 & "=###KRITERGIRIS.EQ_" & trim(AlanAdi) & "###" & vbcrlf & _
					"###ENDIF###"
	Elseif trim(KriterTipi) = "<>" or trim(KriterTipi) = "!=" Then
		CreateWhereStr = "###IF NOTEMPTY KRITERGIRIS.NE_" & trim(Alanadi) & "###" & vbcrlf & _
					"AND " & AlanAdi1 & "<>###KRITERGIRIS.NE_" & trim(AlanAdi) & "###" & vbcrlf & _
					"###ENDIF###"
	Elseif trim(KriterTipi) = "BOÞ" Then
		CreateWhereStr = "AND " & AlanAdi1 & "=''"
	Elseif trim(KriterTipi) = "!BOÞ" Then
		CreateWhereStr = "AND " & AlanAdi1 & "<>''"
	Elseif Left(KriterTipi,1) = "S" Then
		CreateWhereStr = GetCLStr(KriterTipi, Alanadi, AlanAdi1)
	Elseif Left(KriterTipi,2) = "NS" Then
		CreateWhereStr = GetNSStr(KriterTipi, Alanadi, AlanAdi1)
	End if
End Function
Function GetJoinStr()
	GetJoinStr = ""
	if IcmaldenOnceJoinYok Then Exit Function

	JoinStr = ""
	For i = 1 to REP17JT.GetRowCount()
		if Doc.Is_EY1(REP17JT.BEFOREGROUPBY(i)) Then
			if JoinStr <> "" Then
				JoinStr = JoinStr & vbcrlf & "LEFT JOIN "
			Else
				JoinStr = "LEFT JOIN "
			End if
			JoinStr = JoinStr & trim(REP17JT.JOINEDTABLE(i))
			JoinStr = JoinStr & " ON "

			TabloAlan = trim(REP17JT.JOINEDTABLE(i)) & "." & trim(REP17JT.JOINEDFIELD(i))
			if BaglantidanGelenAlanmi(REP17JT.JOINEDFIELD(i)) Then
				TabloAlan = GetOrijinalAlanAdi(REP17JT.JOINEDFIELD(i))
			End if
			JoinStr = JoinStr & TabloAlan
			JoinStr = JoinStr & "="

			TabloAlan = Trim(REP17SE.AU_TABLE) & "." & trim(REP17JT.FIELDNAME1(i))
			if BaglantidanGelenAlanmi(REP17JT.FIELDNAME1(i)) Then
				TabloAlan = GetOrijinalAlanAdi(REP17JT.FIELDNAME1(i))
			End if
			JoinStr = JoinStr & TabloAlan
		End if
	Next
	GetJoinStr = JoinStr
End Function
Function IcmaldenOnceBaglantimi(AlanAdi)
	IcmaldenOnceBaglantimi = True
	For i = 1 to REP17JT.GetRowCount()
		Baglanti = trim(REP17JT.FIELDNAME1(i)) & "_" & trim(REP17JT.JOINEDTABLE(i))
		if len(Baglanti) > Len(AlanAdi) Then
			Exit Function
		Else
			if Left(AlanAdi, Len(Baglanti)) = Baglanti and Doc.Is_HN0(REP17JT.BEFOREGROUPBY(i)) Then
				IcmaldenOnceBaglantimi = False				
				Exit Function
			End if
		End if
	Next
End Function
Function BaglantidanGelenAlanmi(AlanAdi)
	BaglantidanGelenAlanmi = False
	For i = 1 to REP17JT.GetRowCount()
		Baglanti = trim(REP17JT.FIELDNAME1(i)) & "_" & trim(REP17JT.JOINEDTABLE(i))
		if len(Baglanti) > Len(AlanAdi) Then
			Exit Function
		Else
			if Left(AlanAdi, Len(Baglanti)) = Baglanti and Doc.Is_EY1(REP17JT.BEFOREGROUPBY(i)) Then
				BaglantidanGelenAlanmi = True
				Exit Function
			End if
		End if
	Next
End Function
Function GetOrijinalAlanAdi(AlanAdi)
	For i = 1 to REP17JT.GetRowCount()
		Baglanti = trim(REP17JT.FIELDNAME1(i)) & "_" & trim(REP17JT.JOINEDTABLE(i))
		if Left(AlanAdi, Len(Baglanti)) = Baglanti Then
			TabloAdi = trim(REP17JT.JOINEDTABLE(i))
			BaglantidanGelenAlan = Mid(AlanAdi, Len(Baglanti)+2)
			GetOrijinalAlanAdi = TabloAdi & "." & trim(BaglantidanGelenAlan)
		End if
	Next
End Function
Function EklenenAlanmi(AlanAdi)
	EklenenAlanmi = False
	For i = 1 to REP17UT.GetRowCount()
		if trim(REP17UT.FIELDNAME1(i)) = trim(AlanAdi) Then
			EklenenAlanmi = True
			Exit Function
		End if
	Next
End Function
Function IcmaldenOnceJoinYok()
	IcmaldenOnceJoinYok = True
	For i = 1 to REP17JT.GetRowCount()
		if Doc.Is_EY1(REP17JT.BEFOREGROUPBY(i)) Then
			IcmaldenOnceJoinYok = False				
			Exit Function
		End if
	Next
End Function
Function RaporIcerigindeVarmi(AlanAdi)
	RaporIcerigindeVarmi = False
	For i = 1 to REP17RT.GetRowCount()
		if trim(REP17RT.FIELDNAME1(i)) = trim(AlanAdi) Then
			RaporIcerigindeVarmi = True
			Exit Function
		End if
	Next
End Function
Function GetCLStr(KriterTipi, Alanadi, AlanAdi1)
	AlanSayisi = Doc.CDbl(Mid(KriterTipi, 2))
	OrStr = "###IF NOTEMPTY KRITERGIRIS.CL01_" & trim(Alanadi) & "###"  & vbcrlf & "AND (1=0 " & vbcrlf & "###ENDIF###" & vbcrlf & chr(9)
	For i = 1 to AlanSayisi
		OrStr = OrStr & "###IF NOTEMPTY KRITERGIRIS.CL" & Right("00" & i, 2) &"_" & trim(Alanadi) & "###" & vbcrlf & chr(9) & _
			"OR " & AlanAdi1 & "=###KRITERGIRIS.CL" & Right("00" & i, 2) &"_" & trim(AlanAdi) & "###" & vbcrlf & chr(9) & _
			"###ENDIF###" & vbcrlf
		if i <> AlanSayisi Then
			OrStr = OrStr  & chr(9)
		End if
	Next
	OrStr = OrStr & "###IF NOTEMPTY KRITERGIRIS.CL01_" & trim(Alanadi) & "###"  & vbcrlf & ")" & vbcrlf & "###ENDIF###" & vbcrlf
	GetCLStr = OrStr
End Function

	For i = 1 to AlanSayisi
		NSStr = NSStr & "###IF NOTEMPTY KRITERGIRIS.NCL" & Right("00" & i, 2) &"_" & trim(Alanadi) & "###" & vbcrlf & chr(9) & _
			"OR " & AlanAdi1 & "=###KRITERGIRIS.NCL" & Right("00" & i, 2) &"_" & trim(AlanAdi) & "###" & vbcrlf & chr(9) & _
			"###ENDIF###" & vbcrlf
		if i <> AlanSayisi Then
			NSStr = NSStr  & chr(9)
		End if
	Next
	NSStr = NSStr & "###IF NOTEMPTY KRITERGIRIS.NCL01_" & trim(Alanadi) & "###"  & vbcrlf & ")" & vbcrlf & "###ENDIF###" & vbcrlf
	GetNSStr = NSStr
End Function
