Sub Makro1()
	Dosya = OTOSP1E.FULLFILENAME1
	If Trim(Dosya) = "" Then Exit Sub
	Set fso = CreateObject("Scripting.FileSystemObject")
	Set TFile = fso.OpenTextFile(Dosya, 1, True)
	OTOSP1L.Empty

	While Not TFile.AtEndOfStream
		satirIndex = satirIndex + 1
		Doc.StatusBarmessage "Dosya Okunuyor : " & satirIndex
		satir = TFile.ReadLine
		If satirIndex > 1 Then
			ProgramArray = Split(satir, ";")
			If UBound(ProgramArray) > 0 Then
				ParcaKodu = Temizle(ProgramArray(4))
				TeslimTarihi = Temizle(ProgramArray(8))
				Miktar = Doc.CDbl(Temizle(ProgramArray(10)))
				Statu = Temizle(ProgramArray(3))
				Attrib = Temizle(ProgramArray(3))
				If (Trim(Statu) = "Onaylý" or Trim(Statu) = "Açýk") And Miktar <> 0 Then
					OTOSP1L.AddRow()
					satirNo = OTOSP1L.GetRecCount()
					OTOSP1L.SetCurrentRow (satirNo)
					OTOSP1L.D_ID(satirNo) = ParcaKodu
					OTOSP1L.SF_MIKTAR(satirNo) = Miktar
					OTOSP1L.RTESTARIH(satirNo) = TeslimTarihi
					OTOSP1L.REV_RTESTARIH(satirNo) = TeslimTarihi
					OTOSP1L.PROGRAM_NO(satirNo) = Attrib
					if Attrib = "Onaylý" Then
						OTOSP1L.ATTRIB(satirNo) = "H"
					Else
						OTOSP1L.ATTRIB(satirNo) = "G"
					End if
				End If
			End If
		End If
	Wend
	Call SonIrsaliyeBilgileriniGetir()
End Sub
Function Temizle(Yazi)
	Temizle = Replace(Trim(Yazi), """", "")
End Function
Sub SonIrsaliyeBilgileriniGetir()
	For i = 1 to OTOSP1L.GetRowCount()
		Doc.StatusBarMessage "Son Ýrsaliye Bilgileri Getiriliyor : " & i & " / " & OTOSP1L.GetRowCount()
		Set OSP1L = Doc.Select2("wwee", "OTOSP1L", "KOD='"+OTOSP1L.KOD(i)+"'&ASN_TARIHI<>''", "", -1)
		if OSP1L.GetRowCount() > 0 Then
			OTOSP1L.ASN_TARIHI(i) = OSP1L.ASN_TARIHI(1) 
			OTOSP1L.ASN_MIKTARI(i) = OSP1L.ASN_MIKTARI(1) 
		Else
			ASNArr = GetIrsInfo(OTOSP1L.KOD(i))
			OTOSP1L.ASN_TARIHI(i) = ASNArr(0)
			OTOSP1L.ASN_MIKTARI(i) = ASNArr(1)
		End if
	Next
	Doc.DeleteTableObject "wwee"
End Sub
Function GetIrsInfo(StokKodu)
	Dim IrsArray(1)
	IrsArray(0) = ""
	IrsArray(1) = 0
	if trim(StokKodu) = "" Then
		GetIrsInfo = IrsArray
		Exit Function
	End if
	Set S60T = Doc.Select1("wwee233", "STOK60T", "KOD='"+StokKodu+"'&CH_KODU='"+ OTOSP1E.CH_KODU+"'", "", -1)
	if S60T.GetRowCount = 0 Then
		GetIrsInfo = IrsArray
		Doc.DeleteTableObject "wwee233"
		Exit Function
	End if
	S60T.GroupBy("TARIH")

	satirSayisi = S60T.GetRowCount()
	IrsArray(0) = S60T.TARIH(satirSayisi)
	IrsArray(1) = S60T.SF_MIKTAR(satirSayisi)
	GetIrsInfo = IrsArray
	Doc.DeleteTableObject "wwee233"
End Function

Function GetTLFiyat(StokKodu, FOKodu, KVaryant, HVaryant1, HVaryant2, HVaryant3, NVaryant1, NVaryant2, Tarih)
	ret=Doc.Calc_Bom("M","M", StokKodu, 1, 1, "", FOKodu, KVaryant, HVaryant1, HVaryant2, HVaryant3, NVaryant1, NVaryant2) 
	Set Tbmrsys = Doc.GetTableObject("_WBMRSYS_SELECT")
'	CAll SendTableToExcel(Tbmrsys)
	if Tbmrsys.GetRowCount = 0 Then
			GetTLFiyat = GetSatinalmaTLFiyati(StokKodu, Tarih)
		Exit Function
	End if
	TLFiyat = 0
	For i = 1 to Tbmrsys.GetRowCount()
		if Tbmrsys.BOMREC_INPUTTYPE(i) = "H" Then
			if Tbmrsys.IALTSEVIYEBULUNDU(i) = 0 Then
				TLFiyat = TLFiyat + GetSatinalmaTLFiyati(Tbmrsys.BOMREC_KAYNAKCODE(i), Tarih) * Tbmrsys.KAYNAKMIKTART(i)
			Else
				TLFiyat = TLFiyat + GetFasonFiyat(Tbmrsys.BOMREC_KAYNAKCODE(i)) * Tbmrsys.KAYNAKMIKTART(i)
			End if
		Else
			if Left(Tbmrsys.BOMREC_KAYNAKCODE(i), 2) <> "FS" Then
				TLFiyat = TLFiyat + Tbmrsys.BOMREC_KAYNAK0(i) * TezgahSaatUcreti
			End if
		End if
	Next
	GetTLFiyat = TLFiyat
End Function
Function GetSatinalmaTLFiyati(StokKodu, Tarih)
	GetSatinalmaTLFiyati = 0
	Set S48T = Doc.Select1("ddrr", "STOK48T", "KULLANIM_YERI='2'&KOD='"+StokKodu+"'", "", -1)
	if S48T.GetRowCount() > 0 Then
		Kur = Doc.GetKurN(S48T.PRICEUNIT(1), Doc.DefaultPB, Tarih, "", -1)
		Birim = Doc.GetDBFieldValue("STOK00", "IUNIT", StokKodu, "")
		BirimDonusumu = Doc.Stok00_toUnit(StokKodu, Birim, S48T.SF_UNIT(1), -1)
		GetSatinalmaTLFiyati = S48T.PRICE(1) * Kur * BirimDonusumu
		Exit Function
	End if
	GetSatinalmaTLFiyati = GetOncekiSatinalmaFiyat(StokKodu)
	if GetSatinalmaTLFiyati = 0 Then
		ss = S48.AddRow
		S48.KOD(ss) = StokKodu
	End if
End Function
Function GetFasonFiyat(StokKodu)
	GetFasonFiyat = 0
	Set S68T = Doc_Old.Select1("sswweeqq", "STOK10A", "EVRAKTYPE='FG'&KOD='"+trim(StokKodu)+"'&TL_NONMIKTARTUTAR>0", "<TARIH", 1)
	if S68T.GetRowCount() > 0 Then
		GetFasonFiyat = Doc.Div(S68T.TL_NONMIKTARTUTAR(1), S68T.SF_MIKTAR(1))
	End if
End Function
Function GetOncekiSatinalmaFiyat(StokKodu)
	GetOncekiSatinalmaFiyat = 0
	Set S68T = Doc_Old.Select1("sswweeqq", "STOK10A", "EVRAKTYPE='SI'&KOD='"+trim(StokKodu)+"'&TL_TUTAR>0", "<TARIH", 1)
	if S68T.GetRowCount() > 0 Then
		GetOncekiSatinalmaFiyat = Doc.Div(S68T.TL_TUTAR(1)+S68T.TL_NONMIKTARTUTAR(1), S68T.SF_MIKTAR(1))
	End if
End Function
