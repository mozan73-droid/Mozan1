Sub Makro1()

	Set Doc1 = Doc.CreateDocumentNoUI("STOK00","")
	Set T_STOK00 = Doc1.GetTableObject("STOK00")
	Silinen = 0
	xStokKod= ""
	xFlag = 0
	Call TabloAlan()

	For i = 1 to LISTET.GetRecCount()
		SKOD = LISTET.KOD(i)
		xStokKod = Trim(LISTET.KOD(i))
		zSatir = i
		zFlag = 0

		Call NerelerdeKullanilmis()

		If zFlag <> 1 Then
		' -----------------------------------------------------------------
				nretcode = Doc1.Load_Voucher(SKOD)
	  		  If nretcode <> 0 Then
  		  		 MsgBox ("New Error = " & CStr(nretcode) & " - " & CStr(i))
     	  	 LISTET.AD(i) = "Kart Bulunamadý"
				End If

				If LISTET.AD(i) <> "Kart Bulunamadý" Then 'Kart Var ise
		     	nretcode = Doc1.Delete_Voucher()
	  		   	If nretcode <> 0 Then
						LISTET.AD(i) = "Silinemedi"
     			End If		
			   	If nretcode = 0 Then
						LISTET.AD(i) = "OK"
     			   Silinen = Silinen + 1
						 Doc.StatusBarMessage i & " / " & LISTET.GetRecCount() & " - " & Silinen
      			End If	
				End If 'Kart Var ise
		' -----------------------------------------------------------------
		End If
	Next
	msgbox Silinen & " adet stok kodu silindi"
	Silinen = 0
	i=0
End Sub

Sub NerelerdeKullanilmis()
	Dim BatchErrorMode
	Set BatchErrorMode=Doc.CreateSObject("BatchErrorMode") 

	If Trim(xStokKod)="" Then 
		msgbox zSatir&".nci Satýrdaki Stok Kodu Boþ, Ýþlem durduruluyor"
		Exit Sub
	End If

	xFlag = 0

	For j=0 To UBound(Grid, 1)
		NTable = Trim(Grid(j, 0))
		If Trim(NTable)<>"" Then
			NField = Trim(Grid(j, 1))
			xKontrol = Kullanilmismi(xStokKod, NTable, NField)
			xFlag = xFlag + xKontrol
		End If
	Next	
	If xFlag<>0 Then
		Exit Sub
	Else
		'''''	Call UrunAgaciniSil()
		'''''	Call StokKartiniSil()
	End If
End Sub

Function Kullanilmismi(vStokKod, vNTable, vNField)
	Dim zTable
	Dim BatchErrorMode
	Set BatchErrorMode=Doc.CreateSObject("BatchErrorMode") 

	If Trim(vStokKod)="" Then
		doc.d7msgbox zSatir&".nci Satýr Stok Kodu Boþ", "dinamo",MB_OK
		Exit Function
	End If

	If Trim(vNTable)="" Then
		doc.d7msgbox zSatir&".nci Satýr Tablo Adý Boþ", "dinamo",MB_OK
		Exit Function
	End If

	If Trim(vNField)="" Then
		doc.d7msgbox zSatir&".nci Satýr- " & vNTable & " " & "Tablonun Alan Adý Boþ", "dinamo",MB_OK
		Exit Function
	End If


		Set zTable = Doc.Select1("xTable",vNTable, vNField&"='"&vStokKod&"'" ,"", 1) 
		If zTable.GetRowCount>0 Then 
			Kullanilmismi = 1
			xK = "Kullanýmýþ"
			zFlag = 1
		Else
			Kullanilmismi = 0
			xK = "Kullanýlmamýþ"
			zFlag = 0
		End If

	Doc.StatusBarMessage zSatir&".nci Satýr- "&vStokKod&" / "&vNTable&" / "&vNField&" - " &xK

End Function
