		Birim = "EA"

		if trim(S60.KOD(i)) <> "" then
			Ambalaj = S60.FATURASIZMI(i)

'			Birim = S60.SF_SF_UNIT(i)
			PART_CUM_QTY = 0
			PART_PRIOR_CUM_QTY = 0 
			SiparisNo = S60.SIPNO(i)
			MusteriSipNo = Doc.GetDBFieldValue("STOK40E","CHSIPNO",SiparisNo,"")
			PART_PO_NO = trim(MusteriSipNo)
			if trim(Ambalaj) <> "E" then
				KontSayisi = S60.BRUTAGIRLIK(i)
				ParcaTuru = "PP"
				ParcaKodu = trim(StokKodu)
				if trim(cstr(KontSayisi)) = "" or trim(cstr(KontSayisi)) = "0" then
					kontParcaSayisi = Doc.GetDBFieldValue("STOK00","B_B_TECH_1",Kod,"")
					kontParcaSayisi = Doc.Round(kontParcaSayisi,0)
					ParcaKontSayisi = Doc.Div(Miktar,kontParcaSayisi)
					ParcaKontSayisi = Doc.Ceil(ParcaKontSayisi)
				else
					ParcaKontSayisi = Doc.CDbl(KontSayisi)
					kontParcaSayisi = Doc.Div(Miktar,ParcaKontSayisi)
					kontParcaSayisi = Doc.Ceil(kontParcaSayisi)

				ParcaKodu = " " & trim(StokKodu)
				kontParcaSayisi = "0000"
				ParcaKontSayisi = "0000"
				KonteynerTipi = ""
			end if

Function KodKontrol()
	for i = 1 to STOK60T.GetRecCount()
		StokKodu = STOK60T.KOD(i)
		if trim(StokKodu) <> "" then
			CHKodu = STOK60E.CH_KODU
			CHStokKodu = trim(Doc.GetCHStokKodu(StokKodu,CHKodu))
			Ambalaj = STOK60T.FATURASIZMI(i)
			if trim(CHStokKodu) = "" then
				Hata = Hata & StokKodu & vbcrlf
			end if
		end if
	Next
	if Hata <> "" then
		Hata = Hata & "Ford stok kodu karþýlýðý girilmemiþ."
	end if
	KodKontrol = Hata
End Function
Function SandikAdediKontrol()
	for i = 1 to STOK60T.GetRecCount()
		StokKodu = STOK60T.KOD(i)
		if trim(StokKodu) <> "" then
			kontParcaSayisi = Doc.GetDBFieldValue("STOK00","B_B_TECH_1",StokKodu,"")
			Ambalaj = STOK60T.FATURASIZMI(i)
			BrutAgirlik = STOK60T.BRUTAGIRLIK(i)
			if trim(Ambalaj) = "" and kontParcaSayisi = 0 and trim(BrutAgirlik) = "" then
				Hata = Hata & StokKodu & vbcrlf
			end if
		end if
	Next
	if Hata <> "" then
		Hata = Hata & "Konteyner parça adedi girilmemiþ."
	end if
	SandikAdediKontrol = Hata
End Function

Sub Makro2(ListeSatir)
	if trim(Doc.GetGecoValue("STMTYP","KOD","FR",d7c_ErrMsgNo)) = "" then

	end if

		exit sub
	end if
	if trim(Doc.GetGecoValue("STMMUS","KOD","HM",d7c_ErrMsgNo)) = "" then
		ret = GrupKoduYarat("STMMUS","HM","Hammadde Maliyeti")

		Tarih = STOK48T.KRI_TARIH_1(i)
		if trim(Tarih) = "" Then Tarih = STOK48E.REF_TARIH_1
'		If Tarih > "2006/06/01" Then Tarih = "2006/06/01"
		if trim(STOK48T.KOD(i))	<> "" then
			StokKodu = STOK48T.KOD(i)	
			TedarikciStatusu = Doc.GetCHIVal(StokKodu, STOK48T.REF_CHKODU(i), "", "STATU")
			KartTipi = Doc.GetDBFieldValue("STOK00", "KART_TIPI", StokKodu, "")
			if TedarikciStatusu = "T" And KartTipi = "S" Then
				Call StandartMaliyeteYaz(StokKodu, Tarih, i)
		  End if
		end if
	Next
End Sub
Sub StandartMaliyeteYaz(StokKodu, Tarih, i)
	if trim(Tarih) = "" Then Tarih = Doc.BuGun
	rows = Doc.SelectEQ("STDMH0E","KOD",StokKodu)
	if rows = 0 then
		ret = Doc1.New_Voucher()				
		STDMH0E.STDMALTYPE = "FR"
		STDMH0E.KOD = StokKodu
		STDMH0T.AddRow
		satirNo = STDMH0T.GetRecCount
	else
		set ST = Doc.GetTableObject("AU_STDMH0E_SELECT")
		For j = 1 to ST.GetRecCount()
			if trim(ST.STDMALTYPE(j)) = "FR" then
				EvrakNo = ST.EVRAKNO(j)
			end if
		Next
		if trim(EvrakNo) <> "" then
			ret = Doc1.Load_Voucher(EvrakNo)
			if ret <> 0 then
				k = Doc.d7MsgBox(trim(StokKodu) & ": Standart Maliyet tablosu yüklenemedi. Ýþlem kesilmiþtir.","dinamo",MB_OK)
				exit sub
			end if
			satirNo = satirBul(STDMH0T, Tarih)
			if satirNo = 0 then
				STDMH0T.AddRow
				satirNo = STDMH0T.GetRecCount
			End if
		else
			ret = Doc1.New_Voucher()				
			STDMH0E.STDMALTYPE = "FR"
			STDMH0E.KOD = StokKodu
			STDMH0T.AddRow
			satirNo = STDMH0T.GetRecCount
		end if
	end if
	Fiyat = STOK48T.PRICE(i)
	IslemBirimi = STOK48T.SF_UNIT(i)
	TemelBirim = Doc.GetDBFieldValue("STOK00", "IUNIT", STOK48T.KOD(i), "")
	TemelFiyat = Doc.Stok00_toUnit(STOK48T.KOD(i), TemelBirim, IslemBirimi, d7c_ErrMsgNo) * Fiyat
	STDMH0T.VALIDAFTERTARIH(satirNo) = Doc.StrToNorDate(Tarih)
	STDMH0T.UNSUR(satirNo) = "HM"
	STDMH0T.TUTAR(satirNo) = TemelFiyat
	STDMH0T.PARABIRIMI(satirNo) = STOK48T.PRICEUNIT(i)
	ret = Doc1.Save_Voucher()
	if ret <> 0 then
		k = Doc.d7MsgBox(trim(StokKodu) & ": Standart Maliyet tablosu kaydedilemedi. Ýþlem kesilmiþtir.","dinamo",MB_OK)
		ret = Doc1.LoadEmpty_Voucher()
		exit sub
	end if
	ret = Doc1.LoadEmpty_Voucher()
End Sub
Function GrupKoduYarat(GrupKoduTipi,GrupKodu,GrupAciklamasi)

	else
		GrupKoduYarat = 0
	end if
	ret = Doc1.LoadEmpty_Voucher()
End Function

		End if

			StokArray = Split(ElementArray(3), ":")
			StokKodu = StokArray(0)
		End If
		If Trim(IlkElement) = "PAC" Then
			AmbalajArray = Split(Temizle(ElementArray(3)), ":")
			AmbalajKodu = AmbalajArray(0)
			AmbalajAdet = AmbalajArray(2)
		End If

		If Trim(IlkElement) = "SCC" Then
			if uBound(ElementArray) = 3 Then
				Attrib = ElementArray(3)
				if Attrib = "F" Then
					sipAttr = "A"
				Else
					sipAttr = "H"
				End if
			End if
		End If

		If Trim(Element) = "NAD" Then
				AttrArray = Split(ElementArray(2), ":")
				if ElementArray(1) = "MI" Then
					AraSevk= AttrArray(0)
				End if
				if ElementArray(1) = "ST" Then
					MShipTo = AttrArray(0)
				End if
				if ElementArray(1) = "SU" Then
					MSuplier = AttrArray(0)
				End if
		End If

		If Trim(IlkElement) = "DTM" Then
			AttrArray = Split(ElementArray(1), ":")
			If AttrArray(0) = "137" Then
				SiparisTarihi = GetD7Tarih(AttrArray(1))
			ElseIf AttrArray(0) = "158" Then
				TeslimTarihi = GetD7Tarih(AttrArray(1))
			ElseIf AttrArray(0) = "159" Then
				HorizonEndTime = GetD7Tarih(AttrArray(1))
			'ElseIf AttrArray(0) = "10" Then
				'TeslimTarihi= GetD7Tarih(AttrArray(1))
			End If
		End If

		If Trim(IlkElement) = "RFF" Then
			AttrArray = Split(ElementArray(1), ":")
			If AttrArray(0) = "ALM" Then
				PSchNumber = AttrArray(1)
			End If
			If AttrArray(0) = "ON" Then
				SiparisNo = AttrArray(1)
			End If
		End If
		If Trim(IlkElement) = "LOC" Then
			Element2 = Temizle(ElementArray(1))
			If Element2 = "54" Then
				SevkAdresKodu = Temizle(ElementArray(2))
			End If
			If Element2 = "159" Then
				DockKodu = Temizle(ElementArray(2))
			End If
		End If

			AttrArray = Split(ElementArray(1), ":")
			If AttrArray(0) = "1" Then
				Miktar = AttrArray(1)
				If Miktar > -1 Then

					OTOSP1L.AddRow()

					OTOSP1L.LOKASYON(satirNo) = DockKodu
					OTOSP1L.ATTRIB(satirNo) = sipAttr
				End If
			End If
		End If

Function SaatEkle(Ilktarih, IlkZaman, EklenenZaman)
	if EklenenZaman = "" Then 
		SaatEkle = Replace(STOK60E.TARIH, "/", "") & Replace(IlkZaman, ":", "")
	Else
		IlkZamanArr = Split(IlkZaman, ":")
		
		EklenenZamanArr = Split(EklenenZaman, ":")
		BaslangicTarihi = DateSerial(Doc.Cdbl(Left(Ilktarih,4)), Doc.CDbl(Mid(IlkTarih, 6,2)), Doc.CDbl(Mid(IlkTarih, 9,2)))

		YeniSaat = TimeSerial(Doc.Cdbl(IlkZamanArr(0)) + Doc.Cdbl(EklenenZamanArr(0)), Doc.Cdbl(IlkZamanArr(1)) + Doc.Cdbl(EklenenZamanArr(1)), 0)
		YeniTarihSaat = BaslangicTarihi + YeniSaat
		SaatEkle = Year(YeniTarihSaat) & Right("00" & Month(YeniTarihSaat), 2) & Right("00" & Day(YeniTarihSaat), 2) & _
						Right("00" & Hour(YeniTarihSaat),2) & Right("00" & Minute(YeniTarihSaat),2)
	End if
End Function

Function Getd7Tarih2(Tarih)

	Getd7Tarih2 = Gun & "/" & Mid(Tarih, 3, 2) & "/20" & Left(Tarih, 2)

Function Getd7Tarih(Tarih, ByRef DateType)
	ReDim TarihArray(1)
	If Tarih = "000000" Then 'Son kayýt, bundan sonrakiler boþ
		'TarihArray(0) = ""
		'TarihArray(1) = "L"
		TarihArray(0) = Doc.BuGun
		TarihArray(1) = "G"


	ElseIf Tarih = "222222" Then 'ihtiyaç yok
		TarihArray(0) = ""
		TarihArray(1) = "E" 'bundan sonraki kayýtlar boþ
	ElseIf Tarih = "333333" Then 'BAKIYE
		TarihArray(0) = Doc.BuGun
		TarihArray(1) = "G"
	ElseIf Tarih = "444444" Then 'ACÝL
		TarihArray(0) = Doc.BuGun
		TarihArray(1) = "G"
	ElseIf Tarih = "555555" Then
		TarihArray(0) = ""
		TarihArray(1) = "_P"
		DateType = "P"
	Else
		If DateType = "N" Then
			Gun = Mid(Tarih, 5)
			TarihArray(0) = Gun & "/" & Mid(Tarih, 3, 2) & "/20" & Left(Tarih, 2)
			TarihArray(1) = "G"
		ElseIf DateType = "P" Then
			Yil = "20" & Left(Tarih, 2)
			DatePart1 = Mid(Tarih, 3, 2)
			DatePart2 = Mid(Tarih, 5, 2)
			If DatePart2 = "00" Then
				TarihArray(0) = Yil & "/" & DatePart1 & "/01"
				TarihArray(1) = "A"
			Else
				TarihArray(0) = GetHaftaninIlkGunu(Yil, DatePart1)
				'TarihArray(0) = Yil & DatePart2 & "01"
				TarihArray(1) = "H"
			End If
		End If
	End If
	Getd7Tarih = TarihArray
End Function

Function Translate(ByVal InText, xlatTable)

'
	Temp = ""

	Next
	Translate = Temp
End Function

Function HexToStr(ByVal HexStr)
	Temp = ""
	For i = 1 To Len(HexStr) \ 2
		Temp = Temp & Chr(HexToDec(Mid(HexStr, i * 2 - 1, 2)))
	Next
	HexToStr = Temp
End Function
Function HexToDec(strHex)
	Dim lngResult
	Dim intIndex
	Dim strDigit
	Dim intDigit
	Dim intValue

	lngResult = 0
	For intIndex = Len(strHex) To 1 Step -1
		strDigit = Mid(strHex, intIndex, 1)
		intDigit = InStr("0123456789ABCDEF", UCase(strDigit)) - 1
		If intDigit >= 0 Then
			intValue = intDigit * (16 ^ (Len(strHex) - intIndex))
			lngResult = lngResult + intValue
		Else
			lngResult = 0
			intIndex = 0 ' stop the loop
		End If
	Next

	HexToDec = lngResult
End Function

Function Getd7Tarih2(Tarih)
	Gun = Mid(Tarih, 5)
	Getd7Tarih2 = Gun & "/" & Mid(Tarih, 3, 2) & "/20" & Left(Tarih, 2)
End Function
Function Getd7Tarih(Tarih, ByRef DateType)
	ReDim TarihArray(1)
	If Tarih = "000000" Then 'Son kayýt, bundan sonrakiler boþ
		'TarihArray(0) = ""
		'TarihArray(1) = "L"
		TarihArray(0) = Doc.BuGun
		TarihArray(1) = "G"


	ElseIf Tarih = "222222" Then 'ihtiyaç yok
		TarihArray(0) = ""
		TarihArray(1) = "E" 'bundan sonraki kayýtlar boþ
	ElseIf Tarih = "333333" Then 'BAKIYE
		TarihArray(0) = Doc.BuGun
		TarihArray(1) = "G"
	ElseIf Tarih = "444444" Then 'ACÝL
		TarihArray(0) = Doc.BuGun
		TarihArray(1) = "G"
	ElseIf Tarih = "555555" Then
		TarihArray(0) = ""
		TarihArray(1) = "_P"
		DateType = "P"
	Else
		If DateType = "N" Then
			Gun = Mid(Tarih, 5)
			TarihArray(0) = Gun & "/" & Mid(Tarih, 3, 2) & "/20" & Left(Tarih, 2)
			TarihArray(1) = "G"
		ElseIf DateType = "P" Then
			Yil = "20" & Left(Tarih, 2)
			DatePart1 = Mid(Tarih, 3, 2)
			DatePart2 = Mid(Tarih, 5, 2)
			If DatePart2 = "00" Then
				TarihArray(0) = Yil & "/" & DatePart1 & "/01"
				TarihArray(1) = "A"
			Else
				TarihArray(0) = GetHaftaninIlkGunu(Yil, DatePart1)
				'TarihArray(0) = Yil & DatePart2 & "01"
				TarihArray(1) = "H"
			End If
		End If
	End If
	Getd7Tarih = TarihArray
End Function

Function GetSipArray(SipStr, TextLen, ByRef DateType)
	ReDim SipArray(2, 0)
	GetSipArray = SipArray
	Kalan = SipStr
	If Len(SipStr) = 0 Then Exit Function
	While Len(Kalan) > 0
		SipStr = Left(Kalan, TextLen)

		TarihArr = Getd7Tarih(Left(SipStr, 6), DateType)

		If TarihArr(0) <> "" Then
			SipArray(0, UBound(SipArray, 2)) = CDbl(Mid(SipStr, 7))
			SipArray(1, UBound(SipArray, 2)) = TarihArr(0)
			SipArray(2, UBound(SipArray, 2)) = TarihArr(1)
			ReDim Preserve SipArray(2, UBound(SipArray, 2) + 1)
		End If
		Kalan = Mid(Kalan, TextLen + 1)
	Wend
	If UBound(SipArray, 2) > 0 Then ReDim Preserve SipArray(2, UBound(SipArray, 2) - 1)
	GetSipArray = SipArray
End Function

Function Translate(ByVal InText, xlatTable)
'
' Uses a translation table to map InText from one character set to another.
'
	Temp = ""
	For i = 1 To Len(InText)
		SetLocale(1055)
		Temp = Temp & Mid(xlatTable, Asc(Mid(InText, i, 1)) + 1, 1)
	Next
	Translate = Temp
End Function

Function HexToStr(ByVal HexStr)
	Temp = ""
	For i = 1 To Len(HexStr) \ 2
		Temp = Temp & Chr(HexToDec(Mid(HexStr, i * 2 - 1, 2)))
	Next
	HexToStr = Temp
End Function
Function HexToDec(strHex)
	Dim lngResult
	Dim intIndex
	Dim strDigit
	Dim intDigit
	Dim intValue

	lngResult = 0
	For intIndex = Len(strHex) To 1 Step -1
		strDigit = Mid(strHex, intIndex, 1)
		intDigit = InStr("0123456789ABCDEF", UCase(strDigit)) - 1
		If intDigit >= 0 Then
			intValue = intDigit * (16 ^ (Len(strHex) - intIndex))
			lngResult = lngResult + intValue
		Else
			lngResult = 0
			intIndex = 0 ' stop the loop
		End If
	Next

	HexToDec = lngResult
End Function
Function GetHaftaninIlkGunu(Yil, Hafta)

	Set d7Date = Doc.CreateSObject("d7Date")

End Function

Function YeniMi(EdiArray, EdiIndex)
	YeniMi = False
	If EdiIndex = 0 Then Exit Function
	PrevElementArray = Split(EdiArray(EdiIndex - 1), "+")
	Element = Temizle(PrevElementArray(0))
	
	If Trim(Element) = "QTY" Then
		AttribArr = Split(PrevElementArray(1), ":")
		If AttribArr(0) = "131" Then
		Else
			YeniMi = False
		End If
	Else
		YeniMi = True
	End If
End Function

Function GetRotaKodu(EdiArray, EdiIndex)

		End If
	Next
End Function

End Function

Sub AddIfUnique(ByRef Array1, NewItem)
	Dim subArr
	subArr = Filter(Array1, NewItem, 1)
	If UBound(subArr) = -1 Then
		Array1(UBound(Array1)) = NewItem
		ReDim Preserve Array1(UBound(Array1) + 1)
	End If
End Sub

Sub EskiSeriNoQValKopyala(ByRef QValArray, EskiKod, EskiLotNo)

	Set Q02E = Doc.Select1("Q0101ss", "QVAL02E", "KOD='"+EskiKod+"'&LOTNUMBER='"+EskiLotNo+"'", "", -1)

	QArr = Doc.GetQValues_Foy(Foyno)

		AlanIndex = QArr(i,1)

			AlanDegeri = QArr(i,5)
		End if
		Call QValArrayAdd(QValArray, AlanAdi, AlanDegeri)
	Next
	Doc.DeleteTableObject "Q0101ss"
