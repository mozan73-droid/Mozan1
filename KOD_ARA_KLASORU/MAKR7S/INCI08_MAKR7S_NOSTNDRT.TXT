Function isFileExist(Byval fp,Byval fn)

		Exit Function
	End if
	dim fso,f,fc,f1
	Set fso = CreateObject("Scripting.FileSystemObject")

	Next

Function Fixfield(Byval value,length,justfy,filler)
	if len(value)>length Then 
		Fixfield = value
		Exit Function
	End if 
	do while len(value)<length
		if justfy= "R" or justfy= "Right" Then 
			value = Left(filler,1) & value 
		Elseif justfy= "L" or justfy= "Left" Then 
			value = value & Left(filler,1)
		End if
	Loop
	Fixfield = value
End Function

Function Fixfield(Byval value,length,justfy,filler)
	if len(value)>length Then 
		Fixfield = value
		Exit Function
	End if 
	do while len(value)<length
		if justfy= "R" or justfy= "Right" Then 
			value = Left(filler,1) & value 
		Elseif justfy= "L" or justfy= "Left" Then 
			value = value & Left(filler,1)
		End if
	Loop
	Fixfield = value
End Function

Function isFileExist(Byval fp,Byval fn)
	isFileExist=0
	if len(Trim(fn))=0 or len(Trim(fp))=0 Then 
		isFileExist=0
		Exit Function
	End if
	dim fso,f,fc,f1
	Set fso = CreateObject("Scripting.FileSystemObject")
	On Error Resume Next
	Set f = fso.GetFolder(fp)
	if err.number<>0 Then 
		isFileExist = 0
		Exit Function
	End If
	Set fc = f.files
	isFileExist = 0
	For Each f1 in fc
		if f1.Name = fn Then 
			isFileExist = 1
		End if
	Next
End Function

Function TransferCSVtoDinamo(fp,fn)
	dim FullFileName
	dim fso,f
	dim a_file(1000,4)
	dim a_row
	dim row
	dim i 
	FullFileName = fp & "\" & fn
	Set fso = CreateObject("Scripting.FileSystemObject")
	'On Error Resume Next
	Set f = fso.OpenTextFile(FullFileName,1,-2)
	if err.number<>0 Then 
		TransferCSVtoDinamo = -2
		Exit Function
	End if 
	i=0
	Do while f.AtEndOfStream<>True
		row = f.readline()
		if trim(row)="" Then 
			Exit Do
		End if 
		a_row = split(row,";")
		if ubound(a_row)<4 Then 
			TransferCSVtoDinamo = -1
			Exit Function
		End if
		a_file(i,0)=a_row(0)
		a_file(i,1)=a_row(1)
		a_file(i,2)=a_row(2)
		a_file(i,3)=a_row(3)
		a_file(i,4)=a_row(4)

	dim c

	Set TableT = doc.GetTableObject("CARI20T")

	do while Trim(a_file(i,2))<>""
		TableT.AddRow()
		TableT.SetCurrentRow(c)
		TableT.NITELIK(c) = "DK"
		TableT.ACIKLAMA(c) = a_file(i,2)
		Fiyat = a_file(i,3)
		'Fiyat = Replace(Fiyat,",",".")

  		if Csng(Fiyat)<0 Then 
			TableT.TL_ALACAKFIYAT(c) = (-1) * Csng(Fiyat)
		ElseIf Csng(Fiyat)>0 Then 
			TableT.TL_BORCFIYAT(c) =Csng(Fiyat) 
			TableT.MFORCE(c)="V"
			TableT.Kod(c) = "1020902"
			c=c+1
			TableT.AddRow()
			TableT.SetCurrentRow(c)
			TableT.MFORCE(c)="V"
			Aciklama = TableT.ACIKLAMA(c)

		End If
		i=i+1
		c=c+1
	Loop
	TransferCSVtoDinamo = 0
End Function

Function GetCHCodeFromCustID(byval id)

		GetCHCodeFromCustID = Table.KOD(1)
	ElseIf Table.GetRecCount>1 Then 
		GetCHCodeFromCustID = "Mükerrer Banka Kodu"
	Else
		GetCHCodeFromCustID = ""
	End if
	doc.deleteTableObject("ABC")
End Function

Function Fixfield(Byval value,length,justfy,filler)
	if len(value)>length Then 
		Fixfield = value
		Exit Function
	End if 
	do while len(value)<length
		if justfy= "R" or justfy= "Right" Then 
			value = Left(filler,1) & value 
		Elseif justfy= "L" or justfy= "Left" Then 
			value = value & Left(filler,1)
		End if
	Loop
	Fixfield = value
End Function

Function GetAccountID(ByVal xAciklama)
	Dim offset,Tmpid
	GetAccountID = ""
	xAciklama = Trim(xAciklama)
	offset = instr(xAciklama,":")
	if offset<=0 Then 
		Exit Function
	End if
	Tmpid=mid(xAciklama,offset+1,9)
	Tmpid = Trim(Tmpid)
	if isNumeric(Tmpid) Then 
		GetAccountID = Tmpid
	End if 
End Function

Function Fixfield(Byval value,length,justfy,filler)
	if len(value)>length Then 
		Fixfield = value
		Exit Function
	End if 
	do while len(value)<length
		if justfy= "R" or justfy= "Right" Then 
			value = Left(filler,1) & value 
		Elseif justfy= "L" or justfy= "Left" Then 
			value = value & Left(filler,1)
		End if
	Loop
	Fixfield = value
End Function

Sub ProvizyonAl_V(Doc1,EVRAKNO)
	Doc1.LoadEmpty_Voucher
	Dim nret	
	nret=Doc1.Load_Voucher(EVRAKNO)
	If nret<>0 Then Exit Sub
	If Trim(Doc1.GetLoadedVoucher)<>Trim(EVRAKNO) Then Exit Sub
	Dim T_CARI30E
	Set T_CARI30E=Doc1.GetTableObject("CARI30E")
	If T_CARI30E.REQ_TRANS<>"C" Then Exit Sub
	If Doc1.IsBlank(T_CARI30E.KKNO) Then Exit Sub
	Dim EODM_EN
	EODM_EN=d7FindOdemeTuru(Doc1,T_CARI30E.KKNO,T_CARI30E.ODEME_KODU)
	If Doc1.IsBlank(EODM_EN) Then Exit Sub
	Dim Res
	Dim BNK_OnayKodu
	Dim BNK_SonHataKodu
	Dim BNK_SonHataAciklamasi
	Res=d7ProvizyonAl(Doc1,EODM_EN,T_CARI30E.KKUNVAN,T_CARI30E.KKNO,T_CARI30E.KKGK,T_CARI30E.KKGTAY,T_CARI30E.KKGTYIL,T_CARI30E.TL_TUTAR,T_CARI30E.EVRAKNO,BNK_OnayKodu,BNK_SonHataKodu,BNK_SonHataAciklamasi)

	If Res="00" Then
		T_CARI30E.REQ_TRANS="O"	
		T_CARI30E.FN_RUN="1"
		T_CARI30E.FN_STATU="1"
		T_CARI30E.FN_TARIHI=Doc1.Bugun()
		T_CARI30E.FN_SAATI=Doc1.Saat(8)
		T_CARI30E.FN_SONHATAKODU=Res
		T_CARI30E.FN_SONHATAKODU_BNK=""
		T_CARI30E.FN_SONHATATXT_BNK=""
		T_CARI30E.FN_ONAYKODU=Provizyon_Kodu
	Else
		T_CARI30E.FN_RUN="1"
		T_CARI30E.FN_STATU="0"
		T_CARI30E.FN_SONHATAKODU=Res
		T_CARI30E.FN_SONHATAKODU_BNK=Provizyon_SONHATAKODU_BNK
		T_CARI30E.FN_SONHATATXT_BNK=Provizyon_SONHATATXT_BNK
	End If
	nret=Doc1.Save_Voucher
	If nret>0 Then
			Doc1.d7MsgBox Doc1.GetDocType&" "&Trim(CARI30E.EVRAKNO)&" Nolu Evrak Kaydýnda Hata","",-1
	End If
	Doc1.LoadEmpty_Voucher
End Sub

Function StrToArray(Yazi, Ayirac)
  Dim TmpArray()
  Dim AyiracSayisi
  Dim TmpText
  Yazi = Trim(Yazi)
  Yazi = Temizle(Yazi, Ayirac)
  AyiracPos = InStr(1, Yazi, Ayirac)
  If AyiracPos = 0 Then
    ReDim TmpArray(0)
    TmpArray(0) = Yazi
  Else
    pos = 0
    oldpos = 0
    AyiracSayisi = CharNum(Yazi, Ayirac)
    ReDim TmpArray(AyiracSayisi)
    For i = 0 To 10000
      pos = pos + 1
      If pos > Len(Yazi) And oldpos > Len(Yazi) Then Exit For
      pos = InStr(pos, Yazi, Ayirac)
      If pos > 0 Then
        TmpText = Mid(Yazi, oldpos + 1, pos - oldpos - 1)
        TmpArray(i) = TmpText
        oldpos = pos
      Else
        TmpText = Mid(Yazi, oldpos + 1, Len(Yazi) - oldpos)
        TmpArray(i) = TmpText
        Exit For
      End If
    Next
  End If
  StrToArray = TmpArray
End Function
Function Temizle(Yazi, Ayirac)
  Yazi = Trim(Yazi)
  While Right(Yazi, 1) = Ayirac
    Yazi = trim(Left(Yazi, Len(Yazi) - 1))
  Wend
  While Left(Yazi, 1) = Ayirac
    Yazi = trim(right(Yazi, Len(Yazi) - 1))
  Wend
  For i = 1 To 10
    strT = String(2, Ayirac)
    Yazi = Replace(Yazi, strT, Ayirac)
  Next
  Temizle = Yazi
End Function
Function CharNum(Yazi, FindChar)
  If InStr(1, Yazi, FindChar) = 0 Then
    CharNum = 0
  Else
    pos = 0
    For i = 1 To 10000
      pos = pos + 1
      If pos > Len(Yazi) Then Exit For
      pos = InStr(pos, Yazi, FindChar)
      If pos > 0 Then
        CharNum = CharNum + 1
      Else
        Exit For
      End If
    Next
  End If
End Function

Function StrToArray(Yazi, Ayirac)
  Dim TmpArray()
  Dim AyiracSayisi
  Dim TmpText
  Yazi = Trim(Yazi)
  Yazi = Temizle(Yazi, Ayirac)
  AyiracPos = InStr(1, Yazi, Ayirac)
  If AyiracPos = 0 Then
    ReDim TmpArray(0)
    TmpArray(0) = Yazi
  Else
    pos = 0
    oldpos = 0
    AyiracSayisi = CharNum(Yazi, Ayirac)
    ReDim TmpArray(AyiracSayisi)
    For i = 0 To 10000
      pos = pos + 1
      If pos > Len(Yazi) And oldpos > Len(Yazi) Then Exit For
      pos = InStr(pos, Yazi, Ayirac)
      If pos > 0 Then
        TmpText = Mid(Yazi, oldpos + 1, pos - oldpos - 1)
        TmpArray(i) = TmpText
        oldpos = pos
      Else
        TmpText = Mid(Yazi, oldpos + 1, Len(Yazi) - oldpos)
        TmpArray(i) = TmpText
        Exit For
      End If
    Next
  End If
  StrToArray = TmpArray
End Function
Function Temizle(Yazi, Ayirac)
  Yazi = Trim(Yazi)
  While Right(Yazi, 1) = Ayirac
    Yazi = trim(Left(Yazi, Len(Yazi) - 1))
  Wend
  While Left(Yazi, 1) = Ayirac
    Yazi = trim(right(Yazi, Len(Yazi) - 1))
  Wend
  For i = 1 To 10
    strT = String(2, Ayirac)
    Yazi = Replace(Yazi, strT, Ayirac)
  Next
  Temizle = Yazi
End Function
Function CharNum(Yazi, FindChar)
  If InStr(1, Yazi, FindChar) = 0 Then
    CharNum = 0
  Else
    pos = 0
    For i = 1 To 10000
      pos = pos + 1
      If pos > Len(Yazi) Then Exit For
      pos = InStr(pos, Yazi, FindChar)
      If pos > 0 Then
        CharNum = CharNum + 1
      Else
        Exit For
      End If
    Next
  End If
End Function

Function StrToArray(Yazi, Ayirac)
  Dim TmpArray()
  Dim AyiracSayisi
  Dim TmpText
  Yazi = Trim(Yazi)
  Yazi = Temizle(Yazi, Ayirac)
  AyiracPos = InStr(1, Yazi, Ayirac)
  If AyiracPos = 0 Then
    ReDim TmpArray(0)
    TmpArray(0) = Yazi
  Else
    pos = 0
    oldpos = 0
    AyiracSayisi = CharNum(Yazi, Ayirac)
    ReDim TmpArray(AyiracSayisi)
    For i = 0 To 10000
      pos = pos + 1
      If pos > Len(Yazi) And oldpos > Len(Yazi) Then Exit For
      pos = InStr(pos, Yazi, Ayirac)
      If pos > 0 Then
        TmpText = Mid(Yazi, oldpos + 1, pos - oldpos - 1)
        TmpArray(i) = TmpText
        oldpos = pos
      Else
        TmpText = Mid(Yazi, oldpos + 1, Len(Yazi) - oldpos)
        TmpArray(i) = TmpText
        Exit For
      End If
    Next
  End If
  StrToArray = TmpArray
End Function
Function Temizle(Yazi, Ayirac)
  Yazi = Trim(Yazi)
  While Right(Yazi, 1) = Ayirac
    Yazi = trim(Left(Yazi, Len(Yazi) - 1))
  Wend
  While Left(Yazi, 1) = Ayirac
    Yazi = trim(right(Yazi, Len(Yazi) - 1))
  Wend
  For i = 1 To 10
    strT = String(2, Ayirac)
    Yazi = Replace(Yazi, strT, Ayirac)
  Next
  Temizle = Yazi
End Function
Function CharNum(Yazi, FindChar)
  If InStr(1, Yazi, FindChar) = 0 Then
    CharNum = 0
  Else
    pos = 0
    For i = 1 To 10000
      pos = pos + 1
      If pos > Len(Yazi) Then Exit For
      pos = InStr(pos, Yazi, FindChar)
      If pos > 0 Then
        CharNum = CharNum + 1
      Else
        Exit For
      End If
    Next
  End If
End Function

Function StrToArray(Yazi, Ayirac)
  Dim TmpArray()
  Dim AyiracSayisi
  Dim TmpText
  Yazi = Trim(Yazi)
  Yazi = Temizle(Yazi, Ayirac)
  AyiracPos = InStr(1, Yazi, Ayirac)
  If AyiracPos = 0 Then
    ReDim TmpArray(0)
    TmpArray(0) = Yazi
  Else
    pos = 0
    oldpos = 0
    AyiracSayisi = CharNum(Yazi, Ayirac)
    ReDim TmpArray(AyiracSayisi)
    For i = 0 To 10000
      pos = pos + 1
      If pos > Len(Yazi) And oldpos > Len(Yazi) Then Exit For
      pos = InStr(pos, Yazi, Ayirac)
      If pos > 0 Then
        TmpText = Mid(Yazi, oldpos + 1, pos - oldpos - 1)
        TmpArray(i) = TmpText
        oldpos = pos
      Else
        TmpText = Mid(Yazi, oldpos + 1, Len(Yazi) - oldpos)
        TmpArray(i) = TmpText
        Exit For
      End If
    Next
  End If
  StrToArray = TmpArray
End Function
Function Temizle(Yazi, Ayirac)
  Yazi = Trim(Yazi)
  While Right(Yazi, 1) = Ayirac
    Yazi = trim(Left(Yazi, Len(Yazi) - 1))
  Wend
  While Left(Yazi, 1) = Ayirac
    Yazi = trim(right(Yazi, Len(Yazi) - 1))
  Wend
  For i = 1 To 10
    strT = String(2, Ayirac)
    Yazi = Replace(Yazi, strT, Ayirac)
  Next
  Temizle = Yazi
End Function
Function CharNum(Yazi, FindChar)
  If InStr(1, Yazi, FindChar) = 0 Then
    CharNum = 0
  Else
    pos = 0
    For i = 1 To 10000
      pos = pos + 1
      If pos > Len(Yazi) Then Exit For
      pos = InStr(pos, Yazi, FindChar)
      If pos > 0 Then
        CharNum = CharNum + 1
      Else
        Exit For
      End If
    Next
  End If
End Function

Function TransferCSVtoDinamo(fp,fn)

	dim fso,f
	dim a_file(1000,4)
	dim a_row
	dim row
	dim i 
	FullFileName = fp & "\" & fn
	Set fso = CreateObject("Scripting.FileSystemObject")
	'On Error Resume Next
	Set f = fso.OpenTextFile(FullFileName,1,-2)
	if err.number<>0 Then 
		TransferCSVtoDinamo = -2
		Exit Function
	End if 
	i=0
	Do while f.AtEndOfStream<>True
		row = f.readline()
		if trim(row)="" Then 
			Exit Do
		End if 
		a_row = split(row,";")
		if ubound(a_row)<4 Then 
			TransferCSVtoDinamo = -1
			Exit Function
		End if
		a_file(i,0)=a_row(0)
		a_file(i,1)=a_row(1)
		a_file(i,2)=a_row(2)
		a_file(i,3)=a_row(3)
		a_file(i,4)=a_row(4)
		i = i + 1
	Loop
	i=1
	dim TableT,Fiyat
	dim c 
	c=1
	Set TableT = doc.GetTableObject("CARI20T")
	TableT.Empty()
	do while Trim(a_file(i,2))<>""
		TableT.AddRow()
		TableT.SetCurrentRow(c)
		TableT.NITELIK(c) = "DK"
		TableT.ACIKLAMA(c) = a_file(i,2)
		Fiyat = Replace(a_file(i,3),".","")
		Fiyat = Replace(Fiyat,",",".")

  		if Csng(Fiyat)<0 Then 
			TableT.TL_ALACAKFIYAT(c) = (-1) * Csng(Fiyat)
		ElseIf Csng(Fiyat)>0 Then 
			TableT.TL_BORCFIYAT(c) =Csng(Fiyat) 
			TableT.MFORCE(c)="V"
			TableT.Kod(c) = "1020304"
			c=c+1
			TableT.AddRow()
			TableT.SetCurrentRow(c)
			TableT.MFORCE(c)="V"
			Aciklama = TableT.ACIKLAMA(c)
			CHBankID = GetAccountID(Aciklama)
			Chkodu = GetCHCodeFromCustID(CHBankID)
			TableT.Kod(c) = Chkodu
			TableT.NITELIK(c) = "DK"
			TableT.ACIKLAMA(c) = a_file(i,2)
			Fiyat = Replace(a_file(i,3),".","")
			Fiyat = Replace(Fiyat,",",".")
			TableT.TL_ALACAKFIYAT(c) = Csng(Fiyat)
		End If
		i=i+1
		c=c+1
	Loop
	TransferCSVtoDinamo = 0
End Function

Function isFileExist(Byval fp,Byval fn)
	isFileExist=0
	if len(Trim(fn))=0 or len(Trim(fp))=0 Then 
		isFileExist=0
		Exit Function
	End if
	dim fso,f,fc,f1

	On Error Resume Next
	Set f = fso.GetFolder(fp)
	if err.number<>0 Then 
		isFileExist = 0
		Exit Function
	End If
	Set fc = f.files
	isFileExist = 0
	For Each f1 in fc
		if f1.Name = fn Then 
			isFileExist = 1
		End if
	Next
End Function

End Function

Function Fixfield(Byval value,length,justfy,filler)
	if len(value)>length Then 
		Fixfield = value
		Exit Function
	End if 
	do while len(value)<length
		if justfy= "R" or justfy= "Right" Then 
			value = Left(filler,1) & value 
		Elseif justfy= "L" or justfy= "Left" Then 
			value = value & Left(filler,1)
		End if
	Loop
	Fixfield = value
End Function

	Tmpid=mid(xAciklama,offset+1)
	Tmpid = Trim(Tmpid)
	if isNumeric(Tmpid) Then 
		GetAccountID = Tmpid
	End if 
End Function

Function GetMuhCodeFromChCode(Chkodu)
	If Trim(Chkodu)="" Then 
		GetMuhCodeFromChCode = ""
		Exit Function
	End if 
	dim Table 
	dim Kristr
	Kristr = "KOD='" & Chkodu & "'"
	Set Table = doc.Select1("ABC","CARI00",kristr,"",-1)
	if table is nothing  Or table.GetRecCount<>1Then 
		GetMuhCodeFromChCode = ""
		doc.deleteTableObject("ABC")
		Exit Function
	End if 
	GetMuhCodeFromChCode = Table.MUHASEBECODE
	doc.deleteTableObject("ABC")
End Function

Function GetCHCodeFromCustID(byval id)
	Dim Table
	id = fixfield(id,30,"L"," ")
	dim OBJ 
	dim kristr
	kristr = "BANKA_MUSTERIKODU='" & id & "'&BANKA_KODU='0067            '&SUBE_KODU='068             '&HESAPNO='5553            '"
	Set Table= doc.Select1("ABC","BNK00C",kristr,"",-1)	
	if Table.GetRecCount=1 and Len(Trim(id))<>0 Then 
		GetCHCodeFromCustID = Table.KOD(1)
	ElseIf Table.GetRecCount>1 Then 
		GetCHCodeFromCustID = "Mükerrer Banka Kodu"
	Else
		GetCHCodeFromCustID = ""
	End if
	doc.deleteTableObject("ABC")
End Function

Function Fixfield(Byval value,length,justfy,filler)
	if len(value)>length Then 
		Fixfield = value
		Exit Function
	End if 
	do while len(value)<length
		if justfy= "R" or justfy= "Right" Then 
			value = Left(filler,1) & value 
		Elseif justfy= "L" or justfy= "Left" Then 
			value = value & Left(filler,1)
		End if
	Loop
	Fixfield = value
End Function

Function GetAccountID(ByVal xAciklama)
	Dim offset,Tmpid
	GetAccountID = ""
	xAciklama = Trim(xAciklama)
	offset = instr(xAciklama,":")
	if offset<=0 Then 
		Exit Function
	End if
	Tmpid=mid(xAciklama,offset+1)
	Tmpid = Trim(Tmpid)
	if isNumeric(Tmpid) Then 
		GetAccountID = Tmpid
	End if 
End Function
