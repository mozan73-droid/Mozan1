Function GetTPA(Tablo, PersonelKodu, TPAType)
	Dim TPAArray(1)
	TPAArray(0) = 0
	TPAArray(1) = 0
	Set TPA = Doc.Select2("WEQE", Tablo, "KOD='"+PersonelKodu+"'&ODEMETURU='"+TPAType+"'", "", -1)
	TPAArray(1) = TPA.GetRowCount()
	if TPA.GetRowCount() > 0 Then
		TPA.GroupBy("KOD")
		TPAArray(0) = TPA.OR_BORCFIYAT(1)
	End if
	GetTPA = TPAArray
	xret= doc.DeleteTableObject("WEQE")
End Function

	Dim Alan10

	For Sutun=2 To 100

Function GetTPA(Tablo, PersonelKodu, TPAType)
	Dim TPAArray(1)
	TPAArray(0) = 0
	TPAArray(1) = 0
	Set TPA = Doc.Select2("WEQE", Tablo, "KOD='"+PersonelKodu+"'&ODEMETURU='"+TPAType+"'", "", -1)
	TPAArray(1) = TPA.GetRowCount()
	if TPA.GetRowCount() > 0 Then
		TPA.GroupBy("KOD")
		TPAArray(0) = TPA.OR_BORCFIYAT(1)
	End if
	GetTPA = TPAArray
	xret= doc.DeleteTableObject("WEQE")
End Function

Function GetTPA(Tablo, PersonelKodu, TPAType)
	Dim TPAArray(1)
	TPAArray(0) = 0
	TPAArray(1) = 0
	Set TPA = Doc.Select2("WEQE", Tablo, "KOD='"+PersonelKodu+"'&ODEMETURU='"+TPAType+"'", "", -1)
	TPAArray(1) = TPA.GetRowCount()
	if TPA.GetRowCount() > 0 Then
		TPA.GroupBy("KOD")
		TPAArray(0) = TPA.OR_BORCFIYAT(1)
	End if
	GetTPA = TPAArray
	xret= doc.DeleteTableObject("WEQE")
End Function

Function GuvenlikStoguKontrolu(Mesaj, Table)
	Set MevcutTable = Doc.Select2G("fdsfsdfs", Table, "", "", "KOD", -1)
	KayitSayisi = MevcutTable.GetRecCount

	If KayitSayisi > 0 Then
		For i = 1 To KayitSayisi
			StokKodu = Trim(MevcutTable.KOD(i))
			GuvenlikStokDegeri = Doc.GetDBFieldValue("STOK00", "B_GUVENLIKSTOGU", StokKodu, "")
			
			Qstr = "SELECT KOD, SUM(STOK_MIKTAR) STOKMIKTARI FROM "
			Qstr = Qstr & " ("
			Qstr = Qstr & "SELECT KOD, STOK_MIKTAR FROM STOK01"
			Qstr = Qstr &	" WHERE 1=1 "
			Qstr = Qstr &	" AND KOD = '"& Doc.LeftJustify(StokKodu, 24) &"' "
			Qstr = Qstr & " )"
			Qstr = Qstr &	" GROUP BY KOD"

			Set SQL_Tbl = Doc.RunSQLQuery("xxxyyy", Qstr) 
			KayitSayisi = SQL_Tbl.GetRecCount
			
			If KayitSayisi > 0 Then
				GuncelStokMiktari = SQL_Tbl.STOKMIKTARI(1)
			End If

			If GuncelStokMiktari < GuvenlikStokDegeri Then
				BaslikIcerik =  "Güvenlik Stoðu Uyarýsý ; "
				MsgStr = MsgStr & vbcrlf & "[ " & StokKodu & " ] Stok Kodu için Güvenlik Stoðu Uyarýsý... " & vbcrlf & "Güvenlik Stok Miktarý : " & GuvenlikStokDegeri & vbcrlf & _
				"Mevcut Stok Durumu : " & GuncelStokMiktari & vbcrlf
			End If
		Next
	End If
	
	If Trim(MsgStr) <> "" Then
		Call MailAt(MsgStr, BaslikIcerik)
	End If
	
	Set MevcutTable = Nothing
End Function

Function GetTableHTML(Table, ByVal HTMLBaslik, TabloBaslikArray)
	HTMLBaslik = Replace(HTMLBaslik, vbcrlf, "<br>")
	HTMLBaslik = "<p><b><font size=""6"">" & HTMLBaslik & "</font></b></p>"
	FieldArr = Table.GetFields()

	HTMLStr = ""
	HTMLStr = "<html  lang='tr-TR'><body>" & HTMLBaslik
	HTMLStr = HTMLStr & "<table border=1 cellspacing=0 cellpadding=2>"
	if trim(TabloBaslikArray(0)) <> "" Then
		HTMLStr = HTMLStr & "<tr>"
		For each Baslik in TabloBaslikArray
			HTMLStr = HTMLStr & "<th bgcolor=#EEEEFF>" & Baslik & "</th>"
		Next
		HTMLStr = HTMLStr & "</tr>"
	End if

	For i=1 to Table.GetRecCount
		HTMLStr = HTMLStr & "<tr>"
		For j = 0 to uBound(FieldArr,1)
			if j <= uBound(TabloBaslikArray) Then
				AlanTuru = FieldArr(j, 2)
				AlanDegeri = Table.GetFieldValue(FieldArr(j,0), i)
				if InStr(1, FieldArr(j, 0), "_DBL") > 0 Then
					Align = "right"
				Else
					Align = "left"
				End if
				if AlanTuru = "DOUBLE" Then
					Align = "right"
				Elseif AlanTuru = "DATE" Then
					AlanDegeri = Doc.StrToNorDate(AlanDegeri)
				Elseif AlanTuru = "CHAR" Then
					AlanDegeri = rtrim(AlanDegeri)
					AlanDegeri = Replace(AlanDegeri, vbcrlf, "<br>")
					'Call ReplaceEntities(AlanDegeri)
				End if

				HTMLStr = HTMLStr & "<td align=" & Align & ">" & AlanDegeri & "</td>"
			End if
		Next
		HTMLStr = HTMLStr & "</tr>"
	Next
	HTMLStr = HTMLStr & "</table></body></html>"
	GetTableHTML = HTMLStr
End Function

Sub ShowTableHTMLMessage(Table, ByVal HTMLBaslik, TabloBaslikArray)
	HTMStr = GetTableHTML(Table, HTMLBaslik, TabloBaslikArray)
	Call ShowHTML(HTMStr)
End Sub

Sub ReplaceEntities(Str1)
	Str1 = Replace(Str1, " ", "&nbsp;")
	Str1 = Replace(Str1, "<", "&lt;")
	Str1 = Replace(Str1, ">", "&gt;")
	Str1 = Replace(Str1, "&", "&amp;")
	Str1 = Replace(Str1, """", "&quot;")
	Str1 = Replace(Str1, "'", "&apos;")
	Str1 = Replace(Str1, "€", "&euro;")
End Sub

Function GetTableHTML2(Table, ByVal HTMLBaslik, TabloBaslikArray, AlanArr)
	HTMLBaslik = Replace(HTMLBaslik, vbcrlf, "<br>")

	HTMLStr = ""
	HTMLStr = "<html  lang='tr-TR'><body>" & HTMLBaslik
	HTMLStr = HTMLStr & "<table border=1 cellspacing=0 cellpadding=2>"
	HTMLStr = HTMLStr & "<tr>"
	
	For each Baslik in TabloBaslikArray
		HTMLStr = HTMLStr & "<th bgcolor=#EEEEFF>" & Baslik & "</th>"
	Next

	HTMLStr = HTMLStr & "</tr>"

	For i=1 to Table.GetRecCount
		HTMLStr = HTMLStr & "<tr>"
		For j = 0 to uBound(AlanArr)
			AlanAdi = AlanArr(j)
			AlanTuru = GetAlanTuru(Table, AlanAdi)
			AlanDegeri = Table.GetFieldValue(AlanAdi, i)
			Align = "left"
			if AlanTuru = "DOUBLE" Then
				Align = "right"
			Elseif AlanTuru = "DATE" Then
				AlanDegeri = Doc.StrToNorDate(AlanDegeri)
			Elseif AlanTuru = "CHAR" Then
				AlanDegeri = trim(AlanDegeri)
				if AlanDegeri = "" Then
					AlanDegeri = "&nbsp;"
				Elseif TarihMi(AlanDegeri) Then
					AlanDegeri = Doc.StrToNorDate(AlanDegeri)
				End if
			End if

			HTMLStr = HTMLStr & "<td align=" & Align & ">" & AlanDegeri & "</td>"
		Next
		HTMLStr = HTMLStr & "</tr>"
	Next
	HTMLStr = HTMLStr & "</table></body></html>"
	GetTableHTML2 = HTMLStr
End Function

Function GetAlanTuru(Table, AlanAdi)
	Set AlanBilgisi=Table.GetFieldDef(AlanAdi)

	if Not IsObject(AlanBilgisi) Then 
		GetAlanTuru = "CHAR"
		Exit Function
	End if
	
	if AlanBilgisi.IsNumeric() Then
		GetAlanTuru = "DOUBLE"
	Elseif AlanBilgisi.IsDate() Then
		GetAlanTuru = "DATE"
	Else
		GetAlanTuru = "CHAR"
	End if	
End Function

Function TarihMi(ByVal Str1)
	TarihMi = False
	Set d7Date = Doc.CreateSObject("d7Date")
	d7Date.SetDate Str1
	if d7Date.IsValid Then TarihMi = True
End Function

		EvrakNo = Trim(StackTable.EVRAKNO(1))

	Qstr = Qstr & " FROM "
	Qstr = Qstr & " ( "
	Qstr = Qstr & " SELECT "
	Qstr = Qstr & " HM_DEPOSUNDAKI_STOKKODU, "
	Qstr = Qstr & " STOKADI, "

	Qstr = Qstr & " ON A.HM_DEPOSUNDAKI_STOKKODU = C.KOD "
