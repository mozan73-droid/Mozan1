Function GetD7Tarih(TarihStr)
	If Trim(TarihStr) = "" Then
		GetD7Tarih = ""
		Exit Function
	End If
	GetD7Tarih = Mid(TarihStr, 7, 2) & "/" & Mid(TarihStr, 5, 2) & "/" & Left(TarihStr, 4)
End Function

Sub SevkEdilenleriDus1()
	KalanMiktar = 0
	EskiStokKodu = ""
	For i = 1 to OTOSP1L.GetRowCount()
		if EskiStokKodu <> OTOSP1L.KOD(i) Then KalanMiktar = OTOSP1L.ASN_SONSEVKMIK(i)
		if KalanMiktar > 0 Then
			if KalanMiktar >= OTOSP1L.SF_MIKTAR(i) Then
				OTOSP1L.ASN_NETSIPARISMIK(i) = 0
				KalanMiktar = KalanMiktar - OTOSP1L.SF_MIKTAR(i)
			Else
				OTOSP1L.ASN_NETSIPARISMIK(i) = OTOSP1L.SF_MIKTAR(i) - KalanMiktar
				KalanMiktar = 0
			End if
		Else
			OTOSP1L.ASN_NETSIPARISMIK(i) = OTOSP1L.SF_MIKTAR(i)
		End if
		EskiStokKodu = OTOSP1L.KOD(i)
	Next
End Sub
Function SevkEdilenleriDus()
	SevkEdilenleriDus = ""
	Doc.CreatetableObject "WSSS=STOK40T", "TMPCOPY"
	Set S40T = Doc.GetTableObject("WSSS")
	KalanMiktar = 0
	EskiStokKodu = ""
	For i = 1 to OTOSP1L.GetRowCount()
		if OTOSP1L.ASN_SONSEVKMIK(i) < OTOSP1L.ASN_MIKTARI_KUMULE(i) Then
			S40T.AddRow
			satirno = S40T.GetRowCount()
			S40T.KOD(satirNo) = OTOSP1L.KOD(i)
		End if
		if EskiStokKodu <> OTOSP1L.KOD(i) Then KalanMiktar = OTOSP1L.ASN_SONSEVKMIK(i) - OTOSP1L.ASN_MIKTARI_KUMULE(i)
		if KalanMiktar > 0 Then
			if KalanMiktar >= OTOSP1L.SF_MIKTAR(i) Then
				OTOSP1L.ASN_NETSIPARISMIK(i) = 0
				KalanMiktar = KalanMiktar - OTOSP1L.SF_MIKTAR(i)
			Else
				OTOSP1L.ASN_NETSIPARISMIK(i) = OTOSP1L.SF_MIKTAR(i) - KalanMiktar
				KalanMiktar = 0
			End if
		Else
			OTOSP1L.ASN_NETSIPARISMIK(i) = OTOSP1L.SF_MIKTAR(i)
		End if
		EskiStokKodu = OTOSP1L.KOD(i)
	Next
	if S40T.GetRowCount() > 0 Then
		S40T.GroupBy("KOD")
		For i = 1 to S40T.GetRowCount()
			Hata = Hata & S40T.KOD(i) & vbcrlf
		Next
		SevkEdilenleriDus = Hata & "toplam sevk edilen miktar EDI 'de bildirilenden az"
	End if
End Function

Sub SevkEdilenleriDus1()
	KalanMiktar = 0
	EskiStokKodu = ""
	For i = 1 to OTOSP1L.GetRowCount()
		if EskiStokKodu <> OTOSP1L.KOD(i) Then KalanMiktar = OTOSP1L.ASN_SONSEVKMIK(i)
		if KalanMiktar > 0 Then
			if KalanMiktar >= OTOSP1L.SF_MIKTAR(i) Then
				OTOSP1L.ASN_NETSIPARISMIK(i) = 0
				KalanMiktar = KalanMiktar - OTOSP1L.SF_MIKTAR(i)
			Else
				OTOSP1L.ASN_NETSIPARISMIK(i) = OTOSP1L.SF_MIKTAR(i) - KalanMiktar
				KalanMiktar = 0
			End if
		Else
			OTOSP1L.ASN_NETSIPARISMIK(i) = OTOSP1L.SF_MIKTAR(i)
		End if
		EskiStokKodu = OTOSP1L.KOD(i)
	Next
End Sub

Function DahilMi(TezgahKodu)
	DahilMi = False
	if (trim(SFDC20_KRITERLER.KOD1) = "" or TezgahKodu >= SFDC20_KRITERLER.KOD1) _
			and (trim(SFDC20_KRITERLER.KOD2) = "" or TezgahKodu <= SFDC20_KRITERLER.KOD2) Then
		DahilMi1 = True
	Else
		DahilMi1 = False
	End if

	GrupKodu1 = Doc.GetDBFieldValue("IMLT00", "GK_1", TezgahKodu, "")
	if (trim(IMLT00_KRITERLER.GK_11) = "" or GrupKodu1 >= IMLT00_KRITERLER.GK_11) _
			and (trim(IMLT00_KRITERLER.GK_12) = "" or GrupKodu1 <= IMLT00_KRITERLER.GK_12) Then
		DahilMi2 = True
	Else
		DahilMi2 = False
	End if
	DahilMi = DahilMi1 and DahilMi2
End Function

Sub Makro2()
	if trim(Doc.GetGecoValue("STMTYP","KOD","FR",d7c_ErrMsgNo)) = "" then
		ret = GrupKoduYarat("STMTYP","FR","Farel Maliyet Endeksi")
	end if

		k = Doc.d7MsgBox("FR Maliyet endeksi yaratýlamadý. Ýþlem kesilmiþtir.","dinamo",MB_OK)
		exit sub
	end if
	if trim(Doc.GetGecoValue("STMMUS","KOD","HM",d7c_ErrMsgNo)) = "" then
		ret = GrupKoduYarat("STMMUS","HM","Hammadde Maliyeti")
	end if
	if ret <> 0 then
		k = Doc.d7MsgBox("HM masraf unsuru yaratýlamadý. Ýþlem kesilmiþtir.","dinamo",MB_OK)
		exit sub
	end if

	Set Doc1 = Doc.CreateDocumentNoUI("STDMH0","")
	Set STDMH0E = Doc1.GetTableObject("STDMH0E")
	Set STDMH0T = Doc1.GetTableObject("STDMH0T")

			StokKodu = STOK48T.KOD(i)

				ret = Doc1.New_Voucher()

				STDMH0E.KOD = StokKodu
				STDMH0T.AddRow
				satirNo = STDMH0T.GetRecCount

				set ST = Doc.GetTableObject("AU_STDMH0E_SELECT")
				For j = 1 to ST.GetRecCount()
					if trim(ST.STDMALTYPE(j)) = "FR" then
						EvrakNo = ST.EVRAKNO(j)

Function GetCalismaSaati(BaslangicTarih, BitisTarih)
	BaslangicYil = Doc.StrDate_Year(BaslangicTarih)
	BaslangicAy = Doc.StrDate_Month(BaslangicTarih)
	BaslangicGun = Doc.StrDate_Day(BaslangicTarih)
	DakikaAraligi = 15
	AralikSerial = TimeSerial(0,DakikaAraligi ,0)

	Aralik = (Doc.CDateToInt(BitisTarih) - Doc.CDateToInt(BaslangicTarih) + 1) * 24 * 60 / DakikaAraligi
	Baslangic = DateSerial(BaslangicYil, BaslangicAy, BaslangicGun) + TimeSerial(0, 0, 0)
	For i = 1 to Aralik
		BasTarih = Year(Baslangic) & "/" & Right("0" & Month(Baslangic),2) & "/" & Right("0" & Day(Baslangic),2)
		BasSaat = Right("0" & Hour(Baslangic),2) & ":" & Right("0" & Minute(Baslangic),2)
		
		nret = Doc.Takvim_Test_Minute("GENEL", BasTarih, BasSaat, DakikaAraligi)
		if nret = "1" Then
			CalismaSaat = CalismaSaat + DakikaAraligi
		End if
		Baslangic = Baslangic + AralikSerial
	Next
	GetCalismaSaati = Doc.Div(CalismaSaat, 60)
End Function
Function GetCalismaSaati_Tezgah(TezgahKodu, BaslangicTarih, BitisTarih)

	BaslangicAy = Doc.StrDate_Month(BaslangicTarih)
	BaslangicGun = Doc.StrDate_Day(BaslangicTarih)
	DakikaAraligi = 15
	AralikSerial = TimeSerial(0,DakikaAraligi ,0)

	Aralik = (Doc.CDateToInt(BitisTarih) - Doc.CDateToInt(BaslangicTarih) + 1) * 24 * 60 / DakikaAraligi
	Baslangic = DateSerial(BaslangicYil, BaslangicAy, BaslangicGun) + TimeSerial(0, 0, 0)
	For i = 1 to Aralik
		BasTarih = Year(Baslangic) & "/" & Right("0" & Month(Baslangic),2) & "/" & Right("0" & Day(Baslangic),2)
		BasSaat = Right("0" & Hour(Baslangic),2) & ":" & Right("0" & Minute(Baslangic),2)

	Next
	GetCalismaSaati_Tezgah = Doc.Div(CalismaSaat, 60)
End Function
Function GetToplamCalismaSaati(TezgahKodu, BaslangicTarihi, BitisTarihi)
	if trim(BaslangicTarihi) = "" or trim(BitisTarihi) = "" or BaslangicTarihi > BitisTarihi Then
		GetToplamCalismaSaati = 0
		Exit Function
	End if
	Set d7Date = Doc.CreateSObject("d7Date")
	d7Date.SetDate BaslangicTarihi

	ToplamSaat = 0
	Do
		DOW = d7Date.GetDOW
		if DOW = 6 Then
			ToplamSaat = ToplamSaat + Doc.GetDBFieldValue("IMLT00", "B_KAPASITE6", TezgahKodu, "")
		Elseif DOW = 7 Then
			ToplamSaat = ToplamSaat + Doc.GetDBFieldValue("IMLT00", "B_KAPASITE7", TezgahKodu, "")
		Else
			ToplamSaat = ToplamSaat + Doc.GetDBFieldValue("IMLT00", "B_KAPASITE", TezgahKodu, "")
		End if
		d7Date.Increment 1
	Loop Until d7Date.GetRevDate > BitisTarihi
	GetToplamCalismaSaati = ToplamSaat
End Function

Sub Makro1()
	Set Excel = CreateObject("Excel.Application")

	Sh1.Range("G9").Value = Doc.Round(FireArray(0), 2)
	Sh1.Range("J9").Value = Doc.Round(FireArray(1), 2)

	Sh1.Range("G12").Value = KalipDegisimArray(0,0)
	Sh1.Range("J12").Value = KalipDegisimArray(0,1)

	Sh1.Range("G13").Value = Doc.Round(KalipDegisimArray(1,0), 2)
	Sh1.Range("J13").Value = Doc.Round(KalipDegisimArray(1,1), 2)
	Sh1.Range("M13").Value = Doc.Round(KalipDegisimArray(1,2), 2)

	TakozArray = GetTakozFireKG()
	Sh1.Range("G14").Value = Doc.Round(TakozArray(0), 2)
	Sh1.Range("J14").Value = Doc.Round(TakozArray(1), 2)
	Sh1.Range("M14").Value = Doc.Round(TakozArray(2), 2)

	DonemBasi = SFDC20_KRITERLER.RECTARIH11
	DonemSonu = SFDC20_KRITERLER.RECTARIH12

	End if
	Excel.Visible = True
End Sub
Function GetKayipYuzde()
	Dim KayipArray(2)
	Dim SetupDahilDurus(2)

	DurusSuresi = GetToplamDurus()	
	AyarSetupSuresi = GetToplamSetup()

	YemekSuresi = GetToplamSurusSaat("Y")
	MakinaKapamaSure = GetToplamSurusSaat("PY")
	KalipDenemeSure = GetToplamSurusSaat("KD")

	VardiyaCalismaSaati = 8
	MakinaSayisi = GetMakinaSayisi()
	DonemBasi = SFDC20_KRITERLER.RECTARIH11
'	if trim(DonemBasi) = "" Then DonemSonu = Doc.BuGun
	DonemSonu = SFDC20_KRITERLER.RECTARIH12
	if trim(DonemSonu) = "" Then DonemSonu = Doc.BuGun
	CalisilanGun = Doc.CDateToInt(DonemSonu) - Doc.CDateToInt(DonemBasi) + 1

	ToplamCalismaSaati = VardiyaCalismaSaati * MakinaSayisi * CalisilanGun
	
	For i = 0 to 2
		SetupDahilDurus(i) = DurusSuresi(i) - MakinaKapamaSure(i) - YemekSuresi(i) - KalipDenemeSure(i) + AyarSetupSuresi(i)
		KayipArray(i) = Doc.Div(SetupDahilDurus(i), ToplamCalismaSaati) * 100
	Next
	GetKayipYuzde = KayipArray
End Function
Function GetToplamSurusSaat(DurmaSebebi)
	Dim DurusArray(2)
	DurusArray(0) = 0
	DurusArray(1) = 0
	DurusArray(2) = 0

	QStr = "DURMA_SEBEBI='" + DurmaSebebi + "'"
	Set S20T = Doc.Select2("QERW", "SFDC20T_DURUS", QStr, "", -1)
	if S20T.GetRowCount = 0 Then
		GetToplamSurusSaat = DurusArray
		Exit Function
	End if
	S20T.GroupBy("VARDIYA")
	for i = 1 to S20T.GetRowCount()
		if trim(S20T.VARDIYA(i)) = "31" Then
			DurusArray(0) = S20T.SURE(i)
		Elseif trim(S20T.VARDIYA(i)) = "32" Then
			DurusArray(1) = S20T.SURE(i)
		Elseif trim(S20T.VARDIYA(i)) = "33" Then
			DurusArray(2) = S20T.SURE(i)
		End if
	Next
	GetToplamSurusSaat = DurusArray
	Doc.DeleteTableObject "QERW"
End Function

Function GetKalipDegisim()

	KDArray(0, 2) = 0

	KDArray(1, 0) = 0 'baðlama süresi

	KDArray(1, 2) = 0

	Set S20T = Doc.Select2("QERW", "SFDC20T_AYAR", "", "", -1)

		GozSayisi = Doc.getDBFieldValue("STOK00", "B_B_TECH_2", S20T.JOBNO_MMPS10T_R_MAMULKODU(i), "")

			KDArray(1,2) = Doc.Div(S20T.DAKIKA(i), S20T.SURE(i))
		End if
	Next
	GetKalipDegisim = KDArray
	Doc.DeleteTableObject "QERW"
End Function

Function GetKayipYuzde()
	Dim KayipArray(1)
	Dim SetupDahilDurus(1)

	DurusSuresi = GetToplamDurus()	
	AyarSetupSuresi = GetToplamSetup()

	YemekSuresi = GetToplamDurusSaat("Y")
	MakinaKapamaSure = GetToplamDurusSaat("PY")
	KalipDenemeSure = GetToplamDurusSaat("KD")

	VardiyaCalismaSaati = 12
	MakinaSayisi = GetMakinaSayisi()
	DonemBasi = SFDC20_KRITERLER.RECTARIH11
'	if trim(DonemBasi) = "" Then DonemSonu = Doc.BuGun
	DonemSonu = SFDC20_KRITERLER.RECTARIH12
	if trim(DonemSonu) = "" Then DonemSonu = Doc.BuGun
	CalisilanGun = Doc.CDateToInt(DonemSonu) - Doc.CDateToInt(DonemBasi) + 1

	ToplamCalismaSaati = VardiyaCalismaSaati * MakinaSayisi * CalisilanGun
	
	For i = 0 to 1
		SetupDahilDurus(i) = DurusSuresi(i) - MakinaKapamaSure(i) - YemekSuresi(i) - KalipDenemeSure(i) + AyarSetupSuresi(i)
		KayipArray(i) = Doc.Div(SetupDahilDurus(i), ToplamCalismaSaati) * 100
	Next
	GetKayipYuzde = KayipArray
End Function

	FireArray(0) = 0
	FireArray(1) = 0
	'FireArray(2) = 0

	DonemBasi = SFDC20_KRITERLER.RECTARIH11
	DonemSonu = SFDC20_KRITERLER.RECTARIH12
	QStr = "KOD>='11HKT'&KOD<='11HKTzzzz'"
	if trim(DonemBasi) <> "" Then
		QStr = QStr+"&TARIH>='"+DonemBasi+"'"
	End if
	if trim(DonemSonu) <> "" Then
		QStr = QStr+"&TARIH<='"+DonemSonu+"'"
	End if

	Set S20T = Doc.Select1("QERW", "STOK20T", QStr, "", -1)
	if S20T.GetRowCount = 0 Then
		GetTakozFireKG = FireArray
		Exit Function
	End if

	satirSayisi = S20T.GetRowCount()
	For i = 1 to satirSayisi
		Doc.StatusBarMessage "Takoz fire aðýrlýklarý hesaplanýyor : " & i & " / " & satirSayisi
		Vardiya = Doc.GetDBFieldValue("STOK20E", "NITELIK", S20T.EVRAKNO(i), "")
		if trim(Vardiya) = "21" or trim(Vardiya) = "" Then
			FireArray(0) = FireArray(0) + S20T.SF_MIKTAR(i)
		Elseif trim(Vardiya) = "22" Then
			FireArray(1) = FireArray(0) + S20T.SF_MIKTAR(i)
		'Elseif trim(Vardiya) = "33" Then
		'	FireArray(2) = FireArray(0) + S20T.SF_MIKTAR(i)
		End if
	Next
	GetTakozFireKG = FireArray
	Doc.DeleteTableObject "QERW"
End Function
