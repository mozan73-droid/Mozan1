Sub SatirYaz(ASN)
	SevkAdresKodu = STOK60E.SHIPTOCODE
	if trim(SevkAdresKodu) = "D1D4F" then 'GÖLCÜK
		Birim = "EA"
	else
		Birim = "AD"
	end if
	CHKodu = STOK60E.CH_KODU
	for i = 1 to S60.GetRecCount()
		if trim(S60.KOD(i)) <> "" then
			Ambalaj = S60.FATURASIZMI(i)
			Kod = S60.KOD(i)
'			if trim(Ambalaj) = "E" then
'				StokKodu = Kod
'			else
				StokKodu = trim(Doc.GetCHStokKodu(Kod,CHKodu))
'			end if
			Miktar = S60.SF_MIKTAR(i)
'			Birim = S60.SF_SF_UNIT(i)
			PART_CUM_QTY = 0
			PART_PRIOR_CUM_QTY = 0 
			SiparisNo = S60.SIPNO(i)
			MusteriSipNo = Doc.GetDBFieldValue("STOK40E","CHSIPNO",SiparisNo,"")
			PART_PO_NO = trim(MusteriSipNo)
			if trim(Ambalaj) <> "E" then
				KontSayisi = S60.BRUTAGIRLIK(i)
				ParcaTuru = "PP"
				ParcaKodu = trim(StokKodu)
				if trim(cstr(KontSayisi)) = "" or trim(cstr(KontSayisi)) = "0" then
					kontParcaSayisi = Doc.GetDBFieldValue("STOK00","B_B_TECH_1",Kod,"")
					kontParcaSayisi = Doc.Round(kontParcaSayisi,0)
					ParcaKontSayisi = Doc.Div(Miktar,kontParcaSayisi)
					ParcaKontSayisi = Doc.Ceil(ParcaKontSayisi)
				else
					ParcaKontSayisi = Doc.CDbl(KontSayisi)
					kontParcaSayisi = Doc.Div(Miktar,ParcaKontSayisi)
					kontParcaSayisi = Doc.Ceil(kontParcaSayisi)
				end if
				KonteynerTipi = "BOX90"
			else
				ParcaTuru = "CT"
				ParcaKodu = " " & trim(StokKodu)
				kontParcaSayisi = "0000"
				ParcaKontSayisi = "0000"
				KonteynerTipi = ""
			end if
			ASN.WriteLine(cTab(3) & "<PART>")
			ASN.WriteLine(cTab(4) & "<NO_PART>" & ParcaKodu & "</NO_PART>")
			ASN.WriteLine(cTab(4) & "<PART_NO_TYPE>" & ParcaTuru & "</PART_NO_TYPE>")
			ASN.WriteLine(cTab(4) & "<PART_QUANTITY>" & Miktar & "</PART_QUANTITY>")
			ASN.WriteLine(cTab(4) & "<PART_UOM>" & Birim & "</PART_UOM>")
			ASN.WriteLine(cTab(4) & "<PART_CUM_QTY>" & PART_CUM_QTY & "</PART_CUM_QTY>")
			ASN.WriteLine(cTab(4) & "<PART_PRIOR_CUM_QTY>" & PART_PRIOR_CUM_QTY & "</PART_PRIOR_CUM_QTY>")
			ASN.WriteLine(cTab(4) & "<PART_PO_NO>" & PART_PO_NO & "</PART_PO_NO>")
			ASN.WriteLine(cTab(4) & "<CONTAINERS>")
			ASN.WriteLine(cTab(5) & "<CONTAINER>")
			ASN.WriteLine(cTab(6) & "<CONTAINER_TYPE_CODE>" & KonteynerTipi & "</CONTAINER_TYPE_CODE>")
			ASN.WriteLine(cTab(6) & "<PARTS_PER_CONT>" & KontParcaSayisi & "</PARTS_PER_CONT>")
			ASN.WriteLine(cTab(6) & "<CONTS_PER_PART>" & ParcaKontSayisi & "</CONTS_PER_PART>")
			ASN.WriteLine(cTab(5) & "</CONTAINER>")
			ASN.WriteLine(cTab(4) & "</CONTAINERS>")

Function KodKontrol()
	for i = 1 to STOK60T.GetRecCount()
		StokKodu = STOK60T.KOD(i)
		if trim(StokKodu) <> "" then
			CHKodu = STOK60E.CH_KODU
			CHStokKodu = trim(Doc.GetCHStokKodu(StokKodu,CHKodu))
			Ambalaj = STOK60T.FATURASIZMI(i)
			if trim(CHStokKodu) = "" then
				Hata = Hata & StokKodu & vbcrlf
			end if
		end if
	Next
	if Hata <> "" then
		Hata = Hata & "Ford stok kodu karþýlýðý girilmemiþ."
	end if
	KodKontrol = Hata
End Function
Function SandikAdediKontrol()
	for i = 1 to STOK60T.GetRecCount()
		StokKodu = STOK60T.KOD(i)
		if trim(StokKodu) <> "" then
			kontParcaSayisi = Doc.GetDBFieldValue("STOK00","B_B_TECH_1",StokKodu,"")
			Ambalaj = STOK60T.FATURASIZMI(i)
			BrutAgirlik = STOK60T.BRUTAGIRLIK(i)
			if trim(Ambalaj) = "" and kontParcaSayisi = 0 and trim(BrutAgirlik) = "" then
				Hata = Hata & StokKodu & vbcrlf
			end if
		end if
	Next
	if Hata <> "" then
		Hata = Hata & "Konteyner parça adedi girilmemiþ."
	end if
	SandikAdediKontrol = Hata
End Function

Function IEWaitReady(ByVal IEObj)
	IEWaitReady = 0
	While IEObj.Busy or IEObj.ReadyState <> 4
	Wend
End Function
Function IEGoto(ByVal IEObj, ByVal URL)
	IEObj.Navigate (URL)
	IEWaitReady (IEObj)
	IEGoto = 0
End Function

Sub Send_One_Mail(SDW,ouapp,m_subject,m_from,m_to,m_cc,m_bcc,message_type,message,m_attach)
	Dim oumailitem
    Set oumailitem = ouapp.CreateItem(0)

	oumailitem.ExpiryTime = Date + 30
    oumailitem.ReadReceiptRequested = False
    oumailitem.Subject = m_subject
	oumailitem.SentOnBehalfOfName=m_from
    oumailitem.To = m_to
    oumailitem.Cc = m_cc
    oumailitem.Bcc = m_bcc
	If message_type="H" Then
		If Left(ouapp.version,1)>"9" Then oumailitem.Body=""
'		oumailitem.BodyFormat=2
		oumailitem.HtmlBody = "<!DOCTYPE HTML PUBLIC "" -//W3C//DTD HTML 4.0 Transitional//EN"">" + message
	Else
		oumailitem.Body = message
	End If
	If Len(m_attach)>0 Then
		Dim Attachments
		Set Attachments=oumailitem.Attachments
		Attachments.Add m_attach
	End If

	If SDW="W" Then
		oumailitem.Save
	ElseIf SDW="D" Then
		oumailitem.Display
	ElseIf SDW="S" Then
		oumailitem.Send
	End If
End Sub

Sub StandartMaliyeteYaz(StokKodu, Endeks, Tarih, Fiyat, PB, Miktar)
	if trim(Tarih) = "" Then Tarih = Doc.BuGun
	rows = Doc.SelectEQ("STDMH0E","KOD+STDMALTYPE",StokKodu & Endeks)
	if rows = 0 then
		ret = Doc.New_Voucher()				
		STDMH0E.STDMALTYPE = Endeks
		STDMH0E.KOD = StokKodu
		STDMH0T.AddRow
		satirNo = STDMH0T.GetRecCount
		Tarih = Doc.StrDate_Year(Doc.BuGun) & "/01/01"
	else
		set ST = Doc.GetTableObject("AU_STDMH0E_SELECT")
		EvrakNo = ST.EVRAKNO(1)
		ret = Doc.Load_Voucher(EvrakNo)
		if ret <> 0 then
			k = Doc.d7MsgBox(trim(StokKodu) & ": Standart Maliyet tablosu yüklenemedi. Ýþlem kesilmiþtir.","dinamo",MB_OK)
			exit sub
		end if
		satirNo = satirBul(Tarih)
		if satirNo = 0 then
			STDMH0T.AddRow
			satirNo = STDMH0T.GetRecCount
		End if
	end if
	STDMH0T.VALIDAFTERTARIH(satirNo) = Doc.StrToNorDate(Tarih)
	STDMH0T.UNSUR(satirNo) = "HM"
	STDMH0T.TUTAR(satirNo) = Fiyat
	STDMH0T.PARABIRIMI(satirNo) = PB
	STDMH0T.MIKTAR(satirNo) = Miktar
	ret = Doc.Save_Voucher()
	if ret <> 0 then
		k = Doc.d7MsgBox(trim(StokKodu) & ": Standart Maliyet tablosu kaydedilemedi. Ýþlem kesilmiþtir.","dinamo",MB_OK)
		ret = Doc.LoadEmpty_Voucher()
		exit sub
	end if
	ret = Doc.LoadEmpty_Voucher()
End Sub
Function GrupKoduYarat(GrupKoduTipi,GrupKodu,GrupAciklamasi)
	Set Doc1 = Doc.CreateDocumentNoUI("GECO10","")
	Set GT = Doc1.GetTableObject("GECOUST")
	ret = Doc1.Load_Voucher(GrupKoduTipi)
	if ret <> 0 then
		GrupKoduYarat = 1
		ret = Doc1.LoadEmpty_Voucher()
		exit Function
	end if
	GT.AddRow
	satirNo = GT.GetRecCount
	GT.KOD(satirNo) = GrupKodu
	GT.AD(satirNo) = GrupAciklamasi
	ret = Doc1.Save_Voucher()
	if ret <> 0 then
		GrupKoduYarat = 1
	else
		GrupKoduYarat = 0
	end if
	ret = Doc1.LoadEmpty_Voucher()
End Function
Function satirBul(Tarih)

	For i = 1 to STDMH0T.GetRowCount()
		if STDMH0T.VALIDAFTERTARIH(i) = Tarih then
			satirBul = i
			exit function
		end if
	Next
End Function

Makro Kaynak Kodu
Firma Kodu : DLS / Uygulama : MAKR7S
Makro Kodu : UPCARI00
Hangi Tablo için : CNTACT00
Ýlave Hangi Uygulamalar için : 
Açýklama : Cari hesap kodu olanlarý cari kartla karþýlaþtýr farklý ise kaydet
Kod : 
Sub Makro1()
	Dim upflag
	Dim Doc1
	Dim CARI00
	Dim LISTET
	Set Doc1 = Doc.CreateDocumentNOUI("CARI00","")
	If Doc1 Is Nothing Then
		MsgBox "Hata Cari00"
		Exit Sub
	End If
	Set CARI00=Doc1.GetTableObject("CARI00")
	Set LISTET=Doc.GetTableObject("LISTET")
	for i=1 to LISTET.GetRecCount()
		chkod=LISTET.SIRKET_CH_KODU(i)
		sel=LISTET.SELECTED(i)
		if Trim(chkod)<>"" then
			retcode=Doc1.Load_Voucher(chkod)
			if(retcode=0) then
				upflag=0
				if Trim(LISTET.SIRKET_UNVAN1(i))<>Trim(CARI00.AD) then
					CARI00.AD=LISTET.SIRKET_UNVAN1(i)
					upflag=upflag+1
				end if
				if Trim(LISTET.SIRKET_UNVAN2(i))<>Trim(CARI00.NAME2) then
					CARI00.NAME2=LISTET.SIRKET_UNVAN2(i)
					upflag=upflag+1
				end if
				if Trim(LISTET.SIRKET_ADR_1(i))<>Trim(CARI00.ADRES_1) then
					CARI00.ADRES_1=LISTET.SIRKET_ADR_1(i)
					upflag=upflag+1
				end if
				if Trim(LISTET.SIRKET_ADR_2(i))<>Trim(CARI00.ADRES_2) then
					CARI00.ADRES_2=LISTET.SIRKET_ADR_2(i)
					upflag=upflag+1
				end if
				if Trim(LISTET.SIRKET_ADR_3(i))<>Trim(CARI00.ADRES_3) then
					CARI00.ADRES_3=LISTET.SIRKET_ADR_3(i)
					upflag=upflag+1
				end if
				if Trim(LISTET.SIRKET_POSTAKODU(i))<>Trim(CARI00.POSTAKODU) then
					CARI00.POSTAKODU=LISTET.SIRKET_POSTAKODU(i)
					upflag=upflag+1
				end if
				if Trim(LISTET.SIRKET_SEHIR_KODU(i))<>Trim(CARI00.SEHIRKODU) then
					CARI00.SEHIRKODU=LISTET.SIRKET_SEHIR_KODU(i)
					upflag=upflag+1
				end if
				if Trim(LISTET.IS_TELEFONU(i))<>Trim(CARI00.TELEFONNO_1) then
					CARI00.TELEFONNO_1=LISTET.IS_TELEFONU(i)
					upflag=upflag+1
				end if
				if Trim(LISTET.SIRKET_FAX(i))<>Trim(CARI00.FAXNO) then
					CARI00.FAXNO=LISTET.SIRKET_FAX(i)
					upflag=upflag+1
				end if
				if upflag>0 then
					retcode=Doc1.Save_Voucher()
					if retcode<>0 then
						msgbox "Save Error"
					end if
					retcode=Doc1.LoadEmpty_Voucher()
				end if
			end if
		end if
	next
End Sub
