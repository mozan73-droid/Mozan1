Sub ASNGonder()
       Set STOK60E = Doc.GetTableObject("STOK60E")
       Set STOK60T = Doc.GetTableObject("STOK60T")
       Set MTable=Doc.GetTableObject("MAKROTABLE")
       KullaniciAdi = trim(MTable.F1)
       KullaniciSifre = trim(MTable.F2)
       Doc.RUN_APP_CANCEL

       if trim(KullaniciAdi) = "" or trim(KullaniciSifre) = "" Then
               k = doc.d7msgbox("Kullanýcý adý ve þifre girmelisiniz." & vbcrlf & vbcrlf & "Ýþlem tamamlanmadan kesilmiþtir", "dinamo",MB_OK)
               exit sub
       End if
       if Trim(STOK60E.XNITELIK0) = "ASN_OK" then
               k = doc.d7msgbox("Bu irsaliyenin ASN 'si daha önce gönderilmiþ" & vbcrlf & vbcrlf & "Ýþlem tamamlanmadan kesilmiþtir", "dinamo",MB_OK)
               exit sub
       end if
       SeferNumarasi = Doc.GetUDFValue("RI",STOK60E.EVRAKNO,"SEFER_NO",0)
       if trim(SeferNumarasi) = "" then
               k = doc.d7msgbox("Kullanýcý tanýmlý alanlarda sefer numarasý girilmemiþ" & vbcrlf & vbcrlf & "Ýþlem tamamlanmadan kesilmiþtir", "dinamo",MB_OK)
               exit sub
       end if
       PlakaNo = STOK60E.BOX1_3
       if trim(PlakaNo) = "" then
               k = doc.d7msgbox("Plaka Numarasý girilmemiþ" & vbcrlf & vbcrlf & "Ýþlem tamamlanmadan kesilmiþtir", "dinamo", MB_OK)
               exit sub
       end if
       SevkAdresKodu = STOK60E.SHIPTOCODE
       if InStr(1, SevkAdresKodu, "D1D4T") > 0 Then
               FabrikaKodu = "D1D4T"
       Elseif InStr(1, SevkAdresKodu, "D1D4H") > 0 Then
               FabrikaKodu = "D1D4H"
       Elseif InStr(1, SevkAdresKodu, "D1D4F") > 0 Then
               FabrikaKodu = "D1D4F"
       Elseif InStr(1, SevkAdresKodu, "AUF9A") > 0 Then
               FabrikaKodu = "AUF9A"
       Elseif InStr(1, SevkAdresKodu, "CVR1B") > 0 Then
               FabrikaKodu = "CVR1B"
       Elseif InStr(1, SevkAdresKodu, "D1D4Y") > 0 Then
               FabrikaKodu = "D1D4Y"
       else
               k = Doc.d7msgbox("Sevk adresi girilmemiþ veya tanýmsýz" & vbcrlf & vbcrlf & "Ýþlem tamamlanmadan kesilmiþtir.", "dinamo",MB_OK)
               exit sub
       End if
       if not AmbalajVarmi then
               EH = Doc.d7msgbox("Hiç konteyner gönderilmemektedir" & vbcrlf & vbcrlf & "Onaylýyor musunuz.", "dinamo",MB_YESNO)
               if EH <> 6 then
                       k = doc.d7Msgbox("Ýþlem kesilmiþtir.", "dinamo",MB_OK)
                       exit sub
               end if
       End if
       KodDurumu = KodKontrol
       if trim(KodDurumu) <> "" then
               msgbox KodDurumu
               k = doc.d7Msgbox("Ýþlem kesilmiþtir.", "dinamo",MB_OK)
               exit sub
       end if
       SandikDurumu = SandikAdediKontrol
       if trim(SandikDurumu) <> "" then
               msgbox SandikDurumu
               k = doc.d7Msgbox("Ýþlem kesilmiþtir.", "dinamo",MB_OK)
               exit sub
       end if

       Doc.CreateTableObject "ZZZ=STOK60T","TMPCOPY"
       Set S60 = Doc.GetTableObject("ZZZ")
       Set ST60 = Doc.GetTableObject("STOK60T")
       S60.CopyTable(ST60)
       S60.GroupBy("KOD")

       Set Fso = CreateObject("Scripting.FileSystemObject")
       ASNFolder = trim(Doc.Getd7Directory_DataFirm) & "abdat"
       if not fso.FolderExists(ASNFolder) Then
               fso.CreateFolder ASNFolder
       End if
       dosyaAdi = ASNFolder & "\asn" & trim(STOK60E.EVRAKNO) & ".xml"
       Set ASN = Fso.CreateTextFile(dosyaAdi,True)

       Set xmlDoc = CreateObject("Microsoft.XMLDOM")
       xmlDoc.async = False

       Doc.StatusBarMessage "XML Dosyasý oluþturuluyor"
       Call Header(ASN)
       Call SatirYaz(ASN)
       Call Footer(ASN)

       ASN.Close
       xmlDoc.Load(dosyaAdi)
       Set IE = CreateObject("InternetExplorer.Application")
       IE.Navigate("file:\\" & dosyaAdi)
       IE.TheaterMode = true
       IE.Visible = true

       EH = doc.d7Msgbox("Oluþturulan ASN uygun mu?","dinamo",MB_YESNO)
       if EH <> 6 then
               k = doc.d7Msgbox("Ýþlem kesilmiþtir.", "dinamo",MB_OK)
               exit sub
       end if

       Doc.StatusBarMessage "ford.com.tr 'ye baðlanýlýyor"
       cookieStr = FOSNConnect(KullaniciAdi, KullaniciSifre)
       if trim(cookieStr) = "" Then
               k = Doc.d7MsgBox("FOSN 'e baðlanýlamadý", "Hata!", MB_OK)
               Exit Sub
       End if

       Doc.StatusBarMessage "ASN gönderiliyor"
       sonuc = sendData(xmlDoc, cookieStr, dosyaAdi)

       xmlDoc.LoadXML(sonuc)
       if trim(xmldoc.xml) = "" Then
               k = Doc.d7MsgBox("ASN Gönderilemedi , " & ret,"dinamo",MB_OK)
       else
               ret = xmldoc.documentElement.selectSingleNode("RETURN").Text
               if ret <> "0" or ret <> 0 then
                       k = Doc.d7MsgBox("ASN Gönderilemedi , " & ret,"dinamo",MB_OK)
                       k = Doc.d7MsgBox(xmlDoc.xml,"dinamo",MB_OK)
               else
                       k = Doc.d7MsgBox("Ýþlem Tamam","dinamo",MB_OK)
                       STOK60E.XNITELIK0 = "ASN_OK"
                       Doc.RunCommand("MT_SAVE")
               end if
       end if
       Set Fso = Nothing
       Set xmlDoc = Nothing

       Call GetASNPage(KullaniciAdi, KullaniciSifre)
End Sub

Sub SatirYaz(ASN)
       if trim(FabrikaKodu) = "D1D4F" then 'GÖLCÜK
               Birim = "EA"
       else
               Birim = "AD"
       end if
       CHKodu = STOK60E.CH_KODU

               if trim(S60.KOD(i)) <> "" then
                       Ambalaj = S60.FATURASIZMI(i)
                       Kod = S60.KOD(i)
'                       if trim(Ambalaj) = "E" then
'                               StokKodu = Kod
'                       else
                               StokKodu = trim(Doc.GetCHStokKodu(Kod,CHKodu))
'                       end if
                       Miktar = S60.SF_MIKTAR(i)
'                       Birim = S60.SF_SF_UNIT(i)
                       PART_CUM_QTY = 0
                       PART_PRIOR_CUM_QTY = 0
                       SiparisNo = S60.SIPNO(i)
                       MusteriSipNo = Doc.GetDBFieldValue("STOK40E","CHSIPNO",SiparisNo,"")
                       PART_PO_NO = trim(MusteriSipNo)
                       if trim(Ambalaj) <> "E" then
                               KontSayisi = S60.BRUTAGIRLIK(i)
                               ParcaTuru = "PP"
                               ParcaKodu = trim(StokKodu)
                               if trim(cstr(KontSayisi)) = "" or trim(cstr(KontSayisi)) = "0" then
                                       kontParcaSayisi = Doc.GetDBFieldValue("STOK00","B_B_TECH_1",Kod,"")
                                       kontParcaSayisi = Doc.Round(kontParcaSayisi,0)
                                       ParcaKontSayisi = Doc.Div(Miktar,kontParcaSayisi)
                                       ParcaKontSayisi = Doc.Ceil(ParcaKontSayisi)
                               else
                                       ParcaKontSayisi = Doc.CDbl(KontSayisi)
                                       kontParcaSayisi = Doc.Div(Miktar,ParcaKontSayisi)
                                       kontParcaSayisi = Doc.Ceil(kontParcaSayisi)
                               end if
                               KonteynerTipi = "BOX90"
                       else
                               ParcaTuru = "CT"
                               ParcaKodu = " " & trim(StokKodu)
                               kontParcaSayisi = "0000"
                               ParcaKontSayisi = "0000"
                               KonteynerTipi = ""
                       end if
                       ASN.WriteLine(cTab(3) & "<PART>")
                       ASN.WriteLine(cTab(4) & "<NO_PART>" & ParcaKodu & "</NO_PART>")
                       ASN.WriteLine(cTab(4) & "<PART_NO_TYPE>" & ParcaTuru & "</PART_NO_TYPE>")

Function sendData(xmlDoc, cookieStr, xmlFile)

       if Doc.Bugun <= "2007/02/27" Then
               postURL = "https://fosnapp.ford.com.tr/asnposter/postxmlasn.asp"
       Else

       End if

       xmlHTTP.Open "POST", postURL, false
       xmlHTTP.setRequestHeader "Cookie", cookieStr
       xmlHTTP.setRequestHeader "Content-Type", "text/xml"

       xmlhttp.Send xmlDoc

       sendData = xmlHTTP.responseText

Function KodKontrol()
       for i = 1 to STOK60T.GetRecCount()
               StokKodu = STOK60T.KOD(i)
               if trim(StokKodu) <> "" then
                       CHKodu = STOK60E.CH_KODU
                       CHStokKodu = trim(Doc.GetCHStokKodu(StokKodu,CHKodu))
                       Ambalaj = STOK60T.FATURASIZMI(i)
                       if trim(CHStokKodu) = "" then
                               Hata = Hata & StokKodu & vbcrlf
                       end if
               end if
       Next
       if Hata <> "" then
               Hata = Hata & "Ford stok kodu karþýlýðý girilmemiþ."
       end if
       KodKontrol = Hata
End Function
Function SandikAdediKontrol()
       for i = 1 to STOK60T.GetRecCount()
               StokKodu = STOK60T.KOD(i)
               if trim(StokKodu) <> "" then
                       kontParcaSayisi = Doc.GetDBFieldValue("STOK00","B_B_TECH_1",StokKodu,"")
                       Ambalaj = STOK60T.FATURASIZMI(i)
                       BrutAgirlik = STOK60T.BRUTAGIRLIK(i)
                       if trim(Ambalaj) = "" and kontParcaSayisi = 0 and trim(BrutAgirlik) = "" then
                               Hata = Hata & StokKodu & vbcrlf
                       end if
               end if
       Next
       if Hata <> "" then
               Hata = Hata & "Konteyner parça adedi girilmemiþ."
       end if
       SandikAdediKontrol = Hata
End Function

	SagSpace = Right(String(KarakterSayisi, " ") & Trim(Yazi), KarakterSayisi)

Function Ondalikli(Sayi, KarakterSayisi, OndalikSayisi)
  Sayi = Doc.Round(Sayi, OndalikSayisi)
  TamSayi = Fix(Sayi)
  ondalik = Abs(Sayi - TamSayi) * 10 ^ OndalikSayisi
  If TamSayi < 0 Then
    Eksi = "-"
  Else
    Eksi = ""
  End If
  If OndalikSayisi > 0 Then
    Ondalikli = Eksi & SagSpace(Abs(TamSayi), KarakterSayisi - (OndalikSayisi + 1) - Len(Eksi)) & "," & Left(ondalik & "00",OndalikSayisi)
  Else
    Ondalikli = Eksi & SagSpace(Abs(TamSayi), KarakterSayisi - OndalikSayisi - Len(Eksi))
  End If
End Function

Function GetMiktarTarih(StokKodu, Miktar, Tarih, Attrib)
	Dim MiktarArray()
	Redim MiktarArray(1,0)
	if Attrib = "G" then
		MiktarArray(0,0) = Miktar
		MiktarArray(1,0) = Tarih
		GetMiktarTarih = MiktarArray
		exit Function
	end if

	Gun = GetHaftaninGunu(Tarih)
	PartiBuyuklugu = Doc.GetDBFieldValue("STOK00","B_MINPARTIBUYUKLUGU", StokKodu, "")
	if PartiBuyuklugu = 0 then PartiBuyuklugu = Miktar
	if PartiBuyuklugu >= Miktar then
		MiktarArray(0,0) = Miktar
		MiktarArray(1,0) = Tarih
		GetMiktarTarih = MiktarArray
		exit Function
	end if

	if Gun = 1 then
		KacaBolunecek = 3
	Elseif Gun > 1 and Gun <= 3 then 
		KacaBolunecek = 2
	elseif Gun > 3 then
		KacaBolunecek = 1
		MiktarArray(0,0) = Miktar
		if Gun > 5 then
			SipTarih = Tarih
		Else
			SipTarih = GetTarihGun(Tarih, 5)
		end if
		MiktarArray(1,0) = SipTarih
		GetMiktarTarih = MiktarArray

		exit Function		
	end if

	GunlukMiktar = Doc.Div(Miktar, KacaBolunecek)
	GunlukMiktar = Doc.Ceil(Doc.Div(GunlukMiktar, PartiBuyuklugu)) * PartiBuyuklugu

	KalanMiktar = Miktar
	While KalanMiktar > 0
		sirano = siraNo + 1
		if KalanMiktar > GunlukMiktar then
			SipMiktari = GunlukMiktar
		Else
			SipMiktari = KalanMiktar
		end if
		KalanMiktar = KalanMiktar - SipMiktari
		MiktarArray(0, uBound(MiktarArray,2)) = SipMiktari
		if KalanMiktar > 0 then
			Redim Preserve MiktarArray(1, uBound(MiktarArray,2) + 1)
		end if
		if sirano > 100 then
			msgbox "Döngü 100 'den fazla devam etti. Hatayý kontrol ediniz." & vbcrlf & "Ýþlem Kesilmiþtir" 
			Exit Function
		end if
	Wend
	GunSayisi = UBound(MiktarArray, 2) + 1
	if GunSayisi = 2 then
		MiktarArray(1,0) = GetTarihGun(Tarih, 3)
		MiktarArray(1,1) = GetTarihGun(Tarih, 5)
	Elseif GunSayisi = 3 then
		MiktarArray(1,0) = GetTarihGun(Tarih, 1)
		MiktarArray(1,1) = GetTarihGun(Tarih, 3)
		MiktarArray(1,2) = GetTarihGun(Tarih, 5)
	end if
	GetMiktarTarih = MiktarArray
End Function
