Sub Makro1()
	MakaraTipi = "D-TAHTA-0"
	StokKodu = "45 70104B 3V000000010000"
	StokKodu = Doc.LeftJustify(trim(StokKodu), 24)
	SevkBoyu = 400
	
	KabloCapi = Doc.GetUDFValue("STOK00", StokKodu, "DLG095", 0)
	BirimKabloAgirligi = Doc.GetUDFValue("STOK00", StokKodu, "DLG109", 0) / 1000

	Call GetCableInfo(StokKodu, Gerilim, MetalVarMi)

	msgbox GetMakaraAmbalajKodu(MakaraTipi, SevkBoyu, KabloCapi, BirimKabloAgirligi, StokKodu, Gerilim, MetalVarMi)

Function GetMakaraAmbalajKodu(ByVal MakaraTipi, ByVal KabloUzunlugu, ByVal KabloCapi, ByVal BirimKabloAgirligi, ByVal KabloKodu, ByVal Gerilim, ByVal MetalVarMi)
	Dim Kriterler, k, MakaraTasima, MakaraGobek, MakaraKapasite,i,MakaraTable
	
	if KabloUzunlugu > 1500 Then KabloCapi = KabloCapi * 1.05

	Makara = ""
	If trim(MakaraTipi) <> "" Then
		'Set MakaraTable=Doc.CreateRowSetObject("XXX=AMBLJ00","REPORT")
		'Set Kriterler=Doc.CreateSObject("Query")
		'Kriterler.UseTable "AMBLJ00"
		'Kriterler.AddAndCriteria MakaraTable,"MKR_TURU","=",left(MakaraTipi & space(10),10)
		'k = MakaraTable.RunQuery(Kriterler)
		'k = MakaraTable.Sort(">MKR_FLANSCAPI+<MKR_ICHACIM")
		QStr = "SELECT * FROM AMBLJ00"
		QStr = QStr & " WHERE MKR_TURU='" & trim(MakaraTipi) & "'"
		QStr = QStr & " ORDER BY MKR_FLANSCAPI, MKR_ICHACIM DESC, MKR_TASIMAKAP"
		Set MakaraTable = Doc.RunSQLQuery("xxxsdfsdf", QStr)
	Else
		'Set MakaraTable=Doc.CreateRowSetObject("REPORT1=AMBLJ00","REPORT")
		'Set Kriterler=Doc.CreateSObject("Query")
		'Kriterler.UseTable "AMBLJ00"
		'Kriterler.AddAndCriteria MakaraTable,"TEMEL_SINIFI","=",left("MAKARA" & space(10),10)
		'k = MakaraTable.RunQuery(Kriterler)
		'k = MakaraTable.Sort(">MKR_FLANSCAPI+<MKR_ICHACIM")
		QStr = "SELECT * FROM AMBLJ00"
		QStr = QStr & " WHERE TEMEL_SINIFI='MAKARA'"
		QStr = QStr & " ORDER BY MKR_FLANSCAPI, MKR_ICHACIM DESC, MKR_TASIMAKAP"
		Set MakaraTable = Doc.RunSQLQuery("xxxsdfsdf", QStr)
	End If
	satirNo = 1

	satirNo = KapasiteSecim(satirNo, MakaraTipi, KabloUzunlugu, KabloCapi,MakaraTable, KabloKodu)
	if satirNo = -1 Then Exit Function

	satirNo = TasimaSecim(satirNo, MakaraTipi, KabloUzunlugu, BirimKabloAgirligi, MakaraTable, KabloKodu)
	if satirNo = -1 Then Exit Function
	
	satirNo = GobekSecim(satirNo, MakaraTipi, KabloKodu, KabloCapi,MakaraTable, Gerilim, MetalVarMi)

Function GobekSecim(BaslangicSatirNo, MakaraTipi, KabloKodu, KabloCapi, MakaraTable, Gerilim, MetalVarMi)
	If Gerilim < 1 Then
		BukumKatsayisi = 6
	ElseIf Gerilim >= 1 and Gerilim <= 3.3 and not MetalVarMi Then
		BukumKatsayisi =  8
	ElseIf Gerilim >= 1 and Gerilim <= 3.3 and MetalVarMi Then
		BukumKatsayisi =  10
	ElseIf Gerilim > 3.3 Then
		BukumKatsayisi =  12
	Else
		BukumKatsayisi =  12
	End If
	GobekCapi = KabloCapi * BukumKatsayisi

	GobekSecim = GobekBul(KabloKodu, BaslangicSatirNo, MakaraTipi, GobekCapi, MakaraTable)
End Function

Function GobekSecim1(ByVal MakaraTipi, ByVal KabloKodu, ByVal KabloCapi, ByVal MakaraTable)
	Dim Gerilim, Zirh, GobekCapi

		End If
	Else
		GobekCapi = KabloCapi * 15
	End If
	If GobekSecim = 0 Then
		GobekSecim = GobekBul(KabloKodu, MakaraTipi, GobekCapi, MakaraTable)
	End If
End Function

Function KapasiteSecim(satirNo, ByVal MakaraTipi, ByVal KabloUzunlugu, ByVal KabloCapi, ByVal MakaraTable, StokKodu)
	KapasiteSecim = -1
	Dim Pi, FlansCapi, GobekCapi, IcGenislik, YanYana, UstUste, KapamaPayi, FlansYuksekligi, Kapasite,i,k
	Pi = 3.141592654
	
	For i = satirNo To MakaraTable.GetRecCount()
		FlansCapi = MakaraTable.MKR_FLANSCAPI(i)
		GobekCapi = MakaraTable.MKR_GOBEKCAPI(i)
		IcGenislik = MakaraTable.MKR_ICGENISLIK(i)
		
		YanYana = Fix(Doc.Div(IcGenislik, KabloCapi))
		
		If KabloCapi <= 10 Then
			KapamaPayi = 15
		ElseIf KabloCapi <= 20 Then
			KapamaPayi = 30
		ElseIf KabloCapi <= 30 Then
			KapamaPayi = 50
		ElseIf KabloCapi <= 40 Then
			KapamaPayi = 70
		Else
			KapamaPayi = 100
		End If
		FlansYuksekligi = ((FlansCapi - GobekCapi) / 2) - KapamaPayi
		
		UstUste = Doc.Div(FlansYuksekligi,KabloCapi)
		Kapasite = Round(YanYana * UstUste * (GobekCapi + (UstUste * KabloCapi)) * Pi / 1000, 0)
		If Kapasite >= KabloUzunlugu Then
			KapasiteSecim = i
			Exit Function
		End If
	Next
	If KapasiteSecim = -1 Then
		k = Doc.d7msgbox("Stok Kodu : " & trim(StokKodu) & ", Sevk Boyu : " & KabloUzunlugu & " Makara Tipi  : " & trim(MakaraTipi) & ", Bu kablonun sýðabileceði makara bulunamadý","dinamo",MB_OK+MB_ICONWARNING)
	End If
End Function

Sub GetCableInfo(StokKodu, Gerilim, MetalVarMi)
	QStr = "SELECT KTT.ATTRIB, KTT.DEGER"
	QStr = QStr & " FROM KTT_CBATTR0T KTT"
	QStr = QStr & " LEFT JOIN KTT_CBATTR0E KTE ON KTE.EVRAKNO=KTT.EVRAKNO"
	QStr = QStr & " LEFT JOIN KTA_STOK00 S ON S.CB_PLCODE=KTE.PLANTCODE"
	QStr = QStr & " WHERE S.KOD='" & StokKodu & "'"
	QStr = QStr & " AND KTT.ATTRIB IN ('voltagepp','metalliclayer')"
	Set TblAttrib = Doc.RunSQLQuery("sadg3grgdfgsfhha", QStr)
	if TblAttrib.GetRowCount() > 0 Then
		For i = 1 to TblAttrib.GetRowCount()
			if trim(TblAttrib.ATTRIB(i)) = "voltagepp" Then
				Gerilim = Doc.CDbl(TblAttrib.DEGER(i))
			Elseif trim(TblAttrib.ATTRIB(i)) = "metalliclayer" Then
				if UCase(trim(TblAttrib.DEGER(i))) = "YES" Then
					MetalVarMi = True
				Else
					MetalVarMi = False
				End if
			End if
		Next
	Else
		Zirh = Mid(Trim(StokKodu), 17, 1)
		Ekran = Mid(Trim(StokKodu), 15, 1)
		if trim(Zirh) = "0" and trim(Ekran) = "0" Then
			MetalVarMi = False
		Else
			MetalVarMi = True
		End if
		Gerilim = GetGerilim(Mid(Trim(StokKodu), 4, 1))
	End if
End Sub
