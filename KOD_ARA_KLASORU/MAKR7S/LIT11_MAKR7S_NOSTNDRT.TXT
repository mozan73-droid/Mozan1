Sub Makro1()
	MamulKodu = BOMU01E.MAMULCODE
				
	YEn = Doc.GetKartValue("STOK00","B_EN",MamulKodu,-1)

	Gramaj = Doc.CDbl(Doc.GetUDFValue("STOK00",MamulKodu,"GR",0))
	KalipSayisi = Doc.CDbl(Doc.GetUDFValue("STOK00",MamulKodu,"RENKS",0))

	AmbalajMiktari = Doc.CDbl(Doc.GetUDFValue("STOK00",MamulKodu,"AMBALAJMIK",0))
	
	For i = 1 to BOMU01T.GetRecCount()
		if trim(BOMU01T.BOMREC_INPUTTYPE(i)) = "H" then
			MalzemeKodu = BOMU01T.BOMREC_KAYNAKCODE(i)
	 		Boy = Doc.GetKartValue("STOK00","B_BOY",MalzemeKodu,-1)
			Agirlik = Doc.GetKartValue("STOK00","B_AGIRLIK",MalzemeKodu,-1)
 			En = Doc.GetKartValue("STOK00","B_EN",MalzemeKodu,-1)
		end if

		N24 = Doc.GetKartValue("STOK00","B_B_TECH_24",MamulKodu,-1)
		N25 = Doc.GetKartValue("STOK00","B_B_TECH_25",MamulKodu,-1)

		BOMU01E.MAMULMIKTAR = N18*100
		MamulMiktar = BOMU01E.MAMULMIKTAR
		N3 = Doc.GetKartValue("STOK00","B_B_TECH_3",MamulKodu,-1)
		N4 = Doc.GetKartValue("STOK00","B_B_TECH_4",MamulKodu,-1)
		if trim(BOMU01T.BOMREC_INPUTTYPE(i)) = "I" then
			OpKod = BOMU01T.BOMREC_OPERASYON(i)
			AyarSetup = Doc.CDbl(Doc.GetUDFValue("IMLT01",OpKod,"AYARS",0))
			CTP1 = Doc.CDbl(Doc.GetUDFValue("IMLT01",OpKod,"BSURE",0))
			BirimAyiklama = Doc.Cdbl(Doc.GetUdfValue("IMLT01",OpKod,"KUTUS",0))
			Sure = Doc.CDbl(Doc.GetUDFValue("IMLT01",OpKod,"SURE",0))
			Kesim = Doc.CDbl(Doc.GetUDFValue("IMLT01",OpKod,"KESIM",0))
		end if
		if BOMU01T.BOMREC_INPUTTYPE(i) = "H" and Mid(BOMU01T.BOMREC_KAYNAKCODE(i),2,1) = "B" then
			BobinMiktar = N3*N4*Gramaj/100000
			BOMU01T.BOMREC_KAYNAK0(i) = BobinMiktar
		elseif BOMU01T.BOMREC_INPUTTYPE(i) = "H" and Mid(BOMU01T.BOMREC_KAYNAKCODE(i),2,1) = "T" then
			BOMU01T.BOMREC_KAYNAK0(i) = 100
		end if

		if BOMU01T.BOMREC_INPUTTYPE(i) = "I" then
			BOMU01T.BOMREC_KAYNAK1(i) = AyarSetup
		end if	
		if BOMU01T.BOMREC_INPUTTYPE(i) = "I" and left(BOMU01T.BOMREC_KAYNAKCODE(i),4) = "KM01" then
			BOMU01T.BOMREC_KAYNAK0(i) = Kesim
		end if	

		if BOMU01T.BOMREC_INPUTTYPE(i) = "I" and left(BOMU01T.BOMREC_KAYNAKCODE(i),4) = "KM03" then
			BOMU01T.BOMREC_KAYNAK0(i) = Kesim
		end if			
		
		if BOMU01T.BOMREC_INPUTTYPE(i) = "H" and left(BOMU01T.BOMREC_KAYNAKCODE(i),4) = "Y 03" then
			BOMU01T.BOMREC_KAYNAK1(i) = KalipSayisi
		end if
'** deðiþklik yapýlan yer 15.9.2009
		if BOMU01T.BOMREC_INPUTTYPE(i) = "H" and left(BOMU01T.BOMREC_KAYNAKCODE(i),4) = "A 01" then
			BOMU01T.BOMREC_KAYNAK0(i) = Doc.Div(MamulMiktar,KoliMik)

		End if
'***** sonu		
		if BOMU01T.BOMREC_INPUTTYPE(i) = "I" and (left(BOMU01T.BOMREC_KAYNAKCODE(i),2) = "OM" or left(BOMU01T.BOMREC_KAYNAKCODE(i),2) = "TM") Then

			BOMU01T.BOMREC_KAYNAK0(i) = TSure

		if BOMU01T.BOMREC_INPUTTYPE(i) = "H" and Mid(BOMU01T.BOMREC_KAYNAKCODE(i),2,1) = "B" then
			For j = 1 to BOMU01T.GetRecCount
			if BOMU01T.BOMREC_INPUTTYPE(j) = "I" and left(BOMU01T.BOMREC_KAYNAKCODE(j),4) = "KM02" Then
				Kesim11 = (N4*100)/(100*60*80)
				BOMU01T.BOMREC_KAYNAK0(j) = Kesim11
			end if
			Next
		end if
		if BOMU01T.BOMREC_INPUTTYPE(i) = "I" and left(BOMU01T.BOMREC_KAYNAKCODE(i),4) = "CP01" then
			CTP = Doc.Cdbl((KalipSayisi*CTP1))
			BOMU01T.BOMREC_KAYNAK1(i) = CTP+AyarSetup
		end if
		if BOMU01T.BOMREC_INPUTTYPE(i) = "I" and left(BOMU01T.BOMREC_operasyon(i),4) = "OAYK" Then
			AyiklamaSuresi = BirimAyiklama*MamulMiktar
			BOMU01T.BOMREC_KAYNAK0(i) = AyiklamaSuresi
		end if
		if BOMU01T.BOMREC_INPUTTYPE(i) = "H" and left(BOMU01T.BOMREC_KAYNAKCODE(i),4) = "Y 07" then
			Sonuc = (Yuz*N18*100*0.00033)/1000
			BOMU01T.BOMREC_KAYNAK0(i) = Sonuc
		end if
		if BOMU01T.BOMREC_INPUTTYPE(i) = "I" and (trim(BOMU01T.BOMREC_KAYNAKCODE(i)) = "YM01" OR trim(BOMU01T.BOMREC_KAYNAKCODE(i)) = "YM02") then

			Sonuc2 = Doc.Div((N18*100),Sonuc1)
			BOMU01T.BOMREC_KAYNAK0(i) = Sonuc2	
		end if
		if BOMU01T.BOMREC_INPUTTYPE(i) = "H" and left(BOMU01T.BOMREC_KAYNAKCODE(i),4) = "Y 01" then
			BOMU01T.BOMREC_KAYNAK0(i) = N24
		elseif BOMU01T.BOMREC_INPUTTYPE(i) = "H" and left(BOMU01T.BOMREC_KAYNAKCODE(i),4) = "Y 11" then
			BOMU01T.BOMREC_KAYNAK0(i) = N25
		end if
	Next

Function GetTPA(Tablo, PersonelKodu, TPAType)
	Dim TPAArray(1)
	TPAArray(0) = 0
	TPAArray(1) = 0
	Set TPA = Doc.Select2("WEQE", Tablo, "KOD='"+PersonelKodu+"'&ODEMETURU='"+TPAType+"'", "", -1)
	TPAArray(1) = TPA.GetRowCount()
	if TPA.GetRowCount() > 0 Then
		TPA.GroupBy("KOD")
		TPAArray(0) = TPA.OR_BORCFIYAT(1)
	End if
	GetTPA = TPAArray
	xret= doc.DeleteTableObject("WEQE")
End Function

Sub Makro1()
	Kod=trim(BOMU01E.MAMULCODE)
	YEn=Doc.GetKartValue("STOK00","B_EN",Kod,-1)
	YUz=Doc.GetKartValue("STOK00","B_YUKSELIK",Kod,-1)
	YKutuBoyu=Yuz+(2*YEn)+8
	YHiz=Doc.GetUDFValue("STOK00",Kod,"HIZ",0)
	Gramaj=Doc.CDbl(Doc.GetUDFValue("STOK00",Kod,"GR",0))
	koli=Doc.CDbl(Doc.GetUDFValue("STOK00",Kod,"KOLI",0))

	KalipSayisi=Doc.CDbl(Doc.GetUDFValue("STOK00",Kod,"RENKS",0))
	AmbalajKodu=trim(Doc.GetUDFValue("STOK00",Kod,"KOLIKODU",0))
	AmbalajMiktar=Doc.CDbl(Doc.GetUDFValue("STOK00",Kod,"AMBALAJMIK",0))
	Set Stok=Doc.Select1("sdsdsd","STOK00","KOD='"+AmbalajKodu+"'","",-1)
	AmbalajKoduEni=Doc.CDbl(Stok.B_EN)
	AmbalajKoduBoyu=Doc.CDbl(Stok.B_BOY)
	AmbalajKoduAgirlik=Doc.CDbl(Stok.B_AGIRLIK)
	For i= 1 to BOMU01T.GetRecCount()
		if trim(BOMU01T.BOMREC_KAYNAKCODE(i))="H" then
			MKod=trim(BOMU01T.BOMREC_KAYNAKCODE(i))
		end if
		N17=Doc.GetKartValue("STOK00","B_B_TECH_17",Kod,-1)
		N24=Doc.GetKartValue("STOK00","B_B_TECH_24",Kod,-1)
		N25=Doc.GetKartValue("STOK00","B_B_TECH_25",Kod,-1)
		N18=Doc.GetKartValue("STOK00","B_B_TECH_18",Kod,-1)
		N3=Doc.GetKartValue("STOK00","B_B_TECH_3",Kod,-1)
		N4=Doc.GetKartValue("STOK00","B_B_TECH_4",Kod,-1)
		MamulMiktar=N18*100
		BOMU01E.MAMULMIKTAR=MamulMiktar
		if BOMU01T.BOMREC_INPUTTYPE(i)="H" and Mid(trim(BOMU01T.BOMREC_KAYNAKCODE(i)),2,1)="B" then
			BobinMiktar=N3*N4*Gramaj/100000
			BOMU01T.BOMREC_KAYNAK0(i)=BobinMiktar
		elseif BOMU01T.BOMREC_INPUTTYPE(i)="H" and Mid(trim(BOMU01T.BOMREC_KAYNAKCODE(i)),2,1)="T" then
			BOMU01T.BOMREC_KAYNAK0(i)=100
		end if
		if BOMU01T.BOMREC_INPUTTYPE(i)="H" and left(trim(BOMU01T.BOMREC_KAYNAKCODE(i)),4)="Y 03" then
			BOMU01T.BOMREC_KAYNAK1(i)=KalipSayisi
		end if

		if BOMU01T.BOMREC_INPUTTYPE(i)="H" and left(trim(BOMU01T.BOMREC_KAYNAKCODE(i)),4)="A 02" then
			BOMU01T.BOMREC_KAYNAK0(i)=(AmbalajKoduEni*AmbalajKoduBoyu/1000000)*AmbalajKoduAgirlik*AmbalajMiktar*doc.div(BOMU01T.BOMREC_MAMULMIKTAR(i), koli)
		end if
	
		if trim(BOMU01T.BOMREC_INPUTTYPE(i))="I" then
			OpKod=trim(BOMU01T.BOMREC_OPERASYON(i))
			AyarSetup=Doc.CDbl(Doc.GetUDFValue("IMLT01",OpKod,"AYARS",0))
			CTP1=Doc.CDbl(Doc.GetUDFValue("IMLT01",OpKod,"BSURE",0))
			BirimAyiklama=Doc.Cdbl(Doc.GetUdfValue("IMLT01",OpKod,"KUTUS",0))
			Sure=Doc.CDbl(Doc.GetUDFValue("IMLT01",OpKod,"SURE",0))
			Kesim=Doc.CDbl(Doc.GetUDFValue("IMLT01",OpKod,"KESIM",0))
		end if
		if BOMU01T.BOMREC_INPUTTYPE(i)="I" then
			BOMU01T.BOMREC_KAYNAK1(i)=AyarSetup
			BOMU01T.BOMREC_KAYNAK0(i)=Kesim
		end if		
		if BOMU01T.BOMREC_INPUTTYPE(i)="H" and left(trim(BOMU01T.BOMREC_KAYNAKCODE(i)),4)="Y 03" then
			KalipAdet=BOMU01T.BOMREC_KAYNAK1(i)
		end if
		if BOMU01T.BOMREC_INPUTTYPE(i)="I" and trim(left(BOMU01T.BOMREC_KAYNAKCODE(i),2))="OM" or trim(left(BOMU01T.BOMREC_KAYNAKCODE(i),2))="TM" Then
			TSure=Doc.Div((N17*100),Sure)
			BOMU01T.BOMREC_KAYNAK0(i)=TSure
		end if

		if BOMU01T.BOMREC_INPUTTYPE(i)="I" and trim(left(BOMU01T.BOMREC_KAYNAKCODE(i),4))="KM02" Then
			KesimSuresi=(N4*100)/(100*Sure*60)
			BOMU01T.BOMREC_KAYNAK0(i)=KesimSuresi
		end if
		
		if BOMU01T.BOMREC_INPUTTYPE(i)="I" and trim(left(BOMU01T.BOMREC_KAYNAKCODE(i),4))="CP01" Then
			CTP=Doc.CDbl((KalipAdet*CTP1))
			BOMU01T.BOMREC_KAYNAK1(i)=CTP
		end if
		if BOMU01T.BOMREC_INPUTTYPE(i)="I" and trim(left(BOMU01T.BOMREC_operasyon(i),4))="OAYK" Then
			AyiklamaSuresi=BirimAyiklama*MamulMiktar
			BOMU01T.BOMREC_KAYNAK0(i)=AyiklamaSuresi
		end if
		if BOMU01T.BOMREC_INPUTTYPE(i)="H" and left(trim(BOMU01T.BOMREC_KAYNAKCODE(i)),4)="Y 07" then
			Sonuc=(Yuz*N18*100*0.00033)/1000
			BOMU01T.BOMREC_KAYNAK0(i)=Sonuc
		end if
		if BOMU01T.BOMREC_INPUTTYPE(i)="I" and  trim(BOMU01T.BOMREC_KAYNAKCODE(i))="YM01" OR trim(BOMU01T.BOMREC_KAYNAKCODE(i))="YM02" then
			Sonuc1=((YHiz*1000)/YKutuBoyu)*60
			Sonuc2=Doc.Div((N18*100),Sonuc1)
			BOMU01T.BOMREC_KAYNAK0(i)=Sonuc2	
		end if
		if BOMU01T.BOMREC_INPUTTYPE(i)="H" and left(trim(BOMU01T.BOMREC_KAYNAKCODE(i)),4)="Y 01" then
			BOMU01T.BOMREC_KAYNAK0(i)=N24
		elseif BOMU01T.BOMREC_INPUTTYPE(i)="H" and left(trim(BOMU01T.BOMREC_KAYNAKCODE(i)),4)="Y 11" then
			BOMU01T.BOMREC_KAYNAK0(i)=N25
		end if
	Next
End Sub

Sub Makro1()
	call KoliveAmbalajHesapla()

Sub KoliveAmbalajHesapla()
	For i= 1 to STOK20TI.GetRecCount()
		STOK20TI.SetCurrentRow(i)
		HKOD=STOK20TI.KOD(i)
		MKOD=STOK20TI.KARSICODE(i)
		MamulMiktari = 0
		Set S20T = Doc.Select2("d33455t", "STOK20T", "KARSITRNUM='" & STOK20TI.KARSITRNUM(i) & "'", "", -1)
		if S20T.GetRowCount() > 0 Then
			S20T.GroupBy("KARSITRNUM")
			MamulMiktari = S20T.SF_MIKTAR
		End if
		If left(HKOD,4)="A 01" Then
			KoliXxx = Doc.CDbl(Doc.GetUDFValue("STOK00",MKod,"KOLI",0))
			KoliMik = Doc.Ceil(Doc.Div(MamulMiktari,KoliXxx))
			msgbox MamulMiktari
			STOK20TI.SF_MIKTAR(i) = KoliMik
		ElseIf left(HKOD,4)="A 02" Then
			'miktar = Doc.Ceil(Doc.Div(MamulMiktari,KoliMik))
			KTAAmbMik = Doc.CDbl(Doc.GetUDFValue("STOK00",MKod,"AMBALAJMIK",0))
			KoliXxx = Doc.CDbl(Doc.GetUDFValue("STOK00",MKod,"KOLI",0))
			KoliMik = Doc.Ceil(Doc.Div(MamulMiktari,KoliXxx))
			KoliKodu = Doc.GetUDFValue("STOK00",MKod,"KOLIKODU",0)
			En = Doc.GetUDFValue("STOK00",KoliKodu,"B_EN",0)
			Boy = Doc.GetUDFValue("STOK00",KoliKodu,"B_BOY",0)
			Agirlik = Doc.GetUDFValue("STOK00",KoliKodu,"B_AGIRLIK",0)
			AmbMik = KoliMik*En*Boy*Agirlik*KTAAmbMik/1000000
			STOK20TI.SF_MIKTAR(i) = AmbMik
		End If

End Sub
