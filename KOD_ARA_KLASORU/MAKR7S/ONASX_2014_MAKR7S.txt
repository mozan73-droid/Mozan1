Sub SendMail(MailTo, MailHeader, MailMessage, DosyaAdi)
	'on error resume next

	cdoSendUsingPickup = 1

	cdoAnonymous = 0
	cdoBasic = 1
	cdoNTLM = 2 'NTLM
	
	Kullanici = Doc.Getd7UserName

	mailUser = Trim(Doc.GetDBFieldValue("UDEF", "EMAILKULLANICIADI", Kullanici, ""))
	mailPass = Trim(Doc.GetDBFieldValue("UDEF", "DATAMODEMNO", Kullanici, ""))
	mailFrom = Trim(Doc.GetDBFieldValue("UDEF", "EMAIL", Kullanici, ""))
	mailSMTPPort = Trim(Doc.GetDBFieldValue("UDEF", "SMTPSERVERPORT", Kullanici, ""))

	Set CDO = CreateObject("CDO.Message")

	CDO.From = Trim(MailFrom)
	CDO.To = MailTo
	CDO.Subject = MailHeader
	CDO.TextBody = MailMessage
	
	'If Trim(DosyaAdi(3)) <> "" Then
	'	CDO.AddAttachment("file://" & DosyaAdi(3))
	'End if
	
	CDO.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/sendusing") = cdoSendUsingPort 
	
	CDO.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/smtpserver") = mailserver
	
	CDO.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/smtpauthenticate") = cdoBasic
	
	CDO.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/sendusername") = mailUser
	
	CDO.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/sendpassword") = mailPass
	
	CDO.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/smtpserverport") = mailSMTPPort
	
	CDO.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/smtpusessl") = False
	
	CDO.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/smtpconnectiontimeout") = 60
	
	CDO.Configuration.Fields.Update
	
	On Error Resume Next
	CDO.Send
	
	If DosyaAdi(2) = "RafOmru" Then
		If Err.Number = 0 Then
			Set CDO = Nothing

		Else
			Set CDO = Nothing
			Msgbox "RAf Ömrü Bilgi E-maili Gönderilemedi. !!!"
		End If
	Else
		Set CDO = Nothing
	End If

End Sub
