Function StrToArray(Yazi, Ayirac)
  Dim TmpArray()
  Dim AyiracSayisi
  Dim TmpText
  Yazi = Trim(Yazi)
  Yazi = Temizle(Yazi, Ayirac)
  AyiracPos = InStr(1, Yazi, Ayirac)
  If AyiracPos = 0 Then
    ReDim TmpArray(0)
    TmpArray(0) = Yazi
  Else
    pos = 0
    oldpos = 0
    AyiracSayisi = CharNum(Yazi, Ayirac)
    ReDim TmpArray(AyiracSayisi)
    For i = 0 To 10000
      pos = pos + 1
      If pos > Len(Yazi) And oldpos > Len(Yazi) Then Exit For
      pos = InStr(pos, Yazi, Ayirac)
      If pos > 0 Then
        TmpText = Mid(Yazi, oldpos + 1, pos - oldpos - 1)
        TmpArray(i) = TmpText
        oldpos = pos
      Else
        TmpText = Mid(Yazi, oldpos + 1, Len(Yazi) - oldpos)
        TmpArray(i) = TmpText
        Exit For
      End If
    Next
  End If
  StrToArray = TmpArray
End Function
Function Temizle(Yazi, Ayirac)
  Yazi = Trim(Yazi)
  While Right(Yazi, 1) = Ayirac
    Yazi = trim(Left(Yazi, Len(Yazi) - 1))
  Wend
  While Left(Yazi, 1) = Ayirac
    Yazi = trim(right(Yazi, Len(Yazi) - 1))
  Wend
  For i = 1 To 10
    strT = String(2, Ayirac)
    Yazi = Replace(Yazi, strT, Ayirac)
  Next
  Temizle = Yazi
End Function
Function CharNum(Yazi, FindChar)
  If InStr(1, Yazi, FindChar) = 0 Then
    CharNum = 0
  Else
    pos = 0
    For i = 1 To 10000
      pos = pos + 1
      If pos > Len(Yazi) Then Exit For
      pos = InStr(pos, Yazi, FindChar)
      If pos > 0 Then
        CharNum = CharNum + 1
      Else
        Exit For
      End If
    Next
  End If
End Function

Function StrToArray(Yazi, Ayirac)
  Dim TmpArray()
  Dim AyiracSayisi
  Dim TmpText
  Yazi = Trim(Yazi)
  Yazi = Temizle(Yazi, Ayirac)
  AyiracPos = InStr(1, Yazi, Ayirac)
  If AyiracPos = 0 Then
    ReDim TmpArray(0)
    TmpArray(0) = Yazi
  Else
    pos = 0
    oldpos = 0
    AyiracSayisi = CharNum(Yazi, Ayirac)
    ReDim TmpArray(AyiracSayisi)
    For i = 0 To 10000
      pos = pos + 1
      If pos > Len(Yazi) And oldpos > Len(Yazi) Then Exit For
      pos = InStr(pos, Yazi, Ayirac)
      If pos > 0 Then
        TmpText = Mid(Yazi, oldpos + 1, pos - oldpos - 1)
        TmpArray(i) = TmpText
        oldpos = pos
      Else
        TmpText = Mid(Yazi, oldpos + 1, Len(Yazi) - oldpos)
        TmpArray(i) = TmpText
        Exit For
      End If
    Next
  End If
  StrToArray = TmpArray
End Function
Function Temizle(Yazi, Ayirac)
  Yazi = Trim(Yazi)
  While Right(Yazi, 1) = Ayirac
    Yazi = trim(Left(Yazi, Len(Yazi) - 1))
  Wend
  While Left(Yazi, 1) = Ayirac
    Yazi = trim(right(Yazi, Len(Yazi) - 1))
  Wend
  For i = 1 To 10
    strT = String(2, Ayirac)
    Yazi = Replace(Yazi, strT, Ayirac)
  Next
  Temizle = Yazi
End Function
Function CharNum(Yazi, FindChar)
  If InStr(1, Yazi, FindChar) = 0 Then
    CharNum = 0
  Else
    pos = 0
    For i = 1 To 10000
      pos = pos + 1
      If pos > Len(Yazi) Then Exit For
      pos = InStr(pos, Yazi, FindChar)
      If pos > 0 Then
        CharNum = CharNum + 1
      Else
        Exit For
      End If
    Next
  End If
End Function

Function StrToArray(Yazi, Ayirac)
  Dim TmpArray()
  Dim AyiracSayisi
  Dim TmpText
  Yazi = Trim(Yazi)
  Yazi = Temizle(Yazi, Ayirac)
  AyiracPos = InStr(1, Yazi, Ayirac)
  If AyiracPos = 0 Then
    ReDim TmpArray(0)
    TmpArray(0) = Yazi
  Else
    pos = 0
    oldpos = 0
    AyiracSayisi = CharNum(Yazi, Ayirac)
    ReDim TmpArray(AyiracSayisi)
    For i = 0 To 10000
      pos = pos + 1
      If pos > Len(Yazi) And oldpos > Len(Yazi) Then Exit For
      pos = InStr(pos, Yazi, Ayirac)
      If pos > 0 Then
        TmpText = Mid(Yazi, oldpos + 1, pos - oldpos - 1)
        TmpArray(i) = TmpText
        oldpos = pos
      Else
        TmpText = Mid(Yazi, oldpos + 1, Len(Yazi) - oldpos)
        TmpArray(i) = TmpText
        Exit For
      End If
    Next
  End If
  StrToArray = TmpArray
End Function
Function Temizle(Yazi, Ayirac)
  Yazi = Trim(Yazi)
  While Right(Yazi, 1) = Ayirac
    Yazi = trim(Left(Yazi, Len(Yazi) - 1))
  Wend
  While Left(Yazi, 1) = Ayirac
    Yazi = trim(right(Yazi, Len(Yazi) - 1))
  Wend
  For i = 1 To 10
    strT = String(2, Ayirac)
    Yazi = Replace(Yazi, strT, Ayirac)
  Next
  Temizle = Yazi
End Function
Function CharNum(Yazi, FindChar)
  If InStr(1, Yazi, FindChar) = 0 Then
    CharNum = 0
  Else
    pos = 0
    For i = 1 To 10000
      pos = pos + 1
      If pos > Len(Yazi) Then Exit For
      pos = InStr(pos, Yazi, FindChar)
      If pos > 0 Then
        CharNum = CharNum + 1
      Else
        Exit For
      End If
    Next
  End If
End Function
