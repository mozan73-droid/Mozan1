Sub SatirYaz(ASN)
	SevkAdresKodu = STOK60E.SHIPTOCODE
	if trim(SevkAdresKodu) = "D1D4F" then 'GÖLCÜK
		Birim = "EA"
	else
		Birim = "AD"
	end if
	CHKodu = STOK60E.CH_KODU
	for i = 1 to S60.GetRecCount()
		if trim(S60.KOD(i)) <> "" then
			Ambalaj = S60.FATURASIZMI(i)
			Kod = S60.KOD(i)
'			if trim(Ambalaj) = "E" then
'				StokKodu = Kod
'			else
				StokKodu = trim(Doc.GetCHStokKodu(Kod,CHKodu))
'			end if
			Miktar = S60.SF_MIKTAR(i)
'			Birim = S60.SF_SF_UNIT(i)
			PART_CUM_QTY = 0
			PART_PRIOR_CUM_QTY = 0 
			SiparisNo = S60.SIPNO(i)
			MusteriSipNo = Doc.GetDBFieldValue("STOK40E","CHSIPNO",SiparisNo,"")
			PART_PO_NO = trim(MusteriSipNo)
			if trim(Ambalaj) <> "E" then
				KontSayisi = S60.BRUTAGIRLIK(i)
				ParcaTuru = "PP"
				if trim(cstr(KontSayisi)) = "" or trim(cstr(KontSayisi)) = "0" then
					kontParcaSayisi = Doc.GetDBFieldValue("STOK00","B_B_TECH_1",Kod,"")
'					kontParcaSayisi = Doc.GetDBFieldValue("STOK00","B_MINPARTIBUYUKLUGU",Kod,"")
					kontParcaSayisi = Doc.Round(kontParcaSayisi,0)
					ParcaKontSayisi = Doc.Div(Miktar,kontParcaSayisi)
					ParcaKontSayisi = Doc.Ceil(ParcaKontSayisi)
				else
					ParcaKontSayisi = Doc.CDbl(KontSayisi)
					kontParcaSayisi = Doc.Div(Miktar,ParcaKontSayisi)
					kontParcaSayisi = Doc.Ceil(kontParcaSayisi)
				end if
				KonteynerTipi = "BOX90"
			else
				ParcaTuru = "CT"
				kontParcaSayisi = "0000"
				ParcaKontSayisi = "0000"
				KonteynerTipi = ""
			end if
			ASN.WriteLine(cTab(3) & "<PART>")        
			ASN.WriteLine(cTab(4) & "<NO_PART>" & trim(StokKodu) & "</NO_PART>")
			ASN.WriteLine(cTab(4) & "<PART_NO_TYPE>" & ParcaTuru & "</PART_NO_TYPE>")
			ASN.WriteLine(cTab(4) & "<PART_QUANTITY>" & Miktar & "</PART_QUANTITY>")
			ASN.WriteLine(cTab(4) & "<PART_UOM>" & Birim & "</PART_UOM>")
			ASN.WriteLine(cTab(4) & "<PART_CUM_QTY>" & PART_CUM_QTY & "</PART_CUM_QTY>")
			ASN.WriteLine(cTab(4) & "<PART_PRIOR_CUM_QTY>" & PART_PRIOR_CUM_QTY & "</PART_PRIOR_CUM_QTY>")
			ASN.WriteLine(cTab(4) & "<PART_PO_NO>" & PART_PO_NO & "</PART_PO_NO>")
			ASN.WriteLine(cTab(4) & "<CONTAINERS>")
			ASN.WriteLine(cTab(5) & "<CONTAINER>")
			ASN.WriteLine(cTab(6) & "<CONTAINER_TYPE_CODE>" & KonteynerTipi & "</CONTAINER_TYPE_CODE>")
			ASN.WriteLine(cTab(6) & "<PARTS_PER_CONT>" & KontParcaSayisi & "</PARTS_PER_CONT>")
			ASN.WriteLine(cTab(6) & "<CONTS_PER_PART>" & ParcaKontSayisi & "</CONTS_PER_PART>")
			ASN.WriteLine(cTab(5) & "</CONTAINER>")
			ASN.WriteLine(cTab(4) & "</CONTAINERS>")
			ASN.WriteLine(cTab(3) & "</PART>")        
		end if
	Next
End Sub

Function AmbalajVarMi()
	AmbalajVarMi = False
	for i = 1 to STOK60T.GetRecCount()
		Ambalaj = STOK60T.FATURASIZMI(i)
		if trim(Ambalaj) = "E" then
			AmbalajVarMi = True
			Exit Function
		end if
	Next
End Function
Function KodKontrol()
	for i = 1 to STOK60T.GetRecCount()
		StokKodu = STOK60T.KOD(i)
		if trim(StokKodu) <> "" then
			CHKodu = STOK60E.CH_KODU
			CHStokKodu = trim(Doc.GetCHStokKodu(StokKodu,CHKodu))
			Ambalaj = STOK60T.FATURASIZMI(i)
			if trim(CHStokKodu) = "" then
				Hata = Hata & StokKodu & vbcrlf
			end if
		end if
	Next
	if Hata <> "" then
		Hata = Hata & "Ford stok kodu karþýlýðý girilmemiþ."
	end if
	KodKontrol = Hata
End Function
Function SandikAdediKontrol()
	for i = 1 to STOK60T.GetRecCount()
		StokKodu = STOK60T.KOD(i)
		if trim(StokKodu) <> "" then
			kontParcaSayisi = Doc.GetDBFieldValue("STOK00","B_B_TECH_1",StokKodu,"")
			Ambalaj = STOK60T.FATURASIZMI(i)
			BrutAgirlik = STOK60T.BRUTAGIRLIK(i)
			if trim(Ambalaj) = "" and kontParcaSayisi = 0 and trim(BrutAgirlik) = "" then
				Hata = Hata & StokKodu & vbcrlf
			end if
		end if
	Next
	if Hata <> "" then
		Hata = Hata & "Konteyner parça adedi girilmemiþ."
	end if
	SandikAdediKontrol = Hata
End Function

Function GetMiktarTarih(StokKodu, Miktar, Tarih, Attrib)
	Dim MiktarArray()
	Redim MiktarArray(1,0)
	if Attrib = "G" then
		MiktarArray(0,0) = Miktar
		MiktarArray(1,0) = Tarih
		GetMiktarTarih = MiktarArray
		exit Function
	end if

	Gun = GetHaftaninGunu(Tarih)
	PartiBuyuklugu = Doc.GetDBFieldValue("STOK00","B_MINPARTIBUYUKLUGU", StokKodu, "")
	if PartiBuyuklugu = 0 then PartiBuyuklugu = Miktar
	if PartiBuyuklugu >= Miktar then
		MiktarArray(0,0) = Miktar
		MiktarArray(1,0) = Tarih
		GetMiktarTarih = MiktarArray
		exit Function
	end if

	if Gun = 1 then
		KacaBolunecek = 3
	Elseif Gun > 1 and Gun <= 3 then 
		KacaBolunecek = 2
	elseif Gun > 3 then
		KacaBolunecek = 1
		MiktarArray(0,0) = Miktar
		if Gun > 5 then
			SipTarih = Tarih
		Else
			SipTarih = GetTarihGun(Tarih, 5)
		end if
		MiktarArray(1,0) = SipTarih
		GetMiktarTarih = MiktarArray

		exit Function		
	end if

	GunlukMiktar = Doc.Div(Miktar, KacaBolunecek)
	GunlukMiktar = Doc.Ceil(Doc.Div(GunlukMiktar, PartiBuyuklugu)) * PartiBuyuklugu

	KalanMiktar = Miktar
	While KalanMiktar > 0
		sirano = siraNo + 1
		if KalanMiktar > GunlukMiktar then
			SipMiktari = GunlukMiktar
		Else
			SipMiktari = KalanMiktar
		end if
		KalanMiktar = KalanMiktar - SipMiktari
		MiktarArray(0, uBound(MiktarArray,2)) = SipMiktari
		if KalanMiktar > 0 then
			Redim Preserve MiktarArray(1, uBound(MiktarArray,2) + 1)
		end if
		if sirano > 100 then
			msgbox "Döngü 100 'den fazla devam etti. Hatayý kontrol ediniz." & vbcrlf & "Ýþlem Kesilmiþtir" 
			Exit Function
		end if
	Wend
	GunSayisi = UBound(MiktarArray, 2) + 1
	if GunSayisi = 2 then
		MiktarArray(1,0) = GetTarihGun(Tarih, 3)
		MiktarArray(1,1) = GetTarihGun(Tarih, 5)
	Elseif GunSayisi = 3 then
		MiktarArray(1,0) = GetTarihGun(Tarih, 1)
		MiktarArray(1,1) = GetTarihGun(Tarih, 3)
		MiktarArray(1,2) = GetTarihGun(Tarih, 5)
	end if
	GetMiktarTarih = MiktarArray
End Function
Function GetHaftaninGunu(Tarih)
	dim d7Date, k, Gun

Function GetTarihGun(Tarih, KacinciGun)
	HatfaninGunu = GetHaftaninGunu(Tarih)
	Fark = KacinciGun - HatfaninGunu
	GetTarihGun = Doc.StrDate_Increment(Tarih, Fark)
End Function

Function KodKontrol()
	for i = 1 to STOK60T.GetRecCount()
		StokKodu = STOK60T.KOD(i)
		if trim(StokKodu) <> "" then
			CHKodu = STOK60E.CH_KODU
			CHStokKodu = trim(Doc.GetCHStokKodu(StokKodu,CHKodu))
			Ambalaj = STOK60T.FATURASIZMI(i)
			if trim(CHStokKodu) = "" then
				Hata = Hata & StokKodu & vbcrlf
			end if
		end if
	Next
	if Hata <> "" then
		Hata = Hata & "Tofaþ stok kodu karþýlýðý girilmemiþ."
	end if
	KodKontrol = Hata
End Function

Function GetTofasBirim(Birim)
	Birim = UCase(trim(Birim))
	Birim = Replace(Birim, ".", "")
	TBirim = Birim
	if Birim="KGR" or Birim="KILO" or Birim="KÝLO" or Birim="KILOGR" or Birim="KÝLOGR" Then
		TBirim = "KG"
	Elseif Birim="M" or Birim="METRE" or Birim="MTR" Then
		TBirim = "MT"
	Elseif Birim="ADET" or Birim="ADT" Then
		TBirim = "AD"
	Elseif Birim="GRAM" Then
		TBirim = "GR"
	End if
	GetTofasBirim = TBirim
End Function
