SELECT 

' ' SELECTED,
MMPS10T.EVRAKNO MPSNO,
MMPS10E.CREA_TARIH MPSTARIH,

(CASE WHEN MMPS10E.ANAMPS IS NULL THEN
MMPS10TM.MUSTERISIPINO ELSE
M10TMANAMPS.MUSTERISIPINO
END) SIPNO,

MMPS10T.REFTEXT02 HAT,

(CASE WHEN MMPS10E.ANAMPS IS NULL THEN 
CARI00.AD ELSE
COALESCE(ANAMPSCARI00.AD, 'STOK') 
END) MUSTERI ,

R_MAMULKODU KOD,
STOK00.AD,
R_YMK_YMPAKETICERIGI MIKTAR,
VRI_RECETE_A ASAMA,
R_TMYMAMULMIKTAR TMIKTAR,
R_YMK_YMPAKETICERIGI - R_TMYMAMULMIKTAR BMIKTAR,

MMPS10E.ANAMPS ANAMPSNO,

S20T.OPERATOR_1 OPERATOR,
OPAD,

R_LATEST_E_TARIH UTARIH,

R_KAYNAKKODU TEZGAH,

(CASE WHEN RECTIME2 = '' THEN 
	CASE WHEN RECTYPE = 'U1' Then 'Üretim Basladi' ELSE
		CASE WHEN RECTYPE = 'U2' Then 'Üretim Kismi Bitti' ELSE
			CASE WHEN RECTYPE = 'D1' Then 'Durus Basladi' END END END
ELSE
	CASE WHEN RECTYPE = 'D1' Then 'Durus Bitti' END
END) DURUM,

MMPS10E.NOTLAR_1 'NOT',

GECOSST.AD GECIKME,  

(CASE WHEN 
	CONVERT(VARCHAR, R_LATEST_E_TARIH , 112) < FORMAT(GETDATE(), 'dd/MM/yyyy/mm/dd') Then 'RED'
	ELSE 'GREEN' END) R1

FROM MMPS10T

LEFT JOIN CARI00 ON CARI00.KOD = MMPS10T.R_MUSTERIKODU

LEFT JOIN STOK00 ON STOK00.KOD = MMPS10T.R_MAMULKODU
LEFT JOIN MMPS10E ON MMPS10E.EVRAKNO = MMPS10T.EVRAKNO
LEFT JOIN GECOSST ON GECOSST.EVRAKNO='MPSGS1' AND GECOSST.KOD=MMPS10E.REV_SEBEPLERI_1
LEFT JOIN MMPS10TM ON MMPS10TM.EVRAKNO = MMPS10T.EVRAKNO AND MMPS10TM.MAMULSTOKKODU = MMPS10T.R_MAMULKODU
LEFT JOIN MMPS10TM M10TMANAMPS ON M10TMANAMPS.EVRAKNO = MMPS10E.ANAMPS 
LEFT JOIN CARI00 ANAMPSCARI00 ON ANAMPSCARI00.KOD = M10TMANAMPS.MUSTERIKODU

LEFT JOIN (
  SELECT *
  FROM (
    SELECT 
      SFDC20T.OPERATOR_1, 
      PERS00.AD AS OPAD, 
      RECTYPE, 
      D7_ISLEM_KODU, 
      JOBNO,
	  RECTIME2,
      ROW_NUMBER() OVER (PARTITION BY JOBNO ORDER BY EVRAKNO DESC) AS RN
    FROM SFDC20T 
    LEFT JOIN PERS00 ON PERS00.KOD = SFDC20T.OPERATOR_1
  ) AS Ranked
  WHERE RN = 1
) AS S20T ON S20T.JOBNO = MMPS10T.JOBNO

WHERE 1=1

AND MMPS10T.R_KAYNAKKODU = 'MEKANİK'   --MEKANİK
AND MMPS10T.R_KAYNAKTYPE = 'I'
AND MMPS10T.ACIK_KAPALI<>'K'
AND MMPS10T.R_ACIK_KAPALI<>'K'
AND MMPS10T.R_STATUS='R'
--AND MMPS10T.EVRAKNO='    186195'

ORDER BY UTARIH
