Sub Hesapla(satirno)
	Set Doc2 = Doc.CreateDocumentNoUI("BOMU01","")
	
	ListeMamulKodu = LISTET.MAMULCODE(satirno)
	Call yeni_urunagaci_pasif(Listemamulkodu)

	Set Doc1 = Doc.CreateDocumentNoUI("STOK00","")
	Set S00ESKI = Doc1.GetTableObject("STOK00")
	Set S0UESKI = Doc1.GetTableObject("STOK0U")

	S00ESKI.Empty
	S0UESKI.Empty

	Doc1.Load_Voucher Listemamulkodu
	Doc1.Runcommand("MT_DOC_COPY")

	Doc1.LoadEmpty_Voucher()
	Call stokkartpasif(Listemamulkodu)

	Evrakno = LISTET.EVRAKNO(satirno)
	kacincitezgah=1
' ***** ürün aðacý hesapla *********
	rows = Doc.SelectEQ("BOMU01T","EVRAKNO", Evrakno)
	Set BOMESKI = Doc.GetTableObject("AU_BOMU01T_SELECT")
	BOMESKI.sort("BOMREC_INPUTTYPE+SRNUM")
	
	For i=1 to BOMESKI.getreccount
		Mamulmiktar=BOMESKI.BOMREC_MAMULMIKTAR(i)
		tezgah=BOMESKI.BOMREC_KAYNAKCODE(i)
		Tezgah_grup5=trim(doc.getdbfieldvalue("IMLT00", "GK_5", tezgah, ""))
		fo=BOMESKI.BOMREC_FO_OPTION(i)

		If trim(BOMESKI.BOMREC_INPUTTYPE(i))="H" Then
				hammaddesayisi=hammaddesayisi+1
		ElseIf trim(BOMESKI.BOMREC_INPUTTYPE(i))="I" and trim(tezgah)<> trim(Eskitezgah1) And  trim(tezgah_grup5) <> "H" and trim(fo) <> "1" Then	
				eskitezgah1=BOMESKI.BOMREC_KAYNAKCODE(i)
				tezgahsayisi=tezgahsayisi + 1
				if tezgahsayisi=1 then
					eskitezgah=BOMESKI.BOMREC_KAYNAKCODE(i)
				end if
		m=m+1
		if m=1 then
			notx=BOMESKI.BOMREC_NOT(i)
		end if

		End If
	Next
		

'****** tezgah sayýsý sýfýr ise ürün aðacý açma ******
	If tezgahsayisi=0 then
		exit sub
	End If
'*****************************************************

	

	For j=1 To BOMESKI.getreccount
		mamulno1=(tezgahsayisi-kacincitezgah)*10
		Yenimamulno=string(3-len(mamulno1),"0")&mamulno1
		mamulno2=(tezgahsayisi-kacincitezgah+1)*10
		eskimamulno=string(3-len(mamulno2),"0")&mamulno2
		tip=BOMESKI.BOMREC_INPUTTYPE(j)
				gk1=BOMESKI.GK_1(j)

		yenimamulkodu=trim(ListeMamulKodu)&"-"&yenimamulno
		eskimamulkodu=trim(ListeMamulKodu)&"-"&eskimamulno

		Tezgah_grup3=trim(doc.getdbfieldvalue("IMLT00", "GK_3", eskitezgah, ""))
		If trim(tip)="I"  then
				yenitezgah=BOMESKI.BOMREC_KAYNAKCODE(j)
				Tezgah_grup5=trim(doc.getdbfieldvalue("IMLT00", "GK_5", yenitezgah, ""))
				fo=BOMESKI.BOMREC_FO_OPTION(j)
					
			If Trim(tezgah_grup5)<> "H" and trim(fo)<>"1" then
					
				if trim(yenitezgah) <> trim(eskitezgah) then

					kacincitezgah=kacincitezgah + 1
					mamulno1=(tezgahsayisi-kacincitezgah)*10
					Yenimamulno=string(3-len(mamulno1),"0")&mamulno1
					tip=BOMESKI.BOMREC_INPUTTYPE(j)
					yenimamulkodu=trim(ListeMamulKodu)&"-"&yenimamulno

					mamulno2=(tezgahsayisi-kacincitezgah+1)*10
					eskimamulno=string(3-len(mamulno2),"0")&mamulno2
					eskimamulkodu=trim(ListeMamulKodu)&"-"&eskimamulno
					eskitezgah=yenitezgah
					Tezgah_grup3=trim(doc.getdbfieldvalue("IMLT00", "GK_3", eskitezgah, ""))
					sorgu=1
notx=BOMESKI.BOMREC_NOT(j)
					Call stokac(yenimamulkodu, sorgu)
					call UrunAgaciAc(j, yenimamulkodu, sorgu)
					Tezgah_grup4=trim(doc.getdbfieldvalue("IMLT00", "GK_4", yenitezgah, ""))
					Call Stokgrup4ac(eskimamulkodu, sorgu)
					if right(trim(yenimamulkodu), 3)="000" then
						Call Stokgrup4ac(yenimamulkodu, sorgu)
					End If
				else
					If hammaddesayisi=0 then
						sorgu=1 
						call UrunAgaciAc(j, yenimamulkodu, sorgu)
					else
						sorgu=0	
						call UrunAgaciAc(j, yenimamulkodu, sorgu)
					End If
				End If 
			End If
		else
		
			kacincihammadde=kacincihammadde+1
			if kacincihammadde =1 then
				sorgu=1

				Call stokac(yenimamulkodu, sorgu)
				call UrunAgaciAc(j, yenimamulkodu, sorgu)
			else
				sorgu=0
				call UrunAgaciAc(j, yenimamulkodu, sorgu)
			End If
		End If
	Next

Sub Hesapla(satirno)

		Tezgah_grup5=trim(doc.getdbfieldvalue("IMLT00", "GK_5", tezgah, ""))
		fo=BOMESKI.BOMREC_FO_OPTION(i)

		If trim(BOMESKI.BOMREC_INPUTTYPE(i))="H" Then
				hammaddesayisi=hammaddesayisi+1
		ElseIf trim(BOMESKI.BOMREC_INPUTTYPE(i))="I" and trim(tezgah)<> trim(Eskitezgah1) And  trim(tezgah_grup5) <> "H" and trim(fo) <> "1" Then	
				eskitezgah1=BOMESKI.BOMREC_KAYNAKCODE(i)
				tezgahsayisi=tezgahsayisi + 1
				if tezgahsayisi=1 then
					eskitezgah=BOMESKI.BOMREC_KAYNAKCODE(i)
				end if
		m=m+1
		if m=1 then
			notx=BOMESKI.BOMREC_NOT(i)
		end if

		End If
	Next
		

'****** tezgah sayýsý sýfýr ise ürün aðacý açma ******
	If tezgahsayisi=0 then
		exit sub
	End If
'*****************************************************


	For j=1 To BOMESKI.getreccount
		mamulno1=(tezgahsayisi-kacincitezgah)*10
		Yenimamulno=string(3-len(mamulno1),"0")&mamulno1
		mamulno2=(tezgahsayisi-kacincitezgah+1)*10
		eskimamulno=string(3-len(mamulno2),"0")&mamulno2
		tip=BOMESKI.BOMREC_INPUTTYPE(j)
				gk1=BOMESKI.GK_1(j)

		eskimamulkodu=trim(ListeMamulKodu)&"-"&eskimamulno

		If trim(tip)="I"  then
				yenitezgah=BOMESKI.BOMREC_KAYNAKCODE(j)
				Tezgah_grup5=trim(doc.getdbfieldvalue("IMLT00", "GK_5", yenitezgah, ""))
				fo=BOMESKI.BOMREC_FO_OPTION(j)
					
			If Trim(tezgah_grup5)<> "H" and trim(fo)<>"1" then
					
				if trim(yenitezgah) <> trim(eskitezgah) then

					kacincitezgah=kacincitezgah + 1
					mamulno1=(tezgahsayisi-kacincitezgah)*10
					Yenimamulno=string(3-len(mamulno1),"0")&mamulno1
					tip=BOMESKI.BOMREC_INPUTTYPE(j)
					yenimamulkodu=trim(ListeMamulKodu)&"-"&yenimamulno

					mamulno2=(tezgahsayisi-kacincitezgah+1)*10
					eskimamulno=string(3-len(mamulno2),"0")&mamulno2
					eskimamulkodu=trim(ListeMamulKodu)&"-"&eskimamulno
					eskitezgah=yenitezgah
					Tezgah_grup3=trim(doc.getdbfieldvalue("IMLT00", "GK_3", eskitezgah, ""))
					sorgu=1
					notx=BOMESKI.BOMREC_NOT(j)
					Call stokac(yenimamulkodu, sorgu)
					call UrunAgaciAc(j, yenimamulkodu, sorgu)
					Tezgah_grup4=trim(doc.getdbfieldvalue("IMLT00", "GK_4", yenitezgah, ""))
					Call Stokgrup4ac(eskimamulkodu, sorgu)
					if right(trim(yenimamulkodu), 3)="000" then
						Call Stokgrup4ac(yenimamulkodu, sorgu)
					End If
				else
					If hammaddesayisi=0 then
						sorgu=1 
						call UrunAgaciAc(j, yenimamulkodu, sorgu)
					else
						sorgu=0	
						call UrunAgaciAc(j, yenimamulkodu, sorgu)
					End If
				End If 
			End If
		else
		
			kacincihammadde=kacincihammadde+1
			if kacincihammadde =1 then
				sorgu=1

				Call stokac(yenimamulkodu, sorgu)
				call UrunAgaciAc(j, yenimamulkodu, sorgu)
			else
				sorgu=0
				call UrunAgaciAc(j, yenimamulkodu, sorgu)
			End If
		End If
	Next

Sub UrunAgaciAc(satirno, mamulkodu, sorgu)

Sub Hesapla(satirno)
	Set Doc2 = Doc.CreateDocumentNoUI("BOMU01","")
	
	ListeMamulKodu = LISTET.MAMULCODE(satirno)
	Call yeni_urunagaci_pasif(Listemamulkodu)

	Set Doc1 = Doc.CreateDocumentNoUI("STOK00","")
	Set S00ESKI = Doc1.GetTableObject("STOK00")
	Set S0UESKI = Doc1.GetTableObject("STOK0U")

	S00ESKI.Empty
	S0UESKI.Empty

	Doc1.Load_Voucher Listemamulkodu
	Doc1.Runcommand("MT_DOC_COPY")

	Doc1.LoadEmpty_Voucher()
	Call stokkartpasif(Listemamulkodu)

	Evrakno = LISTET.EVRAKNO(satirno)
	kacincitezgah=1
' ***** ürün aðacý hesapla *********
	rows = Doc.SelectEQ("BOMU01T","EVRAKNO", Evrakno)
	Set BOMESKI = Doc.GetTableObject("AU_BOMU01T_SELECT")
	BOMESKI.sort("BOMREC_INPUTTYPE")

	For i=1 to BOMESKI.getreccount
		Mamulmiktar=BOMESKI.BOMREC_MAMULMIKTAR(i)
		tezgah=BOMESKI.BOMREC_KAYNAKCODE(i)
		Tezgah_grup5=trim(doc.getdbfieldvalue("IMLT00", "GK_5", tezgah, ""))
		fo=BOMESKI.BOMREC_FO_OPTION(i)

		If trim(BOMESKI.BOMREC_INPUTTYPE(i))="H" Then
				hammaddesayisi=hammaddesayisi+1
		ElseIf trim(BOMESKI.BOMREC_INPUTTYPE(i))="I" and trim(tezgah)<> trim(Eskitezgah1) And  trim(tezgah_grup5) <> "H" and trim(fo) <> "1" Then	
				eskitezgah1=BOMESKI.BOMREC_KAYNAKCODE(i)
				tezgahsayisi=tezgahsayisi + 1
				if tezgahsayisi=1 then
					eskitezgah=BOMESKI.BOMREC_KAYNAKCODE(i)
				end if
		End If
	Next
		

'****** tezgah sayýsý sýfýr ise ürün aðacý açma ******
	If tezgahsayisi=0 then
		exit sub
	End If
'*****************************************************

	

	For j=1 To BOMESKI.getreccount
		mamulno1=(tezgahsayisi-kacincitezgah)*10
		Yenimamulno=string(3-len(mamulno1),"0")&mamulno1
		mamulno2=(tezgahsayisi-kacincitezgah+1)*10
		eskimamulno=string(3-len(mamulno2),"0")&mamulno2
		tip=BOMESKI.BOMREC_INPUTTYPE(j)

		yenimamulkodu=trim(ListeMamulKodu)&"-"&yenimamulno
		eskimamulkodu=trim(ListeMamulKodu)&"-"&eskimamulno

		Tezgah_grup3=trim(doc.getdbfieldvalue("IMLT00", "GK_3", eskitezgah, ""))
		If trim(tip)="I"  then
				yenitezgah=BOMESKI.BOMREC_KAYNAKCODE(j)
				Tezgah_grup5=trim(doc.getdbfieldvalue("IMLT00", "GK_5", yenitezgah, ""))
				fo=BOMESKI.BOMREC_FO_OPTION(j)
			If Trim(tezgah_grup5)<> "H" and trim(fo)<>"1" then
				if trim(yenitezgah) <> trim(eskitezgah) then
					kacincitezgah=kacincitezgah + 1
					mamulno1=(tezgahsayisi-kacincitezgah)*10
					Yenimamulno=string(3-len(mamulno1),"0")&mamulno1
					tip=BOMESKI.BOMREC_INPUTTYPE(j)
					yenimamulkodu=trim(ListeMamulKodu)&"-"&yenimamulno

					mamulno2=(tezgahsayisi-kacincitezgah+1)*10
					eskimamulno=string(3-len(mamulno2),"0")&mamulno2
					eskimamulkodu=trim(ListeMamulKodu)&"-"&eskimamulno
					eskitezgah=yenitezgah

					Tezgah_grup3=trim(doc.getdbfieldvalue("IMLT00", "GK_3", eskitezgah, ""))
					sorgu=1
					Call stokac(yenimamulkodu, sorgu)
					call UrunAgaciAc(j, yenimamulkodu, sorgu)
					Tezgah_grup4=trim(doc.getdbfieldvalue("IMLT00", "GK_4", yenitezgah, ""))
					Call Stokgrup4ac(eskimamulkodu, sorgu)
					if right(trim(yenimamulkodu), 3)="000" then
						Call Stokgrup4ac(yenimamulkodu, sorgu)
					End If
				else
					If hammaddesayisi=0 then
						sorgu=1 
						call UrunAgaciAc(j, yenimamulkodu, sorgu)
					else
						sorgu=0	
						call UrunAgaciAc(j, yenimamulkodu, sorgu)
					End If
				End If 
			End If
		else
			kacincihammadde=kacincihammadde+1
			if kacincihammadde =1 then
				sorgu=1
				Call stokac(yenimamulkodu, sorgu)
				call UrunAgaciAc(j, yenimamulkodu, sorgu)
			else
				sorgu=0
				call UrunAgaciAc(j, yenimamulkodu, sorgu)
			End If
		End If
	Next

	Dim ImportedVoucher

	ImportedVoucher=""

Sub Makro1()
	Set S46T3 = Doc.Select1("ssddwe", "STOK46T3", "R_KAYNAKTYPE='H'&ACIK_KAPALI=' '", "", -1)
	if S46T3.GetRowCount() = 0 Then Exit Sub
	S46T3.GroupBy("EVRAKNO")
	For i = 1 to S46T3.GetRowCount()
		Doc.Load_Voucher S46T3.EVRAKNO(i)
		Doc.Save_Voucher
	Next
End Sub
