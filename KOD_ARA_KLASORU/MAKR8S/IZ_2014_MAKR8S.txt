Function LOTNUMBER_AfterValidate()
	Hata = ""
	Set S26T = Doc.GetTableObject("STOK26T")
	SatirNo = S26T.GetCurrentRow()
	StokKodu = S26T.KOD(SatirNo)
	LotNo = Trim(S26T.LOTNUMBER(SatirNo))
	
	'STOK KARTINDA RAF ÖMRÜ KRÝTERÝ NEDÝR KONTROLÜ
	Set S00KTA = Doc.Select1("dddBdtyy", "KTA_STOK00", "KOD='" & StokKodu & "'", "", -1)

		End If
	End If
	
End Function

Function FIFOyaUygunMu(StokKodu, LotNo, SatirNo)
	Set S26E = Doc.GetTableObject("STOK26E")
	If Trim(S26E.TARGETAMBCODE) = "D-ALU" OR Trim(S26E.TARGETAMBCODE) = "D-BOYA" OR Trim(S26E.TARGETAMBCODE) = "D-FLYO" OR Trim(S26E.TARGETAMBCODE) = "D-KON" OR Trim(S26E.TARGETAMBCODE) = "D-MAM" OR Trim(S26E.TARGETAMBCODE) = "D-FLYO" Then Exit Function
	
	Set S26T = Doc.GetTableObject("STOK26T")
	If Trim(S26T.AMBCODE_TR(SatirNo)) <> "" Then 
		Depo = S26T.AMBCODE_TR(SatirNo)
	Else
		Depo = S26E.AMBCODE
	End If

	Set S01 = Doc.Select1("sfXXfsfd", "STOK01", "KOD='" + StokKodu + "' & AMBCODE='" + Depo + "'", "", -1)
	KayitSayisi_S01 = S01.GetRecCount
	Redim Lotlar(KayitSayisi_S01)

	If KayitSayisi_S01 > 0 Then
		For n = 1 To KayitSayisi_S01
			If Trim(S01.AMBCODE(n) ) = "" Then 
				y = doc.d7MsgBox("Bu Stok Koduna ait Deposu Tanýmlý Olmayan Veri Var. Lütfen Kontrol Ediniz, FÝFO kontrol Ýþlemi Ýptal Edilmiþtir.","FÝFO Eksik Bilgi",MB_OK)
				Exit Function
				Exit For
			End If

			Lot = Trim(S01.LOTNUMBER(n))
			If Lot = "" Then
				Doc.d7MsgBox "Bu Stok Koduna ait LOT Numarasý Tanýmlanmamýþ Kayýt/lar Var. Lütfen Düzeltiniz. Ýþlem Kesilmiþtir.", "Lot Numarasý Hatasý!", MB_OK
				Exit Function
			Else
				Lotlar(n) = Lot
			End If
		Next
		
		min = Lotlar(1)
		max = Lotlar(1)

		For m = 1 To KayitSayisi_S01
			'array içindeki min deðeri yani Günün Tarihine En Yakýn olan Farký  bulmak için
			If Lotlar(m) =< min Then 
				min = Lotlar(m)		
				Satir = m
			End If
			'array içindeki max deðeri bulmak için
			'If RafOmuruFark(m) > max Then max = RafOmuruFark(m)
		Next
						
		If Satir <> "" Then
			If min <> LotNo Then
				Hata = "Okunacak Ýlk Lot" & vbTab & " : " & Trim(S01.LOTNUMBER(Satir)) & vbcrlf
				Hata = Hata & "Deposu" & vbTab & vbTab & " : " & Depo & vbcrlf
				Hata = Hata & "Lokasyonu" & vbTab & " : " & GetLokasyon(S01, Satir) & vbcrlf
				Hata = Hata & "Miktar" & vbTab & vbTab & " : " & S01.SF_MIKTAR(Satir) & " " & S01.SF_UNIT(Satir) & vbcrlf
			End If
		End If
	End If

	If Trim(Hata) <> "" Then
		FIFOyaUygunMu = Hata
	End if
End Function

Function RAFOMRUNEUygunMu(StokKodu, LotNo, SatirNo)

	If Trim(S26E.TARGETAMBCODE) <> "D-KES" Then Exit Function
	
	GununTarihi = Doc.Cdbl(Replace(Doc.Bugun(),"/",""))
	Fark = LotNo - GununTarihi
	
	Set S26T = Doc.GetTableObject("STOK26T")
	If Trim(S26T.AMBCODE_TR(SatirNo)) <> "" Then 
		Depo = S26T.AMBCODE_TR(SatirNo)
	Else
		Depo = S26E.AMBCODE
	End If

		For n = 1 To KayitSayisi_S01
			If Trim(S01.AMBCODE(n) ) = "" Then 
				y = doc.d7MsgBox("Bu Stok Koduna ait Deposu Tanýmlý Olmayan Veri Var. Lütfen Kontrol Ediniz, FÝFO kontrol Ýþlemi Ýptal Edilmiþtir.","FÝFO Eksik Bilgi",MB_OK)
				Exit Function
				Exit For
			End If

			If Lot = "" Then
				Doc.d7MsgBox "Bu Stok Koduna ait LOT Numarasý Tanýmlanmamýþ Kayýt/lar Var. Lütfen Düzeltiniz. Ýþlem Kesilmiþtir.", "Lot Numarasý Hatasý!", MB_OK
				Exit Function
			Else
				RafOmuruFark(n) = Doc.Cdbl(Lot - GununTarihi)
			End If
		Next
		
		min = RafOmuruFark(1)
		max = RafOmuruFark(1)

		For m = 1 To KayitSayisi_S01
			'array içindeki min deðeri yani Günün Tarihine En Yakýn olan Farký  bulmak için
			If RafOmuruFark(m) =< min Then 
				min = RafOmuruFark(m)

			'array içindeki max deðeri bulmak için
			'If RafOmuruFark(m) > max Then max = RafOmuruFark(m)
		Next
		
		'msgbox "Satir = " & Satir & ", " & "min = " & min & ", " & "Fark = " & Fark  

		If Satir <> "" Then
			If min < Fark Then
				Hata = "Okunacak Ýlk Lot" & vbTab & " : " & Trim(S01.LOTNUMBER(Satir)) & vbcrlf
				Hata = Hata & "Deposu" & vbTab & vbTab & " : " & Depo & vbcrlf

	End if
End Function

Sub SendMail(MailFrom, MailTo,MailHeader,MailMessage, DosyaAdi)

	("http://schemas.microsoft.com/cdo/configuration/smtpserverport") = SMTPPort

	("http://schemas.microsoft.com/cdo/configuration/smtpusessl") = False

	("http://schemas.microsoft.com/cdo/configuration/smtpconnectiontimeout") = 60
