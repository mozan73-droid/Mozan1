Function PSWveAlertOnayiVarMi()
	PSWveAlertOnayiVarMi = True
	
	if Left(STOK29E.CARIHESAPCODE,3) <> "320" Then Exit Function
	if trim(Doc.GetDocType0) <> "STOK29" Then Exit Function
	
	Set d = Doc.CreateSObject("BATCHERRORMODE")
	
	Set STK29T = Doc.Select2G("weffwsfsdf", "STOK29T", "", "", "KOD", -1)
	
	For i=1 To STK29T.GetRecCount()
		If PSWKontroluYapilacakMi(STK29T.KOD(i)) Then
			PSWOnayTarihi = Doc.GetUdfValue("STOK00", STK29T.KOD(i), "TONAYTAR", 0)
			AlertBaslamaTarihi = Getd7Tarih(Doc.GetUdfValue("STOK00", STK29T.KOD(i), "TALERTTARB", 0))
			AlertBitisTarihi = Getd7Tarih(Doc.GetUdfValue("STOK00", STK29T.KOD(i), "TALERTTARS", 0))
			AlertMiktari = Doc.GetUdfValue("STOK00", STK29T.KOD(i), "TALERTMIK", 0)
			if trim(AlertBitisTarihi) <> "" Then AlertMiktari = 99999999999999999999999999
			
			if trim(PSWOnayTarihi) = "" and trim(AlertBaslamaTarihi) = "" Then
				Doc.d7MsgBox "Stok Kodu : " & Trim(STK29T.KOD(i)) & ", PSW Onayý verilmemiþ. Giriþ yapýlamaz", "", MB_OK
				PSWveAlertOnayiVarMi = False
			ElseIf trim(AlertBaslamaTarihi) <> "" and STOK29E.TARIH >= AlertBaslamaTarihi Then
				if Trim(AlertBitisTarihi) <> "" and STOK29E.TARIH > AlertBitisTarihi Then
					Doc.d7MsgBox "Stok Kodu : " & Trim(STK29T.KOD(i)) & ", Alert Bitiþ Tarihi : " & Doc.StrToNorDate(AlertBitisTarihi) & _
					". Bu tarihten sonra giriþ yapýlamaz", "Hata", MB_OK
					PSWveAlertOnayiVarMi = False
				ElseIf AlertMiktari > 0 Then
					ToplamSevk = STK29T.SF_MIKTAR(i)
					QStr = "EVRAKNO<>'" & STOK29E.EVRAKNO & "'&KOD='" & STK29T.KOD(i) & "'&TARIH>='" & AlertBaslamaTarihi & "'"
					Set S29TOld = Doc.Select1G("TempTable", "STOK29T", QStr, "", "KOD", -1)
					If S29TOld.GetRowCount()>0 Then
						ToplamSevk = ToplamSevk + S29TOld.SF_MIKTAR
					End If
					If ToplamSevk > AlertMiktari Then
						Doc.d7MsgBox "Stok Kodu : " & Trim(STK29T.KOD(i)) & " Giriþ Miktarý (" & ToplamSevk & ") Alert Miktarýný (" & AlertMiktari & ") Geçiyor", "", MB_OK
						PSWveAlertOnayiVarMi = False
					End If
				End if
			End If
		End if
	Next
	if not PSWveAlertOnayiVarMi Then
		Doc.d7MsgBox "Evrak Kaydedilmedi", "", MB_OK
	End if
End Function

Function TatilGunuyseGeriAl(Tarih)
	Do
		CalismaSaati = GetToplamCalismaSaati("GENEL", Tarih, Tarih)
		if CalismaSaati > 0 Then Exit Do
		Tarih = Doc.StrDate_Increment(Tarih, -1)

Function GetToplamCalismaSaati(TakvimKodu, BaslangicTarih, BitisTarih)
	BaslangicYil = Doc.StrDate_Year(BaslangicTarih)
	BaslangicAy = Doc.StrDate_Month(BaslangicTarih)
	BaslangicGun = Doc.StrDate_Day(BaslangicTarih)
	DakikaAraligi = 15
	AralikSerial = TimeSerial(0,DakikaAraligi ,0)

	Aralik = (Doc.CDateToInt(BitisTarih) - Doc.CDateToInt(BaslangicTarih) + 1) * 24 * 60 / DakikaAraligi
	Baslangic = DateSerial(BaslangicYil, BaslangicAy, BaslangicGun) + TimeSerial(0, 0, 0)
	For i = 1 to Aralik
		BasTarih = Year(Baslangic) & "/" & Right("0" & Month(Baslangic),2) & "/" & Right("0" & Day(Baslangic),2)
		BasSaat = Right("0" & Hour(Baslangic),2) & ":" & Right("0" & Minute(Baslangic),2)
		nret = Doc.Takvim_Test_Minute(TakvimKodu, BasTarih, BasSaat, DakikaAraligi)
		if nret = "1" Then
			CalismaSaat = CalismaSaat + DakikaAraligi
		End if
		Baslangic = Baslangic + AralikSerial
	Next
	GetToplamCalismaSaati = Doc.Div(CalismaSaat, 60)
End Function
