Sub TuketilenHammaddeMiktarKontrolu(UretimFisEvrakNo)
	If Doc.DataEntryMode <> "GUI" Then Exit Sub
	
	SQL = "SELECT "
	SQL = SQL & " S20T_KARSITRNUM, "
	SQL = SQL & " S20T_EVRAKNO, "
	SQL = SQL & " S20T_URETILEN_KOD, "
	SQL = SQL & " S20T_TOP_URETILEN_MIKTAR, "
	SQL = SQL & " S20T_URETILEN_BIRIM, "
	SQL = SQL & " S20TI_TUKETILEN_KOD, "
	SQL = SQL & " TO_CHAR(S20TI_TOPLAM_TUKETILEN, '999,999.99') S20TI_TOPLAM_TUKETILEN, "
	SQL = SQL & " S20TI_TUKETILEN_BIRIM, "
	SQL = SQL & " B01E_MAMULMIKTAR, "
	SQL = SQL & " B01T_HAMMADDE, "
	SQL = SQL & " B01T_HAMMADDE_MIKTARI, "
	SQL = SQL & " BIRIM_TUKETIM_MIKTARI, "
	SQL = SQL & " TO_CHAR(TOPLAM_HARCANMASI_GEREKEN, '999,999.99') TOPLAM_HARCANMASI_GEREKEN, "

	SQL = SQL & " TO_CHAR((S20TI_TOPLAM_TUKETILEN-TOPLAM_HARCANMASI_GEREKEN), '999,999.99') FARK, "

	SQL = SQL & " (CASE "
	SQL = SQL & " WHEN S20TI_TOPLAM_TUKETILEN > TOPLAM_HARCANMASI_GEREKEN THEN 'FAZLA' "
	SQL = SQL & " WHEN S20TI_TOPLAM_TUKETILEN < TOPLAM_HARCANMASI_GEREKEN THEN 'AZ' "
	SQL = SQL & "   WHEN S20TI_TOPLAM_TUKETILEN = TOPLAM_HARCANMASI_GEREKEN THEN 'AYNI' "
	SQL = SQL & "   WHEN TRIM(B01T_HAMMADDE) IS NULL AND S20TI_TOPLAM_TUKETILEN>0 THEN 'BOM_DA_TANIMSIZ_HAMMADDE' "
	SQL = SQL & "   ELSE 'TUKETILENSIZ' "
	SQL = SQL & " END) DURUM1, "

	SQL = SQL & " (CASE "
	SQL = SQL & " WHEN S20TI_TOPLAM_TUKETILEN > TOPLAM_HARCANMASI_GEREKEN THEN 1 "
	SQL = SQL & " WHEN S20TI_TOPLAM_TUKETILEN < TOPLAM_HARCANMASI_GEREKEN THEN 1 "
	SQL = SQL & "   WHEN S20TI_TOPLAM_TUKETILEN = TOPLAM_HARCANMASI_GEREKEN THEN 0 "
	SQL = SQL & "   WHEN TRIM(B01T_HAMMADDE) IS NULL AND S20TI_TOPLAM_TUKETILEN>0 THEN 1 "
	SQL = SQL & "   ELSE 1 "
	SQL = SQL & " END) DURUM2 "	
	
	SQL = SQL & " FROM "
	SQL = SQL & " ( "
	SQL = SQL & " SELECT "
	SQL = SQL & " S20T_KARSITRNUM, "
	SQL = SQL & " S20T_EVRAKNO, "
	SQL = SQL & " S20T_URETILEN_KOD, "
	SQL = SQL & " TO_CHAR(SUM(S20T_URETILEN_MIKTAR), '999,999.99')  S20T_TOP_URETILEN_MIKTAR, "
	SQL = SQL & " S20T_URETILEN_BIRIM, "

	SQL = SQL & " S20TI_TUKETILEN_KOD, "
	SQL = SQL & " S20TI_TOPLAM_TUKETILEN, "
	SQL = SQL & " S20TI_TUKETILEN_BIRIM, "
	SQL = SQL & " B01E_MAMULMIKTAR, "
	SQL = SQL & " B01T_HAMMADDE, "
	SQL = SQL & " B01T_HAMMADDE_MIKTARI, "
	SQL = SQL & " BIRIM_TUKETIM_MIKTARI, "
	SQL = SQL & " SUM(HARCANMASI_GEREKEN) TOPLAM_HARCANMASI_GEREKEN "

	SQL = SQL & " FROM "
	SQL = SQL & " ( "
	SQL = SQL & " SELECT "
	SQL = SQL & " S20T_KARSITRNUM, "
	SQL = SQL & " S20T_EVRAKNO, "
	SQL = SQL & " S20T_URETILEN_KOD, "
	SQL = SQL & " S20T_URETILEN_MIKTAR, "
	SQL = SQL & " S20T_URETILEN_BIRIM, "

	SQL = SQL & " S20TI_TUKETILEN_KOD, "
	SQL = SQL & " SUM(S20TI_TUKETILEN_MIKTAR) S20TI_TOPLAM_TUKETILEN, "
	SQL = SQL & " S20TI_TUKETILEN_BIRIM, "
	SQL = SQL & " B01E_MAMULMIKTAR, "
	SQL = SQL & " B01T_HAMMADDE, "
	SQL = SQL & " B01T_HAMMADDE_MIKTARI, "
	SQL = SQL & " (B01T_HAMMADDE_MIKTARI/B01E_MAMULMIKTAR) BIRIM_TUKETIM_MIKTARI, "
	SQL = SQL & " (S20T_URETILEN_MIKTAR * (B01T_HAMMADDE_MIKTARI/B01E_MAMULMIKTAR)) HARCANMASI_GEREKEN "

	SQL = SQL & " FROM "

	SQL = SQL & " ( "

	SQL = SQL & " (SELECT KARSITRNUM S20T_KARSITRNUM, EVRAKNO S20T_EVRAKNO, KOD S20T_URETILEN_KOD, SF_MIKTAR S20T_URETILEN_MIKTAR, SF_SF_UNIT S20T_URETILEN_BIRIM FROM STOK20T)A "

	SQL = SQL & " LEFT JOIN "

	SQL = SQL & " (SELECT KARSITRNUM S20TI_KARSITRNUM, EVRAKNO S20TI_EVRAKNO, KOD S20TI_TUKETILEN_KOD, SF_MIKTAR S20TI_TUKETILEN_MIKTAR, SF_SF_UNIT S20TI_TUKETILEN_BIRIM FROM STOK20TI)B "
	SQL = SQL & " ON A.S20T_EVRAKNO = B.S20TI_EVRAKNO AND A.S20T_KARSITRNUM = B.S20TI_KARSITRNUM "

	SQL = SQL & " LEFT JOIN "

	SQL = SQL & " (SELECT MAMULCODE, MAMULMIKTAR B01E_MAMULMIKTAR, AKTIF_PASIF, EVRAKNO B01E_EVRAKNO FROM BOMU01E)C "
	SQL = SQL & " ON A.S20T_URETILEN_KOD = C.MAMULCODE AND C.AKTIF_PASIF = '*' "

	SQL = SQL & " LEFT JOIN "

	SQL = SQL & " (SELECT BOMREC_INPUTTYPE, EVRAKNO BO1T_EVRAKNO, BOMREC_KAYNAKCODE B01T_HAMMADDE, BOMREC_KAYNAK0 B01T_HAMMADDE_MIKTARI FROM BOMU01T)D "
	SQL = SQL & " ON C.B01E_EVRAKNO = D.BO1T_EVRAKNO AND D.BOMREC_INPUTTYPE = 'H' AND D.B01T_HAMMADDE = B.S20TI_TUKETILEN_KOD "

	SQL = SQL & " ) GROUP BY S20T_KARSITRNUM, S20T_EVRAKNO, S20T_URETILEN_KOD, S20T_URETILEN_MIKTAR, S20T_URETILEN_BIRIM, S20TI_TUKETILEN_KOD, S20TI_TUKETILEN_BIRIM, B01E_MAMULMIKTAR, B01T_HAMMADDE, B01T_HAMMADDE_MIKTARI, (B01T_HAMMADDE_MIKTARI/B01E_MAMULMIKTAR), (S20T_URETILEN_MIKTAR * (B01T_HAMMADDE_MIKTARI/B01E_MAMULMIKTAR)) "

	SQL = SQL & " ) GROUP BY S20T_KARSITRNUM, S20T_EVRAKNO, S20T_URETILEN_KOD, S20T_URETILEN_BIRIM, S20TI_TUKETILEN_KOD, S20TI_TOPLAM_TUKETILEN, S20TI_TUKETILEN_BIRIM, B01E_MAMULMIKTAR, B01T_HAMMADDE, B01T_HAMMADDE_MIKTARI, BIRIM_TUKETIM_MIKTARI "

	SQL = SQL & " ) "
	SQL = SQL & " WHERE 1=1 "
	SQL = SQL & " AND TRIM(S20T_EVRAKNO) = '"&UretimFisEvrakNo&"' "  
	'SQL = SQL & " --AND TRIM(MPSNO) = '14_01306' 
	'SQL = SQL & " --AND S20T_KARSITRNUM = '000006'
	SQL = SQL & " ORDER BY S20T_KARSITRNUM "
	
	Set SQLTable = Doc.RunSQLQuery("XXXXX11_TT", SQL)

	KayitSayisi = SQLTable.GetRecCount
	If KayitSayisi = 0 Then Exit Sub
	
	Set S_Table = Doc.Select2("dsadd", "XXXXX11_TT", "DURUM2='1'", "", -1)
	If S_Table.GetRecCount = 0 Then Exit Sub

	k = Doc.d7MsgBox("Tüketilen Hammadde Miktar Uygunsuzluðu." & vbcrlf & vbcrlf & "Detayý Görmek Ýstiyor Musunuz?", "Freedom Uyarý", MB_YESNO)
	If k<>6 Then Exit Sub
		
	Call S_Table.PrintOut("Üretim Fiþi - Tüketilen Miktar Kontrolü","H",FALSE,"S20T_KARSITRNUM, S20T_EVRAKNO, S20T_URETILEN_KOD, S20T_TOP_URETILEN_MIKTAR, S20T_URETILEN_BIRIM, S20TI_TUKETILEN_KOD, S20TI_TOPLAM_TUKETILEN, S20TI_TUKETILEN_BIRIM, TOPLAM_HARCANMASI_GEREKEN, FARK, DURUM1 ")
End Sub
