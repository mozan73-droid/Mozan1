Function GetIrsaliyeTutari(S60,satirNo, Caller)
	GetIrsaliyeTutari = 0
	if S60.SF_MIKTAR(satirNo) <> 0 then
		FaturalanacakMiktar = S60.SF_MIKTAR(satirNo) - S60.SF_FATURALANANMIKTAR(satirNo)
		if S60.TL_KDVDAHILTUTAR(satirNo) = 0 then
			SiparisNo = Doc.GetDBFieldValue("STOK40T","EVRAKNO",S60.SIPARTNO(satirNo),"ARTNO")
			MuhtemelNitelik = Doc.GetDBFieldValue("STOK40E","MUHT_FATURA_NITELIK",SiparisNo,"")
			KurAktarma = Doc.GetDBFieldValue("STOK40E","KURKESINLESTIRME",SiparisNo,"")
			SiparisMiktari = Doc.GetDBFieldValue("STOK40T","SF_MIKTAR",S60.SIPARTNO(satirNo),"ARTNO")
			KDVHaricTutar = Doc.GetDBFieldValue("STOK40T","OR_IVD_KH_TUTAR",S60.SIPARTNO(satirNo),"ARTNO")
			KDVTutari = Doc.GetDBFieldValue("STOK40T","OR_KDVTUT_EFEKTIF",S60.SIPARTNO(satirNo),"ARTNO")
				If MuhtemelNitelik = "XK" Then
				SiparisTutari = KDVHaricTutar + KDVTutari
			Else
				SiparisTutari = KDVHaricTutar
			End If
			SiparisPB = Doc.GetDBFieldValue("STOK40T","PRICEUNIT",S60.SIPARTNO(satirNo),"ARTNO")

			SiparisFiyati = Doc.Div(SiparisTutari, SiparisMiktari)
			if KurAktarma = "E" then
				KurDegeri = Doc.GetDBFieldValue("STOK40T","KUR",S60.SIPARTNO(satirNo),"ARTNO")
			Else
				KurDegeri = Doc.GetKurN(SiparisPB, "YTL",Doc.BuGun,"",-1)
			End if
			GetIrsaliyeTutari = SiparisFiyati * KurDegeri * FaturalanacakMiktar
		else
			GetIrsaliyeTutari = Doc.Div(S60.TL_KDVDAHILTUTAR(satirNo) * FaturalanacakMiktar, S60.SF_MIKTAR(satirNo))
		end if
	end if
End Function
Function GetSiparisTutari()
	for i = 1 to STOK40T.GetRecCount()
		KDVHaricTutar = STOK40T.OR_IVD_KH_TUTAR(i)
		KDVTutari = STOK40T.OR_KDVTUT_EFEKTIF(i)
		MuhtemelNitelik = STOK40E.MUHT_FATURA_NITELIK
		If MuhtemelNitelik = "XK" Then
			SiparisTutari = KDVHaricTutar + KDVTutari
		Else
			SiparisTutari = KDVHaricTutar
		End If
		SiparisPB = STOK40T.PRICEUNIT(i)
		KurDegeri = STOK40T.KUR(i)
		if KurDegeri = 0 then
			KurDegeri = Doc.GetKurN(SiparisPB, Doc.DefaultPB, Doc.BuGun,"",-1)
		End if
		ToplamSipTutari = ToplamSipTutari + SiparisTutari * KurDegeri
	Next
	Set T_STOK40T = Doc.Select1("ABC","STOK40T","MUSTERIKODU='"+STOK40E.MUSTERIKODU+"'&ACIKKAPALI='"+" "+"'&EVRAKNO<>'"+STOK40E.EVRAKNO+"'","",-1)
	for s=1 to T_STOK40T.getreccount
		KDVHaricTutar = T_STOK40T.OR_IVD_KH_TUTAR(s)
		KDVTutari = T_STOK40T.OR_KDVTUT_EFEKTIF(s)
		MuhtemelNitelik = STOK40E.MUHT_FATURA_NITELIK
		If MuhtemelNitelik = "XK" Then
			SiparisTutari = KDVHaricTutar + KDVTutari
		Else
			SiparisTutari = KDVHaricTutar
		End If
		SiparisPB = T_STOK40T.PRICEUNIT(s)
		KurDegeri = T_STOK40T.KUR(s)
		if KurDegeri = 0 then
			KurDegeri = Doc.GetKurN(SiparisPB, Doc.DefaultPB, Doc.BuGun,"",-1)
		End if
		ToplamSipTutari1 = ToplamSipTutari1 + SiparisTutari * KurDegeri
	next
	GetSiparisTutari = ToplamSipTutari + ToplamSipTutari1

End Function

Sub TarihHesapla()
	For i = 1 to STOK20T.GetRowCount()
		if STOK20T.W_EVRAKTYPE(i) = "S4" Then
			sfdcTarih = Doc.GetDBFieldValue("SFDC32E", "RECTARIH1", STOK20T.W_EVRAKNO(i), "")

				STOK20E.TARIH = sfdcTarih

Sub Calc_IscSure()
	For i = 1 to STOK20TS.GetRowCount()
		QStr = "BOMREC_CODE='"+STOK20TS.KARSICODE(i)+"'&AKTIF_PASIF='*'&BOMREC_INPUTTYPE='I'&BOMREC_KAYNAKCODE='" _
					+STOK20TS.KOD(i)+"'"
		Set B01T = Doc.Select1("dsdf", "BOMU01T", QStr, "", 1)
		if B01T.GetRowCount() > 0 Then
			ToplamIsci = B01T.OPERATORSAYISI1(1) + B01T.OPERATORSAYISI2(1) + B01T.OPERATORSAYISI3(1)
			if ToplamIsci = 0 Then
				ToplamIsci = Doc.GetDbFieldValue("IMLT00", "B_ISCILIKSAATI", STOK20TS.KOD(i), "")
			End if
		Else
			ToplamIsci = 0
		End if
		STOK20TS.ISCILIK_SURE(i) = ToplamIsci * STOK20TS.ISLEM_SURE(i)
	Next
End Sub
Function OndalikKontrol()
	Hata = ""
	For i = 1 to STOK20T.GetRowCount()
		if trim(STOK20T.SF_SF_UNIT(i)) = "AD" and Fix(STOK20T.SF_MIKTAR(i)) <> STOK20T.SF_MIKTAR(i) Then
			Hata = Hata & i & ". satýr Kod : " & trim(STOK20T.KOD(i)) & ", Miktar = " & STOK20T.SF_MIKTAR(i) & " " & trim(STOK20T.SF_SF_UNIT(i))
		End if
	Next
	OndalikKontrol = Hata
End Function

Function FIFOyaUygunMu(SeriNo)

	Set S01 = Doc.Select1("sfsfd", "STOK01", "SERINO='"+trim(SeriNo)+"'", "", -1)
	Depo = S01.AMBCODE
	If Depo = "" Then 
		y = doc.d7MsgBox("Deposu Tanýmlý Deðil, iptal","Eksik Barkod Bilgisi",MB_OK)
		Exit Function
	End If
 
	Set S26E = Doc.getTableObject("STOK26E")
	If trim(S26E.TARGETAMBCODE) <> "IM" Then Exit Function

	If S01.GetRowCount() > 0 Then
		Depo = S01.AMBCODE
		LotNo = Left(S01.LOTNUMBER, 6)
		StokKodu = S01.KOD
		Set S01 = Doc.Select1("sfsfd", "STOK01", "KOD='" + StokKodu+"' & AMBCODE='" + Depo+"' & LOTNUMBER<'" + LotNo + "'", "LOTNUMBER", -1)
		For i = 1 to S01.GetRowCount()
			Lot = Left(S01.LOTNUMBER(i), 6)
			if Lot <> EskiLot And i > 1 Then Exit For
			if i = 1 Then Hata = "Okunacak Ýlk Lot:" & trim(S01.LOTNUMBER(i)) & vbcrlf
			Hata = Hata & "L:" & GetLokasyon(S01, i) & ",M:" & S01.SF_MIKTAR(i) & vbcrlf
			EskiLot = Lot
		Next
	End if
	If trim(Hata) <> "" Then
		FIFOyaUygunMu = Hata
	End if
End Function

Sub UretimFisiOlustur(KayitSayisi, S60T)

	Set Doc1 = Doc.CreateDocumentNoUI("STOK20", "")
	Set S20E = Doc1.GetTableObject("STOK20E")
	Set S20T = Doc1.GetTableObject("STOK20T")

	Doc1.New_Voucher
	S20E.AMBCODE = "MA"
	S20E.IMALATAMBCODE = "IM"
	S20E.YANMAMULAMBCODE = "ZZ"
	
	For i = 1 to KayitSayisi
		satirNo = S20T.AddRow()
		S20T.SetCurrentRow(satirNo)

		S20T.MPSNO = ""
		S20T.MPSTM_TRNUM = ""
		S20T.KOD = S60T.KOD(i)
		S20T.LOTNUMBER = S60T.LOTNUMBER(i) 'Mid(Replace(S20E.TARIH, "/", ""), 3)
		S20T.SERINO = S60T.SERINO(i) 'Doc.Sayac0danAl("SERINO")
		S20T.SF_MIKTAR = S60T.SF_MIKTAR(i)
		S20T.W_FIRMA = ""
		S20T.W_APP = ""
		S20T.W_TABLE = ""
		S20T.W_EVRAKTYPE = ""
		S20T.W_EVRAKNO = ""
		S20T.W_TRNUM = ""
	Next

	'Doc1.RunCommand "CMD_SIRALIK_ISLEM_003"
	'Doc1.RunCommand "CMD_STOK01DEBULUNDUMU_BILGISATIRIE"

	Doc1.Save_Voucher
	Doc1.LoadEmpty_Voucher
End Sub

Sub IadeIseSatisEvrakNoF2Degistir()
	TabloStr = "STOK25T"
   AlanAdi = "KTA_MAKARA"
   BrowseTable = "BRW_STOK25T_1"
   BrowseCaption = "Makaralar"

   BrowseQStr = "SELECT T.KOD ""Makara Kodu"", T.AD ""Makara Adi"", " & vbCrLf
   BrowseQStr = BrowseQStr & " T.TEXT_TECH_4" & vbCrLf
   BrowseQStr = BrowseQStr & " FROM STOK00 T" & vbCrLf 
   BrowseQStr = BrowseQStr & " WHERE 1=1 " & vbCrLf
   BrowseQStr = BrowseQStr & " AND T.TEXT_TECH_4='Makara' " & vbCrLf
   BrowseQStr = BrowseQStr & " AND E.CH_KODU=###STOK74E.CH_KODU###" & vbCrLf
   BrowseQStr = BrowseQStr & " ORDER BY T.,T.EVRAKNO"
   BrowseQStr = "'" & Replace(BrowseQStr, "'", "\:q") & "'"
   BrowseFunc = "F2=DO_SQL_BRW(""" & BrowseTable & """,""" & AlanAdi & "=1""," & BrowseQStr & ",""" & BrowseCaption & """)"

   Set Tbl = Doc.GetTableObject(TabloStr)
   Set FieldDef = Tbl.GetFieldDef_E(AlanAdi)

   FieldDef.F2Fnc = BrowseFunc
   Tbl.SetFieldProperties1 AlanAdi,FieldDef
