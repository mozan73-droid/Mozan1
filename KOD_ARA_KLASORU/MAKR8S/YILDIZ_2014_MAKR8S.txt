Sub ParaBirimiKontrol()

		DefKur = Doc.GetDBFieldValue("CARI00", "DEFAULTKUR", CARI20T.KOD, "")
		if trim(DefKur) = "" Then DefKur = Doc.DefaultPB()

		End if
	End if
	if trim(CARI20T.KARSI_CODE) <> "" Then
		DefKur = Doc.GetDBFieldValue("CARI00", "DEFAULTKUR", CARI20T.KARSI_CODE, "")
		if trim(DefKur) = "" Then DefKur = Doc.DefaultPB()
		if trim(CARI20T.CH_KUR) <> trim(DefKur) Then

					trim(CARI20T.CH_KUR) & " para birimi ile iþlem yapmaktasýnýz"
		End if
	End if
	if trim(UyariStr) <> "" Then
		Doc.d7MsgBox UyariStr, "Uyarý", MB_OK+MB_ICONWARNING

Function FIFOyaUygunMu(SeriNo)
	Set STC = Doc.GetTableObject("STACK_PDA")
	if STC.PROPT = "2" Then Exit Function

	if Left(SeriNo, 2) = "--" Then Exit Function

	Set S01_2 = Doc.GetTableObject("STOK01_PDA")
	Set S26E = Doc.GetTableObject("STOK26E") 

	VerenDepoGk4 = Doc.GetDBFieldValue("GDEF00", "GK_4", S26E.AMBCODE, "")
	AlanDepoGk4 = Doc.GetDBFieldValue("GDEF00", "GK_4", S26E.TARGETAMBCODE, "")

	if not(trim(VerenDepoGk4) = "HAMMADDE" and trim(AlanDepoGk4)="URETIM") Then Exit Function

	Set S01 = Doc.Select1("sfsfd", "STOK01", "SERINO='"+trim(SeriNo)+"'", "", -1)
	Farkli = 0
	if S01.GetRowCount() > 0 Then
		Depo = S01.AMBCODE
		LotNo = Left(S01.LOTNUMBER, 6)
		StokKodu = S01.KOD
		Set S01 = Doc.Select1("sfsfd", "STOK01", "KOD='"+StokKodu+"'&AMBCODE='"+Depo+"'&LOTNUMBER<'"+LotNo+"'", "LOTNUMBER", -1)
		for i = 1 to S01.GetRowCount()
			Set S26Tx = Doc.Select2("dff444r55", "STOK26T", "SERINO='" & S01.SERINO(i) & "'", "", -1)
			if S26Tx.GetRowCount() = 0 Then
				Farkli = Farkli + 1
				Lot = Left(S01.LOTNUMBER(i), 6)
				if Lot <> EskiLot And Farkli > 1 Then Exit For
				if Farkli = 1 Then Hata = "Okunacak Ýlk Lot:" & trim(S01.LOTNUMBER(i)) & vbcrlf
				Hata = Hata & "L:" & GetLokasyon(S01, i) & ",M:" & S01.SF_MIKTAR(i) & vbcrlf

				EskiLot = Lot
			End if
		Next

	End if

	if trim(Hata) <> "" Then
		FIFOyaUygunMu = Hata
	End if

End Function

Function MiktarveFireUyumsuzlukKontrol(IsNo)

Function GetPDurusSure(BTarih, BSaat, ETarih, ESaat)
'	Set KK = Doc.CreateSObject("BATCHERRORMODE")
	Dim MolaArray(1,1)
	MolaArray(0,0) = "12:30"
	MolaArray(0,1) = "13:30"
	MolaArray(1,0) = "18:00"
	MolaArray(1,1) = "18:30"
'	MolaArray(2,0) = "04:00"
'	MolaArray(2,1) = "04:30"

	GunSayisi = Doc.CDateToInt(ETarih) - Doc.CDateToInt(BTarih)
	MolaSure = 0
	For i = 0 to uBound(MolaArray, 1)
		For j = 0 to GunSayisi
			if j < GunSayisi Then
				BaslangicSaat = BSaat
				BitisSaat = "23:59"
			Elseif j > 0 Then
				BaslangicSaat = "00:00"
				BitisSaat = ESaat
			Else
				BaslangicSaat = BSaat
				BitisSaat = ESaat
			End if
			Gun = Doc.StrDate_Increment(BTarih, j)
'			Doc.d7MsgBox i &  ", " & j &  ", " & Gun &  ", " & BaslangicSaat &  ", " & BitisSaat & ", " & MolaArray(i,0) & _
'					 ", " & MolaArray(i,1), "", MB_OK
			if not(Gun & BitisSaat < Gun & MolaArray(i,0) or Gun & BaslangicSaat > Gun & MolaArray(i,1)) Then
				if Gun & MolaArray(i,0) >= Gun & BaslangicSaat and Gun & MolaArray(i,1) <= Gun & BitisSaat Then
					MolaSure = MolaSure + Doc.DTDTDif1(Gun, MolaArray(i,0), Gun, MolaArray(i,1))
				Elseif Gun & MolaArray(i,0) >= Gun & BaslangicSaat and Gun & MolaArray(i,1) > Gun & BitisSaat Then
					MolaSure = MolaSure + Doc.DTDTDif1(Gun, MolaArray(i,0), Gun, BitisSaat)
				Elseif Gun & MolaArray(i,0) <= Gun & BaslangicSaat and Gun & MolaArray(i,1) <= Gun & BitisSaat Then
					MolaSure = MolaSure + Doc.DTDTDif1(Gun, BaslangicSaat, Gun, MolaArray(i,1))
				End if
			End if
		Next
	Next
	GetPDurusSure = MolaSure
End Function

Sub TezgahTakvimeGoreSureleriDoldur(satirNo)
	TezgahKodu = TABLO6.KOD(satirNo)
	
	Grid = Doc.GetGridColInfo("", "TABLO6")
	For j = 2 to 40
		BaslaBitis = GetBaslaBitis(Grid, "REPCOL" & j)
		Kapasite = GetCalismaSaati_Tezgah2(TezgahKodu, BaslaBitis(0), BaslaBitis(1))
		Doc.StatusBarMessage "Kapasite hesaplanýyor : " & TezgahKodu & " : " & BaslaBitis(0) & "-" & BaslaBitis(1)

		DonemlikYuk = TABLO6.GetFieldValue("REPCOL" & j, satirNo + 3)
		KumuleYuk = TABLO6.GetFieldValue("REPCOL" & j, satirNo + 4)

		TABLO6.SetFieldValue "REPCOL" & j, satirNo + 1, Kapasite
		KumuleKapasite = TABLO6.GetFieldValue("REPCOL" & j-1, satirNo + 2) + Kapasite
		TABLO6.SetFieldValue "REPCOL" & j, satirNo + 2, KumuleKapasite
		DonemlikBosKapasite = Kapasite - DonemlikYuk
		TABLO6.SetFieldValue "REPCOL" & j, satirNo + 5, DonemlikBosKapasite
		KumulatifBosKapasite = KumuleKapasite - KumuleYuk
		TABLO6.SetFieldValue "REPCOL" & j, satirNo + 6, KumulatifBosKapasite
		KapasiteDolulukYuzdesi = Doc.Div(DonemlikYuk,Kapasite) * 100
		TABLO6.SetFieldValue "REPCOL" & j, satirNo + 7, KapasiteDolulukYuzdesi
		KumuleDolulukYuzdesi = Doc.Div(KumuleYuk,KumuleKapasite) * 100
		TABLO6.SetFieldValue "REPCOL" & j, satirNo + 8, KumuleDolulukYuzdesi
	Next	
End Sub

Function GetPDurusSure(BTarih, BSaat, ETarih, ESaat)
'	Set KK = Doc.CreateSObject("BATCHERRORMODE")
	Dim MolaArray(1,1)
	MolaArray(0,0) = "12:30"
	MolaArray(0,1) = "13:30"
	MolaArray(1,0) = "18:00"
	MolaArray(1,1) = "18:30"
'	MolaArray(2,0) = "04:00"
'	MolaArray(2,1) = "04:30"

	GunSayisi = Doc.CDateToInt(ETarih) - Doc.CDateToInt(BTarih)
	MolaSure = 0
	For i = 0 to uBound(MolaArray, 1)
		For j = 0 to GunSayisi
			if j < GunSayisi Then
				BaslangicSaat = BSaat
				BitisSaat = "23:59"
			Elseif j > 0 Then
				BaslangicSaat = "00:00"
				BitisSaat = ESaat
			Else
				BaslangicSaat = BSaat
				BitisSaat = ESaat
			End if
			Gun = Doc.StrDate_Increment(BTarih, j)
'			Doc.d7MsgBox i &  ", " & j &  ", " & Gun &  ", " & BaslangicSaat &  ", " & BitisSaat & ", " & MolaArray(i,0) & _
'					 ", " & MolaArray(i,1), "", MB_OK
			if not(Gun & BitisSaat < Gun & MolaArray(i,0) or Gun & BaslangicSaat > Gun & MolaArray(i,1)) Then
				if Gun & MolaArray(i,0) >= Gun & BaslangicSaat and Gun & MolaArray(i,1) <= Gun & BitisSaat Then
					MolaSure = MolaSure + Doc.DTDTDif1(Gun, MolaArray(i,0), Gun, MolaArray(i,1))
				Elseif Gun & MolaArray(i,0) >= Gun & BaslangicSaat and Gun & MolaArray(i,1) > Gun & BitisSaat Then
					MolaSure = MolaSure + Doc.DTDTDif1(Gun, MolaArray(i,0), Gun, BitisSaat)
				Elseif Gun & MolaArray(i,0) <= Gun & BaslangicSaat and Gun & MolaArray(i,1) <= Gun & BitisSaat Then
					MolaSure = MolaSure + Doc.DTDTDif1(Gun, BaslangicSaat, Gun, MolaArray(i,1))
				End if
			End if
		Next
	Next
	GetPDurusSure = MolaSure
End Function

	Baslangic = DateSerial(BaslangicYil, BaslangicAy, BaslangicGun) + TimeSerial(BaslangicSaat, BaslangicDakika, 0)
	Bitis = DateSerial(BitisYil, BitisAy, BitisGun) + TimeSerial(BitisSaat, BitisDakika, 0)
	While Baslangic < Bitis
		BasTarih = Year(Baslangic) & "/" & Right("0" & Month(Baslangic),2) & "/" & Right("0" & Day(Baslangic),2)
		BasSaat = Right("0" & Hour(Baslangic),2) & ":" & Right("0" & Minute(Baslangic),2)
		nret = Doc.Takvim_Test_Minute(TakvimKodu, BasTarih, BasSaat, DakikaAraligi)
		if nret = "1" Then
			CalismaDakika = CalismaDakika + DakikaAraligi
		End if
		Baslangic = Baslangic + AralikSerial
	Wend
	GetTakvimCalismaSuresi_dk = CalismaDakika
End Function

Function GetPDurusSure(BTarih, BSaat, ETarih, ESaat)
'	Set KK = Doc.CreateSObject("BATCHERRORMODE")
	GetPDurusSure = 0
	Dim MolaArray(1,1)
	MolaArray(0,0) = "01:00"
	MolaArray(0,1) = "02:00"
	MolaArray(1,0) = "12:00"
	MolaArray(1,1) = "13:00"

	GunSayisi = Doc.CDateToInt(ETarih) - Doc.CDateToInt(BTarih)
	MolaSure = 0
	For i = 0 to uBound(MolaArray, 1)
		For j = 0 to GunSayisi
			if j < GunSayisi Then
				BaslangicSaat = BSaat
				BitisSaat = "23:59"
			Elseif j > 0 Then
				BaslangicSaat = "00:00"
				BitisSaat = ESaat
			Else
				BaslangicSaat = BSaat
				BitisSaat = ESaat
			End if
			Gun = Doc.StrDate_Increment(BTarih, j)
'			Doc.d7MsgBox i &  ", " & j &  ", " & Gun &  ", " & BaslangicSaat &  ", " & BitisSaat & ", " & MolaArray(i,0) & _
'					 ", " & MolaArray(i,1), "", MB_OK
			if not(Gun & BitisSaat < Gun & MolaArray(i,0) or Gun & BaslangicSaat > Gun & MolaArray(i,1)) Then
				if Gun & MolaArray(i,0) >= Gun & BaslangicSaat and Gun & MolaArray(i,1) <= Gun & BitisSaat Then
					MolaSure = MolaSure + Doc.DTDTDif1(Gun, MolaArray(i,0), Gun, MolaArray(i,1))
				Elseif Gun & MolaArray(i,0) >= Gun & BaslangicSaat and Gun & MolaArray(i,1) > Gun & BitisSaat Then

					MolaSure = MolaSure + Doc.DTDTDif1(Gun, BaslangicSaat, Gun, MolaArray(i,1))
				End if
			End if
		Next
	Next
	GetPDurusSure = MolaSure
End Function

Sub GetKurTable(Doc1, objXMLDoc, Tarih)
	Set EXCE = Doc1.GetTableObject("EXCRATE")
	Set EXCT = Doc1.GetTableObject("EXCRATT")

	KURTarih = Doc1.StrDate_Increment(Tarih, -1)
	Call HaftaSonuysaGeriAl(KURTarih )

	xmlTarih = Replace(Left(KURTarih ,7), "/", "") & "/" & Replace(Doc1.StrToNorDate(KURTarih ), "/", "")

	objXMLDoc.Load "http://www.tcmb.gov.tr/kurlar/" & xmlTarih & ".xml"
'	objXMLDoc.Load "c:\today.xml"

	Set Root = objXMLDoc.documentElement
	if Root is Nothing Then
'		k = doc.d7MsgBox("Kurlar alýnamadý " & xmlTarih, "Hata", MB_OK)
		Exit Sub
	End if
	Set KurList = Root.getElementsByTagName("Currency")

	EXCE.EVRAKNOTARIH = Tarih

	EXCT.Empty    
	For Each Kur In KurList
		ParaBirimi = Kur.Attributes(1).NodeValue
		if trim(ParaBirimi) = "EUR" Then ParaBirimi = "EURO"
		if trim(Doc1.GetGecoValue("*PUNIT", "KOD", ParaBirimi, d7c_ErrMsgNo)) <> "" Then
			Birim = GetKurValue(Kur, "Unit")
			DovizAlis = GetKurValue(Kur, "ForexBuying")
			DovizSatis = GetKurValue(Kur, "ForexSelling")
			EfektifAlis = GetKurValue(Kur, "BanknoteBuying")
			EfektifSatis = GetKurValue(Kur, "BanknoteSelling")
		
			satirNo = EXCT.AddRow
			EXCT.SetCurrentRow(satirNo)
			EXCT.CODEFROM(satirNo) = ParaBirimi
			EXCT.CODETO(satirNo) = Doc1.DefaultPB
			EXCT.KURS_1(satirNo) = DovizAlis / Doc1.CDbl(Birim)
			EXCT.KURS_3(satirNo) = DovizAlis / Doc1.CDbl(Birim)
			EXCT.KURS_4(satirNo) = DovizSatis / Doc1.CDbl(Birim)
			EXCT.KURS_5(satirNo) = EfektifAlis / Doc1.CDbl(Birim)
			EXCT.KURS_6(satirNo) = EfektifSatis / Doc1.CDbl(Birim)
		End if
	Next
	Doc1.Save_Voucher

Function NUMUNESATIRKONTROLU
	Kaydedemez = ""
	If S46T_Hafiza_KayitSayisi > 0 Then
		For v = 1 To S46T_Hafiza_KayitSayisi
			Eski_ArtNoDegeri = ArtNoDegeri(v) 
			Eski_StokKodu = StokKodu(v)
			Eski_Miktar = Miktar(v)
			Eski_Birim = Birim(v)
			Eski_Varyant1 = Varyant1(v)
			Eski_Varyant2 = Varyant2(v)
			Eski_TeslimTarihi = TeslimTarihi(v)
			Eski_RTeslimTarihi = RTeslimTarihi(v)
			Eski_Not1 = Not1(v)
			Eski_Not2 = Not2(v)

			Set S46T1 = Doc.Select1("bxxffxxgg", "STOK46T1", "ARTNO='"&Eski_ArtNoDegeri&"'", "", -1)
			If S46T1.GetRecCount > 0 Then
				TalepNo = Trim(S46T1.TALEPNO(1))
				Set S47E = Doc.Select1("yyyy", "STOK47E", "EVRAKNO='"&TalepNo&"'", "", -1)
				If Trim(S47E.GK_3(1)) = "NUMUNEX" Then Exit Function
			End If
			
			Set S46T_Degisen = Doc.Select2("bxxxxgg", "STOK46T", "ARTNO='"&Eski_ArtNoDegeri&"'", "", -1)
						
			If S46T_Degisen.GetRecCount = 0 Then 
				Msgbox "Varyant2 Alanýnda Numune Satiri Silinemez.(1)"
				Kaydedemez = "E"
			Else	
				Set ST46TTT = Doc.GetTableObject("STOK46T")
				For h = 1 TO ST46TTT.GetRecCount
					If ST46TTT.ARTNO(h) = Eski_ArtNoDegeri Then SatirNo = h
				Next
				
				Degisen_StokKodu = S46T_Degisen.KOD(1)
				Degisen_Miktar = S46T_Degisen.SF_MIKTAR(1)
				Degisen_Birim = S46T_Degisen.SF_SF_UNIT(1)

				Degisen_Varyant2 = S46T_Degisen.SF_VRI_VR_R2(1)
				Degisen_TeslimTarihi = S46T_Degisen.RTESTARIH(1)
				Degisen_RTeslimTarihi = S46T_Degisen.REV_RTESTARIH(1)
				Degisen_Not1 = S46T_Degisen.NOTE0(1)
				Degisen_Not2 = S46T_Degisen.NOTE1(1)
				
				If Eski_StokKodu <> Degisen_StokKodu Then
					Msgbox "Varyant2 Alanýnda Numune Satiri Deðiþtirilemez.(2)"
					ST46TTT.KOD(SatirNo) = Eski_StokKodu
					Kaydedemez = "E"
				End If
				
				If Eski_Miktar <> Degisen_Miktar Then
					Msgbox "Varyant2 Alanýnda Numune Satiri Deðiþtirilemez.(2)"
					ST46TTT.SF_MIKTAR(SatirNo) = Eski_Miktar
					Kaydedemez = "E"
				End If
				
				If Eski_Birim <> Degisen_Birim Then
					Msgbox "Varyant2 Alanýnda Numune Satiri Deðiþtirilemez.(2)"
					ST46TTT.SF_SF_UNIT(SatirNo) = Eski_Birim
					Kaydedemez = "E"
				End If
				
				If Eski_Varyant1 <> Degisen_Varyant1 Then
					Msgbox "Varyant2 Alanýnda Numune Satiri Deðiþtirilemez.(2)"
					ST46TTT.SF_VRI_VR_R1(SatirNo) = Eski_Varyant1					
					Kaydedemez = "E"
				End If
				
				If Eski_Varyant2 <> Degisen_Varyant2 Then
					ST46TTT.SF_VRI_VR_R2(SatirNo) = Eski_Varyant2				
					Kaydedemez = "E"
				End If
		
				If Eski_TeslimTarihi <> Degisen_TeslimTarihi Then
					Msgbox "Varyant2 Alanýnda Numune Satiri Deðiþtirilemez.(2)"
					ST46TTT.RTESTARIH(SatirNo) = Eski_TeslimTarihi					
					Kaydedemez = "E"
				End If
				
				If Eski_RTeslimTarihi <> Degisen_RTeslimTarihi Then
					Msgbox "Varyant2 Alanýnda Numune Satiri Deðiþtirilemez.(2)"

					Kaydedemez = "E"

					Kaydedemez = "E"
				End If

					ST46TTT.NOTE1(SatirNo) = Eski_Not2
					Kaydedemez = "E"
				End If
			End If
		Next
	End If
End Function

	cdoAnonymous = 0
	cdoBasic = 1
	cdoNTLM = 2 'NTLM

	mailUser = "freedom@yildizkalip.com"
	mailPass = "sf2552"
	SMTPPort = 587
	mailFrom = "freedom@yildizkalip.com"
	
	Set objMessage = CreateObject("CDO.Message")
	objMessage.From = trim(mailFrom)
	objMessage.To = strTo
	objMessage.Subject = strHeader
	objMessage.TextBody = strMsg
	
	if trim(DosyaAdi) <> "" Then
		objMessage.AddAttachment("file://" & DosyaAdi)
	End if
	
	objMessage.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/sendusing") = cdoSendUsingPort 
	
	objMessage.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/smtpserver") = trim(mailserver)
	
	objMessage.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/smtpauthenticate") = cdoBasic
	
	objMessage.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/sendusername") = trim(mailUser)
	
	objMessage.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/sendpassword") = trim(mailPass)
	
	objMessage.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/smtpserverport") = 587
	
	objMessage.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/smtpusessl") = False
	
	objMessage.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/smtpconnectiontimeout") = 60
	
	objMessage.Configuration.Fields.Update
	objMessage.Send
End Sub
