Sub StandartMaliyeteYaz(StokKodu, Tarih, i)
	If Trim(Tarih) = "" Then Tarih = Doc.BuGun
	
	rows = Doc.SelectEQ("STDMH0E", "KOD", StokKodu)
	If rows = 0 Then
		ret = Doc1.New_Voucher()

		STDMH0E.KOD = StokKodu
		STDMH0T.AddRow
		satirNo = STDMH0T.GetRecCount
	Else
		set ST = Doc.GetTableObject("AU_STDMH0E_SELECT")

			If Trim(ST.STDMALTYPE(j)) = "TK" Then

			End If

		If Trim(EvrakNo) <> "" Then
			ret = Doc1.Load_Voucher(EvrakNo)
			If ret <> 0 Then
				k = Doc.d7MsgBox(trim(StokKodu) & ": Standart Maliyet Tablosu Yüklenemedi. Ýþlem Kesilmiþtir.","Freedom ERP", MB_OK)

Function GrupKoduYarat(GrupKoduTipi,GrupKodu,GrupAciklamasi)

Sub OtomatikDovizKuru()
	KurUSD = Doc.GetKur("USD", Doc.DefaultPB, Doc.BuGun, "", -1)
	if KurUSD > 0 Then Exit Sub

	'if not YeniEvrakKaydedebilirMi("EXCRAT", Doc.Getd7UserName) Then Exit Sub

	Set objXMLDoc = Nothing
	Set objXMLDoc = CreateObject("Microsoft.XMLDOM")
	objXMLDoc.async = False

	Set Doc1 = Doc.CreateDocumentNoUI("EXCRAT", "")
	Call GetKurTable(Doc1, objXMLDoc, Doc.BuGun)	
End Sub

Sub GetKurTable(Doc1, objXMLDoc, Tarih)
	Set EXCE = Doc1.GetTableObject("EXCRATE")
	Set EXCT = Doc1.GetTableObject("EXCRATT")
	
	If Doc.Saat(5) =< "15:30" Then
		KURTarih = Doc1.StrDate_Increment(Tarih, -1)
	Else
		KURTarih = Tarih
	End If
	
	Call HaftaSonuysaGeriAl(KURTarih)

	xmlTarih = Replace(Left(KURTarih ,7), "/", "") & "/" & Replace(Doc1.StrToNorDate(KURTarih), "/", "")
	objXMLDoc.Load "http://www.tcmb.gov.tr/kurlar/" & xmlTarih & ".xml"

	Set Root = objXMLDoc.documentElement
	If Root is Nothing Then
		'k = doc.d7MsgBox("Kurlar alýnamadý " & xmlTarih, "Hata", MB_OK)
		Exit Sub
	End if
	Set KurList = Root.getElementsByTagName("Currency")

	EXCE.EVRAKNOTARIH = Tarih

	EXCT.Empty    
	For Each Kur In KurList
		ParaBirimi = Kur.Attributes(1).NodeValue
		'if trim(ParaBirimi) = "EUR" Then ParaBirimi = "EURO"
		if trim(Doc1.GetGecoValue("*PUNIT", "KOD", ParaBirimi, d7c_ErrMsgNo)) <> "" Then
			If ParaBirimi ="USD" or ParaBirimi ="EUR" Then 
				Birim = GetKurValue(Kur, "Unit")
				DovizAlis = GetKurValue(Kur, "ForexBuying")
				DovizSatis = GetKurValue(Kur, "ForexSelling")
				EfektifAlis = GetKurValue(Kur, "BanknoteBuying")
				EfektifSatis = GetKurValue(Kur, "BanknoteSelling")
			
				satirNo = EXCT.AddRow
				EXCT.SetCurrentRow(satirNo)
				EXCT.CODEFROM(satirNo) = ParaBirimi
				EXCT.CODETO(satirNo) = Doc1.DefaultPB
				EXCT.KURS_1(satirNo) = DovizAlis '/Doc1.CDbl(Birim)
				EXCT.KURS_3(satirNo) = DovizAlis '/Doc1.CDbl(Birim)
				EXCT.KURS_4(satirNo) = DovizSatis '/Doc1.CDbl(Birim)
				EXCT.KURS_5(satirNo) = EfektifAlis '/Doc1.CDbl(Birim)
				EXCT.KURS_6(satirNo) = EfektifSatis '/ Doc1.CDbl(Birim)
			End If
		End if
	Next
	Doc1.Save_Voucher
	Doc1.LoadEmpty_Voucher
End Sub

Function GetKurValue(Kurxml, KurTipi)
	Set KurAlisSatisxml = Kurxml.SelectSingleNode(KurTipi)
	If Not KurAlisSatisxml Is Nothing Then
		If Not KurAlisSatisxml.FirstChild Is Nothing Then
			GetKurValue = KurAlisSatisxml.FirstChild.Data
		End If
	End If
End Function

Sub HaftaSonuysaGeriAl(ByRef Tarih)
	Set d7Date = Doc.CreateSObject("d7Date")
	d7Date.SetDate(Tarih)
	DOW = d7Date.GetDOW()
	if DOW = 6 or DOW = 7 Then
		d7Date.Increment(5 - DOW)
		Tarih = d7Date.GetRevDate
	End if
End Sub
