Sub EtiketPrint()
	Set S01 = Doc.GetTableObject("STOK01")
	BarkodNo = Doc.LeftJustify(S01.SERINO, 30)
	StokKodu = Doc.GetDBFieldValue("D7KIDSLB", "SKOD", BarkodNo, "")
 	MAMULADI	= Trim(Doc.GetKartValue("STOK00","AD",StokKodu,-1))
	MAMULADI2	= Trim(Doc.GetKartValue("STOK00","NAME2",StokKodu,-1))
	Kalinlik	= Doc.GetKartValue("STOK00","B_YUKSELIK",StokKodu,-1)
	En	= Doc.GetKartValue("STOK00","B_EN",StokKodu,-1)
	Boy	= Doc.GetKartValue("STOK00","B_BOY",StokKodu,-1)
	Yogunluk = Doc.GetKartValue("STOK00","B_YOGUNLUK",StokKodu,-1)
	Musteri = Doc.GetDBFieldValue("D7KIDSLB", "CH_MUSTERI", BarkodNo, "") 

	Set tbld7InputBox = Doc.GetTableObject("D7INPUTBOX_SELECT")

	KESILEN = tbld7InputBox.KESILEN
	KALAN = tbld7InputBox.KALAN

	KESILEN_SAC_BILGILERI_KALINLIK	=	tbld7InputBox.SAC_KALINLIK
	KESILEN_SAC_BILGILERI_EN			=	tbld7InputBox.SAC_ENI/1000
	KESILEN_SAC_BILGILERI_BOY			=	tbld7InputBox.SAC_BOYU/1000
	KESILEN_SAC_BILGILERI_YOGUNLUK	=	tbld7InputBox.SAC_YOGUNLUK

	If KALAN <> 0 and KESILEN <> 0 Then 
		msgbox "Sadece KALAN yada KESÝLEN dolu olmalý"
		exit sub
	ElseIf KESILEN <> 0 Then
		HARCANAN_MIKTAR = KESILEN * KESILEN_SAC_BILGILERI_KALINLIK * KESILEN_SAC_BILGILERI_EN * KESILEN_SAC_BILGILERI_BOY * KESILEN_SAC_BILGILERI_YOGUNLUK
		HARCANAN_MIKTAR = Doc.Ceil(HARCANAN_MIKTAR)
	ElseIf KALAN <> 0 Then
		HARCANAN_MIKTAR = KALAN * KESILEN_SAC_BILGILERI_KALINLIK * KESILEN_SAC_BILGILERI_EN * KESILEN_SAC_BILGILERI_BOY * KESILEN_SAC_BILGILERI_YOGUNLUK
		HARCANAN_MIKTAR = Doc.Ceil(HARCANAN_MIKTAR)

		If HARCANAN_MIKTAR =< S01.SF_MIKTAR Then
			HARCANAN_MIKTAR = S01.SF_MIKTAR - HARCANAN_MIKTAR
		Else
			Msgbox "Harcanan Miktar: " & HARCANAN_MIKTAR & " kg," & VBCRLF & "Stokdaki Miktardan: " & S01.SF_MIKTAR & " kg ' dan fazla." & vbcrlf & "Kontrol edin!" 
			HARCANAN_MIKTAR = 0
			Exit Sub
		End If
	Else
		msgbox "giriþleri kontrol et"
	End If
	'-----------------------------------------------------------
	If Left(Trim(StokKodu), 2) = "HS" Then
				S01.SF_PAKETMIKTARI = "1"
				S01.SF_PAKETICERIGI =  HARCANAN_MIKTAR
				S01.SF_PAKETMIKTARI = "1"

	ElseIf Left(Trim(StokKodu), 2) = "HR" Then
				RULOPAKETI_ICINDEKI_RULO_ADETI = InputBox ("Paket içi Rulo adeti?", "PAKET ÝÇÝ RULO ADETÝ","3")
				KESILEN_SAC_BILGILERI_ADET = InputBox ("Kaç adet rulo kesildi?", "KESÝLEN RULO SAC BÝLGÝSÝ","1")
				ORAN = Doc.Div(KESILEN_SAC_BILGILERI_ADET, RULOPAKETI_ICINDEKI_RULO_ADETI)
				S01.SF_PAKETICERIGI = S01.SF_MIKTAR *  ORAN 
				S01.SF_PAKETICERIGI = FormatNumber(S01.SF_PAKETICERIGI,3)
	End If
	Doc.RUN_APP_CANCEL
End Sub

Sub OtomatikDovizKuru()
	KurUSD = Doc.GetKur("USD", Doc.DefaultPB, Doc.BuGun, "", -1)
	
	if KurUSD > 0 Then Exit Sub

	if not YeniEvrakKaydedebilirMi("EXCRAT", Doc.Getd7UserName) Then Exit Sub

	Set objXMLDoc = Nothing
	Set objXMLDoc = CreateObject("Microsoft.XMLDOM")
	objXMLDoc.async = False

	Set Doc1 = Doc.CreateDocumentNoUI("EXCRAT", "")
	Call GetKurTable(Doc1, objXMLDoc, Doc.BuGun)

Sub GetKurTable(Doc1, objXMLDoc, Tarih)
	Set EXCE = Doc1.GetTableObject("EXCRATE")
	Set EXCT = Doc1.GetTableObject("EXCRATT")

	KURTarih = Doc1.StrDate_Increment(Tarih, -1)
	Call HaftaSonuysaGeriAl(KURTarih )

	xmlTarih = Replace(Left(KURTarih ,7), "/", "") & "/" & Replace(Doc1.StrToNorDate(KURTarih ), "/", "")

	objXMLDoc.Load "http://www.tcmb.gov.tr/kurlar/" & xmlTarih & ".xml"
'	objXMLDoc.Load "c:\today.xml"

	Set Root = objXMLDoc.documentElement
	if Root is Nothing Then
'		k = doc.d7MsgBox("Kurlar alýnamadý " & xmlTarih, "Hata", MB_OK)
		Exit Sub
	End if
	Set KurList = Root.getElementsByTagName("Currency")

	EXCE.EVRAKNOTARIH = Tarih

	EXCT.Empty    
	For Each Kur In KurList
		ParaBirimi = Kur.Attributes(1).NodeValue

		If Trim(ParaBirimi) = "USD" OR Trim(ParaBirimi) = "EUR" Then

		'if trim(ParaBirimi) = "EUR" Then ParaBirimi = "EURO"

			if trim(Doc1.GetGecoValue("*PUNIT", "KOD", ParaBirimi, d7c_ErrMsgNo)) <> "" Then
				Birim = GetKurValue(Kur, "Unit")
				DovizAlis = GetKurValue(Kur, "ForexBuying")
				DovizSatis = GetKurValue(Kur, "ForexSelling")

				EfektifAlis = GetKurValue(Kur, "BanknoteBuying")
				EfektifSatis = GetKurValue(Kur, "BanknoteSelling")
		
				satirNo = EXCT.AddRow
				EXCT.SetCurrentRow(satirNo)
				EXCT.CODEFROM(satirNo) = ParaBirimi
				EXCT.CODETO(satirNo) = Doc1.DefaultPB
				EXCT.KURS_1(satirNo) = DovizSatis / Doc.CDbl(Birim)
				EXCT.KURS_3(satirNo) = DovizAlis / Doc.CDbl(Birim)
				EXCT.KURS_4(satirNo) = DovizSatis / Doc.CDbl(Birim)
				EXCT.KURS_5(satirNo) = EfektifAlis / Doc.CDbl(Birim)
				EXCT.KURS_6(satirNo) = EfektifSatis / Doc.CDbl(Birim)
			End If
		End if
	Next
	Doc1.Save_Voucher
	Doc1.LoadEmpty_Voucher
End Sub

Function GetKurValue(Kurxml, KurTipi)

Function GetIrsaliyeTutari(S60,satirNo, Caller)
	GetIrsaliyeTutari = 0
	if S60.SF_MIKTAR(satirNo) <> 0 then
		FaturalanacakMiktar = S60.SF_MIKTAR(satirNo) - S60.SF_FATURALANANMIKTAR(satirNo)
		if S60.TL_KDVDAHILTUTAR(satirNo) = 0 then
			SiparisNo = Doc.GetDBFieldValue("STOK40T","EVRAKNO",S60.SIPARTNO(satirNo),"ARTNO")
			MuhtemelNitelik = Doc.GetDBFieldValue("STOK40E","MUHT_FATURA_NITELIK",SiparisNo,"")
			KurAktarma = Doc.GetDBFieldValue("STOK40E","KURKESINLESTIRME",SiparisNo,"")
			SiparisMiktari = Doc.GetDBFieldValue("STOK40T","SF_MIKTAR",S60.SIPARTNO(satirNo),"ARTNO")
			KDVHaricTutar = Doc.GetDBFieldValue("STOK40T","OR_IVD_KH_TUTAR",S60.SIPARTNO(satirNo),"ARTNO")
			KDVTutari = Doc.GetDBFieldValue("STOK40T","OR_KDVTUT_EFEKTIF",S60.SIPARTNO(satirNo),"ARTNO")
				If MuhtemelNitelik = "XK" Then
				SiparisTutari = KDVHaricTutar + KDVTutari
			Else
				SiparisTutari = KDVHaricTutar
			End If
			SiparisPB = Doc.GetDBFieldValue("STOK40T","PRICEUNIT",S60.SIPARTNO(satirNo),"ARTNO")

			SiparisFiyati = Doc.Div(SiparisTutari, SiparisMiktari)
			if KurAktarma = "E" then
				KurDegeri = Doc.GetDBFieldValue("STOK40T","KUR",S60.SIPARTNO(satirNo),"ARTNO")
			Else
				KurDegeri = Doc.GetKurN(SiparisPB, "YTL",Doc.BuGun,"",-1)
			End if
			GetIrsaliyeTutari = SiparisFiyati * KurDegeri * FaturalanacakMiktar
		else
			GetIrsaliyeTutari = Doc.Div(S60.TL_KDVDAHILTUTAR(satirNo) * FaturalanacakMiktar, S60.SF_MIKTAR(satirNo))
		end if
	end if
End Function
Function GetSiparisTutari()

Function FIFOyaUygunMu(SeriNo)
	if Left(SeriNo, 2) = "--" Then Exit Function

	Set S01_2 = Doc.GetTableObject("STOK01_PDA")
'	if trim(S01_2.PRX_LOCATION1) <> "" Then Exit Function

	Set S26E = Doc.GetTableObject("STOK26E") 

	HedefAmb = GetTargetAmbCode(S01_2.PRX_LOCATION1, S01_2.PRX_LOCATION2, S01_2.PRX_LOCATION3, S01_2.PRX_LOCATION4)
	if HedefAmb = "" Then HedefAmb = S26E.TARGETAMBCODE

	if trim(S26E.TARGETAMBCODE) <> "IM" Then Exit Function
'	if Not(HedefAmb >= "T00" and HedefAmb < "TA") Then Exit Function

	Set S01 = Doc.Select1("sfsfd", "STOK01", "SERINO='"+trim(SeriNo)+"'", "", -1)
	if S01.GetRowCount() > 0 Then
		Depo = S01.AMBCODE
		LotNo = Left(S01.LOTNUMBER, 6)
		StokKodu = S01.KOD
		Set S01 = Doc.Select1("sfsfd", "STOK01", "KOD='"+StokKodu+"'&AMBCODE='"+Depo+"'&LOTNUMBER<'"+LotNo+"'", "LOTNUMBER", -1)
		for i = 1 to S01.GetRowCount()
			Lot = Left(S01.LOTNUMBER(i), 6)
			if Lot <> EskiLot And i > 1 Then Exit For
			if i = 1 Then Hata = "Okunacak Ýlk Lot:" & trim(S01.LOTNUMBER(i)) & vbcrlf
'			Hata = Hata & "Lokasyon : " & GetLokasyon(S01, i) & ", Miktar : " & S01.SF_MIKTAR(i) & vbcrlf
			Hata = Hata & "L:" & GetLokasyon(S01, i) & ",M:" & S01.SF_MIKTAR(i) & vbcrlf

			EskiLot = Lot
		Next

	End if

	if trim(Hata) <> "" Then
'		FIFOyaUygunMu = "Okutulan etiket FIFO 'ya uygun deðil" & vbcrlf & Hata
		FIFOyaUygunMu = Hata
	End if

End Function

		Exit Function

	QStr = QStr & "&LOCATION3='" & L3 & "'"
	QStr = QStr & "&LOCATION4='" & L4 & "'"

	End if

Sub STOKGERIAL
	dim i, durak
	'On Error Resume Next
	Set Doc1 = Doc.CreateDocumentNoUI("STOK21", "")
	Set S21T = Doc1.GetTableObject("STOK21T")
	S21T.SAFEPACK()
	Doc1.Load_Voucher "231       " 
	durak = 0
	For i = 1 to S21T.GetRecCount()
		If Trim(S21T.SERINO(i)) = Trim(BarkodNo) Then 
			S21T.KOD(i) = ""
			S21T.LOTNUMBER(i) = ""
			S21T.SERINO(i) = ""
			S21T.CIKAN_MIKTAR(i) = ""
			S21T.GIREN_MIKTAR(i) = ""
			S21T.KOD_STOK00_AD(i) = ""
			S21T.D_ID_KAYNAK(i) = ""
			S21T.D_ID(i) = ""
			S21T.SF_SF_UNIT(i) = ""
			S21T.STOK_MIKTAR(i) = ""
			S21T.KOD_STOK00_IUNIT(i) = ""
			S21T.NOTES(i) = ""

			Doc1.Save_Voucher
			Doc1.LoadEmpty_Voucher
			Doc1.DeleteDocumentNoUI "STOK21", ""
			Doc1.DeleteTableObject "S21T"
			Exit For
		End If
	Next
	
End Sub

Sub EtiketPrint()
	Set S01 = Doc.GetTableObject("STOK01")
	BarkodNo = Doc.LeftJustify(S01.SERINO, 30)
	StokKodu = Doc.GetDBFieldValue("D7KIDSLB", "SKOD", BarkodNo, "")
 	MAMULADI	= Trim(Doc.GetKartValue("STOK00","AD",StokKodu,-1))
	MAMULADI2	= Trim(Doc.GetKartValue("STOK00","NAME2",StokKodu,-1))
	Kalinlik	= Doc.GetKartValue("STOK00","B_YUKSELIK",StokKodu,-1)
	En	= Doc.GetKartValue("STOK00","B_EN",StokKodu,-1)
	Boy	= Doc.GetKartValue("STOK00","B_BOY",StokKodu,-1)
	Yogunluk = Doc.GetKartValue("STOK00","B_YOGUNLUK",StokKodu,-1)
	Musteri = Doc.GetDBFieldValue("D7KIDSLB", "CH_MUSTERI", BarkodNo, "") 

	Set tbld7InputBox = Doc.GetTableObject("D7INPUTBOX_SELECT")

	KESILEN_SAC_BILGILERI_KALINLIK	=	tbld7InputBox.SAC_KALINLIK
	KESILEN_SAC_BILGILERI_EN			=	tbld7InputBox.SAC_ENI/1000
	KESILEN_SAC_BILGILERI_BOY			=	tbld7InputBox.SAC_BOYU/1000
	KESILEN_SAC_BILGILERI_YOGUNLUK	=	tbld7InputBox.SAC_YOGUNLUK

	If Left(Trim(StokKodu), 2) = "HS" Then
		KESILEN = tbld7InputBox.KESILEN
		KALAN = tbld7InputBox.KALAN
		If KESILEN <> 0 and KALAN <> 0 Then 
			msgbox "Sadece KALAN yada KESÝLEN dolu olmalý"
			exit sub
		ElseIf tbld7InputBox.KESILEN <> 0 Then
			HARCANAN_MIKTAR = KESILEN * KESILEN_SAC_BILGILERI_KALINLIK * KESILEN_SAC_BILGILERI_EN * KESILEN_SAC_BILGILERI_BOY * KESILEN_SAC_BILGILERI_YOGUNLUK
			HARCANAN_MIKTAR = Doc.Ceil(HARCANAN_MIKTAR)
		ElseIf KALAN <> 0 Then
			HARCANAN_MIKTAR = KALAN * KESILEN_SAC_BILGILERI_KALINLIK * KESILEN_SAC_BILGILERI_EN * KESILEN_SAC_BILGILERI_BOY * KESILEN_SAC_BILGILERI_YOGUNLUK
			HARCANAN_MIKTAR = Doc.Ceil(HARCANAN_MIKTAR)
			If HARCANAN_MIKTAR =< S01.SF_MIKTAR Then
				HARCANAN_MIKTAR = S01.SF_MIKTAR - HARCANAN_MIKTAR
			Else
				Msgbox "Harcanan Miktar: " & HARCANAN_MIKTAR & " kg," & VBCRLF & "Stokdaki Miktardan: " & S01.SF_MIKTAR & " kg ' dan fazla." & vbcrlf & "Kontrol edin!" 
				HARCANAN_MIKTAR = 0
				Exit Sub
			End If
		Else
			msgbox "giriþleri kontrol et"
		End If
		S01.SF_PAKETMIKTARI = "1"
		S01.SF_PAKETICERIGI =  HARCANAN_MIKTAR
		S01.SF_PAKETMIKTARI = "1"
	ElseIf Left(Trim(StokKodu), 2) = "HR" Then
		TOPLAMPAKETICIRULO = tbld7InputBox.KESILEN
		HARCANAN = tbld7InputBox.KALAN
		ORAN = Doc.Div(HARCANAN, TOPLAMPAKETICIRULO)
		S01.SF_PAKETMIKTARI = "1"
		S01.SF_PAKETICERIGI = S01.SF_MIKTAR *  ORAN 
		S01.SF_PAKETICERIGI = FormatNumber(S01.SF_PAKETICERIGI,3)
		S01.SF_PAKETMIKTARI = "1"
	Else
		Msgbox "Sadce HR ve HS kodlu saclar için." & vbcrlf & "Ýþlem Ýptal."
		Exit Sub
	End If

	Doc.RUN_APP_CANCEL
End Sub

Sub MinimumStokKontrol()
	Set U12 = Doc.Select1("d4gfy7u", "UDEFT12", "KOD='" & Doc.Getd7UserName() & "'&OZEL_HAK_KODU='GUVENLIK'", "", -1)
	if U12.GetRowCount() = 0 Then Exit Sub

	Set fff = Doc.CreateSObject("BATCHERRORMODE")
	Doc.SetDocParameterValue "BATCHERRORMODE_HEADER_MESSAGE", "Güvenik Stoðunun Altýna Düþenler :"

	Set S00 = Doc.Select1("STOK0011", "STOK00", "B_GUVENLIKSTOGU>0.00001&AP10='1'", "", -1)
	
	For i=1 to S00.GetRecCount()
		TSiparis=0
		TTalep=0

		DEPO_Stok=Doc.Stok00c_SonSerbest(S00.KOD(i),"","")
		
		Set S46T = Doc.Select1("STOK46T11", "STOK46T", "KOD='"& S00.KOD(i) &"'&ACIKKAPALI=' '", "", -1)
		if S46T.GetRowCount() > 0 Then
			S46T.GroupBy("KOD")
			TSiparis = TSiparis + S46T.SF_MIKTAR - S46T.SF_NETKAPANANMIK
		End if

		Set S47T = Doc.Select1("STOK47T11", "STOK47T", "KOD='"& S00.KOD(i) &"'&ACIKKAPALI=' '", "", -1)
		if S47T.GetRowCount() > 0 Then
			S47T.GroupBy("KOD")
			TTalep=TTalep + S47T.SF_MIKTAR - S47T.SF_NETKAPANANMIK
		End if
		
		Fark= DEPO_Stok - S00.B_GUVENLIKSTOGU(i) 
		if Fark < 0 then
			Doc.d7MsgBox trim(S00.KOD(i)) & " - " & trim(S00.AD(i)), "", MB_OK
			Doc.d7MsgBox "      Güvenlik Stoðu = " & _
							S00.B_GUVENLIKSTOGU(i) & ", Depo Mevcudu = " & DEPO_Stok & "  Fark = " & Fark & _
							"  Açýk Talep=" & TTalep & " Açýk Sipariþ=" & TSiparis, "", MB_OK
		end if		
	Next
End Sub
Sub MinimumStokKontrol2(HakKodu)
	If not OzelHakVarMi(HakKodu) Then Exit Sub

	QStr = "SELECT S00.GK_2, S00.KOD, S00.AD, S00.NAME2, S00.B_GUVENLIKSTOGU, S01.SF_MIKTAR, S46T.SIP_MIKTAR, S47T.TALEP_MIKTAR"
	QStr = QStr & " FROM STOK00 S00"
	QStr = QStr & " LEFT JOIN ("
	QStr = QStr & " 	SELECT KOD, SUM(SF_MIKTAR) SF_MIKTAR"
	QStr = QStr & " 	FROM STOK01"
	if HakKodu = "BGUVENLIK" Then
		QStr = QStr & " 	WHERE AMBCODE='70BAK'"
	End if
	QStr = QStr & " 	GROUP BY KOD"
	QStr = QStr & " ) S01 ON S01.KOD=S00.KOD"
	QStr = QStr & " LEFT JOIN ("
	QStr = QStr & " 	SELECT KOD, Sum(SF_MIKTAR-SF_NETKAPANANMIK) SIP_MIKTAR"
	QStr = QStr & " 	FROM STOK46T"
	QStr = QStr & " 	WHERE ACIKKAPALI=' '"
	QStr = QStr & " 	AND (SF_MIKTAR-SF_NETKAPANANMIK)>0"
	QStr = QStr & " 	GROUP BY KOD"
	QStr = QStr & " ) S46T ON S46T.KOD=S00.KOD"
	QStr = QStr & " LEFT JOIN ("
	QStr = QStr & " 	SELECT KOD, Sum(SF_MIKTAR-SF_NETKAPANANMIK) TALEP_MIKTAR"
	QStr = QStr & " 	FROM STOK47T"
	QStr = QStr & " 	WHERE ACIKKAPALI=' '"
	QStr = QStr & " 	AND (SF_MIKTAR-SF_NETKAPANANMIK)>0"
	QStr = QStr & " 	GROUP BY KOD"
	QStr = QStr & " ) S47T ON S47T.KOD=S00.KOD"
	QStr = QStr & " WHERE S00.B_GUVENLIKSTOGU>0.00001"
	QStr = QStr & " AND S00.CODETYPE='D'"
	QStr = QStr & " AND S00.AP10='1'"
	QStr = QStr & " AND COALESCE(S01.SF_MIKTAR,0)<S00.B_GUVENLIKSTOGU"
	if HakKodu = "KGUVENLIK" Then
		QStr = QStr & " AND S00.KOD LIKE 'ZP%'"
	Elseif HakKodu = "BGUVENLIK" Then
		QStr = QStr & " AND S00.GK_2 IN ('MEKANIK','ELEKTRIK')"
	Else
		QStr = QStr & " AND S00.KOD NOT LIKE 'ZP%' AND S00.GK_2 NOT IN ('MEKANIK','ELEKTRIK')"
	End if

	if HakKodu = "BGUVENLIK" Then
		QStr = QStr & " ORDER BY S00.GK_2, S00.KOD"
	End if
	Set S00 = Doc.RunSQLQuery("szdfsehah", QStr)
	if S00.GetRowCount() = 0 Then Exit Sub
	
	MesajBaslik = "Güvenlik Stoðunun Altýna Düþen Malzemeler :"
	BaslikArray = Array("Stok Gk-2", "Stok Kodu","Stok Adý","Stok Adý-2", "Güvenlik Stoðu","Depo Mevcudu","Açýk Sipariþ Miktarý","Açýk Talep Miktarý")
	Doc.RunMacro3 "MAKR7S.FREEMAIL", "ShowTableHTMLMessage", S00, MesajBaslik, BaslikArray
End Sub
