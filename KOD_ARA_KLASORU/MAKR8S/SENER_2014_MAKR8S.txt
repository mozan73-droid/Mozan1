Sub AcikIsemriKontrol()
	if MMPS10E.ACIK_KAPALI = "K" Then Exit Sub
	if trim(MMPS10E.GK_2) = "1" Then Exit Sub
	if Doc.DataEntryMode <> "GUI" Then Exit Sub
	
	Set ddd = Doc.CreateSObject("BATCHERRORMODE")
	rows = Doc.SelectEQ("MMPS10E", "ACIK_KAPALI+MAMULSTOKKODU", " " & MMPS10E.MAMULSTOKKODU)
	MsgStr = ""
	if rows > 0 Then
		Set M10E = Doc.GetTableObject("AU_MMPS10E_SELECT")
		for i = 1 to M10E.GetRowCount()
			if M10E.EVRAKNO(i) <> MMPS10E.EVRAKNO Then
				MsgStr = MsgStr & "MPS No : " & trim(M10E.EVRAKNO(i))
				MsgStr = MsgStr & ", Üretim Tarihi : " & Doc.StrToNorDate(M10E.PLBTARIH_OFO(i))
				MsgStr = MsgStr & ", MPS Miktarý : " & M10E.SF_TOPLAMMIKTAR(i)
				Uretilen = M10E.SF_TOPLAMMIKTAR(i) - M10E.SF_BAKIYEMIKTAR(i)
				MsgStr = MsgStr & ", Üretilen : " & Uretilen
				MsgStr = MsgStr & ", Kalan : " & M10E.SF_BAKIYEMIKTAR(i) & vbcrlf
			End if
		Next
	End if
	if trim(MsgStr) <> "" Then
		doc.d7MsgBox "Bu koda ait açýk iþ emir(ler)i var : " & vbcrlf & MsgStr, "", MB_OK
		MMPS10E.GK_2 = "1"
	End if
End Sub

Function eskifiyatfarkli()
	eskifiyatfarkli = False
	Set S74E = Doc.GetTableObject("STOK74E")
	Set S74T = Doc.GetTableObject("STOK74T")
			msgbox 111
	Set ss = Doc.CreatesObject("BATCHERRORMODE")
	for i = 1 to S74T.GetRecCount()
		Set S74TG = Doc.Select1("daff5agga","STOK74T","KOD = '"& S74T.KOD(i) &"'","<TARIH",-1)
		if S74TG.GetRecCount() > 0 then
			if S74T.OR_FIYAT(i) <> S74TG.OR_FIYAT(1) then
				hata2 = hata2 & S74T.KOD(i) &"Stokkodu bu faturada " & S74T.OR_FIYAT(i) & " "&S74T.PRICEUNIT(i) &" olarak fiyatlandýrýlýrken " & doc.strtonordate(S74TG.TARIH(1)) &" tarihli bir önceki " & S74TG.EVRAKNO(1) & " " & " nolu faturasýnda " & S74TG.OR_FIYAT(1) & " "&S74TG.PRICEUNIT(1) &" olarak fiyatlandýrýlmýþ " &vbcrlf
				Exit for
			End if
		End if
	Next
	if trim(hata2) <> "" then
		Doc.d7msgbox hata2,"",MB_OK
		eskifiyatfarkli = True
		Call EPostaGonder(hata2)
	End if
	Set ss = Nothing
End Function

Function GetPDurusSure(BTarih, BSaat, ETarih, ESaat)
'	Set KK = Doc.CreateSObject("BATCHERRORMODE")
	Dim MolaArray(1,1)
	MolaArray(0,0) = "12:30"
	MolaArray(0,1) = "13:30"
	MolaArray(1,0) = "18:00"
	MolaArray(1,1) = "18:30"
'	MolaArray(2,0) = "04:00"
'	MolaArray(2,1) = "04:30"

	GunSayisi = Doc.CDateToInt(ETarih) - Doc.CDateToInt(BTarih)
	MolaSure = 0
	For i = 0 to uBound(MolaArray, 1)
		For j = 0 to GunSayisi
			if j < GunSayisi Then
				BaslangicSaat = BSaat
				BitisSaat = "23:59"
			Elseif j > 0 Then
				BaslangicSaat = "00:00"
				BitisSaat = ESaat
			Else
				BaslangicSaat = BSaat
				BitisSaat = ESaat
			End if
			Gun = Doc.StrDate_Increment(BTarih, j)
'			Doc.d7MsgBox i &  ", " & j &  ", " & Gun &  ", " & BaslangicSaat &  ", " & BitisSaat & ", " & MolaArray(i,0) & _
'					 ", " & MolaArray(i,1), "", MB_OK
			if not(Gun & BitisSaat < Gun & MolaArray(i,0) or Gun & BaslangicSaat > Gun & MolaArray(i,1)) Then
				if Gun & MolaArray(i,0) >= Gun & BaslangicSaat and Gun & MolaArray(i,1) <= Gun & BitisSaat Then
					MolaSure = MolaSure + Doc.DTDTDif1(Gun, MolaArray(i,0), Gun, MolaArray(i,1))
				Elseif Gun & MolaArray(i,0) >= Gun & BaslangicSaat and Gun & MolaArray(i,1) > Gun & BitisSaat Then
					MolaSure = MolaSure + Doc.DTDTDif1(Gun, MolaArray(i,0), Gun, BitisSaat)
				Elseif Gun & MolaArray(i,0) <= Gun & BaslangicSaat and Gun & MolaArray(i,1) <= Gun & BitisSaat Then
					MolaSure = MolaSure + Doc.DTDTDif1(Gun, BaslangicSaat, Gun, MolaArray(i,1))
				End if
			End if
		Next
	Next
	GetPDurusSure = MolaSure
End Function

				hata2 = hata2 & S74T.KOD(i) &"Stokkodu bu faturada " & S74T.OR_FIYAT(i) & " "&S74T.PRICEUNIT(i) &" olarak fiyatlandýrýlýrken " & doc.strtonordate(S74TG.TARIH(1)) &" tarihli bir önceki " & S74TG.EVRAKNO(1) & " " & " nolu faturasýnda " & S74TG.OR_FIYAT(1) & " "&S74TG.PRICEUNIT(1) &" olarak fiyatlandýrýlmýþ " &vbcrlf
				Exit for
			End if
		End if
	Next
	if trim(hata2) <> "" then
		Doc.d7msgbox hata2,"",MB_OK
		eskifiyatfarkli = True
		Call EPostaGonder(hata2)
	End if
	Set ss = Nothing
End Function
