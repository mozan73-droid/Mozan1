Sub OnLogin()
	if OzelHakVarMi("KALIP") Then
		Set S00C = GetKalipBakimList()
		Call KalipBakimKontrol(S00C)
	Elseif OzelHakVarMi("KALIBRASYO") Then
		Call KalibrasyonKontrol()	
	Else
		Set S00C = GetKalipBakimList()
		if S00C.GetRowCount() > 0 Then
			Call KalipBakimUyariMail(S00C)
		End if
	End if
	Call Raf_omuru()
	
	Call Kalibrasyon_mail()
	
End Sub
Sub KalibrasyonKontrol()
	SonTarih = Doc.StrDate_Increment(Doc.BuGun, 60)
	QStr = "ACIKKAPALI='A'&RECTARIH0<='"& SonTarih &"'"
	Set S00 = Doc.Select1("cccddd", "SRVK00", QStr, "RECTARIH0", -1)
	Doc.SetDocParameterValue "BATCHERRORMODE_HEADER_MESSAGE","Kalibrasyon Zamaný Gelen Cihazlar :"
	Set k = Doc.CreateSObject("BATCHERRORMODE")
	For i = 1 to S00.GetRowCount()
		UyariGunu = Doc.CDbl(Doc.GetDBFieldValue("SRVKC0", "TTANIM3_TEXT", S00.CIHAZKARTNO(i), ""))
		if UyariGunu = 0 Then UyariGunu = 15

		End if
	Next

		Doc.d7MsgBox Uyari, "", MB_OK
	End if
End Sub

Sub KalipBakimUyariMail(S00C)
	if not YeniEvrakKaydedebilirMi("SRVPC0", Doc.Getd7UserName) Then Exit Sub

	Kaliplar = ""
	MsgStr = "Bakým zamaný gelen kalýplar :"
	For i = 1 to S00C.GetRowCount()
		SonMesajTarihi = Doc.GetUDFValue("SRVPC0", S00C.KOD(i), "MTARIH", 0)
		If SonMesajTarihi <> Doc.StrToNorDate(Doc.BuGun) Then
			MsgStr = MsgStr & "Kalýp Kodu : " & S00C.KOD(i) & ", Kalan Ömür : " & Doc.Round(S00C.KALAN_MIKTAR(i),0) & vbcrlf
			Kaliplar = Kaliplar & "," & S00C.KOD(i)
		End if
	Next
	if trim(Kaliplar) = "" Then Exit Sub

	Set U12 = Doc.Select1("de443d44ts", "UDEFT12", "OZEL_HAK_KODU='KALIPMAIL'", "", -1)
	For i = 1 to U12.GetRowCount()
		email = Doc.GetDBFieldValue("UDEF", "EMAIL", U12.KOD(i), "")
		if trim(email) <> "" Then
			Call SendMail(email,"## Bakým Zamaný Gelen Kalýplar ##",MsgStr,"")
		End if
	Next
	if Len(trim(Kaliplar)) > 0 Then
		Call KartKTAGuncelle(Mid(Kaliplar, 2))
	End if
End Sub

Function YeniEvrakKaydedebilirMi(ProgramKodu, KullaniciAdi)
	if trim(KullaniciAdi) = "D7KSUPER" or KullaniciAdi = "FX" Then
		YeniEvrakKaydedebilirMi = True
		Exit Function
	End if

	GrupKullanici = Doc.GetDBFieldValue("UDEF", "AYNIHAK", KullaniciAdi, "")
	if trim(GrupKullanici) <> "" Then KullaniciAdi = GrupKullanici

	YeniEvrakKaydedebilirMi = False
	Set UT1 = Doc.Select1("daef3ff", "UDEFT1", "KOD='" & KullaniciAdi & "'&PROGRAMKODU='" & ProgramKodu & "'", "", -1)
	if UT1.GetRowCount() > 0 Then
		if UT1.RIGHT_W="1" Then
			YeniEvrakKaydedebilirMi = True
			Exit Function
		End if
	Else
		YeniHakki = Doc.GetDBFieldValue("UDEF", "PR_RIGHT_W", KullaniciAdi, "")
		if YeniHakki = "1" Then
			YeniEvrakKaydedebilirMi = True
			Exit Function
		End if
	End if
End Function
Function GetKalipBakimList()

Sub Raf_omuru()

    FreeDomUserName = Doc.Getd7UserName
    Flag = TRUE

    Set A_UDEFT12 = Doc.Select1("LLKKFFTTRR", "UDEFT12", "KOD = '"& FreeDomUserName &"'", "", -1)

    if A_UDEFT12.GetRecCount() = 0 Then Exit Sub    

    for i = 1 TO A_UDEFT12.GetRecCount()
        if trim(A_UDEFT12.OZEL_HAK_KODU(i)) = "RAFO" Then Flag = FALSE
    next

    if     Flag  Then Exit Sub

    SayfaBoyu = 297
    SayfaEni = 300
    SolMargin = 6
    SagMargin = 1
    UstMargin = 20
    AltMargin = 1

    Set pe1 = doc.CreatePrintEngineOutput()
    res = pe1.Open(0,"Order Confirmation", SayfaBoyu, SayfaEni, "")
    res = pe1.SelectFont(Arial, 18)
    res = pe1.SetAlign("CENTER")    
    res = pe1.write(UstMargin , 1 , 10 , SayfaEni - 3, "RAF ÖMRÜ YAKLAÞAN / GEÇEN STOK KODLARI" ) 
    res = pe1.write(pe1.GetTop + 20 , 15 , 10 , 60 , "")

    res = pe1.SetAlign("Left")
    res = pe1.SelectFont(Arial, 12)
    satir_al = 0

    Dim QSTR_1
    QSTR_1 =""

    donem_1 = trim(doc.Customp_GetS("_DV_OFR"))
    donem_2 = mid(donem_1, 1, Len(donem_1) - 4) & Trim(CDbl(mid(donem_1,Len(donem_1) - 3 , 4)) - 1)
    donem_buyil = mid(donem_1, 1, Len(donem_1) - 4) & Trim(CDbl(mid(donem_1,Len(donem_1) - 3 , 4)) + 1)

    QSTR_1 = QSTR_1 &" SELECT S1.KOD , S1.LOTNUMBER, SSTK.TARI_H, ST001.B_B_TECH_14 FROM STOK01 S1 "
    QSTR_1 = QSTR_1 &" INNER JOIN "
    QSTR_1 = QSTR_1 &"     (SELECT STA.KOD , STA.LOTNUMBER , MIN(STA.TARIH) TARI_H FROM ( "
    QSTR_1 = QSTR_1 &"     SELECT KOD,LOTNUMBER, TARIH FROM STOK10A  WHERE LOTNUMBER IN(SELECT LOTNUMBER FROM STOK01 WHERE TRIM(LOTNUMBER)IS NOT NULL) AND EVRAKTYPE IN ('MG','SI','DT', 'DG', 'FG') AND SF_MIKTAR > '0' "
    QSTR_1 = QSTR_1 &" UNION ALL "
    QSTR_1 = QSTR_1 &" SELECT KOD,LOTNUMBER, TARIH FROM "& donem_1 &".STOK10A   WHERE LOTNUMBER IN(SELECT LOTNUMBER FROM "& donem_buyil &".STOK01 WHERE TRIM(LOTNUMBER)IS NOT NULL) AND EVRAKTYPE IN ('MG','SI','DT', 'DG' , 'FG')  AND SF_MIKTAR > '0' "
    QSTR_1 = QSTR_1 &" UNION ALL "
    QSTR_1 = QSTR_1 &" SELECT KOD,LOTNUMBER, TARIH FROM "& donem_2 &".STOK10A  WHERE LOTNUMBER IN(SELECT LOTNUMBER FROM "& donem_buyil &".STOK01 WHERE TRIM(LOTNUMBER)IS NOT NULL) AND EVRAKTYPE IN ('MG','SI','DT', 'DG' , 'FG')  AND SF_MIKTAR > '0' "
    QSTR_1 = QSTR_1 &" )STA GROUP BY STA.KOD, STA.LOTNUMBER) SSTK "
    QSTR_1 = QSTR_1 &" ON  SSTK.LOTNUMBER = S1.LOTNUMBER AND SSTK.KOD = S1.KOD "
    QSTR_1 = QSTR_1 &" LEFT JOIN STOK00 ST001 ON ST001.KOD = S1.KOD "    
    QSTR_1 = QSTR_1 &" GROUP BY S1.KOD , S1.LOTNUMBER, SSTK.TARI_H , ST001.B_B_TECH_14 "
            
    Set A_STOK01 = Doc.RunSqlQuery("SSEERR", QSTR_1)

    if A_STOK01.GetRecCount() = 0 Then Exit Sub

    for i = 1 To A_STOK01.GetRecCount()

        if A_STOK01.B_B_TECH_14(i) > 0 Then 
                raf_omru = 365 * A_STOK01.B_B_TECH_14(i)
                giris_tarihi_sayi =  Doc.CDateToInt(Doc.Bugun()) - Doc.CDateToInt(A_STOK01.TARI_H(i)) 
                if raf_omru - giris_tarihi_sayi <= 30 Then
                    satir_al =     satir_al + 1 
                    if satir_al > 16 Then
                         res = pe1.PageBreak()    
                         satir_al = 0    
                    end if
                    res = pe1.write(pe1.GetTop + 7 , 15 , 10 , 80 , "Kod : " & trim(A_STOK01.KOD(i)) )
                    res = pe1.write(pe1.GetTop , pe1.GetRight , 10 , 60 , "Lot No : " & trim(A_STOK01.LOTNUMBER(i)))
                    res = pe1.write(pe1.GetTop , pe1.GetRight , 10 , 60 , "Kalan Ömrü : " & raf_omru - giris_tarihi_sayi)
                    res = pe1.write(pe1.GetTop , pe1.GetRight , 10 , 60 , "Giriþ Tarihi : " &  Doc.StrToNorDate( A_STOK01.TARI_H(i)))
                    res = pe1.Line(pe1.GetTop + 7, 10, 0, 190)
                end if 
        end if             
    next
    
    if satir_al > 0 Then    res = pe1.Show()
    res = doc.ActivateAfterMacro(1)

End Sub

Function amortismanli_fiyattan_sis_mi()
	amortismanli_fiyattan_sis_mi = FALSE
	Dim bu_fatura_disi_mik, bu_fatura_ici_mik,FLAG
	bu_fatura_disi_mik = 0
	bu_fatura_ici_mik = 0
	FLAG = 0
	Set AMRT_LIST = Doc.CreateTransactionalTableObject("AMRT=STACK","IO")	
	
	AMRT_LIST.Addfield2 "KOD","CHAR",24,0,"L","",""
	AMRT_LIST.Addfield2 "MIKTAR","DOUBLE",12,3,"R","",""
	
	Set A_STOK48E = Doc.Select1("OPOPOPLL", "STOK48E", "NOTES='AMORTÝSMANLI FÝYAT' & KULLANIM_YERI='4' & REF_CHKODU = '"&STOK70E.CH_KODU(1)&"' ", "", -1)

	if A_STOK48E.GetRecCount() = 0 then  Exit Function
	Set AT_STOK48T = Doc.Select1("TRTRRTRT", "STOK48T", "EVRAKNO =  '"&A_STOK48E.EVRAKNO(1)&"' ", "", -1)

	if STOK70T.GetRecCount() = 0 then  Exit Function
	Set msg = Doc.CreateSObject("BATCHERRORMODE")
	FLAG = 0	
	for k = 1 To STOK70T.GetRecCount()
		for p  = 1 To AT_STOK48T.GetRecCount()		
			if STOK70T.KOD(k) = AT_STOK48T.KOD(p) Then
				for i = 1 To AMRT_LIST.GetRecCount() 
					if AMRT_LIST.KOD(i) = STOK70T.KOD(k) Then
						AMRT_LIST.MIKTAR(i) = STOK70T.SF_MIKTAR(k) + AMRT_LIST.MIKTAR(i)
						toplam_miktar = AMRT_LIST.MIKTAR(i)
						FLAG = 1
					end if 
				next 
				if  FLAG = 0 Then
					sat = AMRT_LIST.AddRow()
					AMRT_LIST.KOD(sat) = STOK70T.KOD(k) 
					AMRT_LIST.MIKTAR(sat) = STOK70T.SF_MIKTAR(k)
					toplam_miktar = AMRT_LIST.MIKTAR(sat)
				end if				
				stok_kodu_birimi = trim(Doc.GetDbFieldValue("STOK00", "IUNIT",STOK70T.KOD(k) , "") )
				'/ Amortismana tabi olacak tarihten sonraki mitarý al
				Set B_STOK70T = Doc.Select1G("UYUYUERT", "STOK70T", " EVRAKNO <> '"&STOK70E.EVRAKNO(1)&"' & CH_KODU = '"& STOK70E.CH_KODU(1) &"' & KOD= '"& STOK70T.KOD(k) &"' & TARIH >= '"& AT_STOK48T.REF_TARIH_1(p)&"' ", "", "KOD", -1)
				Set A_STOK70T = Doc.Select2G("OPIOOEWRUE", "STOK70T", "KOD = '"&STOK70T.KOD(k)&"'  ", "", "KOD", -1)				
				if B_STOK70T.GetRecCount() > 0 Then	bu_fatura_disi_mik = B_STOK70T.SF_MIKTAR(1)
				if A_STOK70T.GetRecCount() > 0 Then	bu_fatura_ici_mik = A_STOK70T.SF_MIKTAR(1)	
				faturalaln_miktar =  bu_fatura_disi_mik + toplam_miktar
				aramortiye_tabii_miktar = AT_STOK48T.REF_NUMBER_1(p)
				'Tarih=doc.GetUDFValue("STOK70E",STOK70E.EVRAKNO(1),"FATURA_TRH",0)
				
				'/ Amortiye tabii miktar üzerinde faturalanma mý yapýlýyor ve amortisman fiyatý mý var  			
				if aramortiye_tabii_miktar - faturalaln_miktar < 0 AND AT_STOK48T.REF_TARIH_1(p) <= STOK70E.TARIH(1) Then
					amortismani_gece_miktar = (aramortiye_tabii_miktar - faturalaln_miktar ) * -1
					'msgbox aramortiye_tabii_miktar & " -  " & faturalaln_miktar
					if trim(AT_STOK48T.PRICEUNIT(p)) ="TL" Then
						if AT_STOK48T.PRICE(p) >= STOK70T.TL_FIYAT(k) Then 
							Doc.d7MsgBox "Amortili fiyatý olan " & AT_STOK48T.PRICE(p) & " " & trim(AT_STOK48T.PRICEUNIT(p)) & "  " & trim(STOK70T.KOD(k)) & "   kodunun  bu faturadaki toplam miktarý  " &  toplam_miktar & "  " &stok_kodu_birimi&" ile amorti edilecek miktarý "& amortismani_gece_miktar & "  " &stok_kodu_birimi  & " geçmektedir." , "", MB_OK	
							amortismanli_fiyattan_sis_mi = TRUE
						else
							if STOK70T.SF_MIKTAR(k) - amortismani_gece_miktar > 0 Then
								Doc.d7MsgBox "Amortili fiyatý olan " & AT_STOK48T.PRICE(p) & " " & trim(AT_STOK48T.PRICEUNIT(p)) & "  " & trim(STOK70T.KOD(k)) & "   kodunun   " &  STOK70T.SF_MIKTAR(k) - amortismani_gece_miktar & "  " & stok_kodu_birimi & " miktarý amortili fiyattan olmalýdýr. " , "", MB_OK	
								amortismanli_fiyattan_sis_mi = TRUE
							end if 	
						end if
					else
						if AT_STOK48T.PRICE(p) >= STOK70T.OR_FIYAT(k) Then 
							Doc.d7MsgBox "Amortili fiyatý olan " & AT_STOK48T.PRICE(p) & " " & trim(AT_STOK48T.PRICEUNIT(p)) & "  " & trim(STOK70T.KOD(k)) & "   kodunun  bu faturadaki toplam miktarý  " &  toplam_miktar & "  " &stok_kodu_birimi&" ile amorti edilecek miktarý "& amortismani_gece_miktar & "  " & stok_kodu_birimi & " geçmektedir." , "", MB_OK	
							amortismanli_fiyattan_sis_mi = TRUE
						else
							Doc.d7MsgBox "Amortili fiyatý olan " & AT_STOK48T.PRICE(p) & " " & trim(AT_STOK48T.PRICEUNIT(p)) & "  " & trim(STOK70T.KOD(k)) & "   kodunun   " &  STOK70T.SF_MIKTAR(k) - amortismani_gece_miktar & "  " & stok_kodu_birimi & " miktarý amortili fiyattan olmalýdýr. " , "", MB_OK	
							amortismanli_fiyattan_sis_mi = TRUE
						end if
					end if						
				'/ Amortiye tabii miktar altýnda  faturalanma mý yapýlýyor  ve amortismanýn dýþýnda bir fiyat mý var
				elseif aramortiye_tabii_miktar - faturalaln_miktar >= 0 AND AT_STOK48T.REF_TARIH_1(p) <= STOK70E.TARIH(1) Then
					if trim(AT_STOK48T.PRICEUNIT(p)) ="TL" Then
						if AT_STOK48T.PRICE(p) <> STOK70T.TL_FIYAT(k) Then 
							Doc.d7MsgBox  "Amortismana tabii olan " & trim(STOK70T.KOD(k)) & "   kodu için amortisman fiyatýnýn " & AT_STOK48T.PRICE(p) &"   "& trim(STOK70T.PRICEUNIT(k)) & " dýþýnda " & STOK70T.TL_FIYAT(k) &"   "& trim(STOK70T.PRICEUNIT(k)) & " fiyatlandýrma yapýlmýþ.",  "", MB_OK
							amortismanli_fiyattan_sis_mi = TRUE
						end if 	
					elseif AT_STOK48T.PRICE(p) <> STOK70T.OR_FIYAT(k) Then 
						Doc.d7MsgBox  "Amortismana tabii olan " & trim(STOK70T.KOD(k)) & "   kodu için amortisman fiyatýnýn " & AT_STOK48T.PRICE(p) &"   "& trim(STOK70T.PRICEUNIT(k)) & " dýþýnda " & STOK70T.OR_FIYAT(k) &"   "& trim(STOK70T.PRICEUNIT(k)) & " fiyatlandýrma yapýlmýþ.",  "", MB_OK
						amortismanli_fiyattan_sis_mi = TRUE
					end if						
				end if			 
			end if 
		next	
	next
End Function
