Sub EslesmemisAlacaklariGoster()
	if Not OzelHakVarmi("ESL_UYARI") Then Exit Sub

	QStr = "SELECT CR.EVRAKTYPE, CR.EVRAKNO, TO_CHAR(TO_DATE(CR.TARIH,'YYYY/MM/DD'),'DD/MM/YYYY') TARIH, "
	QStr = QStr & " CR.KOD, C.AD, TO_CHAR(CR.ACK_TUTAR,'999,999.00') ALACAK_DBL"
	QStr = QStr & " FROM CRHDST CR"
	QStr = QStr & " LEFT JOIN CARI00 C ON C.KOD=CR.KOD"
	QStr = QStr & " WHERE CR.B_A='A'"
	QStr = QStr & " AND CR.A_K<>'K'"
	QStr = QStr & " AND CR.KOD LIKE 'M%'"
	QStr = QStr & " AND CR.KOD IN ("
	QStr = QStr & "   SELECT DISTINCT KOD"
	QStr = QStr & "   FROM CRHDST"
	QStr = QStr & "   WHERE B_A='B'"
	QStr = QStr & "   AND A_K<>'K'"
	QStr = QStr & "   AND KOD LIKE 'M%'"
	QStr = QStr & " )"
	QStr = QStr & " ORDER BY KOD"

	Set CRH = Doc.RunSQLQuery("sdfasdhga", QStr)
	HTMLBaslik = "Eþleþmemiþ Alacaklar"
	TabloBaslikArray = Array("Evrak Türü", "Evrak No", "Tarih", "C/H Kodu", "Müþteri", "TL Alacak")
	Doc.RunMacro3 "MAKR7S.FREEMAIL", "ShowTableHTMLMessage", CRH, HTMLBaslik, TabloBaslikArray
End Sub

Function BeforeSave()

	Set M10TM = Doc.GetTableObject("MMPS10TM")
	Set M10E = Doc.GetTableObject("MMPS10E")
	
	If Trim(M10E.HAVUZKODU) = "S1" OR Trim(M10E.HAVUZKODU) = "Z1" Then
		If Trim(M10TM.MUSTERISIPINO) = "" Then
			If Trim(MMPS10E.GK_4) = "" Then
				Msgbox "Manuel Açýlan MPS' lerde MPS Açma Nedeni (GK_4' e) Tanýmlanmalýdýr." & vbcrlf & "Kayýt Ýþlemi Kesilmiþtir."
				BeforeSave = "CANCELWITHOUTERRORMSG"
				Exit Function
			Else
				If Trim(M10E.GK_4) = "DUZELTME" OR Trim(M10E.GK_4) = "NUMUNE" Then
					If Trim(M10TM.MUSTERIKODU(1)) = "" Then
						Msgbox "Manuel Açýlan MPS' lerde GK4' de [ NUMUNE /  DUZELTME ] Seçili ise Müþteri Kodu Girilmelidir." & vbcrlf & "Kayýt Ýþlemi Kesilmiþtir."
						BeforeSave = "CANCELWITHOUTERRORMSG"
						Exit Function
					End If
				End If
			End If
		End If
	End If

	if HammaddeDegismisMi() Then
		BeforeSave = "CANCELWITHOUTERRORMSG"
		Exit Function
	End if

	MMPS10E.TOLERANSTYPE="-+"
	MMPS10E.TOLERANS=5
	MMPS10E.HESAPLANMIS10 = "1"
	
	For i=1 to MMPS10T.Getreccount()
		If trim(MMPS10T.R_OPERASYON(i))="COR" then
			if trim(MMPS10T.R_KAYNAKTYPE(i))="H" then
				formul=formul+MMPS10T.TRANSFERPARTIBUYUKLG(i)
			else
				Miktar=MMPS10T.R_YMK_YMPAKETICERIGI(i)
			end if
		End If
		If trim(MMPS10T.R_OPERASYON(i))="COR1" then
			if trim(MMPS10T.R_KAYNAKTYPE(i))="H" then
				formul1=formul1+MMPS10T.TRANSFERPARTIBUYUKLG(i)
			else
				Miktar1=MMPS10T.R_YMK_YMPAKETICERIGI(i)
			end if
		End If
	Next
	formul_m=100-formul
	formul_m1=100-formul1
	
	For t=1 to MMPS10T.Getreccount()
		If trim(MMPS10T.R_OPERASYON(t))="COR" and MMPS10T.TRANSFERPARTIBUYUKLG(t)>0 then
			if trim(MMPS10T.R_KAYNAKTYPE(t))="H" then
				MMPS10T.R_MIKTAR0(t)=MMPS10T.TRANSFERPARTIBUYUKLG(t)*miktar/100
			else
				MMPS10T.R_MIKTAR0(t)=formul_m*miktar/100
			end if
			MMPS10E.HESAPLANMIS10 = "1"
		end if
		If trim(MMPS10T.R_OPERASYON(t))="COR1" and MMPS10T.TRANSFERPARTIBUYUKLG(t)>0 then
			if trim(MMPS10T.R_KAYNAKTYPE(t))="H" then
				MMPS10T.R_MIKTAR0(t)=MMPS10T.TRANSFERPARTIBUYUKLG(t)*miktar1/100
			else
				MMPS10T.R_MIKTAR0(t)=formul_m1*miktar1/100
			end if
		end if
		MMPS10E.HESAPLANMIS10 = "1"		
	Next
	'Call MinStokMiktariKontrol()
	
	Call Duzeltme()
	Call HammaddeMiktarlariniGuncelle()
	Call CorrectionSayisiYaz()
	'Call YogunlukKontrol()
End Function

Sub AfterSave()
	Call KTAdakiLotlariAktar()
End Sub

Sub MinStokMiktariKontrol()
	if trim(MMPS10E.HAVUZKODU) = "S2" Then Exit Sub
	
	if trim(MMPS10E.GK_2) = "UYARI" Then Exit Sub
	Set s = Doc.CreateSobject("BATCHERRORMODE")
	for i = 1 to MMPS10T.Getreccount()
		if trim(MMPS10T.R_KAYNAKTYPE(i)) = "H" then
			HammaddeKodu = MMPS10T.R_KAYNAKKODU(i)
			GuvenlikStogu = doc.getdbfieldvalue("STOK00","B_GUVENLIKSTOGU",HammaddeKodu,"")
			Depo = Doc.Stok00c_SonMik(HammaddeKodu,"")
			
			MPS_Miktari = MMPS10T.R_MIKTART(i)
			'AÇIK MPSLER
			Dim ttt,ni
			dim Kriter
			Kriter = "ACIK_KAPALI=' '&R_ACIK_KAPALI=' '&R_KAYNAKTYPE='H'&R_KAYNAKKODU='" & HammaddeKodu & "'EVRAKNO<>'" & MMPS10E.EVRAKNO & "'"
			Set ttt = Doc.Select1("ABC","MMPS10T",Kriter,"",-1)
			If Not Doc.IsBlank(ttt) Then
				For ni=1 To ttt.GetRowCount()
					MPS_Miktari = MPS_Miktari + ttt.R_MIKTART(ni)
				Next
			End if
			
			'AÇIK SÝPARÝÞLER
			
			Dim zzz,zi
			dim Kriter1
			Kriter1 = "KOD='" & HammaddeKodu & "'"
			Set zzz = Doc.Select1("ABDC","STOK40T",Kriter1,"<KOD",-1)
			If Not Doc.IsBlank(zzz) Then
				Siparis_Miktari = 0
				For zi=1 To zzz.GetRowCount()
					if trim(zzz.ACIKKAPALI(zi)) <> "K" Then
						Siparis_Miktari = Siparis_Miktari + zzz.SF_MIKTAR(zi)
					End if
				Next
			End if
			
			Ihtiyac = Depo - MPS_Miktari - Siparis_Miktari - GuvenlikStogu
			if Ihtiyac < 0 then
				Doc.d7msgbox trim(HammaddeKodu) & " güvenlik stoðunun altýna düþmüþtür.Net ihtiyaç " & Ihtiyac, "", MB_OK
			End if
		End if
	Next
	MMPS10E.GK_2 = "UYARI"
End Sub

Sub KTAdakiLotlariAktar()
	YeniLot = Doc.GetUDFValue("MMPS10E", MMPS10E.EVRAKNO, "YENILOT", 0)
	if trim(YeniLot) = "" Then
		YeniLot = Doc.GetUDFValue("MMPS10E", MMPS10E.EVRAKNO, "PARTI", 0)
	End if
	Degisti = False
	if trim(YeniLot) <> "" And (trim(YeniLot) <> trim(MMPS10E.NOTLAR_1) or trim(YeniLot) <> trim(MMPS10TM.NOTLAR_1)) Then
		MMPS10E.NOTLAR_1 = YeniLot
		MMPS10TM.NOTLAR_1 = YeniLot
		MMPS10E.HESAPLANMIS10 = "1"
		Degisti = True
	End if
	if Degisti Then 
		MMPS10E.HESAPLANMIS10="1"
		Doc.Save_Voucher()
	End if
End Sub
Sub CorrectionSayisiYaz()
	MMPS10E.SetFieldValueNV "PARTINO", 1, GetCorrIndex
'	MMPS10E.HESAPLANMIS10 = "1"

End Sub
Sub Duzeltme()
	Grade = Doc.GetUDFValue("MMPS10E", MMPS10E.EVRAKNO, "LBC", 0)
	if trim(Grade) = "B" or trim(Grade) = "PR" Then
		MMPS10E.GK_2 = "DUZELTME"
	Else
		MMPS10E.GK_2 = ""
	End if
End Sub
Sub HammaddeMiktarlariniGuncelle()
	if MMPS10E.STATUS <> "R" Then Exit Sub

	GFLS = trim(Doc.Customp_Gets("GFLS_MIK"))
	GFLSArr = Split(GFLS, ".")
	if uBound(GFLSArr) <> 1 Then
		Ondalik = 3
	Else
		Ondalik = Doc.Cdbl(GFLSArr(1))
	End if
	For i = 1 to MMPS10T.GetRowCount()
		if MMPS10T.R_KAYNAKTYPE(i) = "H" Then
			MMPS10T.SetCurrentRow(i)
			MMPS10T.R_MIKTAR0 = Doc.Round(MMPS10T.R_MIKTAR0, Ondalik)
		End if
	Next
	Doc.RunCommand "CMD_HESAPLANMIS_1"
End Sub
Function HammaddeDegismisMi()
	Set M10T = Doc.Select1("sdgrhhdrh", "MMPS10T", "EVRAKNO='" & MMPS10E.EVRAKNO & "'&R_KAYNAKTYPE='H'", "", -1)
	if M10T.GetRowCount() = 0 Then Exit Function
	if M10T.R_STATUS(1) <> "R" Then Exit Function

	HammaddeDegismisMi = False
	For i = 1 to MMPS10T.GetRowCount()
		if (MMPS10T.R_HM_ODONEMNET(i) + MMPS10T.R_HM_TRANSFER_EDILEN(i)) > 0 Then
			Set M10T1 = Doc.Select2("adfe2gwegs", M10T.GetAppName, "JOBNO='" & MMPS10T.JOBNO(i) & "'", "", -1)
			if M10T1.GetRowCount() > 0 Then
				if M10T1.R_KAYNAKKODU(1) <> MMPS10T.R_KAYNAKKODU(i) Then
					Doc.d7MsgBox i & ". Satýr, Transfer yapýlmýþ. Hammadde kodu deðiþtirilemez", "Ýþlem Kesildi", MB_OK
					HammaddeDegismisMi = True
				End if
			End if
		End if
	Next
End Function
Function GetCorrIndex()
	CorrIndex = 0
	if MMPS10E.STATUS <> "R" Then Exit Function
	For i = 1 to MMPS10T.GetRowCount()
		if MMPS10T.R_KAYNAKTYPE(i) = "H" Then
			For j = 1 to 9
				if Doc.Cdbl(Doc.GetUDFValue("MMPS10T", MMPS10T.JOBNO(i), "COR_RCTMIK", j)) = 0 Then Exit For
				if j > CorrIndex Then CorrIndex = j
			Next
		End if
	Next
	GetCorrIndex = CorrIndex
End Function

			Yogunluk = Doc.GetDBFieldValue("STOK00", "B_YOGUNLUK", MMPS10T.R_KAYNAKKODU(i), "")

			End if
		End if
	Next
End Sub

Function GetSiparisTutari(CHKodu)
	GetSiparisTutari = 0
	QStr = "SELECT T.EVRAKNO, T.PRICEUNIT, Sum(T.OR_IVD_KH_TUTAR) OR_IVD_KH_TUTAR, Sum(T.OR_KDVTUT_EFEKTIF) OR_KDVTUT_EFEKTIF, "
	QStr = QStr & " E.MUHT_FATURA_NITELIK, E.KURKESINLESTIRME, Sum(T.SF_MIKTAR - T.SF_NETKAPANANMIK) SIP_BAKIYE,  Sum(T.SF_MIKTAR) SF_MIKTAR"
	QStr = QStr & " FROM STOK40T T"
	QStr = QStr & " LEFT JOIN STOK40E E ON T.EVRAKNO=E.EVRAKNO"
	QStr = QStr & " WHERE T.MUSTERIKODU='" & CHKodu & "'"
	QStr = QStr & " AND T.ACIKKAPALI=' '"
	QStr = QStr & " AND T.SF_MIKTAR > T.SF_NETKAPANANMIK"
	QStr = QStr & " GROUP BY T.EVRAKNO, T.PRICEUNIT, T.KOD, E.MUHT_FATURA_NITELIK, E.KURKESINLESTIRME"
	Set T_STOK40T = Doc.RunSQLQuery("XXX", QStr)

	for i = 1 to T_STOK40T.GetRecCount()

		KurKesinlestirme = T_STOK40T.KURKESINLESTIRME(i)
		If trim(MuhtemelNitelik) = "XK" Then

		Else
			SiparisTutari = T_STOK40T.OR_IVD_KH_TUTAR(i) + T_STOK40T.OR_KDVTUT_EFEKTIF(i)
		End If

		SiparisFiyati = doc.Div(SiparisTutari, T_STOK40T.SF_MIKTAR(i))

		Else
			KurDegeri = Doc.GetKurN(T_STOK40T.PRICEUNIT(i), Doc.DefaultPB, Doc.BuGun, "", -1)
		End If
		GetSiparisTutari = GetSiparisTutari + SiparisFiyati * KurDegeri * T_STOK40T.SIP_BAKIYE(i)

Function CalculateSevkRisk(CHKodu)
	CalculateSevkRisk = 0 'initiate the return value
	SiparisTutari = 0
	SevkQStr = "SELECT STOK40T.MUSTERIKODU Musteri, STOK01.SF_MIKTAR SevkMik, STOK40T.PRICEUNIT ParaBirimi, " & vbCr & _
		"STOK40T.OR_ISKONTOLU_TUTAR IskontoluTutar, STOK40T.OR_KDVTUT_EFEKTIF KDVTutar, " & vbCr & _
		"STOK40T.KUR DovizKur, STOK01.ALLOC_SIPNO SiparisNo, STOK40T.SF_MIKTAR SiparisMik " & vbCr & _
		"FROM STOK01, STOK40T " & vbCr & _
		"WHERE (STOK01.ALLOC_SIPARTNO = STOK40T.ARTNO) AND (STOK01.ALLOC_EVRAKTYPE = 'SV') AND (STOK40T.MUSTERIKODU = '" & CHKodu & "')"
		
	Set S01 = Doc.RunSQLQuery("SevkQuery", SevkQStr)
	For i = 1 To S01.GetRecCount()
		MuhtemelNitelik = doc.GetDbFieldValue("STOK40E", "MUHT_FATURA_NITELIK", S01.SiparisNo(i), "")
		KurAktarma = doc.GetDbFieldValue("STOK40E", "KURKESINLESTIRME", S01.SiparisNo(i), "")
		SiparisPB = S01.ParaBirimi(i)
		KDVHaricTutar = S01.IskontoluTutar(i)

		Else
			SiparisTutari = KDVHaricTutar
		End If

		If KurAktarma = "E" Then
			KurDegeri = S01.DovizKur(i)
		Else
			KurDegeri = doc.GetKurN(SiparisPB, doc.defaultpb, doc.Bugun, "", -1)
		End If

	Next
End Function

	End if

	End if

				Exit Function

			End if
		Else

				if STOK40E.ONAY_SONVERILENNO<> "5" Then
					Doc.RunCommand "MT_DO_ONAYLA_EKINFO"

					Doc.RunCommand "MT_DO_ONAYLA_EKINFO"
				End if
			End if
		End if
	End if
	if OdenmemisBorcKontrol(LimitOp) Then
		STOK40E.ONAY_5KULLANICI = ""
		STOK40E.ONAY_5TARIH = ""
		STOK40E.ONAY_5SAAT = ""

		STOK40E.ONAY_SONTARIH = ""

			Exit Function

		Exit Function
	End if

		Exit Function
	End if
	if OnaylanmisSiparisteDegisiklikYapiliyor() Then
		BeforeSave = "CANCELWITHOUTERRORMSG"
		Exit Function
	End if
End Function

		MuhtemelNitelik = STOK40E.MUHT_FATURA_NITELIK
		KurKesinlestirme = STOK40E.KURKESINLESTIRME
		If trim(MuhtemelNitelik) = "XK" Then
			SiparisTutari = T_STOK40T.OR_IVD_KH_TUTAR(i)
		Else
			SiparisTutari = T_STOK40T.OR_IVD_KH_TUTAR(i) + T_STOK40T.OR_KDVTUT_EFEKTIF(i)
		End If

		SiparisFiyati = doc.Div(SiparisTutari, T_STOK40T.SF_MIKTAR(i))
		If trim(KurKesinlestirme) = "E" Then
			KurDegeri = T_STOK40T.KUR(i)
		Else
			KurDegeri = Doc.GetKurN(T_STOK40T.PRICEUNIT(i), Doc.DefaultPB, Doc.BuGun, "", -1)
		End If
		GetBuSiparisTutari = GetBuSiparisTutari + SiparisFiyati * KurDegeri * (T_STOK40T.SF_MIKTAR(i) - T_STOK40T.SF_NETKAPANANMIK(i))
	Next
End Function
Function CalculateSevkRisk(CHKodu)
	CalculateSevkRisk = 0 'initiate the return value
	SiparisTutari = 0
	SevkQStr = "SELECT STOK40T.MUSTERIKODU Musteri, STOK01.SF_MIKTAR SevkMik, STOK40T.PRICEUNIT ParaBirimi, " & vbCr & _
		"STOK40T.OR_ISKONTOLU_TUTAR IskontoluTutar, STOK40T.OR_KDVTUT_EFEKTIF KDVTutar, " & vbCr & _
		"STOK40T.KUR DovizKur, STOK01.ALLOC_SIPNO SiparisNo, STOK40T.SF_MIKTAR SiparisMik " & vbCr & _
		"FROM STOK01, STOK40T " & vbCr & _
		"WHERE (STOK01.ALLOC_SIPARTNO = STOK40T.ARTNO) AND (STOK01.ALLOC_EVRAKTYPE = 'SV') AND (STOK40T.MUSTERIKODU = '" & CHKodu & "')"
		
	Set S01 = Doc.RunSQLQuery("SevkQuery", SevkQStr)
	For i = 1 To S01.GetRecCount()
		MuhtemelNitelik = doc.GetDbFieldValue("STOK40E", "MUHT_FATURA_NITELIK", S01.SiparisNo(i), "")
		KurAktarma = doc.GetDbFieldValue("STOK40E", "KURKESINLESTIRME", S01.SiparisNo(i), "")
		SiparisPB = S01.ParaBirimi(i)
		KDVHaricTutar = S01.IskontoluTutar(i)
		KDVTatari = S01.KDVTutar(i)
	
		If MuhtemelNitelik = "XK" Then
			SiparisTutari = KDVHaricTutar + KDVTutari
		Else
			SiparisTutari = KDVHaricTutar
		End If
		SiparisFiyati = doc.Div(SiparisTutari, S01.SiparisMik(i))
		If KurAktarma = "E" Then
			KurDegeri = S01.DovizKur(i)
		Else
			KurDegeri = doc.GetKurN(SiparisPB, doc.defaultpb, doc.Bugun, "", -1)
		End If
		CalculateSevkRisk = CalculateSevkRisk + SiparisFiyati * KurDegeri * S01.SevkMik(i)
	Next
End Function
Function GetIrsaliyeRiski(CHKodu)

Function GetSiparisTutari(CHKodu)
	GetSiparisTutari = 0
	QStr = "SELECT T.EVRAKNO, T.PRICEUNIT, Sum(T.OR_IVD_KH_TUTAR) OR_IVD_KH_TUTAR, Sum(T.OR_KDVTUT_EFEKTIF) OR_KDVTUT_EFEKTIF, "
	QStr = QStr & " E.MUHT_FATURA_NITELIK, E.KURKESINLESTIRME, Sum(T.SF_MIKTAR - T.SF_NETKAPANANMIK) SIP_BAKIYE,  Sum(T.SF_MIKTAR) SF_MIKTAR"
	QStr = QStr & " FROM STOK40T T"
	QStr = QStr & " LEFT JOIN STOK40E E ON T.EVRAKNO=E.EVRAKNO"
	QStr = QStr & " WHERE T.MUSTERIKODU='" & CHKodu & "'"
	QStr = QStr & " AND T.ACIKKAPALI=' '"
	QStr = QStr & " AND T.SF_MIKTAR > T.SF_NETKAPANANMIK"
	QStr = QStr & " GROUP BY T.EVRAKNO, T.PRICEUNIT, T.KOD, E.MUHT_FATURA_NITELIK, E.KURKESINLESTIRME"
	Set T_STOK40T = Doc.RunSQLQuery("XXX", QStr)

	QStr "SELECT ARTNO, SUM(SF_MIKTAR) SF_MIKTAR"
	QStr = QStr & " FROM STOK60T"
	QStr = QStr & " WHERE EVRAKNO='" & STOK60E.EVRAKNO & "'"
	QStr = QStr & " AND TRIM(SIPARTNO)  IS NOT NULL"
	QStr = QStr & " GROUP BY SIPARTNO"
	Set S60T = Doc.RunSQLQuery("asdfasdg", QStr)

	for i = 1 to T_STOK40T.GetRecCount()
		Set S60T1 = Doc.Select2("asdfasdf", S60T.GetAppName, "SIPARTNO='" & STOK40T.ARTNO(i) & "'", "", -1)
		if S60T1.GetRowCount() > 0 Then
			KayitliIrsMik = S60T1.SF_MIKTAR
		Else
			KayitliIrsMik = 0
		End if

		Set S60T1 = Doc.Select2("asdfasdf", "STOK60T", "SIPARTNO='" & STOK40T.ARTNO(i) & "'", "", -1)
		if S60T1.GetRowCount() > 0 Then
			BuIrsMik = S60T1.SF_MIKTAR
		Else
			BuIrsMik = 0
		End if

		RiskMiktar = T_STOK40T.SIP_BAKIYE(i) + KayitliIrsMik - BuIrsMik
		if RiskMiktar > 0 Then
			MuhtemelNitelik = T_STOK40T.MUHT_FATURA_NITELIK(i)
			KurKesinlestirme = T_STOK40T.KURKESINLESTIRME(i)
			If trim(MuhtemelNitelik) = "XK" Then
				SiparisTutari = T_STOK40T.OR_IVD_KH_TUTAR(i)
			Else
				SiparisTutari = T_STOK40T.OR_IVD_KH_TUTAR(i) + T_STOK40T.OR_KDVTUT_EFEKTIF(i)
			End If

			SiparisFiyati = doc.Div(SiparisTutari, T_STOK40T.SF_MIKTAR(i))
			If trim(KurKesinlestirme) = "E" Then
				KurDegeri = T_STOK40T.KUR(i)
			Else
				KurDegeri = Doc.GetKurN(T_STOK40T.PRICEUNIT(i), Doc.DefaultPB, Doc.BuGun, "", -1)
			End If
			GetSiparisTutari = GetSiparisTutari + SiparisFiyati * KurDegeri * RiskMiktar
		End if
	Next
End Function
Function CalculateSevkRisk(CHKodu)
	CalculateSevkRisk = 0 'initiate the return value
	SiparisTutari = 0
	SevkQStr = "SELECT STOK40T.MUSTERIKODU Musteri, STOK01.SF_MIKTAR SevkMik, STOK40T.PRICEUNIT ParaBirimi, " & vbCr & _
		"STOK40T.OR_ISKONTOLU_TUTAR IskontoluTutar, STOK40T.OR_KDVTUT_EFEKTIF KDVTutar, " & vbCr & _
		"STOK40T.KUR DovizKur, STOK01.ALLOC_SIPNO SiparisNo, STOK40T.SF_MIKTAR SiparisMik " & vbCr & _
		"FROM STOK01, STOK40T " & vbCr & _
		"WHERE (STOK01.ALLOC_SIPARTNO = STOK40T.ARTNO) AND (STOK01.ALLOC_EVRAKTYPE = 'SV') AND (STOK40T.MUSTERIKODU = '" & CHKodu & "')"
		
	Set S01 = Doc.RunSQLQuery("SevkQuery", SevkQStr)
	For i = 1 To S01.GetRecCount()
		MuhtemelNitelik = doc.GetDbFieldValue("STOK40E", "MUHT_FATURA_NITELIK", S01.SiparisNo(i), "")
		KurAktarma = doc.GetDbFieldValue("STOK40E", "KURKESINLESTIRME", S01.SiparisNo(i), "")
		SiparisPB = S01.ParaBirimi(i)
		KDVHaricTutar = S01.IskontoluTutar(i)
		KDVTatari = S01.KDVTutar(i)
	
		If MuhtemelNitelik = "XK" Then
			SiparisTutari = KDVHaricTutar + KDVTutari
		Else
			SiparisTutari = KDVHaricTutar
		End If
		SiparisFiyati = doc.Div(SiparisTutari, S01.SiparisMik(i))
		If KurAktarma = "E" Then
			KurDegeri = S01.DovizKur(i)
		Else
			KurDegeri = doc.GetKurN(SiparisPB, doc.defaultpb, doc.Bugun, "", -1)
		End If
		CalculateSevkRisk = CalculateSevkRisk + SiparisFiyati * KurDegeri * S01.SevkMik(i)
	Next
End Function
Function GetIrsaliyeRiski(CHKodu)
	GetIrsaliyeRiski = 0
	QStr = "SELECT (STOK60T.SF_MIKTAR - STOK60T.SF_FATURALANANMIKTAR) MIKTAR, STOK40E.MUHT_FATURA_NITELIK, STOK40E.KURKESINLESTIRME,"
	QStr = QStr & " (STOK40T.OR_ISKONTOLU_TUTAR + STOK40T.OR_KDVTUT_EFEKTIF) SIP_KDV_DAHIL_TUTAR, STOK40T.OR_ISKONTOLU_TUTAR SIP_TUTAR, STOK40T.KUR SIP_KUR, " 
	QStr = QStr & " STOK40T.SF_MIKTAR SIP_MIKTAR, STOK40T.PRICEUNIT SIP_PB "
	QStr = QStr & " FROM STOK60T"
	QStr = QStr & " INNER JOIN STOK40T ON STOK60T.SIPARTNO=STOK40T.ARTNO"
	QStr = QStr & " INNER JOIN STOK40E ON STOK60T.SIPNO=STOK40E.EVRAKNO"
	QStr = QStr & " WHERE STOK60T.CH_KODU='" & CHKodu & "'"
	QStr = QStr & " AND STOK60T.SF_MIKTAR > STOK60T.SF_FATURALANANMIKTAR"
	QStr = QStr & " AND STOK60T.EVRAKNO<>'" & STOK60E.EVRAKNO & "'"
	Set S60T = Doc.RunSQLQuery("aef23g", QStr)
	For i = 1 to S60T.GetRowCount()
		If trim(S60T.MUHT_FATURA_NITELIK(i)) = "XK" Then
			SiparisTutari = S60T.SIP_TUTAR(i)
		Else
			SiparisTutari = S60T.SIP_KDV_DAHIL_TUTAR(i)
		End If

		SiparisFiyati = doc.Div(SiparisTutari, S60T.SIP_MIKTAR(i))
		If trim(S60T.KURKESINLESTIRME(i)) = "E" Then
			KurDegeri = S60T.SIP_KUR(i)
		Else
			KurDegeri = Doc.GetKurN(S60T.SIP_PB(i), Doc.DefaultPB, Doc.BuGun, "", -1)
		End If
		GetIrsaliyeRiski = GetIrsaliyeRiski + SiparisFiyati * KurDegeri * S60T.MIKTAR(i)
	Next
End Function

Sub GeCustomerLastCollectionInfo(CHKodu, IslemTarihi, Tutar, OrtalamaValorTarihi)
	IslemTarihi = ""
	Tutar = 0
	OrtalamaValorTarihi = ""
	QStr = "SELECT TARIH, SUM(ALACAKFIYAT) ALACAKFIYAT, "
	QStr = QStr & " TO_CHAR(TO_DATE(ROUND(Sum(TO_NUMBER(TO_CHAR(TO_DATE(VALORTARIH,'YYYY/MM/DD'),'J')) * ALACAKFIYAT) / Sum(ALACAKFIYAT),0), 'J'),'YYYY/MM/DD') ORT_VALOR"
	QStr = QStr & " FROM CARI10T"
	QStr = QStr & " WHERE KOD='" & CHKodu & "'"
	QStr = QStr & " AND ALACAKFIYAT>0"
	QStr = QStr & " AND TARIH = ("
	QStr = QStr & " 	SELECT MAX(TARIH) FROM CARI10T WHERE KOD='" & CHKodu & "'"
	QStr = QStr & " 	AND ALACAKFIYAT>0"
	QStr = QStr & " )"
	QStr = QStr & " GROUP BY TARIH"
	Set C10T = Doc.RunSQLQuery("sadfasdah", QStr)
	if C10T.GetRowCount() = 0 Then Exit Sub

	IslemTarihi = C10T.TARIH
	Tutar = C10T.ALACAKFIYAT
	OrtalamaValorTarihi = C10T.ORT_VALOR
End Sub

Function GetIrsaliyeTutari(S60,satirNo)
	GetIrsaliyeTutari = 0
	if S60.SF_MIKTAR(satirNo) <> 0 then
		FaturalanacakMiktar = S60.SF_MIKTAR(satirNo) - S60.SF_FATURALANANMIKTAR(satirNo)
		if S60.TL_KDVDAHILTUTAR(satirNo) = 0 then
			SiparisNo = Doc.GetDBFieldValue("STOK40T","EVRAKNO",S60.SIPARTNO(satirNo),"ARTNO")
			MuhtemelNitelik = Doc.GetDBFieldValue("STOK40E","MUHT_FATURA_NITELIK",SiparisNo,"")
			KurAktarma = Doc.GetDBFieldValue("STOK40E","KURKESINLESTIRME",SiparisNo,"")
			SiparisMiktari = Doc.GetDBFieldValue("STOK40T","SF_MIKTAR",S60.SIPARTNO(satirNo),"ARTNO")
			KDVHaricTutar = Doc.GetDBFieldValue("STOK40T","OR_IVD_KH_TUTAR",S60.SIPARTNO(satirNo),"ARTNO")
			KDVTutari = Doc.GetDBFieldValue("STOK40T","OR_KDVTUT_EFEKTIF",S60.SIPARTNO(satirNo),"ARTNO")
				If MuhtemelNitelik = "XK" Then
				SiparisTutari = KDVHaricTutar + KDVTutari
			Else
				SiparisTutari = KDVHaricTutar
			End If
			SiparisPB = Doc.GetDBFieldValue("STOK40T","PRICEUNIT",S60.SIPARTNO(satirNo),"ARTNO")

			SiparisFiyati = Doc.Div(SiparisTutari, SiparisMiktari)
			if KurAktarma = "E" then
				KurDegeri = Doc.GetDBFieldValue("STOK40T","KUR",S60.SIPARTNO(satirNo),"ARTNO")
			Else
				KurDegeri = Doc.GetKurN(SiparisPB, Doc.DefaultPB, Doc.BuGun, "", -1)
			End if
			GetIrsaliyeTutari = SiparisFiyati * KurDegeri * FaturalanacakMiktar
		else
			GetIrsaliyeTutari = Doc.Div(S60.TL_KDVDAHILTUTAR(satirNo) * FaturalanacakMiktar, S60.SF_MIKTAR(satirNo))
		end if
	end if
End Function

Function B_DEFAULT_VADEGUN_AfterValidate()
	Set C00 = Doc.GetTableObject("CARI00")
	VadeGunu = C00.B_DEFAULT_VADEGUN
	
	CariKod = 	Trim(C00.KOD)
	
	PatermVar = False
	Set Tbl = Doc.Select1("dsfsdfsd", "GECOUST", "EVRAKNO='!PAYTERMS'  ", "", -1)		
	If Tbl.GetRecCount > 0 Then
		For i = 1 To Tbl.GetRecCount
			If Trim(VadeGunu) = Trim(Tbl.KOD(i)) Then
				PatermVar = True
			End If
		Next
	End If

		C00.B_DEFAULT_VADEGUN = SistemdekiVadeGunu
		Exit Function
	End If

	Call Doc.SetSaveable(TRUE)
	Doc.UFLD00_CreateEditTable "Xulfd", "CARI00", CariKod
	Set KTAYAZ = Doc.GetTableObject("Xulfd")
	
	KTAYAZ.PAYTERMS = VadeGunu
	KTAYAZ.Exec "SAVE" 	
End Function
