Function BeforeSave()
	if Doc.IsBlank(SRVP00) Then Set SRVP00 = Doc.GetTableObject("SRVP00")
	if Doc.IsBlank(SRVP00O) Then Set SRVP00O = Doc.GetTableObject("SRVP00O")
	if trim(SRVP00.SABLONTURU)="OB" Then SRVP00.TELEPEDEN = "00201"

		if trim(SRVP00.TELEPEDEN) = "" Then
			Doc.d7msgBox "Talep Eden Doldurulmalýdýr", "Ýþlem Kesildi", MB_OK
			BeforeSave = "CANCELWITHOUTERRORMSG"
			Exit Function
		End if

		if Pasifkalip() Then
		Doc.d7msgBox "Pasif kalýplara iþ emri açýlamaz.", "Ýþlem Kesildi", MB_OK
		BeforeSave = "CANCELWITHOUTERRORMSG"
		Exit Function
	End if
	
		if trim(SRVP00.PROBLEMISTEK) = "" Then
			Doc.d7msgBox "Problem/Arýza Doldurulmalýdýr", "Ýþlem Kesildi", MB_OK
			BeforeSave = "CANCELWITHOUTERRORMSG"
			Exit Function
		End if

		if trim(SRVP00.PROBLEMISTEK_TEXT) = "" Then
			Doc.d7msgBox "Yapýlmasý Ýstenen Doldurulmalýdýr", "Ýþlem Kesildi", MB_OK
			BeforeSave = "CANCELWITHOUTERRORMSG"
			Exit Function
		End if

			Doc.d7msgBox "Grup Kodu-1 Doldurulmalýdýr", "Ýþlem Kesildi", MB_OK
			BeforeSave = "CANCELWITHOUTERRORMSG"
			Exit Function
		End if

			Doc.d7msgBox "Grup Kodu-2 Doldurulmalýdýr", "Ýþlem Kesildi", MB_OK
			BeforeSave = "CANCELWITHOUTERRORMSG"
			Exit Function
		End if

			Doc.d7msgBox "Hücre Adý Doldurulmalýdýr", "Ýþlem Kesildi", MB_OK
			BeforeSave = "CANCELWITHOUTERRORMSG"
			Exit Function
		End if
	End if
	if IsEmriKapatmaYetkisiYok() Then
		BeforeSave = "CANCELWITHOUTERRORMSG"
		Exit Function
	End if

	Set ddd=Doc.CreateSObject("d7Date")
	FOR Z = 1 TO SRVP00O.GETRECCOUNT()
		ddd.SetDate(SRVP00O.RECTARIH1(Z))
		L = ddd.GetDOWStr()
		IF TRIM(SRVP00O.RECTARIH1(Z)) = TRIM(SRVP00O.RECTARIH2(Z)) THEN
			MOLA = 0 
			CALL HESAPLA(SRVP00O.RECTIME1(Z),SRVP00O.RECTIME2(Z))
		ELSE 
			MOLA = 0
			CALL HESAPLA(SRVP00O.RECTIME1(Z),"23:59")
			CALL HESAPLA("00:00",SRVP00O.RECTIME2(Z))
		END IF

		SRVP00O.MOLA_DK(Z) = MOLA 
		molax = DOC.DIV(MOLA,60)
		SRVP00O.MOLA_SAAT(Z) = MOLAX
	NEXT

	if ZamaniGecenIsEmirlerindeDegisiklik() Then
		BeforeSave = "CANCELWITHOUTERRORMSG"
		Exit Function
	End if

	if IsEmriKapatmaYetkisiYok() Then
		BeforeSave = "CANCELWITHOUTERRORMSG"
		Exit Function
	End if
	


	Call MTBFMTTRAyarla()
	Call BaslangicBitisTarihleriniHesapla()
	
End Function


SUB HESAPLA (T1,T2)
	if Doc.IsBlank(SRVP00) Then Set SRVP00 = Doc.GetTableObject("SRVP00")

  IF TRIM(T1) <= "10:00" AND TRIM(T2) >= "10:10" THEN 
    MOLA = MOLA + 10  
  END IF 
 
'********cuma günü ise
  IF TRIM(L) = "Cum" THEN

    IF TRIM(T1) <= "12:40" AND TRIM(T2) >= "14:10" THEN 
      MOLA = MOLA + 90  
    END IF

    IF TRIM(T1) <= "16:00" AND TRIM(T2) >= "16:10" THEN 
      MOLA = MOLA + 10  
    END IF
   ELSE 
'*****cum deðil ise
     IF TRIM(T1) <= "12:40" AND TRIM(T2) >= "13:10" THEN 
       MOLA = MOLA + 30  
     END IF
     IF TRIM(T1) <= "15:00" AND TRIM(T2) >= "15:10" THEN 
       MOLA = MOLA + 10  
     END IF
   END IF 
   IF TRIM(T1) <= "01:00" AND TRIM(T2) >= "01:10" THEN 
     MOLA = MOLA + 10  
   END IF
   IF TRIM(T1) <= "03:00" AND TRIM(T2) >= "03:30" THEN 
     MOLA = MOLA + 30  
   END IF
   IF TRIM(T1) <= "06:00" AND TRIM(T2) >= "06:10" THEN 
     MOLA = MOLA + 10  
   END IF
END SUB
Function ZamaniGecenIsEmirlerindeDegisiklik()
	if Doc.IsBlank(SRVP00) Then Set SRVP00 = Doc.GetTableObject("SRVP00")
	if Doc.IsBlank(SRVP00O) Then Set SRVP00O = Doc.GetTableObject("SRVP00O")

	SonIslemTarihi = trim(Doc.Customp_Gets("!BKMTAR"))
	if trim(SonIslemTarihi) = "" then Exit Function

	Set d7Date = Doc.CreateSObject("d7Date")
	d7Date.SetDate trim(SonIslemTarihi)
	if Not d7Date.IsValid Then Exit Function
	SonIslemTarihi = d7Date.GetRevDate

	QStr = "SELECT * FROM SRVP00O WHERE KOD='" & SRVP00.KOD & "'"
	Set S00O = Doc.RunSQLQuery("sdfsdh", QStr)

	'Insert
	SRVP00O.SafePack()
	For i=1 to SRVP00O.GetRecCount()
		Set S00 = Doc.Select2("sadfsdgs", S00O.GetAppName, "TRNUM='" & SRVP00O.TRNUM(i) & "'", "", -1)
		if S00.GetRowCount() = 0 and (SRVP00O.RECTARIH1(i)<=SonIslemTarihi or SRVP00O.RECTARIH2(i)<=SonIslemTarihi) Then
			Doc.d7MsgBox Doc.StrToNorDate(SonIslemTarihi) & " tarihinden öncesine iþlem girilemez", "Ýþlem Kesildi", MB_OK
			ZamaniGecenIsEmirlerindeDegisiklik = True
			Exit Function
		End if
	Next

	'Update/Delete
	QStr = "SELECT * FROM SRVP00O WHERE KOD='" & SRVP00.KOD & "'"
	QStr = QStr & " AND (RECTARIH1<='" & SonIslemTarihi & "' OR RECTARIH2<='" & SonIslemTarihi & "')"
	Set S00O = Doc.RunSQLQuery("sdfsdh", QStr)
	if S00O.GetRowCount() = 0 Then Exit Function

	For i=1 to S00O.GetRecCount()
		Set S00 = Doc.Select2("sadfsdgs", "SRVP00O", "TRNUM='" & S00O.TRNUM(i) & "'", "", -1)
		if S00.GetRowCount() = 0 Then
			Doc.d7MsgBox Doc.StrToNorDate(SonIslemTarihi) & " tarihinden önceki iþlemler silinemez", "Ýþlem Kesildi", MB_OK
			ZamaniGecenIsEmirlerindeDegisiklik = True
			Exit Function
		Else
			if S00.RECTARIH1 & S00.RECTIME1 & S00.RECTARIH2 & S00.RECTIME2 & S00.CALISTIGISURE <> _
				S00O.RECTARIH1(i) & S00O.RECTIME1(i) & S00O.RECTARIH2(i) & S00O.RECTIME2(i) & S00O.CALISTIGISURE(i) Then

				Doc.d7MsgBox Doc.StrToNorDate(SonIslemTarihi) & " tarihinden önceki iþlemlerde deðiþiklik yapýlamaz.", "Ýþlem Kesildi", MB_OK
				ZamaniGecenIsEmirlerindeDegisiklik = True
				Exit Function
			End if
		End if
	Next
End Function

Sub MinimumStokKontrol2(HakKodu)
	If not OzelHakVarMi(HakKodu) Then Exit Sub

	QStr = "SELECT S00.GK_2, S00.KOD, S00.AD, S00.NAME2, S00.B_GUVENLIKSTOGU, S01.SF_MIKTAR, S46T.SIP_MIKTAR, S47T.TALEP_MIKTAR"
	QStr = QStr & " FROM STOK00 S00"
	QStr = QStr & " LEFT JOIN ("
	QStr = QStr & " 	SELECT KOD, SUM(SF_MIKTAR) SF_MIKTAR"
	QStr = QStr & " 	FROM STOK01"
	if HakKodu = "BGUVENLIK" Then
		QStr = QStr & " 	WHERE AMBCODE='70BAK'"
	End if
	QStr = QStr & " 	GROUP BY KOD"
	QStr = QStr & " ) S01 ON S01.KOD=S00.KOD"
	QStr = QStr & " LEFT JOIN ("
	QStr = QStr & " 	SELECT KOD, Sum(SF_MIKTAR-SF_NETKAPANANMIK) SIP_MIKTAR"
	QStr = QStr & " 	FROM STOK46T"
	QStr = QStr & " 	WHERE ACIKKAPALI=' '"
	QStr = QStr & " 	AND (SF_MIKTAR-SF_NETKAPANANMIK)>0"
	QStr = QStr & " 	GROUP BY KOD"
	QStr = QStr & " ) S46T ON S46T.KOD=S00.KOD"
	QStr = QStr & " LEFT JOIN ("
	QStr = QStr & " 	SELECT KOD, Sum(SF_MIKTAR-SF_NETKAPANANMIK) TALEP_MIKTAR"
	QStr = QStr & " 	FROM STOK47T"
	QStr = QStr & " 	WHERE ACIKKAPALI=' '"
	QStr = QStr & " 	AND (SF_MIKTAR-SF_NETKAPANANMIK)>0"
	QStr = QStr & " 	GROUP BY KOD"
	QStr = QStr & " ) S47T ON S47T.KOD=S00.KOD"

	QStr = QStr & " AND S00.CODETYPE='D'"

Function FIFOyaUygunMu(SeriNo)

	if trim(CST.D_TEXT(1)) <> "E" Then Exit Function

	if STC.PROPT = "2" Then Exit Function

	if Left(SeriNo, 2) = "--" Then Exit Function

	Set S01_2 = Doc.GetTableObject("STOK01_PDA")
	Set S60E = Doc.GetTableObject("STOK60E") 

	if trim(S60E.AMBCODE) <> "01MER" Then Exit Function
	
	Set S01 = Doc.Select1("sfsfd", "STOK01", "SERINO='"+trim(SeriNo)+"'", "", -1)
	Farkli = 0
	if S01.GetRowCount() > 0 Then
		Depo = S01.AMBCODE
		LotNo = Left(S01.LOTNUMBER, 6)
		StokKodu = S01.KOD
		Set S01 = Doc.Select1("sfsfd", "STOK01", "KOD='"+StokKodu+"'&AMBCODE='"+Depo+"'&LOTNUMBER<'"+LotNo+"'", "LOTNUMBER", -1)
		for i = 1 to S01.GetRowCount()
			Set S60Tx = Doc.Select2("dff444r55", "STOK60T", "SERINO='" & S01.SERINO(i) & "'", "", -1)
			if S60Tx.GetRowCount() = 0 Then
				Farkli = Farkli + 1
				Lot = Left(S01.LOTNUMBER(i), 6)
				if Lot <> EskiLot And Farkli > 1 Then Exit For
				if Farkli = 1 Then Hata = "Okunacak Ýlk Lot:" & trim(S01.LOTNUMBER(i)) & vbcrlf
				Hata = Hata & "L:" & GetLokasyon(S01, i) & ",M:" & S01.SF_MIKTAR(i) & vbcrlf

				EskiLot = Lot
			End if
		Next

	End if

	if trim(Hata) <> "" Then
		FIFOyaUygunMu = Hata
	End if
End Function

Function GetMRP_Hmd(HammaddeKodu, IhtiyacMiktari, TesisKodu, MamulMiktari)
	GetMRP_Hmd = ""
	BaslangicTarihi1 = Doc.BuGun
	SonTarih = Doc.StrDate_Increment(BaslangicTarihi1, 7)
	QStr = "R_KAYNAKTYPE='H'&R_KAYNAKKODU='" & HammaddeKodu & "'&EVRAKNO<>'" & MMPS10E.EVRAKNO & "'&R_ACIK_KAPALI<>'K'&ACIK_KAPALI<>'K'&R_STATUS='R'&R_LATEST_B_TARIH<'"&SonTarih&"'"

	MRPStr = ""
	HmdBirim = Doc.GetDBFieldValue("STOK00", "IUNIT", HammaddeKodu, "")
	For i = 1 to M10T.GetRowCount()
		Ihtiyac = M10T.R_BAKIYEKAYNAK(i)
		IhtiyacTarihi = M10T.R_LATEST_B_TARIH(i)
		ToplamIhtiyac = ToplamIhtiyac + Ihtiyac
		MRPStr = MRPStr & "MPS No : " & M10T.EVRAKNO(i) & _
		", Ýhtiyaç : " & Ihtiyac & " " & HmdBirim & _
		", Tarih : " & Doc.StrtonorDate(IhtiyacTarihi) & vbcrlf
	Next
	TesisStok = GetToplamStok(Hammaddekodu)
	Fark = TesisStok - (ToplamIhtiyac+IhtiyacMiktari)
	if Fark < 0 Then
		HammaddeAdi = Doc.GetDBFieldValue("STOK00", "AD", HammaddeKodu, "")
		BuMPSHaricFark = TesisStok - ToplamIhtiyac
		if BuMPSHaricFark > 0 Then
			Uretilebilecekmiktar = Doc.Round(Doc.Div(BuMPSHaricFark, IhtiyacMiktari) * MamulMiktari,0)
		Else
			Uretilebilecekmiktar = 0
		End if
		MRPStr = "Hammadde : " & trim(HammaddeKodu) & "  " & HammaddeAdi & vbcrlf & _
		"Tesis Toplam Stoðu : " & TesisStok & " " & HmdBirim & vbcrlf & _
		"Önceki MPS 'lerin Ýhtiyacý : " & ToplamIhtiyac & " " & HmdBirim & vbcrlf & _
		"Bu MPS 'in Ýhtiyacý : " & IhtiyacMiktari & " " & HmdBirim & vbcrlf & _
		"Toplam Ýhtiyaç : " & ToplamIhtiyac + IhtiyacMiktari & " " & HmdBirim & vbcrlf & _
		"Fark : " & Fark & " " & HmdBirim & vbcrlf & _
		"Üretilebilecek Miktar : " & Uretilebilecekmiktar & vbcrlf & _
		"Ýhtiyaç duyan diðer MPS 'ler : " & vbcrlf & MRPStr

Function OEEOnay()
	OEEOnay = 0
	PlanliDurusSuresi = 0
	PlansizDurusSuresi = 0
	FireMiktari = 0
	Set S32D = Doc.Select3("wf4fwd", "SFDC32D", "", "",-1)
	For i = 1 to S32D.GetRowCount()
		Set T73T = Doc.Select1("wd23fsdf", "TABL73T", "AKTIF_PASIF='*'&DURMA_SEBEBI='" & S32D.DURMA_SEBEBI(i) & "'", "", -1)
		if T73T.GetRowCount() = 0 Then
			PlansizDurusSuresi = PlansizDurusSuresi + S32D.SURE(i)
		Else
			if trim(T73T.STATU) = "P" Then
				PlanliDurusSuresi = PlanliDurusSuresi + S32D.SURE(i)
			Else
				PlansizDurusSuresi = PlansizDurusSuresi + S32D.SURE(i)
			End if
		End if
	Next
	AyarSuresi = SFDC32E.AYAR_SURESI
	PlansizDurusSuresi = PlansizDurusSuresi + AyarSuresi

	Set S20S = Doc.Select3G("wf4fwdSS", "SFDC20S", "", "","EVRAKNO",-1)
	if S20S.GetRowCount() > 0 Then FireMiktari = S20S.SF_MIKTAR

	NetHazirEkipmanZamani = SFDC32E.SURE - PlanliDurusSuresi
	EkipmanHazirBulunmaOrani = Doc.Div(NetHazirEkipmanZamani - PlansizDurusSuresi, NetHazirEkipmanZamani)

	Set M10T = Doc.Select1("sf234fdfg", "MMPS10T", "JOBNO='" & SFDC32E.JOBNO & "'", "", -1)
	IdealCevrimSuresi_sn = Doc.Div(M10T.R_MIKTAR0 * 3600, M10T.R_YMAMULMIKTAR)
	GerceklesenCevrimSuresi_sn = Doc.Div((NetHazirEkipmanZamani - PlansizDurusSuresi) * 3600, SFDC32E.SF_MIKTAR)
	PerformansVerimliligi = Doc.Div(IdealCevrimSuresi_sn, GerceklesenCevrimSuresi_sn)

	KalitePerformansi = Doc.Div(SFDC32E.SF_MIKTAR, SFDC32E.SF_MIKTAR + FireMiktari)

	OEE = EkipmanHazirBulunmaOrani * PerformansVerimliligi * KalitePerformansi

	Str1 = 				"Duruþ Performansý : % " & Doc.Round(EkipmanHazirBulunmaOrani * 100, 1) & vbcrlf
	Str1 = Str1 & "Hýz Performansý      : % " & Doc.Round(PerformansVerimliligi * 100, 1) & vbcrlf
	Str1 = Str1 & "Kalite Performansý  : % " & Doc.Round(KalitePerformansi * 100, 1) & vbcrlf & vbcrlf
	Str1 = Str1 & "OEE                        : % " & Doc.Round(OEE * 100, 1) & vbclf & vbcrlf 

'	k = Doc.d7MsgBox(Str1 & vbcrlf & "Onaylýyor musunuz", "OEE", MB_YESNO+MB_ICONINFORMATION)
	k = Doc.d7MsgBox(Str1, "OEE Bilgi", MB_OK)
'	if k <> 6 Then
'		OEEOnay = -1
'	End if
End Function
Sub OperatorSureleriniHesapla()

	For i = 1 to SFDC32O.GetRowCount()

		if trim(SFDC32E.RECTIME1) <> "" and trim(SFDC32O.RECTIME1(i)) = "" Then SFDC32O.RECTIME1(i) = SFDC32E.RECTIME1

		BSaat = SFDC32O.RECTIME1(i)

		ESaat = SFDC32O.RECTIME2(i)
		MolaSuresi = Doc.RunMacro4("MAKR7S.SFDC321", "GetPDurusSure", BTarih, BSaat, ETarih, ESaat)

		SFDC32O.DAKIKA(i) = SFDC32O.SURE(i) * 60
	Next
	
	SFDC32E.TOPLAM_OPRTR_S_ST = SFDC32O.SumOf("SURE")
End Sub

Sub GetToplamOpSure(EvrakNo, OpUretimSuresi, OpAyarSuresi, OpDurusSuresi)
	OpUretimSuresi = 0
	OpAyarSuresi = 0
	OpDurusSuresi = 0
	ToplamOpUretimSuresi = 0

	AyarBTarih = Doc.GetUDFValue("SFDC32E", EvrakNo, "AYR_BTARIH", 0)
	AyarBSaat = Doc.GetUDFValue("SFDC32E", EvrakNo, "AYR_BSAAT", 0)
	AyarETarih = Doc.GetUDFValue("SFDC32E", EvrakNo, "AYR_ETARIH", 0)
	AyarESaat = Doc.GetUDFValue("SFDC32E", EvrakNo, "AYR_ESAAT", 0)

	Set S32O = Doc.Select1("asdasdfha", "SFDC32O", "EVRAKNO='" & EvrakNo & "'", "", -1)
	Set S32D = Doc.Select1("afffffsdasdfha", "SFDC32D", "EVRAKNO='" & EvrakNo & "'", "", -1)
	For i = 1 to S32O.GetRowCount()
		ToplamOpUretimSuresi = ToplamOpUretimSuresi + Doc.DTDTDif1(S32O.RECTARIH1(i), S32O.RECTIME1(i), S32O.RECTARIH2(i), S32O.RECTIME2(i))
		OpAyarSuresi = OpAyarSuresi + GetKesisimSuresi(AyarBTarih, AyarBSaat, AyarETarih, AyarESaat, S32O.RECTARIH1(i), S32O.RECTIME1(i), S32O.RECTARIH2(i), S32O.RECTIME2(i))
		For j = 1 to S32D.GetRowCount()
			OpDurusSuresi = OpDurusSuresi + GetKesisimSuresi(S32D.RECTARIH1(j), S32D.RECTIME1(j), S32D.RECTARIH2(j), S32D.RECTIME2(j), S32O.RECTARIH1(i), S32O.RECTIME1(i), S32O.RECTARIH2(i), S32O.RECTIME2(i))
		Next
	Next
	OpUretimSuresi = ToplamOpUretimSuresi - OpDurusSuresi - OpAyarSuresi
End Sub
Function GetKesisimSuresi(BTarih1, BSaat1, ETarih1, ESaat1, BTarih2, BSaat2, ETarih2, ESaat2)
	GetKesisimSuresi = 0
	if ETarih1 & ESaat1 < BTarih2 & BSaat2 Then Exit Function
	if BTarih1 & BSaat1 > ETarih2 & ESaat2 Then Exit Function

	if BTarih1 & BSaat1 < BTarih2 & BSaat2 Then
		BTarih = BTarih2 
		BSaat = BSaat2
	Else
		BTarih = BTarih1 
		BSaat = BSaat1
	End if

	if ETarih1 & ESaat1 < ETarih2 & ESaat2 Then
		ETarih = ETarih1
		ESaat = ESaat1
	Else
		ETarih = ETarih2
		ESaat = ESaat2
	End if
	GetKesisimSuresi = Doc.DTDTDif1(BTarih, BSaat, ETarih, ESaat)
End Function

Sub MolaSureleriniHesapla()
	if Doc.IsBlank(SRVB00) Then Set SRVB00 = Doc.GetTableObject("SRVB00")
	if Doc.IsBlank(SRVB00O) Then Set SRVB00O = Doc.GetTableObject("SRVB00O")
	
	Set ddd=Doc.CreateSObject("d7Date")
	FOR Z = 1 TO SRVB00O.GETRECCOUNT()
		ddd.SetDate(SRVB00O.RECTARIH1(Z))
		L = ddd.GetDOWStr()
		IF TRIM(SRVB00O.RECTARIH1(Z)) = TRIM(SRVB00O.RECTARIH2(Z)) THEN
			MOLA = 0 
			CALL HESAPLA(SRVB00O.RECTIME1(Z),SRVB00O.RECTIME2(Z))
		ELSE 
			MOLA = 0
			CALL HESAPLA(SRVB00O.RECTIME1(Z),"23:59")
			CALL HESAPLA("00:00",SRVB00O.RECTIME2(Z))
		END IF

		SRVB00O.MOLA_DK(Z) = MOLA 
		molax = DOC.DIV(MOLA,60)
		SRVB00O.MOLA_SAAT(Z) = MOLAX
	NEXT
End Sub
SUB HESAPLA (T1,T2)
  IF TRIM(T1) <= "10:00" AND TRIM(T2) >= "10:10" THEN 
    MOLA = MOLA + 10  
  END IF 
 
'********cuma günü ise
  IF TRIM(L) = "Cum" THEN

    IF TRIM(T1) <= "12:40" AND TRIM(T2) >= "14:10" THEN 
      MOLA = MOLA + 90  
    END IF

    IF TRIM(T1) <= "16:00" AND TRIM(T2) >= "16:10" THEN 
      MOLA = MOLA + 10  
    END IF
   ELSE 
'*****cum deðil ise
     IF TRIM(T1) <= "12:40" AND TRIM(T2) >= "13:10" THEN 
       MOLA = MOLA + 30  
     END IF
     IF TRIM(T1) <= "15:00" AND TRIM(T2) >= "15:10" THEN 
       MOLA = MOLA + 10  
     END IF
   END IF 
   IF TRIM(T1) <= "01:00" AND TRIM(T2) >= "01:10" THEN 
     MOLA = MOLA + 10  
   END IF
   IF TRIM(T1) <= "03:00" AND TRIM(T2) >= "03:30" THEN 
     MOLA = MOLA + 30  
   END IF
   IF TRIM(T1) <= "06:00" AND TRIM(T2) >= "06:10" THEN 
     MOLA = MOLA + 10  
   END IF
END SUB

Sub IadeGoster(StokKodu)
	QStr = "SELECT S60T.KOD, S00.AD KOD_STOK00_AD, S60T.CH_KODU, S29T.EVRAKNO GIRIS_IRSNO, S29T.TARIH, SUM(SF_MIKTAR) IADE_MIK"
	QStr = QStr & " FROM STOK60T S60T"
	QStr = QStr & " INNER JOIN STOK00 S00 ON S00.KOD=S60T.KOD"
	QStr = QStr & " INNER JOIN ("
	QStr = QStr & "   SELECT EVRAKNO, KOD, SERINO, TARIH"
	QStr = QStr & "   FROM STOK29T"
	QStr = QStr & "   WHERE IADEMI<>'1'"
	QStr = QStr & "   AND CARIHESAPCODE='" & STOK29E.CARIHESAPCODE & "'"
	QStr = QStr & "   AND KOD='" & StokKodu & "'"
	QStr = QStr & "   AND TRIM(SERINO) IS NOT NULL"
	QStr = QStr & "   GROUP BY EVRAKNO, KOD, SERINO, TARIH"
	QStr = QStr & " ) S29T ON S29T.SERINO=S60T.SERINO AND S29T.KOD=S60T.KOD"
	QStr = QStr & " WHERE S60T.IADEMI='1'"
	QStr = QStr & " AND S29T.EVRAKNO IN ("
	QStr = QStr & "  SELECT EVRAKNO"
	QStr = QStr & "  FROM ("
	QStr = QStr & "   SELECT EVRAKNO"
	QStr = QStr & "   FROM STOK29T"
	QStr = QStr & "    WHERE CARIHESAPCODE='" & STOK29E.CARIHESAPCODE & "'"
	QStr = QStr & "    AND KOD='" & StokKodu & "'"
	QStr = QStr & "    AND EVRAKNO<>'" & STOK29E.EVRAKNO & "'"
	QStr = QStr & "    AND TARIH<='" & STOK29E.TARIH & "'"
	QStr = QStr & "    GROUP BY TARIH, EVRAKNO"
	QStr = QStr & "    ORDER BY TARIH DESC, EVRAKNO DESC"
	QStr = QStr & "  )"
	QStr = QStr & "  WHERE ROWNUM<=5"
	QStr = QStr & " )"
	QStr = QStr & " AND S60T.CH_KODU='" & STOK29E.CARIHESAPCODE & "'"
	QStr = QStr & " AND S60T.KOD='" & StokKodu & "'"
	QStr = QStr & " GROUP BY S29T.TARIH, S60T.KOD, S00.AD, S60T.CH_KODU, S29T.EVRAKNO"
	QStr = QStr & " ORDER BY S29T.TARIH DESC"
	Set S60T = Doc.RunSQLQuery("asdfsdg", QStr)
	if S60T.GetRowCount() = 0 Then Exit Sub
	
	For i = 1 to S60T.GetRowCount()
		Doc.d7MsgBox "Stok Kodu : " & trim(S60T.KOD(i)) & " - " & trim(S60T.KOD_STOK00_AD(i)) & ", Giriþ Ýrsaliye No : " & trim(S60T.GIRIS_IRSNO(i)) & ",  Giriþ Tarihi : " & Doc.StrToNorDate(S60T.TARIH(i)), "", MB_OK
	Next
End Sub
