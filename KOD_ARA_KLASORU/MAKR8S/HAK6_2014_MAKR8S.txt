Function SF_PMIKTAR_OnValidate
	' her zaman iþlem miktarý üzerinden hesaplama yapýldýðýndan 
	' kaynak alan iþlem miktarý olacak þekilde ayarlandý.
	
	Set S40T = Doc.GetTableObject("STOK40T")
	PaketSayisi = S40T.SF_PMIKTAR
	IslemMiktari = S40T.SF_MIKTAR
	SatirNo = S40T.GetCurrentRow

	S40T.SetFieldValueNV "SF_PMIKTAR", satirNo, PaketSayisi
	
	If IslemMiktari <> 0 Then
		PaketIciMiktar = Doc.Div(IslemMiktari, PaketSayisi)
	Else
		IslemMiktari = PaketSayisi * PaketIciMiktar
	End If

	S40T.SetFieldValueNV "SF_IMIKTAR", satirNo, PaketIciMiktar	
	S40T.SetFieldValueNV "SF_MIKTAR", satirNo, IslemMiktari
End Function
'--------------------------------------------------------------------------

	IslemMiktari = S40T.SF_MIKTAR
	PaketSayisi = S40T.SF_PMIKTAR
	PaketIciMiktar = Doc.Div(IslemMiktari, PaketSayisi)

	S40T.SetFieldValueNV "SF_PMIKTAR", satirNo, PaketSayisi
	
	S40T.SetFieldValueNV "SF_IMIKTAR", satirNo, PaketIciMiktar
	S40T.SetFieldValueNV "SF_MIKTAR", satirNo, IslemMiktari
End Function

Function SF_IMIKTAR_OnValidate
	Set S40T = Doc.GetTableObject("STOK40T")
	SatirNo = S40T.GetCurrentRow
	
	PaketSayisi = S40T.SF_PMIKTAR
	PaketIciMiktar = S40T.SF_IMIKTAR

		If IslemMiktari <> 0 Then

	S40T.SetFieldValueNV "SF_IMIKTAR", satirNo, PaketIciMiktar
	S40T.SetFieldValueNV "SF_MIKTAR", satirNo, IslemMiktari
End Function

Sub OtomatikDovizKuru()
	KurUSD = Doc.GetKur("USD", Doc.DefaultPB, Doc.BuGun, "", -1)
	if KurUSD > 0 Then Exit Sub

	if not YeniEvrakKaydedebilirMi("EXCRAT", Doc.Getd7UserName) Then Exit Sub

	Set objXMLDoc = Nothing
	Set objXMLDoc = CreateObject("Microsoft.XMLDOM")
	objXMLDoc.async = False

	Set Doc1 = Doc.CreateDocumentNoUI("EXCRAT", "")
	Call GetKurTable(Doc1, objXMLDoc, Doc.BuGun)	
End Sub
Sub GetKurTable(Doc1, objXMLDoc, Tarih)
	Set EXCE = Doc1.GetTableObject("EXCRATE")
	Set EXCT = Doc1.GetTableObject("EXCRATT")

	KURTarih = Doc1.StrDate_Increment(Tarih, -1)
	Call HaftaSonuysaGeriAl(KURTarih )

	xmlTarih = Replace(Left(KURTarih ,7), "/", "") & "/" & Replace(Doc1.StrToNorDate(KURTarih ), "/", "")

	objXMLDoc.Load "http://www.tcmb.gov.tr/kurlar/" & xmlTarih & ".xml"
'	objXMLDoc.Load "c:\today.xml"

	Set Root = objXMLDoc.documentElement
	if Root is Nothing Then
'		k = doc.d7MsgBox("Kurlar alýnamadý " & xmlTarih, "Hata", MB_OK)
		Exit Sub
	End if
	Set KurList = Root.getElementsByTagName("Currency")

	EXCE.EVRAKNOTARIH = Tarih

	EXCT.Empty    
	For Each Kur In KurList
		ParaBirimi = Kur.Attributes(1).NodeValue
		if trim(Doc1.GetGecoValue("*PUNIT", "KOD", ParaBirimi, d7c_ErrMsgNo)) <> "" Then
			Birim = GetKurValue(Kur, "Unit")
			DovizAlis = GetKurValue(Kur, "ForexBuying")
			DovizSatis = GetKurValue(Kur, "ForexSelling")
			EfektifAlis = GetKurValue(Kur, "BanknoteBuying")
			EfektifSatis = GetKurValue(Kur, "BanknoteSelling")
		
			satirNo = EXCT.AddRow
			EXCT.SetCurrentRow(satirNo)
			EXCT.CODEFROM(satirNo) = ParaBirimi
			EXCT.CODETO(satirNo) = Doc1.DefaultPB
			EXCT.KURS_1(satirNo) = DovizAlis / Doc1.CDbl(Birim)
			EXCT.KURS_3(satirNo) = DovizAlis / Doc1.CDbl(Birim)
			EXCT.KURS_4(satirNo) = DovizSatis / Doc1.CDbl(Birim)
			EXCT.KURS_5(satirNo) = EfektifAlis / Doc1.CDbl(Birim)
			EXCT.KURS_6(satirNo) = EfektifSatis / Doc1.CDbl(Birim)
		End if
	Next
	Doc1.Save_Voucher
	Doc1.LoadEmpty_Voucher
End Sub
Function GetKurValue(Kurxml, KurTipi)
	Set KurAlisSatisxml = Kurxml.SelectSingleNode(KurTipi)
	If Not KurAlisSatisxml Is Nothing Then
		If Not KurAlisSatisxml.FirstChild Is Nothing Then
			GetKurValue = KurAlisSatisxml.FirstChild.Data
		End If
	End If
End Function

Sub AfterLoad
	If Doc.DataEntryMode <> "GUI" Then Exit Sub
	Set S20TI = Doc.GetTableObject("STOK20TI")
	Set Tbl_S01 = Doc.GetTableObject("STOK01_PDA")
	
	TuketilecekKod = Trim(S20TI.KOD(1))
	
	Tbl_S01.Empty
	If TuketilecekKod <> "31 10 20 312" And TuketilecekKod <> "31 10 20 412" Then Exit Sub
		
	QStr = "SELECT * FROM STOK01"
	QStr = QStr & " WHERE Trim(KOD)='"&TuketilecekKod&"' "
	QStr = QStr & " AND Trim(ALLOC_SYSSTATUS) IS NULL"
	QStr = QStr & " ORDER BY AMBCODE, VRI_VR_R3 DESC"		

	Set S01 = Doc.RunSQLQuery("RESULT_TBL", Qstr) 

	KayitSayisi = S01.GetRecCount
	
	If KayitSayisi > 0 Then 
			For i = 1 To KayitSayisi
			SatirNo = Tbl_S01.AddRow	
			Tbl_S01.SetCurrentRow(SatirNo)
			HammaddeKodu = Trim(S01.KOD(i))
			SeriNo = Trim(S01.SERINO(i))
			Depo = Trim(S01.AMBCODE(i))
			BukumYonu = Trim(S01.VRI_VR_C(i))
			Makara = Trim(S01.VRI_VR_R1(i))
			Nitelik = Trim(S01.VRI_VR_R2(i))
			Varyant3 = Trim(S01.VRI_VR_R3(i))
			Miktar = S01.SF_MIKTAR(i)
			Birimi = S01.SF_UNIT(i)
		
			Tbl_S01.KOD = HammaddeKodu
			Tbl_S01.SERINO = SeriNo
			Tbl_S01.AMBCODE = Depo
			Tbl_S01.VRI_VR_C = BukumYonu
			Tbl_S01.VRI_VR_R1 = Makara
			Tbl_S01.VRI_VR_R2 = Nitelik
			Tbl_S01.VRI_VR_R3 = Varyant3			
			Tbl_S01.SF_MIKTAR = Miktar
			Tbl_S01.SF_UNIT = Birimi
		Next

	End If
