Function ErdemirFiyatKontrol()
	if trim(STOK74E.CH_KODU) <> "ERDEMIR" Then exit Function
	Tolerans = 5
	Hata = ""
	For i = 1 to STOK74T1.GetRecCount()
		SiparisFiyati = Doc.GetDBFieldValue("STOK46T", "OR_FIYAT", STOK74T1.SIPARTNO(i), "ARTNO")
		FaturaFiyati = GetFaturaFiyat(STOK74T1.OR_TRNUM(i))
		Fark = (FaturaFiyati - SiparisFiyati) / SiparisFiyati * 100
		if Abs(Fark) > Tolerans then

	Next
	ErdemirFiyatKontrol = Hata
