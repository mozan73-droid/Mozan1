Sub OnLogin()

		Set S00C = GetKalipBakimList()
		Call KalipBakimKontrol(S00C)
	Elseif OzelHakVarMi("KALIBRASYO") Then
		Call KalibrasyonKontrol()

		Set S00C = GetKalipBakimList()

	End if

		if UyariGunu = 0 Then UyariGunu = 15

End Sub

Sub KalipBakimUyariMail(S00C)
	if not YeniEvrakKaydedebilirMi("SRVPC0", Doc.Getd7UserName) Then Exit Sub

	Kaliplar = ""
	MsgStr = "Bakým zamaný gelen kalýplar :"
	For i = 1 to S00C.GetRowCount()
		SonMesajTarihi = Doc.GetUDFValue("SRVPC0", S00C.KOD(i), "MTARIH", 0)
		If SonMesajTarihi <> Doc.StrToNorDate(Doc.BuGun) Then
			MsgStr = MsgStr & "Kalýp Kodu : " & S00C.KOD(i) & ", Kalan Ömür : " & Doc.Round(S00C.KALAN_MIKTAR(i),0)
			Kaliplar = Kaliplar & "," & S00C.KOD(i)
		End if
	Next
	if trim(Kaliplar) = "" Then Exit Sub

	Set U12 = Doc.Select1("de443d44ts", "UDEFT12", "OZEL_HAK_KODU='KALIPMAIL'", "", -1)
	For i = 1 to U12.GetRowCount()
		email = Doc.GetDBFieldValue("UDEF", "EMAIL", U12.KOD(i), "")
		if trim(email) <> "" Then
			Call SendMail(email,"## Bakým Zamaný Gelen Kalýplar ##",MsgStr,"")
		End if
	Next
	if Len(trim(Kaliplar)) > 0 Then
		Call KartKTAGuncelle(Mid(Kaliplar, 2))
	End if
End Sub
Sub SendMail(strTo,strHeader,strMsg,DosyaAdi)
	cdoSendUsingPickup = 1 
	cdoSendUsingPort = 2 
	cdoAnonymous = 0
	cdoBasic = 1
	cdoNTLM = 2 'NTLM

	mailserver = "mail.gungorotomobil.com.tr"
	mailUser = "freedom@gungorotomobil.com.tr"
	mailPass = "tteknoloji"
	mailFrom = "freedom@gungorotomobil.com.tr"
	
	Set objMessage = CreateObject("CDO.Message")
	objMessage.From = trim(mailFrom)
	objMessage.To = strTo
	objMessage.Subject = strHeader
	objMessage.TextBody = strMsg
	
	if trim(DosyaAdi) <> "" Then
		objMessage.AddAttachment("file://" & DosyaAdi)
	End if
	objMessage.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/sendusing") = cdoSendUsingPort 
	
	objMessage.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/smtpserver") = trim(mailserver)
	
	objMessage.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/smtpauthenticate") = cdoBasic
	
	objMessage.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/sendusername") = trim(mailUser)
	
	objMessage.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/sendpassword") = trim(mailPass)
	
	objMessage.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/smtpserverport") = 25
	
	objMessage.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/smtpusessl") = False
	
	objMessage.Configuration.Fields.Item _
	("http://schemas.microsoft.com/cdo/configuration/smtpconnectiontimeout") = 60
	
	objMessage.Configuration.Fields.Update
	objMessage.Send
End Sub

Function YeniEvrakKaydedebilirMi(ProgramKodu, KullaniciAdi)
	if trim(KullaniciAdi) = "D7KSUPER" or KullaniciAdi = "FX" Then
		YeniEvrakKaydedebilirMi = True
		Exit Function
	End if

	GrupKullanici = Doc.GetDBFieldValue("UDEF", "AYNIHAK", KullaniciAdi, "")
	if trim(GrupKullanici) <> "" Then KullaniciAdi = GrupKullanici

	YeniEvrakKaydedebilirMi = False
	Set UT1 = Doc.Select1("daef3ff", "UDEFT1", "KOD='" & KullaniciAdi & "'&PROGRAMKODU='" & ProgramKodu & "'", "", -1)
	if UT1.GetRowCount() > 0 Then
		if UT1.RIGHT_W="1" Then
			YeniEvrakKaydedebilirMi = True
			Exit Function
		End if
	Else
		YeniHakki = Doc.GetDBFieldValue("UDEF", "PR_RIGHT_W", KullaniciAdi, "")
		if YeniHakki = "1" Then
			YeniEvrakKaydedebilirMi = True
			Exit Function
		End if
	End if
End Function
Function GetKalipBakimList()
	Set GetKalipBakimList = Doc.Select1("sxxsee_dd", "SRVP00C", "KOD='zsesew32ed'", "", -1)

	Set S00C = Doc.Select1("ssee_dd", "SRVP00C", "KALAN_MIKTAR<" & 10000, "KALAN_MIKTAR", -1)
	For i = 1 to S00C.GetRowCount()
		KalipOmru = Doc.GetDBFieldValue("SRVPC0", "OB_FREKANSI", S00C.KOD(i), "")
		If S00C.KALAN_MIKTAR(i) < (0.1 * KalipOmru) and Not AcikOnleyiciBakimIsEmriVar(S00C.KOD(i)) Then
			satirNo = GetKalipBakimList.AddRow()
			GetKalipBakimList.KOD(satirNo) = S00C.KOD(i)
		End if
	Next
End Function
