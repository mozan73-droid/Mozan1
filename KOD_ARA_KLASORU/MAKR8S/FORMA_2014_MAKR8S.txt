Sub KartTuruHesapla()
	For i = 1 to MMPS10T.GetRowCount()
		if MMPS10T.R_KAYNAKTYPE(i) = "H" Then
			MMPS10T.R_KAYNAKKODU_STOK00_KART_TIPI(i) = Doc.GetDBFieldValue("STOK00", "KART_TIPI", MMPS10T.R_KAYNAKKODU(i), "")
		End if
	Next
	Doc.RunCommand "CMD_HESAPLANMIS_1"
End Sub
Sub KaynakTarihleriniHesapla()
	if Left(MMPS10E.MAMULSTOKKODU, 3) <> "152" Then Exit Sub
	Call OperasyonTarihleriniHesapla()
	Call HammaddeIhtiyacTarihleriniHesapla()
	Doc.RunCommand "CMD_HESAPLANMIS_1"
End Sub
Sub HammaddeIhtiyacTarihleriniHesapla()
	Set M10TH = Doc.Select2("sdsdgash", "MMPS10T", "R_KAYNAKTYPE='H'", "SRNUM", -1)
	Call MontajSiralariniHesapla(M10TH)

	if M10TH.GetRowCount() = 0 Then
		Doc.d7MsgBox MMPS10E.MAMULSTOKKODU & " ürün aðacýnda hammadde bulunamadý", "", MB_OK
		Exit Sub
	End if
	MaxMontajSiraNo = M10TH.R_HM_BAKIYE(1)
	For i = 1 to M10TH.GetRowCount()
		satirNo = Doc.Cdbl(M10TH.SRNUM(i))
		if M10TH.R_HM_BAKIYE(i) = MaxMontajSiraNo Then
			BitisTarihi = GetIlkOpBaslangicTarihi()
		Else
			BitisTarihi = GetOncekiMontajBaslanicTarihi(M10TH, i)
		End if
		UretimSuresi = GetYMUretimSuresi(M10TH.R_KAYNAKKODU(i), M10TH.R_MIKTAR0(i))
		BaslangicTarihi = GetBaslangicTarihi(BitisTarihi, UretimSuresi)

		M10TH.R_LATEST_B_TARIH(i) = BaslangicTarihi
		MMPS10T.R_LATEST_B_TARIH(satirNo) = BaslangicTarihi
'		MMPS10T.R_HM_BAKIYE(satirNo) = M10TH.R_HM_BAKIYE(i)
	Next
End Sub
Sub MontajSiralariniHesapla(M10TH)
	if M10TH.GetRowCount() = 0 Then Exit Sub
	QStr = "SELECT BOMREC_KAYNAKCODE, REFNUM01"
	QStr = QStr & " FROM BOMU01T "
	QStr = QStr & " WHERE BOMREC_CODE='" & M10TH.R_MAMULKODU(1) & "' AND AKTIF_PASIF='*' AND BOMREC_INPUTTYPE='H'"
	Set B01T = Doc.RunSQLQuery("dferhshsh", QStr)

	For i = 1 to M10TH.GetRowCount()
		Set B01T1 = Doc.Select2("sfasdfhad", B01T.GetAppName, "BOMREC_KAYNAKCODE='" & M10TH.R_KAYNAKKODU(i) & "'", "", -1)
		if B01T1.GetRowCount() > 0 Then
			M10TH.R_HM_BAKIYE(i) = B01T1.REFNUM01(1)
		Else
			M10TH.R_HM_BAKIYE(i) = 0
		End if
	Next
	M10TH.Sort("<R_HM_BAKIYE")
End Sub
Function GetIhtiyacTarihi(HammaddeKodu, MPSTeslimTarihi)
	GetIhtiyacTarihi = MPSTeslimTarihi
	Set S00 = Doc.RunSQLQuery("sdfsds", "SELECT B_HAZIRLIKSURESI FROM STOK00 WHERE KOD='" & HammaddeKodu & "'")
	if S00.GetRowCount() = 0 Then Exit Function
	if S00.B_HAZIRLIKSURESI(1) = 0 Then Exit Function

	GetIhtiyacTarihi = GetIsGunu(Doc.StrDate_Increment(MPSTeslimTarihi, -1 * S00.B_HAZIRLIKSURESI(1)))
End Function
Function GetIsGunu(Tarih)
	GetIsGunu = Tarih
	Set d7Date = Doc.CreateSObject("d7Date")
	d7Date.SetDate Tarih
	if d7Date.GetDOW > 5 Then
		d7Date.Increment (d7Date.GetDOW - 7)
		GetIsGunu = d7Date.GetRevDate
	End if
End Function
Sub OperasyonTarihleriniHesapla()
	Set M10TI = Doc.Select2("sdsdgash", "MMPS10T", "R_KAYNAKTYPE='I'", "<SRNUM", -1)
	For i = 1 to M10TI.GetRowCount()
		satirNo = Doc.Cdbl(M10TI.SRNUM(i))
		if i = 1 Then
			MMPS10T.R_LATEST_E_TARIH(satirNo) = MMPS10E.REVURETIMDENTESTARIH
		Else
			MMPS10T.R_LATEST_E_TARIH(satirNo) = M10TI.R_LATEST_B_TARIH(i-1)
		End if
		IslemSuresiGun = GetSureGun(M10TI.R_KAYNAKKODU(i), M10TI.R_MIKTART(i))
		M10TI.R_LATEST_B_TARIH(i) = GetBaslangicTarihi(MMPS10T.R_LATEST_E_TARIH(satirNo), IslemSuresiGun)
		MMPS10T.R_LATEST_B_TARIH(satirNo) = M10TI.R_LATEST_B_TARIH(i)
	Next
End Sub
Function GetBaslangicTarihi(BitisTarihi, UretimSuresi)
	BaslangicTarihi = Doc.StrDate_Increment(BitisTarihi, -1 * UretimSuresi)
	TatilGunSayisi = GetTatilGunSayisi(BaslangicTarihi, BitisTarihi)

	GetBaslangicTarihi = GetIsGunu(Doc.StrDate_Increment(BaslangicTarihi, -1 * TatilGunSayisi))
End Function
Function GetSureGun(TezgahKodu, SureSaat)
	GetSureGun = 0
	Set I00 = Doc.RunSQLQuery("dfgsdfgs", "SELECT B_KAPASITE, B_KAPASITE_PERMPS FROM IMLT00 WHERE KOD='" & TezgahKodu & "'")
	if I00.GetRowCount() = 0 Then Exit Function
	GunlukKapasite = I00.B_KAPASITE_PERMPS
	if GunlukKapasite = 0 Then GunlukKapasite = I00.B_KAPASITE
	if GunlukKapasite = 0 Then Exit Function

	GetSureGun = Doc.Ceil(Doc.Div(SureSaat, GunlukKapasite))
End Function
Function GetTatilGunSayisi(BaslangicTarihi, ByVal BitisTarihi)
	GetTatilGunSayisi = 0
	Set d7Date = Doc.CreateSObject("d7Date")
	d7Date.SetDate BitisTarihi
	if Not d7Date.IsValid Then Exit Function
	BitisTarihi = d7Date.GetRevDate

	d7Date.SetDate BaslangicTarihi
	if Not d7Date.IsValid Then Exit Function

	While d7Date.GetRevDate <= BitisTarihi
		if d7Date.GetDOW > 5 Then
			GetTatilGunSayisi = GetTatilGunSayisi + 1
		End if
		d7Date.Increment 1
	Wend
End Function
Function GetIlkOpBaslangicTarihi()

		GetIlkOpBaslangicTarihi = M10TI1.R_LATEST_B_TARIH(1)

	End if
End Function
Function GetYMUretimSuresi(YMKodu, Miktar)
	GetYMUretimSuresi = 0
	if trim(YMKodu) = "" Then Exit Function
	if Doc.Cdbl(Miktar) = 0 Then Exit Function

	QStr = "SELECT BOMREC_KAYNAKCODE, (" & Doc.Cdbl(Miktar) & "*BOMREC_KAYNAK0/BOMREC_MAMULMIKTAR + BOMREC_KAYNAK1) SURE"
	QStr = QStr & " FROM BOMU01T"
	QStr = QStr & " WHERE BOMREC_CODE='" & YMKodu & "' AND BOMREC_INPUTTYPE='I' AND BOMREC_ALTTURU<'1' AND AKTIF_PASIF='*'"
	Set B01T = Doc.RunSQLQuery("asdferuqsgsj", QStr)
	For i = 1 to B01T.GetRowCount()
		GetYMUretimSuresi = GetYMUretimSuresi + GetSureGun(B01T.BOMREC_KAYNAKCODE(i), B01T.SURE(i))
	Next
End Function
Function GetOncekiMontajBaslanicTarihi(M10TH, satirNo)

	End if
End Function
Sub MusteriHesapla()
	if MMPS10TM.GetRowCount() = 0 Then Exit Sub
	if trim(MMPS10TM.KAYNAKMPS_EVRAKTYPE) <> "UP" Then Exit Sub

	QStr = "SELECT M.MUSTERIKODU, M.MUSTERISIPINO, M.MUSTERISIPIARTNO, C.AD"
	QStr = QStr & " FROM MMPS10TM M"
	QStr = QStr & " LEFT JOIN CARI00 C ON C.KOD=M.MUSTERIKODU"
	QStr = QStr & " WHERE M.EVRAKNO='" & MMPS10TM.KAYNAKMPS & "'"
	Set M10TM = Doc.RunSQLQuery("sdfgsdfhsdfj", QStr)
	if M10TM.GetRowCount() = 0 Then Exit Sub
	MMPS10TM.MUSTERIKODU(1) = M10TM.MUSTERIKODU(1)
	MMPS10TM.MUSTERIKODU_CARI00_AD(1) = M10TM.AD(1)
	MMPS10TM.MUSTERISIPINO(1) = M10TM.MUSTERISIPINO(1)
	MMPS10TM.MUSTERISIPIARTNO(1) = M10TM.MUSTERISIPIARTNO(1)

	MMPS10E.MUSTERIKODU = M10TM.MUSTERIKODU(1)
	MMPS10E.MUSTERIKODU_CARI00_AD = M10TM.AD(1)
	MMPS10E.MUSTERISIPINO = M10TM.MUSTERISIPINO(1)
	MMPS10E.MUSTERISIPIARTNO = M10TM.MUSTERISIPIARTNO(1)

	Doc.RunCommand "CMD_HESAPLANMIS_1"
End Sub
Function BaglantiliMPSSorun()
	BaglantiliMPSSorun = False
	QStr = "SELECT MMPS10TM.EVRAKNO, MMPS10TM.MAMULSTOKKODU, STOK00.AD FROM MMPS10TM"
	QStr = QStr & " LEFT JOIN STOK00 ON STOK00.KOD=MMPS10TM.MAMULSTOKKODU"
	QStr = QStr & " WHERE KAYNAKMPS_EVRAKTYPE='UP' and KAYNAKMPS='" & MMPS10E.EVRAKNO & "'"
	QStr = QStr & " AND KAYNAKMPS_TRNUM IN ("
	QStr = QStr & "     SELECT TRNUM FROM MMPS10T WHERE EVRAKNO='" & MMPS10E.EVRAKNO & "'"
	QStr = QStr & "     AND TRNUM NOT IN ("
	For i = 1 to MMPS10T.GetRowCount()
		if i > 1 Then QStr = QStr & ","
		QStr = QStr & "'" & MMPS10T.TRNUM(i) & "'"
	Next
	QStr = QStr & "     )"
	QStr = QStr & " )"
	Set M10TM = Doc.RunSQLQuery("dfgsfhs", QStr)
	if M10TM.GetRowCount() = 0 Then Exit Function

	BaglantiliMPSSorun = False
	Set ggg = Doc.CreateSObject("BATCHERRORMODE")
	Doc.d7MsgBox "Bu MPS 'e ait baðlantýlý MPS 'ler var, hammadde satýrlarý silinemez. Baðlantýlý MPS 'ler : ", "", MB_OK

	For i = 1 to M10TM.GetRowCount()
		Doc.d7MsgBox "MPS No : " & trim(M10TM.EVRAKNO(i)) & ", Stok Kodu : " & Trim(M10TM.MAMULSTOKKODU(i)) & " - " & Trim(M10TM.AD(i)), "", MB_OK
	Next
End Function

		Call MusteriAdlariniHesapla()
	Elseif cmd = "CMD_SUZ_MMPS10T" Then
		Call MusteriAdlariniHesapla()

Sub MusteriAdlariniHesapla()
	For i = 1 to MMPS10T_LISTESI.GetRowCount()
		Doc.StatusBarMessage "Müþteriler Hesaplanýyor : " & i & " / " & MMPS10T_LISTESI.GetRowCount()
		if i = 1 Then
			QStr = "SELECT C.AD FROM MMPS10TM TM LEFT JOIN CARI00 C ON C.KOD=TM.MUSTERIKODU WHERE TM.EVRAKNO='" & MMPS10T_LISTESI.EVRAKNO(i) & "'"
			Set M10TM = Doc.RunSQLQuery("sdfasdg", QStr)
			if M10TM.GetRowCount() > 0 Then
				MMPS10T_LISTESI.RECETENOTU(i) = M10TM.AD(1)
			End if
		Else
			if MMPS10T_LISTESI.EVRAKNO(i) = MMPS10T_LISTESI.EVRAKNO(i-1) Then
				MMPS10T_LISTESI.RECETENOTU(i) = MMPS10T_LISTESI.RECETENOTU(i-1)
			Else
				QStr = "SELECT C.AD FROM MMPS10TM TM LEFT JOIN CARI00 C ON C.KOD=TM.MUSTERIKODU WHERE TM.EVRAKNO='" & MMPS10T_LISTESI.EVRAKNO(i) & "'"
				Set M10TM = Doc.RunSQLQuery("sdfasdg", QStr)
				if M10TM.GetRowCount() > 0 Then
					MMPS10T_LISTESI.RECETENOTU(i) = M10TM.AD(1)
				End if
			End if
		End if
	Next
End Sub

	End if

Sub D_ID2_AfterValidate()

Sub ArtNoEkle()
	if Doc.IsBlank(STOK29T) Then Set STOK29T = Doc.GetTableObject("STOK29T")
	if trim(STOK29T.D_ID) <> "" Then
		SipArtNo = Doc.GetDBFieldValue("D7KIDSLB", "SIPARTNO_46", Doc.LeftJustify(trim(STOK29T.D_ID), 30), "")
		if trim(SipArtNo) <> "" Then
			STOK29T.SIPARTNO = SipArtNo
		End if
	End if	
End Sub
Sub MiktarlariGuncelle()

Sub GetKurTable(Doc1, objXMLDoc, Tarih)
	Set EXCE = Doc1.GetTableObject("EXCRATE")
	Set EXCT = Doc1.GetTableObject("EXCRATT")

	KURTarih = Doc1.StrDate_Increment(Tarih, -1)
	Call HaftaSonuysaGeriAl(KURTarih )

	xmlTarih = Replace(Left(KURTarih ,7), "/", "") & "/" & Replace(Doc1.StrToNorDate(KURTarih ), "/", "")

	objXMLDoc.Load "http://www.tcmb.gov.tr/kurlar/" & xmlTarih & ".xml"
'	objXMLDoc.Load "c:\today.xml"

	Set Root = objXMLDoc.documentElement
	if Root is Nothing Then
'		k = doc.d7MsgBox("Kurlar alýnamadý " & xmlTarih, "Hata", MB_OK)
		Exit Sub
	End if
	Set KurList = Root.getElementsByTagName("Currency")

	EXCE.EVRAKNOTARIH = Tarih

	EXCT.Empty    
	For Each Kur In KurList
		ParaBirimi = Kur.Attributes(1).NodeValue
		if trim(Doc1.GetGecoValue("*PUNIT", "KOD", ParaBirimi, d7c_ErrMsgNo)) <> "" Then
			Birim = GetKurValue(Kur, "Unit")
			DovizAlis = GetKurValue(Kur, "ForexBuying")
			DovizSatis = GetKurValue(Kur, "ForexSelling")
			EfektifAlis = GetKurValue(Kur, "BanknoteBuying")
			EfektifSatis = GetKurValue(Kur, "BanknoteSelling")
		
			satirNo = EXCT.AddRow
			EXCT.SetCurrentRow(satirNo)
			EXCT.CODEFROM(satirNo) = ParaBirimi
			EXCT.CODETO(satirNo) = Doc1.DefaultPB
			EXCT.KURS_1(satirNo) = DovizAlis / Doc1.CDbl(Birim)
			EXCT.KURS_3(satirNo) = DovizAlis / Doc1.CDbl(Birim)
			EXCT.KURS_4(satirNo) = DovizSatis / Doc1.CDbl(Birim)
			EXCT.KURS_5(satirNo) = EfektifAlis / Doc1.CDbl(Birim)
			EXCT.KURS_6(satirNo) = EfektifSatis / Doc1.CDbl(Birim)
		End if
	Next
	Doc1.Save_Voucher
	Doc1.LoadEmpty_Voucher
End Sub

		End If
	End If
End Function

		QStr = QStr & " M10T.R_LATEST_B_TARIH, M10T.R_BAKIYEKAYNAK "
		QStr = QStr & " FROM MMPS10T M10T"
		QStr = QStr & " LEFT JOIN STOK00 ON STOK00.KOD=M10T.R_MAMULKODU"
		QStr = QStr & " LEFT JOIN STOK00 S00 ON S00.KOD=M10T.R_KAYNAKKODU"
		QStr = QStr & " WHERE M10T.ACIK_KAPALI<>'K' AND M10T.R_STATUS='R' AND M10T.R_BAKIYEKAYNAK>0 "
		QStr = QStr & " AND M10T.R_KAYNAKKODU='" & S20T.KOD(i) & "'"
		QStr = QStr & " ORDER BY M10T.R_LATEST_B_TARIH, M10T.EVRAKNO"

		Set M10T = Doc.RunSQLQuery("dfgdsfhs", QStr)

		QStr = "SELECT Sum(SF_MIKTAR) SF_MIKTAR"
		QStr = QStr & " FROM STOK01 "
		QStr = QStr & " WHERE KOD='" & S20T.KOD(i) & "'"
		QStr = QStr & " AND AMBCODE IN ('01', '02')"
		Set S01 = Doc.RunSQLQuery("sdfgddsdfhs", QStr)
		if S01.GetRowCOunt() > 0 Then 
			DepoStok = S01.SF_MIKTAR(1)
		Else
			DepoStok = 0
		End if
		satirNo = 1

		Do
			if DepoStok = 0 Then Exit Do
			if satirNo > M10T.GetRowCount() Then Exit Do

			if DepoStok > M10T.R_BAKIYEKAYNAK(satirNo) Then
				Dagitilan = M10T.R_BAKIYEKAYNAK(satirNo)
			Else
				Dagitilan = DepoStok
			End if
			DepoStok = DepoStok - Dagitilan
			M10T.R_BAKIYEKAYNAK(satirNo) = M10T.R_BAKIYEKAYNAK(satirNo) - Dagitilan
			if M10T.R_BAKIYEKAYNAK(satirNo) <= 0 Then satirNo = SatirNo + 1
		Loop
		if satirNo <= M10T.GetRowCount() Then
			MsgStr = MsgStr & vbcrlf & "Hammadde Kodu : " & trim(S20T.KOD(i)) & " " & trim(M10T.HMD_ADI(satirNo)) & _
							", Giriþ Miktarý : " & S20T.SF_STOK_MIKTAR(i) & " " & M10T.HMD_UNIT(satirNo)
			MGMiktar = S20T.SF_STOK_MIKTAR(i)
			Do
				if MGMiktar = 0 Then Exit Do
				if satirNo > M10T.GetRowCount() Then Exit Do

				if MGMiktar > M10T.R_BAKIYEKAYNAK(satirNo) Then
					Dagitilan = M10T.R_BAKIYEKAYNAK(satirNo)
				Else
					Dagitilan = MGMiktar
				End if
				MGMiktar = MGMiktar - Dagitilan
				M10T.R_BAKIYEKAYNAK(satirNo) = M10T.R_BAKIYEKAYNAK(satirNo) - Dagitilan
				MsgStr = MsgStr & vbcrlf & "    MPS No : " & trim(M10T.EVRAKNO(satirNo)) & _
							", Mamul : " & trim(M10T.R_MAMULKODU(satirNo)) & " " & trim(M10T.MAMUL_ADI(satirNo)) & _
							", Ýhtiyaç Miktarý : " & Dagitilan & _
							", Ýhtiyaç Tarihi : " & Doc.StrToNorDate(M10T.R_LATEST_B_TARIH(satirNo))

				if M10T.R_BAKIYEKAYNAK(satirNo) <= 0 Then satirNo = SatirNo + 1
			Loop
		End if
	Next

	if trim(MsgStr) = "" Then Exit Sub

	MailFrom = "erp@formamakina.com" 
	MailTo = GetMailReceiver("MGMAIL")
	if MailTo = "" Then Exit Sub
	MailHeader = "Üretim Giriþi Bilgilendirme Mesajý - Fiþ No : " & trim(STOK20E.EVRAKNO)
	MailMessage = MsgStr
	DosyaAdi = ""

	Doc.RunMacro5 "MAKR7S.FREEMAIL", "SendMail", MailFrom, MailTo,MailHeader,MailMessage, DosyaAdi

End Sub
Function GetMailReceiver(KullaniciGrubu)
	MailStr = ""
	QStr = "SELECT EMAIL FROM UDEF"
	QStr = QStr & " WHERE KOD IN (SELECT KOD FROM UDEFT12 WHERE OZEL_HAK_KODU='" & trim(KullaniciGrubu) & "')"
	QStr = QStr & " AND TRIM(UDEF.EMAIL) IS NOT NULL"
	
	Set UDF = Doc.RunSQLQuery("sdfsdfhsdh", QStr)
	For i = 1 to UDF.GetRowCount()
		if trim(MailStr) <> "" Then MailStr = MailStr & ","
		MailStr = MailStr & trim(UDF.EMAIL(i))
	Next
	GetMailReceiver = MailStr
End Function
