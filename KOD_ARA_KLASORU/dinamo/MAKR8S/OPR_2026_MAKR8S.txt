Function GetTableHTML(Table, ByVal HTMLBaslik, TabloBaslikArray)
	HTMLBaslik = Replace(HTMLBaslik, vbcrlf, "<br>")
	HTMLBaslik = "<p><b><font size=""3"">" & HTMLBaslik & "</font></b></p>"
	FieldArr = Table.GetFields()

	HTMLStr = ""
	HTMLStr = "<html  lang='tr-TR'><meta http-equiv="&Chr(34)&"refresh"&Chr(34)& "content="&Chr(34)&"30"&Chr(34)&"><body>" & HTMLBaslik
	HTMLStr = HTMLStr & "<table border=1 cellspacing=0 cellpadding=2>"
	if trim(TabloBaslikArray(0)) <> "" Then
		HTMLStr = HTMLStr & "<tr>"
		For each Baslik in TabloBaslikArray
			HTMLStr = HTMLStr & "<th bgcolor=#EEEEFF>" & Baslik & "</th>"
		Next
		HTMLStr = HTMLStr & "</tr>"
	End if

	For i=1 to Table.GetRecCount
		HTMLStr = HTMLStr & "<tr>"
		For j = 0 to uBound(FieldArr,1)
			if j <= uBound(TabloBaslikArray) Then
				AlanTuru = FieldArr(j, 2)
				AlanDegeri = Table.GetFieldValue(FieldArr(j,0), i)
				if InStr(1, FieldArr(j, 0), "_DBL") > 0 Then
					Align = "right"
				Else
					Align = "left"
				End if
				if AlanTuru = "DOUBLE" Then
					Align = "right"
				Elseif AlanTuru = "DATE" Then
					AlanDegeri = Doc.StrToNorDate(AlanDegeri)
				Elseif AlanTuru = "CHAR" Then
					AlanDegeri = trim(AlanDegeri)
					AlanDegeri = Replace(AlanDegeri, vbcrlf, "<br>")
					'Call ReplaceEntities(AlanDegeri)
				End if

				HTMLStr = HTMLStr & "<td align=" & Align & ">" & AlanDegeri & "</td>"
			End if
		Next
		HTMLStr = HTMLStr & "</tr>"
	Next
	HTMLStr = HTMLStr & "</table></body></html>"
	GetTableHTML = HTMLStr
End Function
